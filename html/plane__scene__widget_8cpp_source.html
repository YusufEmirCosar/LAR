30bool validGroundPosition(double latitude, double longitude) noexcept {
31 constexpr double Pi = 3.14159265358979323846;
32 constexpr double HalfPi = Pi * 0.5;
33 return std::isfinite(latitude) && std::isfinite(longitude) && latitude >= -HalfPi &&
34 latitude <= HalfPi && longitude >= -Pi && longitude <= Pi;
35}

77void PlaneSceneWidget::setSceneState(const LarSceneState &state) {
78 if (!m_surfaceGroundOrigin && state.hasScene &&
79 fieldAvailable(state.availableFields, StateField::Location0) &&
80 fieldAvailable(state.availableFields, StateField::Location1) &&
81 validGroundPosition(state.plane.location[0], state.plane.location[1])) {
82 m_surfaceGroundOrigin =
83 GeoCoordinateRadians{state.plane.location[0], state.plane.location[1]};
84 }
85 m_scene = state;
86 refreshSurfaceState();
87 updateTerrainPlacement();
88 requestTerrainIfNeeded();
89 const QQuaternion orientation = PlaneAttitudeTransform::orientation(
90 m_scene.plane, m_scene.availableFields, &m_attitudeComplete);
91 Q_UNUSED(orientation)
92 update();

185void PlaneSceneWidget::initializeGL() {
186 if (context() != nullptr) {
187 m_contextConnection = connect(context(), &QOpenGLContext::aboutToBeDestroyed, this,
188 &PlaneSceneWidget::cleanupGL, Qt::UniqueConnection);
189 }
190 if (m_model == nullptr) {
191 const QString modelPath =

193 if (!loadModelFromFile(modelPath)) {
194 return;
195 }
196 } else {
197 m_renderer.setModel(m_model);
198 }
199 m_renderer.setSurfaceVisible(m_surfaceVisible);
200
201 if (m_skyboxes.count() > 0) {
202 loadSkybox(std::clamp(m_skyboxIndex, 0, m_skyboxes.count() - 1));
203 } else {
204 emit diagnosticRaised(QStringLiteral("No packaged skyboxes were found."));
205 }
206 if (m_skyboxes.rejectedSetCount() > 0) {
207 emit diagnosticRaised(QStringLiteral("Skipped %1 incomplete or mismatched cubemap set(s).")
208 .arg(m_skyboxes.rejectedSetCount()));
209 }
210 QString error;
211 if (!m_renderer.initialize(&error)) {
212 setDiagnostic(error);
213 return;
214 }
215 m_diagnostic.clear();

226void PlaneSceneWidget::paintGL() {
227 if (m_renderer.ready()) {
228 const double cameraDistance = m_orbitCamera.distance();
229 const float nearClip = static_cast<float>(std::max(0.1, cameraDistance / 100'000.0));
230 const double surfaceDistance =
231 std::hypot(static_cast<double>(m_surfaceState.surfaceHalfExtent),
232 static_cast<double>(m_surfaceState.surfaceHeight));
233 double terrainDistance = 0.0;
234 if (m_terrainVisible && m_terrainPatch) {
235 double altitudeMeters = m_terrainPatch->centerElevationMeters -
236 static_cast<double>(m_surfaceState.surfaceHeight) *
237 m_terrainPatch->metersPerSceneUnit;
238 if (fieldAvailable(m_scene.availableFields, StateField::Location2) &&
239 std::isfinite(m_scene.plane.location[2])) {
240 altitudeMeters = m_scene.plane.location[2];
241 }
242 const double verticalMeters =
243 std::max(std::abs(m_terrainPatch->minimumElevationMeters - altitudeMeters),
244 std::abs(m_terrainPatch->maximumElevationMeters - altitudeMeters));
245 terrainDistance = std::hypot(m_terrainPatch->halfExtentMeters * 1.5, verticalMeters) /
246 m_terrainPatch->metersPerSceneUnit;
247 }
248 const float farClip = static_cast<float>(
249 std::max({250.0, cameraDistance * 4.0, surfaceDistance * 2.0, terrainDistance * 1.5}));
250 QMatrix4x4 projection;
251 projection.perspective(45.0F,
252 static_cast<float>(std::max(1, width())) /
253 static_cast<float>(std::max(1, height())),
254 nearClip, farClip);
255 const QMatrix4x4 view = m_orbitCamera.viewMatrix(headingRadians());
256 const QQuaternion attitude = PlaneAttitudeTransform::orientation(
257 m_scene.plane, m_scene.availableFields, &m_attitudeComplete);
258 QString error;
259 if (!m_renderer.draw(view, projection, attitude, &error)) {
260 setDiagnostic(error.isEmpty() ? QStringLiteral("Plane GPU upload failed.") : error);
261 }
262 }
263
264 QPainter painter(this);
265 painter.setRenderHint(QPainter::Antialiasing);
266 if (!m_diagnostic.isEmpty()) {
267 painter.setPen(QColor(QStringLiteral("#ffb4a8")));
268 painter.setBrush(QColor(8, 12, 20, 220));
269 painter.drawText(rect().adjusted(30, 70, -30, -30),
270 Qt::AlignHCenter | Qt::AlignTop | Qt::TextWordWrap, m_diagnostic);
271 } else if (!m_scene.hasScene) {
272 painter.setPen(QColor(QStringLiteral("#e5ece9")));
273 painter.drawText(rect().adjusted(30, 80, -30, -30), Qt::AlignCenter,
274 QStringLiteral("WAITING FOR PLANE DATA"));
275 } else if (!m_attitudeComplete) {
276 painter.setPen(QColor(QStringLiteral("#ffd27a")));
277 painter.drawText(rect().adjusted(24, 72, -24, -24), Qt::AlignHCenter | Qt::AlignTop,
278 QStringLiteral("ATTITUDE DATA INCOMPLETE"));
279 }
280 if (m_surfaceVisible || m_terrainVisible) {
281 const QFontMetrics metrics(painter.font());
282 const QString terrainSourceLabel =
283 QStringLiteral("DT%1").arg(dtedLevelNumber(m_terrainDataset.level));
284 QString statusLabel;
285 if (m_surfaceVisible && m_terrainVisible) {
286 statusLabel =
287 QStringLiteral("Grid: %1 • Terrain: %2")
288 .arg(GridGeometryBuilder::formatDistance(m_surfaceState.gridSpacingMeters),
289 terrainPatchReady() ? terrainSourceLabel : QStringLiteral("Loading"));
290 } else if (m_surfaceVisible) {
291 statusLabel =
292 QStringLiteral("Grid: %1")
293 .arg(GridGeometryBuilder::formatDistance(m_surfaceState.gridSpacingMeters));
294 } else {
295 statusLabel = terrainPatchReady()
296 ? QStringLiteral("Terrain: %1").arg(terrainSourceLabel)
297 : QStringLiteral("Terrain: Loading");
298 }
299 const int labelWidth =
300 std::min(metrics.horizontalAdvance(statusLabel) + 22, std::max(0, width() - 36));
301 const QRect labelRect(std::max(0, (width() - labelWidth) / 2), height() - 49, labelWidth,
302 31);
303 painter.setPen(QColor(235, 242, 239, 235));
304 painter.setBrush(QColor(8, 12, 20, 190));
305 painter.drawRoundedRect(labelRect, 7.0, 7.0);
306 painter.drawText(labelRect, Qt::AlignCenter, statusLabel);
307 }
308 painter.end();
309 emit frameRendered();
310}
