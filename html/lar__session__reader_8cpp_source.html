56bool LarSessionReader::parseStream(QIODevice *device, QString *error) {
57 if (device->size() < lar::session::HeaderSize || device->read(4) != QByteArrayLiteral("LAR1")) {
58 if (error)
59 *error = QStringLiteral("File magic is not LAR1");
60 return false;
61 }
62
63 QDataStream stream(device);
64 configureStream(stream);
65 quint32 packetOffset = 0;
66 stream >> packetOffset;
67
68 if (stream.status() != QDataStream::Ok || packetOffset <= lar::session::HeaderSize ||
69 packetOffset > quint64(device->size()) ||
70 packetOffset - lar::session::HeaderSize > lar::session::MaximumMappingSize) {
71 if (error)
72 *error = QStringLiteral("Invalid UDP package-section offset");
73 return false;
74 }
75
76 const QByteArray mappingJson = device->read(qint64(packetOffset) - lar::session::HeaderSize);
77 PacketMapping mapping;
78 JsonMappingRepository repo;
79 QString mappingError;
80 if (!repo.loadJson(mappingJson, &mapping, &mappingError)) {
81 if (error)
82 *error = QStringLiteral("Invalid embedded mapping: %1").arg(mappingError);
83 return false;
84 }
85
86 if (device->pos() != packetOffset && !device->seek(packetOffset)) {
87 if (error)
88 *error = QStringLiteral("Cannot seek to the UDP package section");
89 return false;
90 }
91
92 quint64 previousTime = 0;
93 bool hasPreviousTime = false;
94 QVector<RecordIndex> recordIndex;
95 const qint64 estimatedRecordSize = qMax<qint64>(
96 1, qint64(mapping.minimumPacketSize()) + qint64(sizeof(quint64) + sizeof(quint32)));
97 const qint64 estimatedCount =
98 qMax<qint64>(0, (device->size() - packetOffset) / estimatedRecordSize);
99 recordIndex.reserve(
100 int(qMin<qint64>(qint64(lar::session::MaximumRecordCount), estimatedCount)));
101 while (!device->atEnd()) {
102 if (recordIndex.size() >= qsizetype(lar::session::MaximumRecordCount)) {
103 if (error)
104 *error = QStringLiteral("Session contains too many UDP packages");
105 return false;
106 }
107 if (device->size() - device->pos() < qint64(sizeof(quint64) + sizeof(quint32))) {
108 if (error)
109 *error = QStringLiteral("Truncated UDP package record header");
110 return false;
111 }
112 quint64 relativeTimeMs = 0;
113 quint32 packetSize = 0;
114 stream >> relativeTimeMs >> packetSize;
115 if (stream.status() != QDataStream::Ok) {
116 if (error)
117 *error = QStringLiteral("Invalid UDP package record header");
118 return false;
119 }
120 if (relativeTimeMs > lar::session::MaximumDurationMilliseconds) {
121 if (error) {
122 *error = QStringLiteral("Session timestamp exceeds the supported 365-day limit");
123 }
124 return false;
125 }
126 if (hasPreviousTime && relativeTimeMs < previousTime) {
127 if (error) {
128 *error = QStringLiteral("Session package time is not monotonic");
129 }
130 return false;
131 }
132 if (packetSize == 0 || packetSize > lar::session::MaximumPacketSize ||
133 qint64(packetSize) > device->size() - device->pos()) {
134 if (error)
135 *error = QStringLiteral("Invalid or truncated UDP package payload");
136 return false;
137 }
138 const qint64 payloadOffset = device->pos();
139 const QByteArray packet = device->read(packetSize);
140 SessionStateItem item;
141 const auto timestamp = SessionTimestamp::fromStoredMilliseconds(relativeTimeMs);
142 if (!timestamp) {
143 if (error) {
144 *error = QStringLiteral("Session timestamp exceeds the supported 365-day limit");
145 }
146 return false;
147 }
148 item.timestamp = *timestamp;
149 QString packetError;
150 if (!mapping.decode(packet, &item.state, &packetError)) {
151 if (error) {
152 *error = QStringLiteral("Stored UDP package %1 is invalid: %2")
153 .arg(recordIndex.size())
154 .arg(packetError);
155 }
156 return false;
157 }
158 previousTime = relativeTimeMs;
159 hasPreviousTime = true;
160 recordIndex.append(RecordIndex{item.timestamp, payloadOffset, packetSize});
161 }
162
163 m_mapping = std::move(mapping);
164 m_mappingJson = mappingJson;
165 m_recordIndex = std::move(recordIndex);
166 m_duration =
167 m_recordIndex.isEmpty() ? SessionTimestamp::zero() : m_recordIndex.constLast().timestamp;
168 m_isValid = true;
169 return true;
170}

185bool LarSessionReader::recordAt(int index, SessionStateItem *item, QString *error) const {
186 if (!item) {
187 if (error)
188 *error = QStringLiteral("Session record output is null");
189 return false;
190 }
191 if (!m_isValid) {
192 if (error)
193 *error = QStringLiteral("Session reader is not valid");
194 return false;
195 }
196 if (index < 0 || index >= m_recordIndex.size()) {
197 if (error)
198 *error = QStringLiteral("Session record index is out of range");
199 return false;
200 }
201 const RecordIndex &record = m_recordIndex.at(index);
202
203 QByteArray packet;
204 if (m_sourceFile) {
205 if (!m_sourceFile->seek(record.payloadOffset)) {
206 if (error)
207 *error = m_sourceFile->errorString();
208 return false;
209 }
210 packet = m_sourceFile->read(record.payloadSize);
211 } else {
212 if (record.payloadOffset < 0 ||
213 record.payloadOffset > qint64(m_sourceData.size()) - qint64(record.payloadSize)) {
214
215 if (error)
216 *error = QStringLiteral("Session record range is invalid");
217 return false;
218 }
219 packet = m_sourceData.mid(record.payloadOffset, record.payloadSize);
220 }
221 if (packet.size() != qint64(record.payloadSize)) {
222 if (error) {
223 *error = QStringLiteral("Session record payload could not be read completely");
224 }
225 return false;
226 }
227
228 SessionStateItem decoded;
229 decoded.timestamp = record.timestamp;
230 QString decodeError;
231 if (!m_mapping.decode(packet, &decoded.state, &decodeError)) {
232 if (error) {
233 *error = QStringLiteral("Session record cannot be decoded: %1").arg(decodeError);
234 }
235 return false;
236 }
237 *item = std::move(decoded);
238 return true;
239}
