19bool LarSessionReader::loadFile(const QString &path, QString *error) {
20 close();
21
22 if (path.isEmpty()) {
23 if (error)
24 *error = QStringLiteral("Session file path is empty");
25 return false;
26 }
27
28 auto file = std::make_unique<QFile>(path);
29 if (!file->open(QIODevice::ReadOnly)) {
30 if (error)
31 *error = file->errorString();
32 return false;
33 }
34
35 // Most recordings are small enough to become immutable playback
36 // snapshots. Besides removing platform-specific QFile seek costs, this
37 // ensures playback observes exactly the bytes that passed validation.
38 const qint64 sourceSize = file->size();
39 if (sourceSize >= 0 && sourceSize <= lar::session::MaximumResidentSourceBytes) {
40 QByteArray source = file->readAll();
41 if (qint64(source.size()) != sourceSize) {
42 if (error)
43 *error = file->errorString().isEmpty()
44 ? QStringLiteral("Session file could not be cached completely")
45 : file->errorString();
46 return false;
47 }
48 QBuffer buffer(&source);
49 if (!buffer.open(QIODevice::ReadOnly)) {
50 if (error)
51 *error = QStringLiteral("Cannot open the cached session data");
52 return false;
53 }
54 if (!parseStream(&buffer, error))
55 return false;
56 m_sourceData = std::move(source);
57 return true;
58 }
59
60 if (!parseStream(file.get(), error))
61 return false;
62 m_sourceFile = std::move(file);
63 return true;

82bool LarSessionReader::parseStream(QIODevice *device, QString *error) {
83 if (device->size() < lar::session::HeaderSize || device->read(4) != QByteArrayLiteral("LAR1")) {
84 if (error)
85 *error = QStringLiteral("File magic is not LAR1");
86 return false;
87 }
88
89 QDataStream stream(device);
90 configureStream(stream);
91 quint32 packetOffset = 0;
92 stream >> packetOffset;
93
94 if (stream.status() != QDataStream::Ok || packetOffset <= lar::session::HeaderSize ||
95 packetOffset > quint64(device->size()) ||
96 packetOffset - lar::session::HeaderSize > lar::session::MaximumMappingSize) {
97 if (error)
98 *error = QStringLiteral("Invalid UDP package-section offset");
99 return false;
100 }
101
102 const QByteArray mappingJson = device->read(qint64(packetOffset) - lar::session::HeaderSize);
103 PacketMapping mapping;
104 JsonMappingRepository repo;
105 QString mappingError;
106 if (!repo.loadJson(mappingJson, &mapping, &mappingError)) {
107 if (error)
108 *error = QStringLiteral("Invalid embedded mapping: %1").arg(mappingError);
109 return false;
110 }
111
112 if (device->pos() != packetOffset && !device->seek(packetOffset)) {
113 if (error)
114 *error = QStringLiteral("Cannot seek to the UDP package section");
115 return false;
116 }
117
118 quint64 previousTime = 0;
119 bool hasPreviousTime = false;
120 qint64 recordCount = 0;
121 SessionTimestamp duration;
122 QVector<Checkpoint> checkpoints;
123 QVector<RecordIndex> residentIndex;
124 const qint64 maximumResidentRecords =
125 lar::session::MaximumResidentRecordIndexBytes / qint64(sizeof(RecordIndex));
126 constexpr qint64 IndexReserveChunkRecords = 262'144;
127 const qint64 minimumRecordBytes =
128 qint64(sizeof(quint64) + sizeof(quint32)) + qMax(1, mapping.minimumPacketSize());
129 const qint64 possibleRecordCount =
130 qMax<qint64>(0, device->size() - qint64(packetOffset)) / minimumRecordBytes;
131 residentIndex.reserve(static_cast<qsizetype>(
132 qMin(qMin(possibleRecordCount, maximumResidentRecords), IndexReserveChunkRecords)));
133 bool retainResidentIndex = true;
134 while (!device->atEnd()) {
135 const qint64 headerOffset = device->pos();
136 if (device->size() - device->pos() < qint64(sizeof(quint64) + sizeof(quint32))) {
137 if (error)
138 *error = QStringLiteral("Truncated UDP package record header");
139 return false;
140 }
141 quint64 relativeTimeMs = 0;
142 quint32 packetSize = 0;
143 stream >> relativeTimeMs >> packetSize;
144 if (stream.status() != QDataStream::Ok) {
145 if (error)
146 *error = QStringLiteral("Invalid UDP package record header");
147 return false;
148 }
149 if (relativeTimeMs > lar::session::MaximumDurationMilliseconds) {
150 if (error) {
151 *error = QStringLiteral("Session timestamp exceeds the supported 365-day limit");
152 }
153 return false;
154 }
155 if (hasPreviousTime && relativeTimeMs < previousTime) {
156 if (error) {
157 *error = QStringLiteral("Session package time is not monotonic");
158 }
159 return false;
160 }
161 if (packetSize == 0 || packetSize > lar::session::MaximumPacketSize ||
162 qint64(packetSize) > device->size() - device->pos()) {
163 if (error)
164 *error = QStringLiteral("Invalid or truncated UDP package payload");
165 return false;
166 }
167 const qint64 payloadOffset = device->pos();
168 const QByteArray packet = device->read(packetSize);
169 SessionStateItem item;
170 const auto timestamp = SessionTimestamp::fromStoredMilliseconds(relativeTimeMs);
171 if (!timestamp) {
172 if (error) {
173 *error = QStringLiteral("Session timestamp exceeds the supported 365-day limit");
174 }
175 return false;
176 }
177 item.timestamp = *timestamp;
178 QString packetError;
179 if (!mapping.decode(packet, &item.state, &packetError)) {
180 if (error) {
181 *error = QStringLiteral("Stored UDP package %1 is invalid: %2")
182 .arg(recordCount)
183 .arg(packetError);
184 }
185 return false;
186 }
187 previousTime = relativeTimeMs;
188 hasPreviousTime = true;
189 duration = item.timestamp;
190 if (recordCount % RecordsPerPage == 0)
191 checkpoints.append(Checkpoint{recordCount, headerOffset, item.timestamp});
192 if (retainResidentIndex) {
193 if (recordCount >= maximumResidentRecords) {
194 // Preserve support for very large valid recordings without an
195 // unbounded index allocation. Their sparse fallback remains
196 // correct, while ordinary recordings keep constant-time access.
197 residentIndex.clear();
198 residentIndex.squeeze();
199 retainResidentIndex = false;
200 } else {
201 if (residentIndex.size() == residentIndex.capacity()) {
202 const qint64 nextCapacity =
203 qMin(maximumResidentRecords,
204 qint64(residentIndex.capacity()) + IndexReserveChunkRecords);
205 residentIndex.reserve(static_cast<qsizetype>(nextCapacity));
206 }
207 residentIndex.append(RecordIndex{item.timestamp, payloadOffset, packetSize});
208 }
209 }
210 ++recordCount;
211 }
212
213 m_mapping = std::move(mapping);
214 m_mappingJson = mappingJson;
215 m_checkpoints = std::move(checkpoints);
216 m_residentIndex = std::move(residentIndex);
217 m_residentIndexAvailable = retainResidentIndex;
218 m_page.clear();
219 m_pageFirstRecord = -1;
220 m_sourceSize = device->size();
221 m_recordCount = recordCount;
222 m_duration = duration;
223 m_isValid = true;
224 return true;
225}

246bool LarSessionReader::ensurePage(qint64 index, QString *error) const {
247 if (index < 0 || index >= m_recordCount) {
248 if (error)
249 *error = QStringLiteral("Session record index is out of range");
250 return false;
251 }
252
253 const qint64 firstRecord = (index / RecordsPerPage) * RecordsPerPage;
254 if (m_pageFirstRecord == firstRecord && index - firstRecord < m_page.size())
255 return true;
256
257 const qint64 checkpointIndex = index / RecordsPerPage;
258 if (checkpointIndex < 0 || checkpointIndex >= m_checkpoints.size()) {
259 if (error)
260 *error = QStringLiteral("Session checkpoint index is invalid");
261 return false;
262 }
263 const Checkpoint checkpoint = m_checkpoints.at(static_cast<qsizetype>(checkpointIndex));
264 if (checkpoint.recordIndex != firstRecord) {
265 if (error)
266 *error = QStringLiteral("Session checkpoint table is inconsistent");
267 return false;
268 }
269
270 QBuffer memoryDevice;
271 QIODevice *device = nullptr;
272 if (m_sourceFile) {
273 device = m_sourceFile.get();
274 } else {
275 if (m_sourceData.size() != m_sourceSize) {
276 if (error)
277 *error = QStringLiteral("Session data changed after validation");
278 return false;
279 }
280 memoryDevice.setData(m_sourceData);
281 if (!memoryDevice.open(QIODevice::ReadOnly)) {
282 if (error)
283 *error = QStringLiteral("Cannot open session data cache");
284 return false;
285 }
286 device = &memoryDevice;
287 }
288 if (!device->isOpen() || device->size() != m_sourceSize ||
289 !device->seek(checkpoint.headerOffset)) {
290 if (error) {
291 *error = m_sourceFile ? QStringLiteral("Session file changed after validation: %1")
292 .arg(m_sourceFile->errorString())
293 : QStringLiteral("Session checkpoint cannot be reached");
294 }
295 return false;
296 }
297
298 QVector<RecordIndex> page;
299 page.reserve(static_cast<qsizetype>(RecordsPerPage));
300 QDataStream stream(device);
301 configureStream(stream);
302 const qint64 endRecord = qMin(m_recordCount, firstRecord + RecordsPerPage);
303 for (qint64 current = firstRecord; current < endRecord; ++current) {
304 if (device->size() - device->pos() < qint64(sizeof(quint64) + sizeof(quint32))) {
305 if (error)
306 *error = QStringLiteral("Session record header changed after validation");
307 return false;
308 }
309 quint64 relativeTimeMs = 0;
310 quint32 packetSize = 0;
311 stream >> relativeTimeMs >> packetSize;
312 const qint64 payloadOffset = device->pos();
313 const auto timestamp = SessionTimestamp::fromStoredMilliseconds(relativeTimeMs);
314 if (stream.status() != QDataStream::Ok || !timestamp || packetSize == 0 ||
315 packetSize > lar::session::MaximumPacketSize ||
316 qint64(packetSize) > device->size() - payloadOffset) {
317 if (error)
318 *error = QStringLiteral("Session record changed after validation");
319 return false;
320 }
321 page.append(RecordIndex{*timestamp, payloadOffset, packetSize});
322 if (!device->seek(payloadOffset + qint64(packetSize))) {
323 if (error)
324 *error = QStringLiteral("Session record payload cannot be skipped");
325 return false;
326 }
327 }
328 if (device->size() != m_sourceSize) {
329 if (error)
330 *error = QStringLiteral("Session source size changed during access");
331 return false;
332 }
333 m_page = std::move(page);
334 m_pageFirstRecord = firstRecord;
335 return true;
336}

339 QString *error) const {
340 if (!m_isValid) {
341 if (error)

369bool LarSessionReader::findRecordAtOrBefore(SessionTimestamp position, qint64 *index,
370 QString *error) const {
371 if (!index) {
372 if (error)
373 *error = QStringLiteral("Session record-index output is null");
374 return false;
375 }
376 if (!m_isValid) {
377 if (error)
378 *error = QStringLiteral("Session reader is not valid");
379 return false;
380 }
381 if (m_recordCount <= 0 || m_checkpoints.isEmpty()) {
382 if (error)
383 *error = QStringLiteral("Session contains no records");
384 return false;
385 }
386
387 if (m_residentIndexAvailable) {
388 qsizetype low = 0;
389 qsizetype high = m_residentIndex.size();
390 while (low < high) {
391 const qsizetype middle = low + (high - low) / 2;
392 if (m_residentIndex.at(middle).timestamp <= position)
393 low = middle + 1;
394 else
395 high = middle;
396 }
397 *index = static_cast<qint64>(low == 0 ? 0 : low - 1);
398 return true;
399 }
400
401 // Checkpoint timestamps form a compact first-level index. Choosing the
402 // last page whose first timestamp is eligible avoids the former pattern
403 // of rebuilding a different 4,096-record page for every binary-search
404 // probe across the complete file.
405 qsizetype checkpointLow = 0;
406 qsizetype checkpointHigh = m_checkpoints.size();
407 while (checkpointLow < checkpointHigh) {
408 const qsizetype middle = checkpointLow + (checkpointHigh - checkpointLow) / 2;
409 if (m_checkpoints.at(middle).timestamp <= position)
410 checkpointLow = middle + 1;
411 else
412 checkpointHigh = middle;
413 }
414 const qsizetype checkpointIndex = checkpointLow == 0 ? 0 : checkpointLow - 1;
415 const Checkpoint &checkpoint = m_checkpoints.at(checkpointIndex);
416 if (!ensurePage(checkpoint.recordIndex, error))
417 return false;
418
419 // The second-level search is entirely in the one bounded location page.
420 // Upper-bound semantics select the last duplicate timestamp, including
421 // duplicates that cross a page boundary.
422 qsizetype pageLow = 0;
423 qsizetype pageHigh = m_page.size();
424 while (pageLow < pageHigh) {
425 const qsizetype middle = pageLow + (pageHigh - pageLow) / 2;
426 if (m_page.at(middle).timestamp <= position)
427 pageLow = middle + 1;
428 else
429 pageHigh = middle;
430 }
431 const qsizetype pageIndex = pageLow == 0 ? 0 : pageLow - 1;
432 *index = m_pageFirstRecord + static_cast<qint64>(pageIndex);
433 return true;
434}

436bool LarSessionReader::recordAt(qint64 index, SessionStateItem *item, QString *error) const {
437 if (!item) {
438 if (error)
439 *error = QStringLiteral("Session record output is null");
440 return false;
441 }
442 const RecordIndex *location = locationAt(index, error);
443 if (!location)
444 return false;
445 const RecordIndex record = *location;
446
447 QByteArray packet;
448 if (m_sourceFile) {
449 if (!m_sourceFile->seek(record.payloadOffset)) {
450 if (error)
451 *error = m_sourceFile->errorString();
452 return false;
453 }
454 packet = m_sourceFile->read(record.payloadSize);
455 } else {
456 if (record.payloadOffset < 0 ||
457 record.payloadOffset > qint64(m_sourceData.size()) - qint64(record.payloadSize)) {
458
459 if (error)
460 *error = QStringLiteral("Session record range is invalid");
461 return false;
462 }
463 packet = m_sourceData.mid(record.payloadOffset, record.payloadSize);
464 }
465 if (packet.size() != qint64(record.payloadSize)) {
466 if (error) {
467 *error = QStringLiteral("Session record payload could not be read completely");
468 }
469 return false;
470 }
471
472 SessionStateItem decoded;
473 decoded.timestamp = record.timestamp;
474 QString decodeError;
475 if (!m_mapping.decode(packet, &decoded.state, &decodeError)) {
476 if (error) {
477 *error = QStringLiteral("Session record cannot be decoded: %1").arg(decodeError);
478 }
479 return false;
480 }
481 *item = std::move(decoded);
482 return true;
483}
