25bool pointInTriangle(const std::array<double, 2> &a, const std::array<double, 2> &b,
26 const std::array<double, 2> &c, double x, double y) noexcept {
27 constexpr double Epsilon = 1.0e-8;
28 const double first = edge(a[0], a[1], b[0], b[1], x, y);
29 const double second = edge(b[0], b[1], c[0], c[1], x, y);
30 const double third = edge(c[0], c[1], a[0], a[1], x, y);
31 const bool negative = first < -Epsilon || second < -Epsilon || third < -Epsilon;
32 const bool positive = first > Epsilon || second > Epsilon || third > Epsilon;
33 return !(negative && positive);
34}

53bool PlaneLandMask::landAtLocal(double eastMeters, double northMeters) const noexcept {
54 if (!valid() || !std::isfinite(eastMeters) || !std::isfinite(northMeters)) {
55 return false;

57 if (coverage == PlaneLandCoverage::AllLand) {
58 return true;

63 return false;

76int PlaneLandMaskBuilder::resolutionFor(double halfExtentMeters) noexcept {
77 if (!std::isfinite(halfExtentMeters) || halfExtentMeters <= 0.0) {
78 return 0;
79 }
80 const double desired = std::ceil(2.0 * halfExtentMeters / DesiredMetersPerPixel);
81 int resolution = MinimumMaskResolution;
82 while (resolution < MaximumMaskResolution && static_cast<double>(resolution) < desired) {
83 resolution *= 2;
84 }
85 return std::clamp(resolution, MinimumMaskResolution, MaximumMaskResolution);
86}
