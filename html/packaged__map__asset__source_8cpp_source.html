39bool openTrustedDirectChild(const QString &canonicalDirectory, const QString &path, QFile *file) {
40 const QFileInfo before(path);
41 if (!isDirectChild(canonicalDirectory, before)) {
42 return false;
43 }
44 file->setFileName(before.absoluteFilePath());
45 if (!file->open(QIODevice::ReadOnly)) {
46 return false;
47 }
48 const QFileInfo after(path);
49 if (!isDirectChild(canonicalDirectory, after)) {
50 return false;
51 }
52#ifdef Q_OS_UNIX
53 struct stat pathStatus{};
54 struct stat openedStatus{};
55 const QByteArray nativePath = QFile::encodeName(after.absoluteFilePath());
56 if (::lstat(nativePath.constData(), &pathStatus) != 0 ||
57 ::fstat(file->handle(), &openedStatus) != 0 || !S_ISREG(pathStatus.st_mode) ||
58 !S_ISREG(openedStatus.st_mode) || pathStatus.st_dev != openedStatus.st_dev ||
59 pathStatus.st_ino != openedStatus.st_ino) {
60 return false;
61 }
62#endif
63 return true;
64}

88MapAssetReadResult PackagedMapAssetSource::load() const {
89 const QFileInfo directoryInfo(m_packageDirectory);
90 const QString canonicalDirectory = directoryInfo.canonicalFilePath();
91 if (!directoryInfo.exists() || !directoryInfo.isDir() || directoryInfo.isSymLink() ||
92 canonicalDirectory.isEmpty()) {
93
94 return failure(MapAssetError::Io,
95 QStringLiteral("The packaged map directory is unavailable."));
96 }
97
98 QFile assetFile;
99 QFile manifestFile;
100 if (!openTrustedDirectChild(canonicalDirectory, assetPath(), &assetFile) ||
101 !openTrustedDirectChild(canonicalDirectory, manifestPath(), &manifestFile)) {
102 return failure(MapAssetError::Integrity,
103 QStringLiteral("The packaged map location is not trusted."));
104 }
105 const qint64 assetSize = assetFile.size();
106 if (assetSize < static_cast<qint64>(format::HeaderSize) ||
107 assetSize > limits::MaximumAssetBytes) {
108 return failure(MapAssetError::Size,
109 QStringLiteral("The packaged map has an invalid size."));
110 }
111 if (manifestFile.size() <= 0 || manifestFile.size() > MaximumManifestBytes) {
112 return failure(MapAssetError::Integrity,
113 QStringLiteral("The packaged map manifest has an invalid size."));
114 }
115
116 QByteArray manifestBytes;
117 if (!readBounded(&manifestFile, MaximumManifestBytes, &manifestBytes)) {
118 return failure(MapAssetError::Io,
119 QStringLiteral("The packaged map manifest could not be read safely."));
120 }
121 QJsonParseError parseError;
122 const QJsonDocument manifestDocument = QJsonDocument::fromJson(manifestBytes, &parseError);
123 if (parseError.error != QJsonParseError::NoError || !manifestDocument.isObject()) {
124 return failure(MapAssetError::Integrity,
125 QStringLiteral("The packaged map manifest is invalid."));
126 }
127 const QJsonObject manifest = manifestDocument.object();
128 const QJsonValue sizeValue = manifest.value(QStringLiteral("bytes"));
129 const QJsonValue versionValue = manifest.value(QStringLiteral("formatVersion"));
130 const QJsonValue shaValue = manifest.value(QStringLiteral("sha256"));
131 if (!sizeValue.isDouble() || !versionValue.isDouble() || !shaValue.isString()) {
132 return failure(MapAssetError::Integrity,
133 QStringLiteral("The packaged map manifest is invalid."));
134 }
135 const qint64 expectedSize = sizeValue.toInteger(-1);
136 const int expectedVersion = versionValue.toInt(-1);
137 const QByteArray expectedSha = shaValue.toString().toLatin1().toLower();
138 if (expectedSize != assetSize || expectedVersion != static_cast<int>(format::Version) ||
139 expectedSha.size() != 64) {
140
141 return failure(MapAssetError::Integrity,
142 QStringLiteral("The packaged map manifest does not match."));
143 }
144
145 QByteArray assetBytes;
146 if (!readBounded(&assetFile, limits::MaximumAssetBytes, &assetBytes) ||
147 assetBytes.size() != expectedSize) {
148 return failure(MapAssetError::Io,
149 QStringLiteral("The packaged map could not be read completely."));
150 }
151 const QByteArray actualSha =
152 QCryptographicHash::hash(assetBytes, QCryptographicHash::Sha256).toHex();
153 if (actualSha != expectedSha) {
154 return failure(MapAssetError::Integrity,
155 QStringLiteral("The packaged map integrity check failed."));
156 }
157
158 return MapAssetReader::read(QByteArrayView(assetBytes));
159}
