36void HudRenderer::drawNoTrack(QPainter &painter, const QRectF &bounds) const {
37 painter.save();
38 painter.fillRect(bounds, HudBlack);
39 painter.setPen(QPen(HudGreenDim, 1.0));
40 QFont font = QFontDatabase::systemFont(QFontDatabase::FixedFont);
41 font.setPixelSize(std::max(12, static_cast<int>(bounds.height() * 0.035)));
42 painter.setFont(font);
43 painter.drawText(bounds, Qt::AlignCenter, QStringLiteral("NO TRACK"));
44 painter.restore();
45}

56 painter.setPen(NezAmber);

58 std::max(20.0, bounds.width() * 0.10), std::max(20.0, bounds.height() * 0.10),
59 -std::max(20.0, bounds.width() * 0.10), -std::max(20.0, bounds.height() * 0.10));
60 painter.drawText(textBounds, Qt::AlignCenter | Qt::TextWordWrap,
61 message.trimmed().isEmpty() ? QStringLiteral("DLZ input is invalid")
62 : message.trimmed());
63 painter.restore();

66void HudRenderer::draw(QPainter &painter, const QRectF &bounds, const dlz::Solution &solution,
67 const dlz::HudState &hudState, double displayedRangeNm) const {
68 const bool finiteInputs =
69 hudState.mode == dlz::HudMode::Prelaunch &&
70 std::isfinite(solution.aerodynamicMaximumRangeNm) &&
71 std::isfinite(solution.interceptRangeNm) && std::isfinite(solution.noEscapeRangeNm) &&
72 std::isfinite(solution.turnAndRunRangeNm) && std::isfinite(solution.minimumRangeNm) &&
73 std::isfinite(solution.timeOfFlightSeconds) && std::isfinite(hudState.currentRangeNm) &&
74 std::isfinite(hudState.rangeRateKnots) && std::isfinite(hudState.scaleMinimumNm) &&
75 std::isfinite(hudState.scaleMaximumNm) &&
76 std::isfinite(hudState.timeToImpactRemainingSeconds) && std::isfinite(displayedRangeNm) &&
77 hudState.scaleMaximumNm > hudState.scaleMinimumNm;
78 if (!finiteInputs) {
79 drawNoTrack(painter, bounds);
80 return;
81 }
82 painter.save();
83 painter.fillRect(bounds, HudBlack);
84 painter.setRenderHint(QPainter::Antialiasing, true);
85
86 const double marginX = std::max(20.0, bounds.width() * 0.08);
87 const double top = bounds.top() + std::max(28.0, bounds.height() * 0.10);
88 const double bottom = bounds.bottom() - std::max(28.0, bounds.height() * 0.10);
89 // Keep the primary DLZ scale centered in the HUD canvas. The range labels
90 // are laid out to its right, while the current-range caret stays adjacent
91 // to the scale so it reads as a marker on that same axis.
92 const double scaleX = bounds.center().x();
93 const double stapleWidth = std::max(18.0, std::min(42.0, bounds.width() * 0.075));
94 const double tickLength = std::max(16.0, std::min(32.0, bounds.width() * 0.055));
95 const double lineWidth = std::max(1.0, bounds.width() / 700.0);
96 const double caretX = scaleX - std::max(3.0, lineWidth * 2.0);
97 const auto yFor = [&](double range) {
98 return rangeToY(range, hudState.scaleMinimumNm, hudState.scaleMaximumNm, top, bottom);
99 };
100
101 const double yAero = yFor(solution.aerodynamicMaximumRangeNm);
102 const double yPi = yFor(solution.interceptRangeNm);
103 const double yNe = yFor(solution.noEscapeRangeNm);
104 const double yMin = yFor(solution.minimumRangeNm);
105 const double yCaret = yFor(displayedRangeNm);
106
107 painter.setPen(QPen(HudGreen, lineWidth, Qt::SolidLine, Qt::SquareCap));
108 painter.drawLine(QPointF(scaleX, top), QPointF(scaleX, bottom));
109
110 painter.setPen(QPen(HudGreen, lineWidth, Qt::SolidLine, Qt::SquareCap));
111 painter.drawLine(QPointF(scaleX, yAero), QPointF(scaleX + stapleWidth, yAero));
112 painter.drawLine(QPointF(scaleX + stapleWidth, yAero), QPointF(scaleX + stapleWidth, yMin));
113 painter.drawLine(QPointF(scaleX + stapleWidth, yMin), QPointF(scaleX, yMin));
114
115 const double bandTop = std::min(yNe, yMin);
116 const double bandBottom = std::max(yNe, yMin);
117 painter.fillRect(QRectF(scaleX - lineWidth * 0.5, bandTop, stapleWidth + lineWidth,
118 std::max(0.0, bandBottom - bandTop)),
119 QBrush(QColor(NezAmber.red(), NezAmber.green(), NezAmber.blue(), 70)));
120 painter.setPen(QPen(NezAmber, lineWidth, Qt::DashLine, Qt::SquareCap));
121 for (double y = bandTop + 4.0; y < bandBottom; y += 8.0) {
122 painter.drawLine(QPointF(scaleX, y), QPointF(scaleX + stapleWidth, y));
123 }
124
125 drawTick(painter, scaleX, yAero, tickLength, HudGreen, lineWidth);
126 drawTick(painter, scaleX, yPi, tickLength, HudGreen, lineWidth);
127 drawTick(painter, scaleX, yNe, tickLength, NezAmber, lineWidth);
128 drawTick(painter, scaleX, yMin, tickLength, HudGreen, lineWidth);
129
130 QPainterPath caret;
131 caret.moveTo(caretX, yCaret);
132 caret.lineTo(caretX - std::max(7.0, tickLength * 0.40),
133 yCaret - std::max(5.0, tickLength * 0.28));
134 caret.lineTo(caretX - std::max(7.0, tickLength * 0.40),
135 yCaret + std::max(5.0, tickLength * 0.28));
136 caret.closeSubpath();
137 painter.setPen(QPen(HudGreen, lineWidth, Qt::SolidLine, Qt::RoundCap));
138 painter.setBrush(HudGreen);
139 painter.drawPath(caret);
140
141 QFont font = QFontDatabase::systemFont(QFontDatabase::FixedFont);
142 font.setPixelSize(std::max(10, static_cast<int>(bounds.height() * 0.028)));
143 font.setBold(true);
144 painter.setFont(font);
145 painter.setPen(HudGreen);
146 const double currentRangeTextWidth = std::max(70.0, bounds.width() * 0.26);
147 // Keep the numeric marker visibly separate from the caret's left edge.
148 const double currentRangeTextGap = std::max(12.0, tickLength * 0.50);
149 painter.drawText(QRectF(caretX - currentRangeTextWidth - currentRangeTextGap,
150 yCaret - font.pixelSize() * 0.8, currentRangeTextWidth,
151 font.pixelSize() * 1.6),
152 Qt::AlignRight | Qt::AlignVCenter, rangeText(displayedRangeNm));
153
154 QFont labelFont = font;
155 labelFont.setPixelSize(std::max(9, static_cast<int>(bounds.height() * 0.022)));
156 painter.setFont(labelFont);
157 painter.setPen(HudGreenDim);
158 const auto label = [&](double y, const QString &name, double value) {
159 painter.drawText(
160 QRectF(scaleX + stapleWidth + 8.0, y - labelFont.pixelSize() * 0.75,
161 bounds.right() - scaleX - stapleWidth - 10.0, labelFont.pixelSize() * 1.5),
162 Qt::AlignLeft | Qt::AlignVCenter, QStringLiteral("%1 %2").arg(name, rangeText(value)));
163 };
164 label(yAero, QStringLiteral("RMAX"), solution.aerodynamicMaximumRangeNm);
165 label(yPi, QStringLiteral("RPI"), solution.interceptRangeNm);
166 painter.setPen(NezAmber);
167 label(yNe, QStringLiteral("RNE"), solution.noEscapeRangeNm);
168 label(yMin, QStringLiteral("RMIN"), solution.minimumRangeNm);
169
170 painter.setPen(HudGreenDim);
171 painter.drawText(QRectF(scaleX - 20.0, bottom + labelFont.pixelSize() * 0.5, 80.0,
172 labelFont.pixelSize() * 1.5),
173 Qt::AlignLeft,
174 QStringLiteral("0 / %1 NM").arg(hudState.scaleMaximumNm, 0, 'f', 0));
175
176 if (hudState.shootFlash) {
177 painter.setPen(QPen(HudGreen, lineWidth * 1.4, Qt::SolidLine, Qt::SquareCap));
178 painter.drawText(QRectF(bounds.left() + marginX, top - labelFont.pixelSize() * 1.8,
179 bounds.width() * 0.38, labelFont.pixelSize() * 1.6),
180 Qt::AlignLeft | Qt::AlignVCenter, QStringLiteral("SHOOT"));
181 }
182 painter.restore();
183}
