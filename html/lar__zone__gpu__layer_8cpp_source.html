174bool meshIsValid(const LarZoneMesh &mesh) noexcept {
175 if (mesh.vertices.size() % 3U != 0U ||
176 mesh.vertices.size() / 3U > LarZoneMeshBuilder::MaximumVertexCount ||
177 mesh.indices.size() > LarZoneMeshBuilder::MaximumIndexCount) {
178
179 return false;
180 }
181 for (std::size_t offset = 0U; offset + 2U < mesh.vertices.size(); offset += 3U) {
182 if (!std::isfinite(mesh.vertices[offset]) || !std::isfinite(mesh.vertices[offset + 1U]) ||
183 !std::isfinite(mesh.vertices[offset + 2U]) ||
184 std::abs(mesh.vertices[offset]) > lar::map::limits::MaximumAbsoluteLongitude ||
185 std::abs(mesh.vertices[offset + 1U]) > lar::map::limits::MaximumAbsoluteLatitude ||
186 std::abs(mesh.vertices[offset + 2U]) > lar::map::limits::MaximumAbsoluteMercatorY) {
187 return false;
188 }
189 }
190 const std::size_t vertexCount = mesh.vertices.size() / 3U;
191 if (std::any_of(mesh.indices.begin(), mesh.indices.end(), [vertexCount](std::uint32_t index) {
192 return static_cast<std::size_t>(index) >= vertexCount;
193 })) {
194
195 return false;
196 }
197 return rangeIsValid(mesh.inRangeFill, mesh.indices.size(), 3U) &&
198 rangeIsValid(mesh.inZoneFill, mesh.indices.size(), 3U) &&
199 rangeIsValid(mesh.inRangeLines, mesh.indices.size(), 2U) &&
200 rangeIsValid(mesh.inZoneLines, mesh.indices.size(), 2U);
201}

304bool LarZoneGpuLayer::writeBuffer(QOpenGLBuffer &buffer, int &capacityBytes, const void *data,
305 std::size_t sizeBytes, const QString &description) {
306 if (sizeBytes > static_cast<std::size_t>(std::numeric_limits<int>::max())) {
307 emit diagnosticRaised(
308 QStringLiteral("%1 exceeds the OpenGL upload limit.").arg(description));
309 return false;
310 }
311 const int requiredBytes = static_cast<int>(sizeBytes);
312 if (!buffer.bind()) {
313 emit diagnosticRaised(QStringLiteral("Could not bind %1.").arg(description));
314 return false;
315 }
316 if (requiredBytes > capacityBytes) {
317 capacityBytes = retainedCapacity(requiredBytes);
318 buffer.allocate(capacityBytes);
319 }
320 if (requiredBytes > 0) {
321 buffer.write(0, data, requiredBytes);
322 }
323 return true;

326bool LarZoneGpuLayer::upload(const LarZoneMesh &mesh) {
327 m_indexCount = 0U;
328 if (mesh.empty()) {
329 return true;
330 }
331 if (!meshIsValid(mesh)) {
332 emit diagnosticRaised(QStringLiteral("The LAR overlay mesh exceeds renderer limits."));
333 return false;
334 }
335 if (m_program == nullptr || !m_vertexArray.isCreated()) {
336 emit diagnosticRaised(QStringLiteral("LAR overlay GPU resources are unavailable."));
337 return false;
338 }
339
340 QOpenGLVertexArrayObject::Binder binder(&m_vertexArray);
341 if (!writeBuffer(m_vertexBuffer, m_vertexCapacityBytes, mesh.vertices.data(),
342 mesh.vertices.size() * sizeof(float),
343 QStringLiteral("the LAR overlay vertex buffer")) ||
344 !writeBuffer(m_indexBuffer, m_indexCapacityBytes, mesh.indices.data(),
345 mesh.indices.size() * sizeof(std::uint32_t),
346 QStringLiteral("the LAR overlay index buffer"))) {
347
348 return false;
349 }
350
351 m_inRangeFill = mesh.inRangeFill;
352 m_inZoneFill = mesh.inZoneFill;
353 m_inRangeLines = mesh.inRangeLines;
354 m_inZoneLines = mesh.inZoneLines;
355 m_indexCount = mesh.indices.size();
356 return true;
357}
