61void EarthOverlayCoordinator::complete(quint64 revision, const LarZoneMesh &mesh) {
62 if (!m_enabled || revision != m_revision)
63 return;
64 m_pending = false;
65 m_dirty = false;
66 m_mesh = mesh;
67 m_prepared = true;
68 m_longitudeBounds.reset();
69 if (!mesh.empty()) {
70 double minimum = std::numeric_limits<double>::infinity();
71 double maximum = -std::numeric_limits<double>::infinity();
72 for (std::size_t offset = 0; offset + 2U < mesh.vertices.size(); offset += 3U) {
73 const double longitude =
74 static_cast<double>(mesh.vertices[offset]) +
75 (mesh.coordinateSpace == LarZoneCoordinateSpace::GeographicDegrees
76 ? 0.0
77 : mesh.coordinateOrigin.x());
78 minimum = std::min(minimum, longitude);
79 maximum = std::max(maximum, longitude);
80 }
81 if (minimum <= maximum)
82 m_longitudeBounds = {{minimum, maximum}};
83 }
84 emit meshAvailable(mesh.inputRejected);
85}
