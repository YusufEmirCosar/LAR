59void copyLittleEndianArray(std::vector<Value> &destination, const unsigned char *source,
60 std::size_t &cursor) {
61#if Q_BYTE_ORDER == Q_LITTLE_ENDIAN
62 const std::size_t byteCount = destination.size() * sizeof(Value);
63 if (byteCount > 0U) {
64 std::memcpy(destination.data(), source + cursor, byteCount);
65 }
66 cursor += byteCount;
67#else
68 for (Value &value : destination) {
69 if constexpr (std::is_same_v<Value, float>) {
70 const std::uint32_t bits = readUint32(source + cursor);
71 std::memcpy(&value, &bits, sizeof(value));
72 } else {
73 value = readUint32(source + cursor);
74 }
75 cursor += sizeof(Value);
76 }
77#endif
78}

80bool verticesAreValid(const MapMesh &mesh) noexcept {
81 for (std::size_t offset = 0U; offset + 2U < mesh.vertices.size(); offset += 3U) {
82 const double longitude = mesh.vertices[offset];
83 const double latitude = mesh.vertices[offset + 1U];
84 const double mercatorY = mesh.vertices[offset + 2U];
85 if (!std::isfinite(longitude) || !std::isfinite(latitude) || !std::isfinite(mercatorY) ||
86 std::abs(longitude) > limits::MaximumAbsoluteLongitude ||
87 std::abs(latitude) > limits::MaximumAbsoluteLatitude ||
88 std::abs(mercatorY) > limits::MaximumAbsoluteMercatorY) {
89
90 return false;
91 }
92 }
93 return true;

107MapAssetReadResult MapAssetReader::read(QByteArrayView bytes) {
108 if (bytes.size() < static_cast<qsizetype>(format::HeaderSize) ||
109 bytes.size() > limits::MaximumAssetBytes) {
110
111 return failure(MapAssetError::Size,
112 QStringLiteral("The packaged map has an invalid size."));
113 }
114
115 const auto *data = reinterpret_cast<const unsigned char *>(bytes.data());
116 if (std::memcmp(data + format::MagicOffset, format::Magic.data(), format::Magic.size()) != 0) {
117 return failure(MapAssetError::Format,
118 QStringLiteral("The packaged map has an unsupported signature."));
119 }
120
121 const std::uint32_t version = readUint32(data + format::VersionOffset);
122 const std::uint32_t headerSize = readUint32(data + format::HeaderSizeOffset);
123 const std::uint32_t flags = readUint32(data + format::FlagsOffset);
124 if (version != format::Version || headerSize != format::HeaderSize ||
125 flags != format::SupportedFlags || readUint32(data + format::ReservedOffset) != 0U) {
126
127 return failure(MapAssetError::Format,
128 QStringLiteral("The packaged map format is not supported."));
129 }
130
131 const std::uint64_t vertexCount = readUint64(data + format::VertexCountOffset);
132 const std::uint64_t mercatorIndexCount = readUint64(data + format::MercatorIndexCountOffset);
133 const std::uint64_t sphereIndexCount = readUint64(data + format::SphereIndexCountOffset);
134 const std::uint64_t borderIndexCount = readUint64(data + format::BorderIndexCountOffset);
135 const std::uint64_t declaredPayloadSize = readUint64(data + format::PayloadSizeOffset);
136 const std::uint32_t expectedCrc = readUint32(data + format::PayloadCrcOffset);
137
138 if (!validCounts(vertexCount, mercatorIndexCount, sphereIndexCount, borderIndexCount)) {
139 return failure(MapAssetError::Limits,
140 QStringLiteral("The packaged map exceeds geometry limits."));
141 }
142 std::size_t vertexFloatCount = 0U;
143 std::size_t vertexBytes = 0U;
144 std::size_t mercatorBytes = 0U;
145 std::size_t sphereBytes = 0U;
146 std::size_t borderBytes = 0U;
147 std::size_t expectedPayloadSize = 0U;
148 std::size_t expectedFileSize = 0U;
149 if (!checkedMultiply(static_cast<std::size_t>(vertexCount), 3U, vertexFloatCount) ||
150 !checkedMultiply(vertexFloatCount, sizeof(float), vertexBytes) ||
151 !checkedMultiply(static_cast<std::size_t>(mercatorIndexCount), sizeof(std::uint32_t),
152 mercatorBytes) ||
153 !checkedMultiply(static_cast<std::size_t>(sphereIndexCount), sizeof(std::uint32_t),
154 sphereBytes) ||
155 !checkedMultiply(static_cast<std::size_t>(borderIndexCount), sizeof(std::uint32_t),
156 borderBytes) ||
157 !checkedAdd(vertexBytes, mercatorBytes, expectedPayloadSize) ||
158 !checkedAdd(expectedPayloadSize, sphereBytes, expectedPayloadSize) ||
159 !checkedAdd(expectedPayloadSize, borderBytes, expectedPayloadSize) ||
160 expectedPayloadSize > limits::MaximumPayloadBytes ||
161 declaredPayloadSize != expectedPayloadSize ||
162 !checkedAdd(static_cast<std::size_t>(format::HeaderSize), expectedPayloadSize,
163 expectedFileSize) ||
164 expectedFileSize != static_cast<std::size_t>(bytes.size())) {
165
166 return failure(MapAssetError::Format,
167 QStringLiteral("The packaged map sections are inconsistent."));
168 }
169
170 const unsigned char *payload = data + format::HeaderSize;
171 if (MapChecksum::crc32(payload, expectedPayloadSize) != expectedCrc) {
172 return failure(MapAssetError::Integrity,
173 QStringLiteral("The packaged map checksum does not match."));
174 }
175
176 try {
177 auto mutableMesh = std::make_shared<MapMesh>();
178 mutableMesh->vertices.resize(vertexFloatCount);
179 mutableMesh->mercatorFillIndices.resize(static_cast<std::size_t>(mercatorIndexCount));
180 mutableMesh->sphereFillIndices.resize(static_cast<std::size_t>(sphereIndexCount));
181 mutableMesh->borderIndices.resize(static_cast<std::size_t>(borderIndexCount));
182
183 std::size_t cursor = 0U;
184 copyLittleEndianArray(mutableMesh->vertices, payload, cursor);
185 copyLittleEndianArray(mutableMesh->mercatorFillIndices, payload, cursor);
186 copyLittleEndianArray(mutableMesh->sphereFillIndices, payload, cursor);
187 copyLittleEndianArray(mutableMesh->borderIndices, payload, cursor);
188
189 if (cursor != expectedPayloadSize || !verticesAreValid(*mutableMesh)) {
190 return failure(MapAssetError::Format,
191 QStringLiteral("The packaged map contains invalid vertices."));
192 }
193 const std::size_t loadedVertexCount = mutableMesh->vertexCount();
194 if (!indicesAreValid(mutableMesh->mercatorFillIndices, loadedVertexCount) ||
195 !indicesAreValid(mutableMesh->sphereFillIndices, loadedVertexCount) ||
196 !indicesAreValid(mutableMesh->borderIndices, loadedVertexCount)) {
197
198 return failure(MapAssetError::Format,
199 QStringLiteral("The packaged map contains invalid indices."));
200 }
201
202 return {std::shared_ptr<const MapMesh>(std::move(mutableMesh)), MapAssetError::None, {}};
203 } catch (const std::bad_alloc &) {
204 return failure(MapAssetError::Allocation,
205 QStringLiteral("There is not enough memory to load the map."));
206 } catch (const std::length_error &) {
207 return failure(MapAssetError::Limits,
208 QStringLiteral("The packaged map exceeds container limits."));
209 }
210}
