49void HudWorkspace::setInputMode(ControlPanel::InputMode mode) {
50 if (m_inputMode == mode) {
51 m_controlPanel->setInputMode(mode);
52 return;
53 }
54 m_inputMode = mode;
55 m_controlPanel->setInputMode(mode);

66 m_externalInputs.altitudeFeet, 0.90, 0.95};
67 rebuild(true);
68}

70void HudWorkspace::setExternalInputs(const dlz::TelemetryInputs &inputs, bool available,
71 const QString &source) {
72 m_externalInputs = inputs;
73 m_externalAvailable = available;
74 m_controlPanel->setExternalInputs(inputs, available, source);
75
76 // Incoming frames are cached while the local calculation mode is active;
77 // they must not alter the slider-driven drawing or its temporal state.
78 if (m_inputMode != ControlPanel::InputMode::UdpOrReplay) {
79 return;
80 }
81 if (!available) {
82 clearExternalInputs();
83 return;
84 }
85 m_inputs = {inputs.rangeNm, inputs.aspectDegrees, inputs.altitudeFeet, 0.90, 0.95};
86 // A sequential external sample retains filter, scale, and cue state. The
87 // controller is reset only when the source/mode changes or the stream is
88 // cleared; resetTemporal is reserved for those discontinuities.
89 rebuild(false);
90}

104void HudWorkspace::rebuildFromControls() {
105 if (m_inputMode != ControlPanel::InputMode::SliderTest) {
106 return;
107 }
108 m_calculationInputs = m_controlPanel->inputs();
109 m_inputs = m_calculationInputs;

115void HudWorkspace::rebuild(bool resetTemporal) {
116 if (resetTemporal) {
117 m_controller.reset();
118 }
119 if (m_inputMode == ControlPanel::InputMode::UdpOrReplay && !m_externalAvailable) {
120 m_hasFrame = false;
121 m_controller.clear();
122 m_hudView->clearFrame();
123 return;
124 }
125
126 dlz::ScenarioFrame frame;
127 QString error;
128 if (!dlz::ScenarioAdapter::build(m_inputs, &frame, &error)) {
129 // Keep finite geometry and candidate readouts live while withholding
130 // an invalid solver result from the renderer.
131 if (frame.geometry.rangeNm > 0.0) {
132 m_controlPanel->setGeometryReadout(frame.geometry, frame.solution);
133 }
134 m_hasFrame = false;
135 m_controller.clear();
136 m_hudView->setError(
137 error.isEmpty() ? QStringLiteral("DLZ calculation input is outside its ordered domain")
138 : error);
139 return;
140 }
141 m_frame = frame;
142 m_hasFrame = true;
143 m_controlPanel->setGeometryReadout(m_frame.geometry, m_frame.solution);
144 present(0.0);
145}

147void HudWorkspace::present(double dtSeconds) {
148 if (!m_hasFrame) {
149 return;
150 }
151 m_controller.update(m_frame.solution, m_frame.geometry.rangeNm, m_frame.geometry.rangeRateKnots,
152 dlz::HudMode::Prelaunch, dtSeconds);
153 m_hudView->setFrame(m_frame.solution, m_controller.state(), m_controller.displayedRangeNm());
154}

156void HudWorkspace::tick() {
157 const qint64 now = m_clock.nsecsElapsed();
158 const double dt = static_cast<double>(now - m_lastTickNanoseconds) / 1.0e9;
159 m_lastTickNanoseconds = now;
160 present(std::clamp(dt, 0.0, 0.25));
161}
