43SphereProjectionParameters MapCamera::sphereProjectionParameters() const noexcept {
44 const auto split = [](double value) {
45 const float high = static_cast<float>(value);
46 return std::pair{high, static_cast<float>(value - static_cast<double>(high))};
47 };
48 const auto [longitudeHigh, longitudeLow] = split(m_sphereLongitude);
49 const auto [latitudeHigh, latitudeLow] = split(m_sphereLatitude);
50 const double latitudeRadians = qDegreesToRadians(m_sphereLatitude);
51 return {{longitudeHigh, latitudeHigh},
52 {longitudeLow, latitudeLow},
53 {static_cast<float>(std::sin(latitudeRadians)),
54 static_cast<float>(std::cos(latitudeRadians))}};
55}

175MercatorViewport MapCamera::mercatorViewport(int viewportWidth, int viewportHeight) const noexcept {
176 const int safeWidth = std::max(1, viewportWidth);
177 const int safeHeight = std::max(1, viewportHeight);
178 const double halfWidth = m_mercatorWorldBounds.width() * 0.5;
179 const double halfHeight = m_mercatorWorldBounds.height() * 0.5;
180 const double aspect = static_cast<double>(safeWidth) / static_cast<double>(safeHeight);
181 const double contentAspect = halfWidth / halfHeight;
182
183 double viewHalfWidth = halfWidth;
184 double viewHalfHeight = halfHeight;
185 if (aspect >= contentAspect) {
186 viewHalfWidth = halfHeight * aspect;
187 } else {
188 viewHalfHeight = halfWidth / aspect;
189 }
190 viewHalfWidth /= static_cast<double>(m_mercatorZoom);
191 viewHalfHeight /= static_cast<double>(m_mercatorZoom);
192
193 return {viewHalfWidth,
194 viewHalfHeight,
195 m_mercatorCenter.x() - viewHalfWidth,
196 m_mercatorCenter.x() + viewHalfWidth,
197 m_mercatorCenter.y() - viewHalfHeight,
198 m_mercatorCenter.y() + viewHalfHeight};
199}

210WorldCopyRange MapCamera::visibleWorldCopiesForBounds(double minimumLongitude,
211 double maximumLongitude, int viewportWidth,
212 int viewportHeight) const noexcept {
213 if (m_presentation != MapPresentation::Mercator || !std::isfinite(minimumLongitude) ||
214 !std::isfinite(maximumLongitude) || minimumLongitude > maximumLongitude) {
215
216 return m_presentation == MapPresentation::Mercator ? WorldCopyRange{0, -1}
217 : WorldCopyRange{0, 0};
218 }
219
220 const QPointF corners[] = {
221 screenToMercator({0.0, 0.0}, viewportWidth, viewportHeight),
222 screenToMercator({static_cast<double>(viewportWidth), 0.0}, viewportWidth, viewportHeight),
223 screenToMercator({static_cast<double>(viewportWidth), static_cast<double>(viewportHeight)},
224 viewportWidth, viewportHeight),
225 screenToMercator({0.0, static_cast<double>(viewportHeight)}, viewportWidth,
226 viewportHeight)};
227
228 double viewportLeft = corners[0].x();
229 double viewportRight = corners[0].x();
230 for (const QPointF &corner : corners) {
231 viewportLeft = std::min(viewportLeft, corner.x());
232 viewportRight = std::max(viewportRight, corner.x());
233 }
234
235 // Copy k occupies [minimum + 360k, maximum + 360k]. Solving its
236 // intersection with the viewport yields these inclusive ceil/floor bounds;
237 // epsilon keeps a copy touching the edge visible despite projection noise.
238 constexpr double Epsilon = 1.0e-6;
239 const double firstValue =
240 std::ceil((viewportLeft - maximumLongitude - Epsilon) / MapProjection::WorldWidthDegrees);
241 const double lastValue =
242 std::floor((viewportRight - minimumLongitude + Epsilon) / MapProjection::WorldWidthDegrees);
243 if (firstValue < std::numeric_limits<int>::min() ||
244 firstValue > std::numeric_limits<int>::max() ||
245 lastValue < std::numeric_limits<int>::min() ||
246 lastValue > std::numeric_limits<int>::max()) {
247
248 return {0, 0};
249 }
250 const int first = static_cast<int>(firstValue);
251 const int last = static_cast<int>(lastValue);
252 return first <= last ? WorldCopyRange{first, last} : WorldCopyRange{0, -1};
253}

255QPointF MapCamera::screenToMercator(const QPointF &screenPosition, int viewportWidth,
256 int viewportHeight) const noexcept {
257 const int safeWidth = std::max(1, viewportWidth);
258 const int safeHeight = std::max(1, viewportHeight);
259 const MercatorViewport viewport = mercatorViewport(safeWidth, safeHeight);
260 const double normalizedX = screenPosition.x() / static_cast<double>(safeWidth) * 2.0 - 1.0;
261 const double normalizedY = 1.0 - screenPosition.y() / static_cast<double>(safeHeight) * 2.0;
262 const double viewDeltaX = normalizedX * viewport.halfWidth;
263 const double viewDeltaY = normalizedY * viewport.halfHeight;
264 const double bearing = qDegreesToRadians(m_bearingDegrees);
265 const double cosine = std::cos(bearing);
266 const double sine = std::sin(bearing);
267 return {m_mercatorCenter.x() + viewDeltaX * cosine + viewDeltaY * sine,
268 m_mercatorCenter.y() - viewDeltaX * sine + viewDeltaY * cosine};
269}

271bool MapCamera::panMercator(const QPoint &pixelDelta, int viewportWidth,
272 int viewportHeight) noexcept {
273 const int safeWidth = std::max(1, viewportWidth);
274 const int safeHeight = std::max(1, viewportHeight);
275 const MercatorViewport viewport = mercatorViewport(safeWidth, safeHeight);
276 const double viewOffsetX = static_cast<double>(pixelDelta.x()) * 2.0 * viewport.halfWidth /
277 static_cast<double>(safeWidth);
278 const double viewOffsetY = -static_cast<double>(pixelDelta.y()) * 2.0 * viewport.halfHeight /
279 static_cast<double>(safeHeight);
280 const double bearing = qDegreesToRadians(m_bearingDegrees);
281 const double cosine = std::cos(bearing);
282 const double sine = std::sin(bearing);
283 const QPointF previous = m_mercatorCenter;
284 m_mercatorCenter.rx() -= viewOffsetX * cosine + viewOffsetY * sine;
285 m_mercatorCenter.ry() -= -viewOffsetX * sine + viewOffsetY * cosine;
286 constrainMercatorCenter(viewportWidth, viewportHeight);
287 return previous != m_mercatorCenter;
288}

290bool MapCamera::zoomAt(const QPointF &screenPosition, int wheelDelta, int viewportWidth,
291 int viewportHeight) noexcept {
292 if (wheelDelta == 0) {
293 return false;
294 }
295 const float factor = std::pow(1.0015F, static_cast<float>(wheelDelta));
296 if (m_presentation == MapPresentation::Sphere) {
297 return setSphereZoom(m_sphereZoom * factor);
298 }
299
300 // Preserve the geographic point under the cursor: measure it before and
301 // after the zoom, then translate the center by the projection-space drift.
302 const QPointF before = screenToMercator(screenPosition, viewportWidth, viewportHeight);
303 const float previousZoom = m_mercatorZoom;
304 m_mercatorZoom = std::clamp(m_mercatorZoom * factor, MinimumMercatorZoom, MaximumMercatorZoom);
305 const QPointF after = screenToMercator(screenPosition, viewportWidth, viewportHeight);
306 m_mercatorCenter += before - after;
307 constrainMercatorCenter(viewportWidth, viewportHeight);
308 return !equal(previousZoom, m_mercatorZoom);
309}

315void MapCamera::clampMercatorCenterY(int viewportWidth, int viewportHeight) noexcept {
316 const MercatorViewport viewport = mercatorViewport(viewportWidth, viewportHeight);
317 const double boundsMinimum =
318 std::min(m_mercatorWorldBounds.top(), m_mercatorWorldBounds.bottom());
319 const double boundsMaximum =
320 std::max(m_mercatorWorldBounds.top(), m_mercatorWorldBounds.bottom());
321 const double bearing = qDegreesToRadians(m_bearingDegrees);
322 // Rotation projects both viewport half-axes onto world y. Their absolute
323 // contributions form the furthest vertical reach that must remain in bounds.
324 const double verticalExtent = std::abs(std::sin(bearing)) * viewport.halfWidth +
325 std::abs(std::cos(bearing)) * viewport.halfHeight;
326 const double minimum = boundsMinimum + verticalExtent;
327 const double maximum = boundsMaximum - verticalExtent;
328 if (minimum >= maximum) {
329 m_mercatorCenter.setY(m_mercatorWorldBounds.center().y());
330 } else {
331 m_mercatorCenter.setY(std::clamp(m_mercatorCenter.y(), minimum, maximum));
332 }
333}

340QMatrix4x4 MapCamera::projectionMatrix(int viewportWidth, int viewportHeight) const noexcept {
341 QMatrix4x4 projection;
342 const float aspect =
343 viewportHeight > 0 ? static_cast<float>(viewportWidth) / static_cast<float>(viewportHeight)
344 : 1.0F;
345 if (m_presentation == MapPresentation::Sphere) {
346 float horizontalScale = m_sphereZoom;
347 float verticalScale = m_sphereZoom;
348 if (aspect > 1.0F) {
349 horizontalScale /= aspect;
350 } else if (aspect > 0.0F && aspect < 1.0F) {
351 verticalScale *= aspect;
352 }
353 projection.scale(horizontalScale, verticalScale, 1.0F);
354 } else {
355 const MercatorViewport viewport = mercatorViewport(viewportWidth, viewportHeight);
356 projection.ortho(static_cast<float>(m_mercatorCenter.x() - viewport.halfWidth),
357 static_cast<float>(m_mercatorCenter.x() + viewport.halfWidth),
358 static_cast<float>(m_mercatorCenter.y() - viewport.halfHeight),
359 static_cast<float>(m_mercatorCenter.y() + viewport.halfHeight), -1.0F,
360 1.0F);
361 }
362 return projection;
363}

365bool MapCamera::projectGeoToScreen(double longitudeDegrees, double latitudeDegrees,
366 int viewportWidth, int viewportHeight, QPointF &screenPosition,
367 bool &visible) const noexcept {
368 if (!std::isfinite(longitudeDegrees) || !std::isfinite(latitudeDegrees) ||
369 latitudeDegrees < -90.0 || latitudeDegrees > 90.0 || viewportWidth <= 0 ||
370 viewportHeight <= 0) {
371
372 visible = false;
373 return false;
374 }
375
376 const double bearing = qDegreesToRadians(m_bearingDegrees);
377 const double cosine = std::cos(bearing);
378 const double sine = std::sin(bearing);
379 if (m_presentation == MapPresentation::Mercator) {
380 const MercatorViewport viewport = mercatorViewport(viewportWidth, viewportHeight);
381 const double longitude = MapProjection::unwrapLongitude(
382 MapProjection::wrapLongitude(longitudeDegrees), m_mercatorCenter.x());
383 const double projectedY = MapProjection::projectLatitude(latitudeDegrees);
384 const double deltaX = longitude - m_mercatorCenter.x();
385 const double deltaY = projectedY - m_mercatorCenter.y();
386 const double viewX = deltaX * cosine - deltaY * sine;
387 const double viewY = deltaX * sine + deltaY * cosine;
388 const double normalizedX = viewX / viewport.halfWidth;
389 const double normalizedY = viewY / viewport.halfHeight;
390
391 screenPosition = {(normalizedX + 1.0) * 0.5 * viewportWidth,
392 (1.0 - normalizedY) * 0.5 * viewportHeight};
393
394 visible = std::abs(latitudeDegrees) <= MapProjection::MaximumMercatorLatitudeDegrees &&
395 normalizedX >= -1.0 && normalizedX <= 1.0 && normalizedY >= -1.0 &&
396 normalizedY <= 1.0;
397 return true;
398 }
399
400 const double longitude = qDegreesToRadians(longitudeDegrees);
401 const double latitude = qDegreesToRadians(latitudeDegrees);
402 const double centerLongitude = qDegreesToRadians(m_sphereLongitude);
403 const double centerLatitude = qDegreesToRadians(m_sphereLatitude);
404 const double x = std::cos(latitude) * std::sin(longitude - centerLongitude);
405 const double y =
406 std::cos(centerLatitude) * std::sin(latitude) -
407 std::sin(centerLatitude) * std::cos(latitude) * std::cos(longitude - centerLongitude);
408 const double z =
409 std::sin(centerLatitude) * std::sin(latitude) +
410 std::cos(centerLatitude) * std::cos(latitude) * std::cos(longitude - centerLongitude);
411 const double viewX = x * cosine - y * sine;
412 const double viewY = x * sine + y * cosine;
413 const double aspect = static_cast<double>(viewportWidth) / static_cast<double>(viewportHeight);
414 double horizontalScale = m_sphereZoom;
415 double verticalScale = m_sphereZoom;
416 if (aspect > 1.0) {
417 horizontalScale /= aspect;
418 } else if (aspect < 1.0) {
419 verticalScale *= aspect;
420 }
421 const double normalizedX = viewX * horizontalScale;
422 const double normalizedY = viewY * verticalScale;
423
424 screenPosition = {(normalizedX + 1.0) * 0.5 * viewportWidth,
425 (1.0 - normalizedY) * 0.5 * viewportHeight};
426
427 visible = z >= 0.0 && normalizedX >= -1.0 && normalizedX <= 1.0 && normalizedY >= -1.0 &&
428 normalizedY <= 1.0;
429 return true;
430}

432bool MapCamera::sphereCoordinateAt(const QPointF &screenPosition, int viewportWidth,
433 int viewportHeight, double &longitudeDegrees,
434 double &latitudeDegrees) const noexcept {
435 const int safeWidth = std::max(1, viewportWidth);
436 const int safeHeight = std::max(1, viewportHeight);
437 const float aspect = static_cast<float>(safeWidth) / static_cast<float>(safeHeight);
438 float horizontalScale = m_sphereZoom;
439 float verticalScale = m_sphereZoom;
440 if (aspect > 1.0F) {
441 horizontalScale /= aspect;
442 } else if (aspect < 1.0F) {
443 verticalScale *= aspect;
444 }
445
446 const double normalizedX = screenPosition.x() / static_cast<double>(safeWidth) * 2.0 - 1.0;
447 const double normalizedY = 1.0 - screenPosition.y() / static_cast<double>(safeHeight) * 2.0;
448 const double viewX = normalizedX / horizontalScale;
449 const double viewY = normalizedY / verticalScale;
450 const double bearing = qDegreesToRadians(m_bearingDegrees);
451 const double cosine = std::cos(bearing);
452 const double sine = std::sin(bearing);
453 const double sphereX = viewX * cosine + viewY * sine;
454 const double sphereY = -viewX * sine + viewY * cosine;
455 const double radiusSquared = sphereX * sphereX + sphereY * sphereY;
456 if (radiusSquared > 1.0) {
457 return false;
458 }
459
460 // Orthographic projection drops z. Choosing the positive square root
461 // reconstructs the visible hemisphere; points outside the unit disk have
462 // no corresponding surface coordinate.
463 const double sphereZ = std::sqrt(std::max(0.0, 1.0 - radiusSquared));
464 const double centerLatitude = qDegreesToRadians(m_sphereLatitude);
465 const double sinLatitude =
466 std::cos(centerLatitude) * sphereY + std::sin(centerLatitude) * sphereZ;
467 latitudeDegrees = qRadiansToDegrees(std::asin(std::clamp(sinLatitude, -1.0, 1.0)));
468 const double longitudeDelta = std::atan2(sphereX, -std::sin(centerLatitude) * sphereY +
469 std::cos(centerLatitude) * sphereZ);
470 longitudeDegrees =
471 MapProjection::wrapLongitude(m_sphereLongitude + qRadiansToDegrees(longitudeDelta));
472 return true;
473}
