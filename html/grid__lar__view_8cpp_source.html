37void GridLarView::setSceneState(const LarSceneState &state) {
38 const auto fieldChanged = [this, &state](int field) {
39 const bool wasAvailable = m_scene.hasScene && field >= 0 &&
40 field < m_scene.availableFields.size() &&
41 m_scene.availableFields.testBit(field);
42 const bool isAvailable = state.hasScene && field >= 0 &&
43 field < state.availableFields.size() &&
44 state.availableFields.testBit(field);
45 return wasAvailable != isAvailable ||
46 (isAvailable && StateField::value(m_scene.plane, m_scene.target, field) !=
47 StateField::value(state.plane, state.target, field));
48 };
49 static constexpr std::array<int, 12> VisibleFields{
50 StateField::Location0, StateField::Location1, StateField::Euler0, StateField::IrPos0,
51 StateField::IrPos1, StateField::IrR, StateField::IzPos0, StateField::IzPos1,
52 StateField::IzTheta1, StateField::IzTheta2, StateField::IzR1, StateField::IzR2};
53 const bool visibleStateChanged =
54 m_scene.hasScene != state.hasScene ||
55 std::any_of(VisibleFields.begin(), VisibleFields.end(), fieldChanged);
56 m_scene = state;
57 if (!m_cameraTransform.hasOrigin() &&
58 hasFields({StateField::Location0, StateField::Location1})) {
59 m_cameraTransform.setOrigin(m_scene.plane.location[0], m_scene.plane.location[1]);
60 }
61 updateTrackedCamera();
62 if (visibleStateChanged) {
63 update();
64 }
65}

115void GridLarView::fitToData() {
116 if (!m_scene.hasScene || width() <= 0 || height() <= 0) {
117 return;
118 }
119 updateTrackedCamera();
120 const LarZoneValidationResult validation =
121 LarZoneInputValidator().validate(m_scene.target, m_scene.availableFields);
122 double horizontalExtent = 1000.0;
123 double verticalExtent = 1000.0;
124 if (m_cameraTransform.hasLocation() && validation.inRange) {
125
126 const QPointF center = geographicToCameraWorld(m_scene.target.ir_pos);
127 horizontalExtent =
128 qMax(horizontalExtent, qAbs(center.x()) + validation.inRange->outerRadiusMeters);
129 verticalExtent =
130 qMax(verticalExtent, qAbs(center.y()) + validation.inRange->outerRadiusMeters);
131 }
132 if (m_cameraTransform.hasLocation() && validation.inZone) {
133
134 const QPointF center = geographicToCameraWorld(m_scene.target.iz_pos);
135 horizontalExtent =
136 qMax(horizontalExtent, qAbs(center.x()) + validation.inZone->outerRadiusMeters);
137 verticalExtent =
138 qMax(verticalExtent, qAbs(center.y()) + validation.inZone->outerRadiusMeters);
139 }
140 if (m_cameraTransform.hasLocation() &&
141 hasFields({StateField::Location0, StateField::Location1})) {
142
143 const QPointF plane = geographicToCameraWorld(m_scene.plane.location);
144 horizontalExtent = qMax(horizontalExtent, qAbs(plane.x()) + 50.0);
145 verticalExtent = qMax(verticalExtent, qAbs(plane.y()) + 50.0);
146 }
147 m_cameraTransform.setScale(
148 std::clamp(std::min((width() * 0.5 - 45.0) / qMax(1.0, horizontalExtent),
149 (height() * 0.5 - 45.0) / qMax(1.0, verticalExtent)),
150 0.00001, 50.0));
151 update();
152}

172void GridLarView::drawGrid(QPainter &painter) const {
173 const double targetWorldSpacing = 90.0 / m_cameraTransform.scale();
174 painter.setPen(QPen(QColor(QStringLiteral("#d9ddda")), 1));
175
176 if (!m_scene.hasScene || !m_cameraTransform.hasLocation() || !m_cameraTransform.hasOrigin()) {
177
178 const double step = GridGeometryBuilder::gridStep(targetWorldSpacing);
179 const QPointF topLeft = screenToWorld(QPointF(0, 0));
180 const QPointF bottomRight = screenToWorld(QPointF(width(), height()));
181 for (double x = std::floor(topLeft.x() / step) * step; x <= bottomRight.x(); x += step) {
182 const double screenX = worldToScreen(QPointF(x, 0)).x();
183 painter.drawLine(QPointF(screenX, 0), QPointF(screenX, height()));
184 }
185 for (double y = std::floor(bottomRight.y() / step) * step; y <= topLeft.y(); y += step) {
186 const double screenY = worldToScreen(QPointF(0, y)).y();
187 painter.drawLine(QPointF(0, screenY), QPointF(width(), screenY));
188 }
189 return;
190 }
191
192 const double bearing = effectiveCameraBearing();
193 const double cosineBearing = std::cos(bearing);
194 const double sineBearing = std::sin(bearing);
195 const QPointF corners[] = {screenToWorld(QPointF(0, 0)), screenToWorld(QPointF(width(), 0)),
196 screenToWorld(QPointF(width(), height())),
197 screenToWorld(QPointF(0, height()))};
198
199 const auto groundOffset = [cosineBearing, sineBearing](const QPointF &world) {
200 return QPointF(world.x() * cosineBearing + world.y() * sineBearing,
201 -world.x() * sineBearing + world.y() * cosineBearing);
202 };
203 const QPointF firstGroundCorner = groundOffset(corners[0]);
204 double minimumEast = firstGroundCorner.x();
205 double maximumEast = firstGroundCorner.x();
206 double minimumNorth = firstGroundCorner.y();
207 double maximumNorth = firstGroundCorner.y();
208 for (int index = 1; index < 4; ++index) {
209 const QPointF ground = groundOffset(corners[index]);
210 minimumEast = qMin(minimumEast, ground.x());
211 maximumEast = qMax(maximumEast, ground.x());
212 minimumNorth = qMin(minimumNorth, ground.y());
213 maximumNorth = qMax(maximumNorth, ground.y());
214 }
215
216 const double spacingMeters = GridGeometryBuilder::gridStep(targetWorldSpacing);
217 const double originCosineLatitude = qMax(0.01, std::cos(m_cameraTransform.originLatitude()));
218 const auto groundCoordinates = [this, originCosineLatitude](double latitude, double longitude) {
219 double longitudeDelta = longitude - m_cameraTransform.originLongitude();
220 while (longitudeDelta > Pi) {
221 longitudeDelta -= 2.0 * Pi;
222 }
223 while (longitudeDelta < -Pi) {
224 longitudeDelta += 2.0 * Pi;
225 }
226 return QPointF(EarthRadiusMeters * longitudeDelta * originCosineLatitude,
227 EarthRadiusMeters * (latitude - m_cameraTransform.originLatitude()));
228 };
229 const auto &cameraLocation = m_cameraTransform.location();
230 const QPointF cameraGround = groundCoordinates(cameraLocation[0], cameraLocation[1]);
231 const double minimumGroundEast = cameraGround.x() + minimumEast;
232 const double maximumGroundEast = cameraGround.x() + maximumEast;
233 const double minimumGroundNorth = cameraGround.y() + minimumNorth;
234 const double maximumGroundNorth = cameraGround.y() + maximumNorth;
235 const auto groundToScreen = [this, cosineBearing, sineBearing, cameraGround](double east,
236 double north) {
237 const double relativeEast = east - cameraGround.x();
238 const double relativeNorth = north - cameraGround.y();
239 return worldToScreen(QPointF(relativeEast * cosineBearing - relativeNorth * sineBearing,
240 relativeEast * sineBearing + relativeNorth * cosineBearing));
241 };
242
243 const qint64 firstNorthIndex = qint64(std::floor(minimumGroundNorth / spacingMeters));
244 const qint64 lastNorthIndex = qint64(std::ceil(maximumGroundNorth / spacingMeters));
245 for (qint64 index = firstNorthIndex; index <= lastNorthIndex; ++index) {
246 const double north = static_cast<double>(index) * spacingMeters;
247 painter.drawLine(groundToScreen(minimumGroundEast, north),
248 groundToScreen(maximumGroundEast, north));
249 }
250 const qint64 firstEastIndex = qint64(std::floor(minimumGroundEast / spacingMeters));
251 const qint64 lastEastIndex = qint64(std::ceil(maximumGroundEast / spacingMeters));
252 for (qint64 index = firstEastIndex; index <= lastEastIndex; ++index) {
253 const double east = static_cast<double>(index) * spacingMeters;
254 painter.drawLine(groundToScreen(east, minimumGroundNorth),
255 groundToScreen(east, maximumGroundNorth));
256 }
257}

273void GridLarView::drawInZone(QPainter &painter) const {
274 const LarZoneValidationResult validation =
275 LarZoneInputValidator().validate(m_scene.target, m_scene.availableFields);
276 if (!m_cameraTransform.hasLocation() || !validation.inZone) {
277
278 return;
279 }
280
281 const QPointF center = geographicToCameraWorld(m_scene.target.iz_pos);
282 const LarZoneDefinition &zone = *validation.inZone;
283 const QPolygonF worldPolygon = LarGeometryBuilder::inZoneWedgePolygon(
284 center, zone.innerRadiusMeters, zone.outerRadiusMeters, zone.startBearingRadians,
285 zone.startBearingRadians + zone.spanRadians, effectiveCameraBearing());
286 QPolygonF screenPolygon;
287 screenPolygon.reserve(worldPolygon.size());
288 for (const QPointF &point : worldPolygon) {
289 screenPolygon.append(worldToScreen(point));
290 }
291 QPainterPath path;
292 path.addPolygon(screenPolygon);
293 path.closeSubpath();
294 painter.fillPath(path, QColor(20, 131, 102, 76));
295 painter.setPen(QPen(QColor(QStringLiteral("#0e755b")), 2));
296 painter.drawPath(path);

299void GridLarView::drawMarkers(QPainter &painter) const {

301 const QPointF targetPosition =
302 worldToScreen(geographicToCameraWorld(m_scene.target.iz_pos));

305 const QPolygonF targetShape{targetPosition + QPointF(0, -10),
306 targetPosition + QPointF(8.66, 5),
307 targetPosition + QPointF(-8.66, 5)};
308 painter.drawPolygon(targetShape);
309 }
310 if (m_cameraTransform.hasLocation() &&
311 hasFields({StateField::Location0, StateField::Location1})) {
312 const QPointF planePosition =
313 worldToScreen(geographicToCameraWorld(m_scene.plane.location));
314 painter.save();
315 painter.translate(planePosition);
316 const double relativeHeading =

318 effectiveCameraBearing();
319 painter.rotate(std::remainder(relativeHeading, LarGeodesicGeometry::TwoPi) * 180.0 / Pi);
320 const QPolygonF planeShape{QPointF(0, -15), QPointF(9, 11), QPointF(0, 7), QPointF(-9, 11)};
321 painter.setPen(QPen(QColor(QStringLiteral("#173f54")), 2));
322 painter.setBrush(QColor(QStringLiteral("#3f87a6")));
323 painter.drawPolygon(planeShape);

328void GridLarView::drawNavigation(QPainter &painter) const {
329 constexpr double NavigatorSize = 25.0;
330 constexpr double BottomMargin = 16.0;
331 constexpr double ScaleMaximumWidth = 110.0;
332 constexpr double NavigatorGap = 16.0;
333 constexpr double HorizontalPadding = 14.0;
334 constexpr double VerticalPadding = 10.0;
335
336 const double distance =
337 GridGeometryBuilder::scaleBarDistance(ScaleMaximumWidth / m_cameraTransform.scale());
338 const double scaleWidth = distance * m_cameraTransform.scale();
339 const double contentWidth = ScaleMaximumWidth + NavigatorGap + NavigatorSize;
340 const QSizeF boxSize(contentWidth + HorizontalPadding * 2.0,
341 NavigatorSize + VerticalPadding * 2.0);
342 const QRectF box((width() - boxSize.width()) * 0.5, height() - BottomMargin - boxSize.height(),
343 boxSize.width(), boxSize.height());
344 const QRectF navigatorRect(box.right() - HorizontalPadding - NavigatorSize,
345 box.top() + VerticalPadding, NavigatorSize, NavigatorSize);
346 const double scaleSlotLeft = box.left() + HorizontalPadding;
347 const double scaleSlotRight = navigatorRect.left() - NavigatorGap;
348 const double scaleCenter = (scaleSlotLeft + scaleSlotRight) * 0.5;
349 const double scaleLeft = scaleCenter - scaleWidth * 0.5;
350 const double scaleRight = scaleCenter + scaleWidth * 0.5;
351 const double scaleY = box.center().y() + 9.0;
352 const double bearing = m_scene.hasScene ? effectiveCameraBearing() : 0.0;
353
354 painter.save();
355 painter.setPen(QPen(QColor(QStringLiteral("#101814")), 1.5));
356 painter.setBrush(Qt::white);
357 painter.drawRoundedRect(box, 8.0, 8.0);
358 painter.restore();
359
360 painter.save();
361 painter.setRenderHint(QPainter::SmoothPixmapTransform);
362 painter.translate(navigatorRect.center());
363 painter.rotate(-bearing * 180.0 / Pi);
364 painter.drawPixmap(
365 QRectF(-NavigatorSize * 0.5, -NavigatorSize * 0.5, NavigatorSize, NavigatorSize),
366 m_navigator, QRectF(m_navigator.rect()));
367 painter.restore();
368
369 painter.save();
370 QFont labelFont = painter.font();
371 labelFont.setPixelSize(11);
372 labelFont.setBold(true);
373 painter.setFont(labelFont);
374 painter.setPen(QColor(QStringLiteral("#28352f")));
375 painter.drawText(QRectF(scaleLeft - 8.0, scaleY - 24.0, scaleWidth + 16.0, 16.0),
376 Qt::AlignCenter, GridGeometryBuilder::formatDistance(distance));
377 painter.setPen(QPen(QColor(QStringLiteral("#28352f")), 2));
378 painter.drawLine(QPointF(scaleLeft, scaleY), QPointF(scaleRight, scaleY));
379 painter.drawLine(QPointF(scaleLeft, scaleY - 6.0), QPointF(scaleLeft, scaleY + 1.0));
380 painter.drawLine(QPointF(scaleRight, scaleY - 6.0), QPointF(scaleRight, scaleY + 1.0));
381 painter.restore();
382}
