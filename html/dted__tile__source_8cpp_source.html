31bool DtedTileSource::containsFile(const QString &path) const {
32 if (m_canonicalRootDirectory.isEmpty()) {
33 return false;
34 }
35 const QFileInfo fileInfo(path);
36 const QString canonicalFile = fileInfo.canonicalFilePath();
37 if (!fileInfo.isFile() || canonicalFile.isEmpty()) {
38 return false;
39 }
40
41 // QDir paths use forward slashes internally even on Windows, whereas
42 // QDir::separator() returns the native backslash there. Comparing a path
43 // against a prefix built with the native separator therefore rejected
44 // valid Windows tiles. A canonical relative path also gives the boundary
45 // check platform-aware drive-letter and case handling without admitting a
46 // sibling directory or a symlink target above the selected root.
47 const QString relative = QDir::cleanPath(
48 QDir::fromNativeSeparators(QDir(m_canonicalRootDirectory).relativeFilePath(canonicalFile)));
49 return !relative.isEmpty() && !QDir::isAbsolutePath(relative) &&
50 relative != QStringLiteral(".") && relative != QStringLiteral("..") &&
51 !relative.startsWith(QStringLiteral("../"));
52}

54QString DtedTileSource::pathFor(const DtedCellKey &key) const {
55 const QChar longitudeHemisphere =
56 key.longitudeDegrees < 0 ? QLatin1Char('w') : QLatin1Char('e');
57 const QChar latitudeHemisphere = key.latitudeDegrees < 0 ? QLatin1Char('s') : QLatin1Char('n');
58 const QString longitudeDirectory =
59 QStringLiteral("%1%2")
60 .arg(longitudeHemisphere)
61 .arg(std::abs(key.longitudeDegrees), 3, 10, QLatin1Char('0'));
62 const QString latitudeFile = QStringLiteral("%1%2%3")
63 .arg(latitudeHemisphere)
64 .arg(std::abs(key.latitudeDegrees), 2, 10, QLatin1Char('0'))
65 .arg(dtedFileSuffix(m_level));
66 const QString uppercaseLongitudeDirectory = longitudeDirectory.toUpper();
67 const QString uppercaseLatitudeFile = latitudeFile.toUpper();
68 const std::array<QString, 4> relativeCandidates{
69 longitudeDirectory + QLatin1Char('/') + latitudeFile,
70 uppercaseLongitudeDirectory + QLatin1Char('/') + uppercaseLatitudeFile,
71 longitudeDirectory + QLatin1Char('/') + uppercaseLatitudeFile,
72 uppercaseLongitudeDirectory + QLatin1Char('/') + latitudeFile};
73 const QDir root(m_rootDirectory);
74 for (const QString &relative : relativeCandidates) {
75 const QString candidate = root.filePath(relative);
76 if (containsFile(candidate)) {
77 return candidate;
78 }
79 }
80 return root.filePath(relativeCandidates.front());
81}

83DtedCellReadResult DtedTileSource::load(const DtedCellKey &key) const {
84 if (key.longitudeDegrees < -180 || key.longitudeDegrees > 179 || key.latitudeDegrees < -90 ||
85 key.latitudeDegrees > 89) {
86 return {nullptr, QStringLiteral("%1 cell key is outside the WGS84 tile range.")
87 .arg(dtedLevelDisplayName(m_level))};
88 }
89 const QString path = pathFor(key);
90 if (QFileInfo(path).isFile() && !containsFile(path)) {
91 return {nullptr, QStringLiteral("%1 tile '%2' resolves outside the selected root.")
92 .arg(dtedLevelDisplayName(m_level), path)};
93 }
94 DtedCellReadResult result = DtedCellReader::readFile(path, m_level);
95 if (result.succeeded() && !(result.cell->key == key)) {
96 return {nullptr, QStringLiteral("%1 tile '%2' declares a different cell origin.")
97 .arg(dtedLevelDisplayName(m_level), path)};
98 }
99 return result;
100}

102std::optional<DtedCellKey> DtedTileSource::keyForRadians(double latitudeRadians,

112 return DtedCellKey{static_cast<int>(std::floor(longitudeDegrees)),
113 static_cast<int>(std::floor(clampedLatitude))};
