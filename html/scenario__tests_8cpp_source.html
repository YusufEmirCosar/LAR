35double larCenterSeparationMeters(const Target &target) {
36 const double latitudeDelta = target.ir_pos[0] - target.iz_pos[0];
37 const double longitudeDelta = std::remainder(target.ir_pos[1] - target.iz_pos[1], 2.0 * Pi);
38 const double sinHalfLatitude = std::sin(latitudeDelta * 0.5);
39 const double sinHalfLongitude = std::sin(longitudeDelta * 0.5);
40 const double haversine =
41 std::clamp(sinHalfLatitude * sinHalfLatitude + std::cos(target.ir_pos[0]) *
42 std::cos(target.iz_pos[0]) *
43 sinHalfLongitude * sinHalfLongitude,
44 0.0, 1.0);
45 return 2.0 * EarthRadiusMeters * std::asin(std::sqrt(haversine));
46}

48bool checkState(const QString &scenario, int packetIndex, double elapsedSeconds,
49 const PacketMapping &floatMapping, const PacketMapping &dlzMapping,
50 const PacketMapping &combinedMapping) {
51 Plane firstPlane{};
52 Target firstTarget{};
53 dlz::TelemetryInputs firstDlz;
54 TestSenderScenarios::initialize(scenario, &firstPlane, &firstTarget, &firstDlz);
55 TestSenderScenarios::update(scenario, packetIndex, elapsedSeconds, &firstPlane, &firstTarget,
56 &firstDlz);
57
58 QBitArray allFields(StateField::Count, true);
59 QString validationError;
60 if (!StateValidator::validate(firstPlane, firstTarget, allFields, &validationError)) {
61 QTextStream(stderr) << scenario << " packet " << packetIndex
62 << " is invalid: " << validationError << '\n';
63 return false;
64 }
65 DecodedState firstDecoded;
66 firstDecoded.dlzInputs = firstDlz;
67 firstDecoded.availableFields = QBitArray(StateField::Count);
68 firstDecoded.availableFields.setBit(StateField::DlzRangeNm);
69 firstDecoded.availableFields.setBit(StateField::DlzAspectDegrees);
70 firstDecoded.availableFields.setBit(StateField::DlzAltitudeFeet);
71 if (!StateValidator::validate(firstDecoded, &validationError)) {
72 QTextStream(stderr) << scenario << " packet " << packetIndex
73 << " has invalid DLZ telemetry: " << validationError << '\n';
74 return false;
75 }
76 const double centerSeparation = larCenterSeparationMeters(firstTarget);
77 if (!std::isfinite(centerSeparation) || centerSeparation > MaximumLarCenterSeparationMeters) {
78 QTextStream(stderr) << scenario << " packet " << packetIndex
79 << " separates IR and IZ centers by " << centerSeparation
80 << " meters\n";
81 return false;
82 }
83
84 Plane mappedPlane{};
85 Target mappedTarget{};
86 QBitArray mappedFields;
87 if (!floatMapping.decode(floatMapping.encode(firstPlane, firstTarget), &mappedPlane,
88 &mappedTarget, &mappedFields, &validationError)) {
89 QTextStream(stderr) << scenario << " packet " << packetIndex
90 << " fails float mapping round-trip: " << validationError << '\n';
91 return false;
92 }
93
94 Plane secondPlane{};
95 Target secondTarget{};
96 dlz::TelemetryInputs secondDlz;
97 TestSenderScenarios::initialize(scenario, &secondPlane, &secondTarget, &secondDlz);
98 TestSenderScenarios::update(scenario, packetIndex, elapsedSeconds, &secondPlane, &secondTarget,
99 &secondDlz);
100 if (!statesEqual(firstPlane, firstTarget, secondPlane, secondTarget) ||
101 !dlzValuesEqual(firstDlz, secondDlz)) {
102 QTextStream(stderr) << scenario << " packet " << packetIndex << " is not deterministic\n";
103 return false;
104 }
105
106 DecodedState aggregate;
107 aggregate.plane = firstPlane;
108 aggregate.target = firstTarget;
109 aggregate.dlzInputs = firstDlz;
110 for (const PacketMapping *mapping : {&dlzMapping, &combinedMapping}) {
111 aggregate.availableFields = mapping->availableFields();
112 const QByteArray encoded = mapping->encode(aggregate);
113 DecodedState decoded;
114 if (!mapping->decode(encoded, &decoded, &validationError)) {
115 QTextStream(stderr) << scenario << " packet " << packetIndex
116 << " fails DLZ mapping round-trip: " << validationError << '\n';
117 return false;
118 }
119 if (decoded.dlzInputs.rangeNm != firstDlz.rangeNm ||
120 decoded.dlzInputs.aspectDegrees != firstDlz.aspectDegrees ||
121 decoded.dlzInputs.altitudeFeet != firstDlz.altitudeFeet) {
122 QTextStream(stderr) << scenario << " packet " << packetIndex
123 << " changes DLZ telemetry during round-trip\n";
124 return false;
125 }
126 }
127 return true;
128}
