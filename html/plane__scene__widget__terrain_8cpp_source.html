71 const QPointF eastNorth = LarProjection::geographicToPlaneWorld(

82int terrainResolution(double halfExtentMeters) noexcept {
83 constexpr double NominalPostSpacingMeters = 900.0;
84 int resolution =
85 static_cast<int>(std::ceil((halfExtentMeters * 2.0) / NominalPostSpacingMeters)) + 1;
86 resolution = std::clamp(resolution, 49, 129);
87 if (resolution % 2 == 0) {
88 resolution += resolution < 129 ? 1 : -1;
89 }
90 return resolution;
91}

148void PlaneSceneWidget::requestTerrainIfNeeded(bool force) {
149 if (!m_terrainVisible || !m_terrainAvailable || m_terrainWorker == nullptr ||
150 !m_scene.hasScene || !fieldAvailable(m_scene.availableFields, StateField::Location0) ||
151 !fieldAvailable(m_scene.availableFields, StateField::Location1) ||
152 !validGroundPosition(m_scene.plane.location[0], m_scene.plane.location[1])) {
153 return;
154 }
155 const GeoCoordinateRadians anchor{m_scene.plane.location[0], m_scene.plane.location[1]};
156 const double halfExtent = terrainHalfExtentMeters(m_surfaceState);
157 const double scale = m_surfaceState.metersPerSceneUnit;
158 const auto covers = [this, halfExtent, scale](const GeoCoordinateRadians &candidate,
159 double candidateHalfExtent,
160 double candidateScale) {
161 return std::abs(candidateScale - scale) <= scale * 1.0e-6 &&
162 candidateHalfExtent >= halfExtent * 0.9 &&
163 distanceFromAircraftMeters(candidate, m_scene.plane) <= candidateHalfExtent * 0.35;
164 };
165 if (!force && m_terrainPatch &&
166 covers({m_terrainPatch->anchorLatitudeRadians, m_terrainPatch->anchorLongitudeRadians},
167 m_terrainPatch->halfExtentMeters, m_terrainPatch->metersPerSceneUnit)) {
168 return;
169 }
170 if (!force && m_terrainPending && m_terrainRequestAnchor &&
171 covers(*m_terrainRequestAnchor, m_pendingTerrainHalfExtent, m_pendingTerrainScale)) {
172 return;
173 }
174 if (!force && m_failedTerrainAnchor &&
175 distanceFromAircraftMeters(*m_failedTerrainAnchor, m_scene.plane) <=
176 std::max(5000.0, m_failedTerrainHalfExtent * 0.35)) {
177 return;
178 }
179 if (m_terrainPatch &&
180 distanceFromAircraftMeters(
181 {m_terrainPatch->anchorLatitudeRadians, m_terrainPatch->anchorLongitudeRadians},
182 m_scene.plane) > m_terrainPatch->halfExtentMeters * 0.8) {
183 m_terrainPatch.reset();
184 m_renderer.setTerrainPatch(nullptr);
185 emit terrainPatchChanged(false);
186 }
187
188 PlaneTerrainBuildRequest request;
189 request.revision = ++m_terrainRevision;
190 request.latitudeRadians = anchor.latitude;
191 request.longitudeRadians = anchor.longitude;
192 request.halfExtentMeters = halfExtent;
193 request.metersPerSceneUnit = scale;
194 request.resolution = terrainResolution(halfExtent);
195 m_terrainRequestAnchor = anchor;
196 m_pendingTerrainHalfExtent = halfExtent;
197 m_pendingTerrainScale = scale;
198 m_terrainPending = true;
199 m_failedTerrainAnchor.reset();
200 m_terrainWorker->submit(request);
201}

203void PlaneSceneWidget::updateTerrainPlacement() {
204 if (!m_terrainPatch || !m_scene.hasScene ||
205 !fieldAvailable(m_scene.availableFields, StateField::Location0) ||
206 !fieldAvailable(m_scene.availableFields, StateField::Location1) ||
207 !validGroundPosition(m_scene.plane.location[0], m_scene.plane.location[1])) {
208 m_renderer.setTerrainPlacement({}, 0.0F);
209 return;
210 }
211 const double anchorPosition[3]{m_terrainPatch->anchorLatitudeRadians,
212 m_terrainPatch->anchorLongitudeRadians, 0.0};
213 const QPointF eastNorth = LarProjection::geographicToPlaneWorld(
214 anchorPosition, m_scene.plane.location, 0.0, m_scene.plane.location[0], true);
215 const double scale = m_terrainPatch->metersPerSceneUnit;
216 double altitudeMeters = m_terrainPatch->centerElevationMeters -
217 static_cast<double>(m_surfaceState.surfaceHeight) * scale;
218 if (fieldAvailable(m_scene.availableFields, StateField::Location2) &&
219 std::isfinite(m_scene.plane.location[2])) {
220 altitudeMeters = m_scene.plane.location[2];
221 }
222 m_renderer.setTerrainPlacement(
223 {static_cast<float>(eastNorth.x() / scale), static_cast<float>(-eastNorth.y() / scale)},
224 static_cast<float>(altitudeMeters / scale));
225}

227void PlaneSceneWidget::completeTerrainPatch(quint64 revision, PlaneTerrainPatchPtr patch,
228 const QString &message) {
229 if (revision != m_terrainRevision) {
230 return;
231 }
232 m_terrainPending = false;
233 if (patch && !patch->empty()) {
234 m_terrainPatch = std::move(patch);
235 m_failedTerrainAnchor.reset();
236 m_renderer.setTerrainPatch(m_terrainPatch);
237 updateTerrainPlacement();
238 if (m_diagnostic.startsWith(QStringLiteral("DTED terrain"))) {
239 m_diagnostic.clear();
240 }
241 emit terrainPatchChanged(true);
242 } else {
243 m_terrainPatch.reset();
244 m_renderer.setTerrainPatch(nullptr);
245 m_failedTerrainAnchor = m_terrainRequestAnchor;
246 m_failedTerrainHalfExtent = m_pendingTerrainHalfExtent;
247 setDiagnostic(message.isEmpty()
248 ? QStringLiteral("DTED terrain is unavailable here; using flat ground.")
249 : QStringLiteral("DTED terrain: %1").arg(message));
250 emit terrainPatchChanged(false);
251 }
252 update();
253}
