66int terrainResolution(double halfExtentMeters, DtedLevel level) noexcept {
67 int resolution = static_cast<int>(std::ceil((halfExtentMeters * 2.0) /
68 dtedNominalPostSpacingMeters(level))) +
69 1;
70 resolution = std::clamp(resolution, 49, 257);
71 if (resolution % 2 == 0) {
72 resolution += resolution < 257 ? 1 : -1;
73 }
74 return resolution;
75}

77std::optional<DtedCellKey> keyForCandidatePath(const QString &path, DtedLevel level) {
78 static const QRegularExpression LongitudePattern(QStringLiteral("^([eEwW])(\\d{3})$"));
79 static const QRegularExpression LatitudePattern(
80 QStringLiteral("^([nNsS])(\\d{2})\\.[dD][tT]([012])$"));
81 const QFileInfo fileInfo(path);
82 const QRegularExpressionMatch longitudeMatch = LongitudePattern.match(fileInfo.dir().dirName());
83 const QRegularExpressionMatch latitudeMatch = LatitudePattern.match(fileInfo.fileName());
84 if (!longitudeMatch.hasMatch() || !latitudeMatch.hasMatch() ||
85 latitudeMatch.captured(3).toInt() != dtedLevelNumber(level)) {
86 return std::nullopt;
87 }
88 int longitude = longitudeMatch.captured(2).toInt();
89 int latitude = latitudeMatch.captured(2).toInt();
90 if (longitudeMatch.captured(1).compare(QStringLiteral("w"), Qt::CaseInsensitive) == 0) {
91 longitude = -longitude;
92 }
93 if (latitudeMatch.captured(1).compare(QStringLiteral("s"), Qt::CaseInsensitive) == 0) {
94 latitude = -latitude;
95 }
96 if (longitude < -180 || longitude > 179 || latitude < -90 || latitude > 89) {
97 return std::nullopt;
98 }
99 return DtedCellKey{longitude, latitude};
100}

102QString validateTerrainDataset(const DtedDataset &dataset) {
103 const DtedTileSource source(dataset.rootDirectory, dataset.level);
104 if (!source.isAvailable()) {
105 return QStringLiteral("The selected DTED folder is not a readable directory.");
106 }
107 const QString suffix = dtedFileSuffix(dataset.level);
108 QDirIterator directoryIterator(dataset.rootDirectory, QDir::Dirs | QDir::NoDotAndDotDot,
109 QDirIterator::NoIteratorFlags);
110 constexpr int MaximumProbeDirectories = 1024;
111 constexpr int MaximumProbeCandidates = 512;
112 int directories = 0;
113 int candidates = 0;
114 QString firstFailure;
115 while (directoryIterator.hasNext() && directories < MaximumProbeDirectories &&
116 candidates < MaximumProbeCandidates) {
117 const QString directoryPath = directoryIterator.next();
118 ++directories;
119 QDirIterator fileIterator(
120 directoryPath,
121 {QStringLiteral("*%1").arg(suffix), QStringLiteral("*%1").arg(suffix.toUpper())},
122 QDir::Files | QDir::Readable, QDirIterator::NoIteratorFlags);
123 while (fileIterator.hasNext() && candidates < MaximumProbeCandidates) {
124 const QString candidatePath = fileIterator.next();
125 ++candidates;
126 const std::optional<DtedCellKey> key =
127 keyForCandidatePath(candidatePath, dataset.level);
128 if (!key) {
129 continue;
130 }
131 const QString addressedPath = source.pathFor(*key);
132 const QString candidateCanonical = QFileInfo(candidatePath).canonicalFilePath();
133 const QString addressedCanonical = QFileInfo(addressedPath).canonicalFilePath();
134 if (candidateCanonical.isEmpty() || candidateCanonical != addressedCanonical) {
135 continue;
136 }
137 const DtedCellReadResult result = source.load(*key);
138 if (result.succeeded()) {
139 return {};
140 }
141 if (firstFailure.isEmpty()) {
142 firstFailure = result.message;
143 }
144 }
145 }
146 if (!firstFailure.isEmpty()) {
147 return firstFailure;
148 }
149 return QStringLiteral("No valid %1 tile was found under the selected folder. Expected "
150 "{e|w}DDD/{n|s}DD%2.")
151 .arg(dtedLevelDisplayName(dataset.level), suffix);
152}

224 m_terrainRequestAnchor.reset();
225 m_failedTerrainAnchor.reset();
226 m_terrainPatch.reset();

248void PlaneSceneWidget::requestTerrainIfNeeded(bool force) {
249 if (!m_terrainVisible || !m_terrainAvailable || m_terrainWorker == nullptr ||
250 !m_scene.hasScene || !fieldAvailable(m_scene.availableFields, StateField::Location0) ||
251 !fieldAvailable(m_scene.availableFields, StateField::Location1) ||
252 !validGroundPosition(m_scene.plane.location[0], m_scene.plane.location[1])) {
253 return;
254 }
255 const GeoCoordinateRadians anchor{m_scene.plane.location[0], m_scene.plane.location[1]};
256 const double halfExtent = terrainHalfExtentMeters(m_surfaceState);
257 const double scale = m_surfaceState.metersPerSceneUnit;
258 const auto covers = [this, halfExtent, scale](const GeoCoordinateRadians &candidate,
259 double candidateHalfExtent,
260 double candidateScale) {
261 return std::abs(candidateScale - scale) <= scale * 1.0e-6 &&
262 candidateHalfExtent >= halfExtent * 0.9 &&
263 distanceFromAircraftMeters(candidate, m_scene.plane) <= candidateHalfExtent * 0.35;
264 };
265 if (!force && m_terrainPatch &&
266 covers({m_terrainPatch->anchorLatitudeRadians, m_terrainPatch->anchorLongitudeRadians},
267 m_terrainPatch->halfExtentMeters, m_terrainPatch->metersPerSceneUnit)) {
268 return;
269 }
270 if (!force && m_terrainPending && m_terrainRequestAnchor &&
271 covers(*m_terrainRequestAnchor, m_pendingTerrainHalfExtent, m_pendingTerrainScale)) {
272 return;
273 }
274 if (!force && m_failedTerrainAnchor &&
275 distanceFromAircraftMeters(*m_failedTerrainAnchor, m_scene.plane) <=
276 std::max(5000.0, m_failedTerrainHalfExtent * 0.35)) {
277 return;
278 }
279 if (m_terrainPatch &&
280 distanceFromAircraftMeters(
281 {m_terrainPatch->anchorLatitudeRadians, m_terrainPatch->anchorLongitudeRadians},
282 m_scene.plane) > m_terrainPatch->halfExtentMeters * 0.8) {
283 m_terrainPatch.reset();
284 m_renderer.setTerrainPatch(nullptr);
285 emit terrainPatchChanged(false);
286 }
287
288 PlaneTerrainBuildRequest request;
289 request.revision = ++m_terrainRevision;
290 request.latitudeRadians = anchor.latitude;
291 request.longitudeRadians = anchor.longitude;
292 request.projectionOriginLatitudeRadians = anchor.latitude;
293 request.halfExtentMeters = halfExtent;
294 request.metersPerSceneUnit = scale;
295 request.resolution = terrainResolution(halfExtent, m_terrainDataset.level);
296 m_terrainRequestAnchor = anchor;
297 m_pendingTerrainHalfExtent = halfExtent;
298 m_pendingTerrainScale = scale;
299 m_terrainPending = true;
300 m_failedTerrainAnchor.reset();
301 m_terrainWorker->submit(request);
302}

304void PlaneSceneWidget::updateTerrainPlacement() {
305 if (!m_terrainPatch || !m_scene.hasScene ||
306 !fieldAvailable(m_scene.availableFields, StateField::Location0) ||
307 !fieldAvailable(m_scene.availableFields, StateField::Location1) ||
308 !validGroundPosition(m_scene.plane.location[0], m_scene.plane.location[1])) {
309 m_renderer.setTerrainPlacement({}, {1.0F, 1.0F}, 0.0F);
310 return;
311 }
312 const double anchorPosition[3]{m_terrainPatch->anchorLatitudeRadians,
313 m_terrainPatch->anchorLongitudeRadians, 0.0};
314 const QPointF eastNorth = LarProjection::geographicToPlaneWorld(
315 anchorPosition, m_scene.plane.location, 0.0, m_scene.plane.location[0], true);
316 const double patchLatitude = std::isfinite(m_terrainPatch->projectionOriginLatitudeRadians)
317 ? m_terrainPatch->projectionOriginLatitudeRadians
318 : m_terrainPatch->anchorLatitudeRadians;
319 const double currentCosineLatitude = std::max(0.01, std::cos(m_scene.plane.location[0]));
320 const double patchCosineLatitude = std::max(0.01, std::cos(patchLatitude));
321 const double scale = m_terrainPatch->metersPerSceneUnit;
322 double altitudeMeters = m_terrainPatch->centerElevationMeters -
323 static_cast<double>(m_surfaceState.surfaceHeight) * scale;
324 if (fieldAvailable(m_scene.availableFields, StateField::Location2) &&
325 std::isfinite(m_scene.plane.location[2])) {
326 altitudeMeters = m_scene.plane.location[2];
327 }
328 m_renderer.setTerrainPlacement(
329 {static_cast<float>(eastNorth.x() / scale), static_cast<float>(-eastNorth.y() / scale)},
330 {static_cast<float>(currentCosineLatitude / patchCosineLatitude), 1.0F},
331 static_cast<float>(altitudeMeters / scale));
332}

334void PlaneSceneWidget::completeTerrainPatch(quint64 revision, PlaneTerrainPatchPtr patch,
335 const QString &message) {
336 if (revision != m_terrainRevision) {
337 return;
338 }
339 m_terrainPending = false;
340 if (patch && !patch->empty()) {
341 m_terrainPatch = std::move(patch);
342 m_failedTerrainAnchor.reset();
343 m_renderer.setTerrainPatch(m_terrainPatch);
344 updateTerrainPlacement();
345 if (m_diagnostic.startsWith(QStringLiteral("DTED terrain"))) {
346 m_diagnostic.clear();
347 }
348 emit terrainPatchChanged(true);
349 } else {
350 m_terrainPatch.reset();
351 m_renderer.setTerrainPatch(nullptr);
352 m_failedTerrainAnchor = m_terrainRequestAnchor;
353 m_failedTerrainHalfExtent = m_pendingTerrainHalfExtent;
354 setDiagnostic(
355 message.isEmpty()
356 ? QStringLiteral("DTED terrain is unavailable here; terrain surface is hidden.")
357 : QStringLiteral("DTED terrain: %1").arg(message));
358 emit terrainPatchChanged(false);
359 }
360 update();
361}
