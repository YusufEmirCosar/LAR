166void OfflinePanel::renderPlayback(const ApplicationViewModel &viewModel) {
167 m_duration = viewModel.playbackDuration();
168 const SessionTimestamp position = viewModel.playbackPosition();
169 m_timeLabel->setText(PlaybackTimeFormatter::format(position) + QStringLiteral(" / ") +
170 PlaybackTimeFormatter::format(m_duration));
171 if (!m_timeline->isSliderDown() && m_duration != SessionTimestamp::zero()) {
172 m_timeline->setValue(
173 PlaybackTimelineMapper::toSlider(position, m_duration, m_timeline->maximum()));
174 }

177void OfflinePanel::applyPlaybackRate() {
178 QString text = m_rate->text().trimmed();
179 if (text.endsWith(QLatin1Char('x'), Qt::CaseInsensitive))
180 text.chop(1);
181 bool valid = false;
182 const double rate = text.toDouble(&valid);
183 if (!valid || !std::isfinite(rate) || rate <= 0.0 || !m_application.setPlaybackRate(rate)) {
184 m_rate->setText(QStringLiteral("1"));
185 m_application.setPlaybackRate(1.0);
186 return;
187 }
188 m_rate->setText(QString::number(rate));
189}
