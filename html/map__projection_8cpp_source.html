39double MapProjection::unwrapLongitude(double longitudeDegrees,
40 double referenceLongitudeDegrees) noexcept {
41 if (!std::isfinite(longitudeDegrees) || !std::isfinite(referenceLongitudeDegrees)) {
42 return longitudeDegrees;
43 }
44 while (longitudeDegrees - referenceLongitudeDegrees > HalfWorldDegrees) {
45 longitudeDegrees -= WorldWidthDegrees;
46 }
47 while (longitudeDegrees - referenceLongitudeDegrees < -HalfWorldDegrees) {
48 longitudeDegrees += WorldWidthDegrees;

53double MapProjection::wrapPeriodic(double value, double minimumInclusive, double period) noexcept {

55 period <= 0.0) {
56
