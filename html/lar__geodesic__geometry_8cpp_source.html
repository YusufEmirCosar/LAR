8double LarGeodesicGeometry::wrapLongitude(double longitudeRadians) noexcept {
9 if (!std::isfinite(longitudeRadians)) {
10 return std::numeric_limits<double>::quiet_NaN();
11 }
12 double wrapped = std::remainder(longitudeRadians, TwoPi);
13 if (wrapped >= Pi) {
14 wrapped -= TwoPi;
15 }
16 if (wrapped < 0.0) {
17 return wrapped < -Pi ? wrapped + TwoPi : wrapped;
18 }

20}

22double LarGeodesicGeometry::positiveAngularSpan(double startRadians, double endRadians) noexcept {
23 if (!std::isfinite(startRadians) || !std::isfinite(endRadians)) {
24 return std::numeric_limits<double>::quiet_NaN();
25 }
26 const double normalizedStart = std::remainder(startRadians, TwoPi);
27 const double normalizedEnd = std::remainder(endRadians, TwoPi);
28 double span = std::fmod(normalizedEnd - normalizedStart, TwoPi);
29 if (span < 0.0) {
30 span += TwoPi;
31 }
32 if (span < 1.0e-12) {
33 return TwoPi;
34 }
35 return std::min(span, TwoPi);
36}

39 double distanceMeters,

41 if (!std::isfinite(center.latitude) || !std::isfinite(center.longitude) ||

43
44 return {};
45 }
46
47 const double angularDistance = distanceMeters / EarthRadiusMeters;
48 const double sinLatitude = std::sin(center.latitude);
49 const double cosLatitude = std::cos(center.latitude);
50 const double sinDistance = std::sin(angularDistance);
51 const double cosDistance = std::cos(angularDistance);
52
53 const double destinationLatitude = std::asin(
54 std::clamp(sinLatitude * cosDistance + cosLatitude * sinDistance * std::cos(bearingRadians),
55 -1.0, 1.0));
56 const double longitudeOffset =
57 std::atan2(std::sin(bearingRadians) * sinDistance * cosLatitude,
58 cosDistance - sinLatitude * std::sin(destinationLatitude));
59
60 return {destinationLatitude, wrapLongitude(center.longitude + longitudeOffset)};
61}
