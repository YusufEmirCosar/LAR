39bool decodeDataUri(const QString &uri, qint64 maximumBytes, QByteArray *result) {
40 if (result == nullptr || maximumBytes < 0 ||
41 !uri.startsWith(QStringLiteral("data:"), Qt::CaseInsensitive)) {
42 return false;
43 }
44 const qsizetype separator = uri.indexOf(QLatin1Char(','));
45 if (separator <= 5)
46 return false;
47 const QByteArray metadata = uri.left(separator).toUtf8().toLower();
48 const QByteArray payload = uri.mid(separator + 1).toUtf8();
49 if (payload.size() > maximumBytes * 4 / 3 + 16)
50 return false;
51
52 QByteArray decoded;
53 if (metadata.split(';').contains(QByteArrayLiteral("base64"))) {
54 const auto decoding =
55 QByteArray::fromBase64Encoding(payload, QByteArray::AbortOnBase64DecodingErrors);
56 if (decoding.decodingStatus != QByteArray::Base64DecodingStatus::Ok)
57 return false;
58 decoded = decoding.decoded;
59 } else {
60 decoded = QByteArray::fromPercentEncoding(payload);
61 }
62 if (decoded.size() > maximumBytes)
63 return false;
64 *result = std::move(decoded);
65 return true;
66}

69 QString *canonicalPath) {
70 if (canonicalPath == nullptr || uri.isEmpty() || sourceDirectory.isEmpty())

72 const QUrl parsed = QUrl::fromEncoded(uri.toUtf8(), QUrl::StrictMode);

75 return false;

80 return false;

95 if (QFileInfo(componentPath).isSymLink())
96 return false;
97 }

100 const QString canonicalCandidate = candidate.canonicalFilePath();
101 if (canonicalCandidate.isEmpty() || !candidate.isFile() || candidate.isSymLink())
102 return false;

104 if (QDir::isAbsolutePath(contained) || contained == QStringLiteral("..") ||
105 contained.startsWith(QStringLiteral("../"))) {
106 return false;
107 }

119bool GlbResourceReader::splitGlb(const QByteArray &bytes, GlbContainer *container) {
120 if (container == nullptr || bytes.size() < 20 || littleU32(bytes.constData()) != GlbMagic ||
121 littleU32(bytes.constData() + 4) != 2U ||
122 littleU32(bytes.constData() + 8) != static_cast<quint32>(bytes.size())) {
123 return false;
124 }
125
126 GlbContainer parsed;
127 bool foundJson = false;
128 qint64 offset = 12;
129 while (offset + 8 <= bytes.size()) {
130 const quint32 chunkLength = littleU32(bytes.constData() + static_cast<qsizetype>(offset));
131 const quint32 chunkType = littleU32(bytes.constData() + static_cast<qsizetype>(offset + 4));
132 offset += 8;
133 if (chunkLength == 0U || offset + static_cast<qint64>(chunkLength) > bytes.size())
134 return false;
135 const QByteArray chunk =
136 bytes.mid(static_cast<qsizetype>(offset), static_cast<qsizetype>(chunkLength));
137 if (chunkType == JsonChunk && !foundJson) {
138 parsed.json = chunk;
139 foundJson = true;
140 } else if (chunkType == BinaryChunk && !parsed.hasBinary) {
141 parsed.binary = chunk;
142 parsed.hasBinary = true;
143 }
144 offset += static_cast<qint64>(chunkLength);
145 }
146 if (offset != bytes.size() || !foundJson)
147 return false;
148 while (!parsed.json.isEmpty() && parsed.json.back() == '\0')
149 parsed.json.chop(1);
150 if (parsed.json.isEmpty())
151 return false;
152 *container = std::move(parsed);
153 return true;
154}

156bool GlbResourceReader::readUri(const QString &uri, const QString &sourceDirectory,
157 qint64 maximumBytes, QByteArray *result) {
158 QByteArray bytes;
159 if (decodeDataUri(uri, maximumBytes, &bytes)) {
160 *result = std::move(bytes);
161 return true;
162 }
163 if (result == nullptr || maximumBytes < 0 ||
164 uri.startsWith(QStringLiteral("data:"), Qt::CaseInsensitive)) {
165 return false;
166 }
167
168 QString path;
169 if (!containedRegularFile(uri, sourceDirectory, &path))
170 return false;
171 QFile file(path);
172 if (!file.open(QIODevice::ReadOnly) || file.size() <= 0 || file.size() > maximumBytes)
173 return false;
174 bytes = file.readAll();
175 if (bytes.size() != file.size())
176 return false;
177 // Re-check after opening to narrow the path-replacement window.
178 QString verifiedPath;
179 if (!containedRegularFile(uri, sourceDirectory, &verifiedPath) || verifiedPath != path)
180 return false;
181 *result = std::move(bytes);
182 return true;
183}

185bool GlbResourceReader::resolveBuffers(const QJsonArray &bufferTable,
186 const QString &sourceDirectory, const QByteArray &glbBinary,
187 bool hasGlbBinary, std::vector<QByteArray> *binaries) {
188 if (binaries == nullptr || bufferTable.isEmpty() ||
189 bufferTable.size() > limits::MaximumBufferCount) {
190 return false;
191 }
192
193 qint64 aggregateBytes = 0;
194 for (const QJsonValue &candidate : bufferTable) {
195 int byteLength = -1;
196 if (!candidate.isObject() ||
197 !integerValue(candidate.toObject(), QStringLiteral("byteLength"), &byteLength) ||
198 byteLength < 0 || byteLength > limits::MaximumResourceBytes ||
199 aggregateBytes > limits::MaximumAggregateBufferBytes - byteLength) {
200 return false;
201 }
202 aggregateBytes += byteLength;
203 }
204
205 std::vector<QByteArray> resolved;
206 resolved.reserve(static_cast<std::size_t>(bufferTable.size()));
207 for (qsizetype index = 0; index < bufferTable.size(); ++index) {
208 const QJsonObject bufferObject = bufferTable[index].toObject();
209 int byteLength = -1;
210 if (!integerValue(bufferObject, QStringLiteral("byteLength"), &byteLength))
211 return false;
212 const QJsonValue uriValue = bufferObject.value(QStringLiteral("uri"));
213 QByteArray binary;
214 if (uriValue.isString()) {
215 if (!readUri(uriValue.toString(), sourceDirectory, limits::MaximumResourceBytes,
216 &binary)) {
217 return false;
218 }
219 } else if (!uriValue.isUndefined() || !hasGlbBinary || index != 0) {
220 return false;
221 } else {
222 binary = glbBinary;
223 }
224 if (binary.size() < byteLength)
225 return false;
226 binary.truncate(byteLength);
227 resolved.push_back(std::move(binary));
228 }
229 *binaries = std::move(resolved);
230 return true;
231}
