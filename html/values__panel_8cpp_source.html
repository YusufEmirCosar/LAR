168void ValuesPanel::render(const ApplicationViewModel &viewModel) {
169 for (int id = 0; id < m_valueLabels.size(); ++id) {
170 QLabel *label = m_valueLabels[id];
171 if (!label)
172 continue;
173 const bool available = viewModel.hasState() && id < viewModel.availableFields().size() &&
174 viewModel.availableFields().testBit(id);
175 if (!available) {
176 label->setText(QStringLiteral("N/A"));
177 continue;
178 }
179 const auto value = StateField::tryValue(viewModel.decodedState(), id);
180 label->setText(value ? StateValueFormatter::format(id, *value) : QStringLiteral("N/A"));
181 }
