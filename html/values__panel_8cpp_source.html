47bool contains(const LarZoneDefinition &zone, const QPointF &zoneCenterOffset) noexcept {
48 constexpr double BoundaryToleranceMeters = 1.0e-6;
49 constexpr double BoundaryToleranceRadians = 1.0e-12;
50 const double distance = std::hypot(zoneCenterOffset.x(), zoneCenterOffset.y());
51 if (!std::isfinite(distance) || distance < zone.innerRadiusMeters - BoundaryToleranceMeters ||
52 distance > zone.outerRadiusMeters + BoundaryToleranceMeters) {
53 return false;
54 }
55 if (zone.spanRadians >= LarGeodesicGeometry::TwoPi - BoundaryToleranceRadians ||
56 distance <= BoundaryToleranceMeters) {
57 return true;
58 }
59
60 // The aircraft is the Plane scene origin; negate the center offset to get the
61 // bearing from the zone center to the aircraft in the renderer's convention.
62 const double bearing = std::atan2(-zoneCenterOffset.x(), -zoneCenterOffset.y());
63 double bearingOffset =
64 std::remainder(bearing - zone.startBearingRadians, LarGeodesicGeometry::TwoPi);
65 if (bearingOffset < 0.0) {
66 bearingOffset += LarGeodesicGeometry::TwoPi;
67 }
68 return bearingOffset <= zone.spanRadians + BoundaryToleranceRadians;
69}

71PlaneZoneStatus planeZoneStatus(const ApplicationViewModel &viewModel) noexcept {
72 if (!viewModel.hasState() ||
73 !hasFields(viewModel.availableFields(), {StateField::Location0, StateField::Location1}) ||
74 !validPlanePosition(viewModel.plane().location[0], viewModel.plane().location[1])) {
75 return PlaneZoneStatus::Outside;
76 }
77
78 // Plane mode uses this same local flat metric for its rings and sectors.
79 const LarZoneValidationResult validation =
80 LarZoneInputValidator().validate(viewModel.target(), viewModel.availableFields());
81 if (validation.inZone &&
82 contains(*validation.inZone, planeWorldFor(validation.inZone->center, viewModel.plane()))) {
83 return PlaneZoneStatus::InZone;
84 }
85 if (validation.inRange &&
86 contains(*validation.inRange,
87 planeWorldFor(validation.inRange->center, viewModel.plane()))) {
88 return PlaneZoneStatus::InRange;
89 }
90 return PlaneZoneStatus::Outside;
91}

93void setPlaneZoneIndicator(QLabel *indicator, PlaneZoneStatus status) {
94 QString text;
95 QString background;
96 QString border;
97 switch (status) {
98 case PlaneZoneStatus::InZone:
99 text = QStringLiteral("IZ");
100 background = QStringLiteral("#0e755b");
101 border = QStringLiteral("#79d5b7");
102 break;
103 case PlaneZoneStatus::InRange:
104 text = QStringLiteral("IR");
105 background = QStringLiteral("#346d91");
106 border = QStringLiteral("#8ac8e8");
107 break;
108 case PlaneZoneStatus::Outside:
109 text = QStringLiteral("OUTSIDE");
110 background = QStringLiteral("#6b3f47");
111 border = QStringLiteral("#e0a3ad");
112 break;
113 }
114 if (indicator->property("zoneStatus").toString() == text)
115 return;
116 indicator->setText(text);
117 indicator->setToolTip(QStringLiteral("Plane area: %1").arg(text));
118 indicator->setStyleSheet(
119 QStringLiteral("QLabel#planeZoneIndicator { background-color: %1; color: #ffffff; "
120 "border: 1px solid %2; border-radius: 0px; font-weight: 700; }")
121 .arg(background, border));
122 indicator->setProperty("zoneStatus", text);
123}

290void ValuesPanel::render(const ApplicationViewModel &viewModel) {
291 if (m_planeZoneIndicator != nullptr) {
292 setPlaneZoneIndicator(m_planeZoneIndicator, planeZoneStatus(viewModel));
293 }
294 for (int id = 0; id < m_valueLabels.size(); ++id) {
295 QLabel *label = m_valueLabels[id];
296 if (!label)
297 continue;
298 const bool available = viewModel.hasState() && id < viewModel.availableFields().size() &&
299 viewModel.availableFields().testBit(id);
300 if (!available) {
301 if (label->text() != QStringLiteral("N/A"))
302 label->setText(QStringLiteral("N/A"));
303 continue;
304 }
305 const auto value = StateField::tryValue(viewModel.decodedState(), id);
306 const QString text =
307 value ? StateValueFormatter::format(id, *value) : QStringLiteral("N/A");
308 if (label->text() != text)
309 label->setText(text);
310 }
