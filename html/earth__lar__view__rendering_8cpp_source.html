49LarZoneRenderState EarthLarView::zoneRenderState() const {
50 const bool sphere = m_viewMode == LarViewMode::Sphere;
51 QMatrix4x4 projection;
52 if (sphere) {
53 projection = getProjectionMatrix(width(), height());
54 } else {
55 const lar::map::MercatorViewport viewport = flatViewport(width(), height());
56 projection.ortho(static_cast<float>(-viewport.halfWidth),
57 static_cast<float>(viewport.halfWidth),
58 static_cast<float>(-viewport.halfHeight),
59 static_cast<float>(viewport.halfHeight), -1.0F, 1.0F);
60 }
61 lar::map::WorldCopyRange copies =
62 sphere ? lar::map::WorldCopyRange{0, 0} : visibleFlatWorldCopies(width(), height());
63 std::optional<std::pair<double, double>> bounds;
64 if (!sphere && m_useParametricBackend && m_parametricZoneGpuLayer) {
65 double minimum = 0.0;
66 double maximum = 0.0;
67 if (m_parametricZoneGpuLayer->longitudeBounds(&minimum, &maximum)) {
68 bounds = {{minimum, maximum}};
69 }
70 } else if (!sphere && m_overlayCoordinator) {
71 bounds = m_overlayCoordinator->longitudeBounds();
72 }
73 if (bounds) {
74 const auto zoneCopies = mapCamera().visibleWorldCopiesForBounds(
75 bounds->first, bounds->second, width(), height());
76 if (zoneCopies.count() > 0) {
77 if (copies.count() == 0)
78 copies = zoneCopies;
79 else {
80 copies.first = std::min(copies.first, zoneCopies.first);
81 copies.last = std::max(copies.last, zoneCopies.last);
82 }
83 }
84 }
85 return {sphere,
86 mapCamera().sphereProjectionParameters(),
87 mapCamera().sphereCenter(),
88 qDegreesToRadians(cameraBearing()),
89 flatCenterProjected(),
90 projection,
91 copies};
92}
