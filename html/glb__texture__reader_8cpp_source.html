22bool integerValue(const QJsonObject &object, const QString &key, int *value, int fallback = -1) {
23 const QJsonValue candidate = object.value(key);
24 if (candidate.isUndefined()) {
25 *value = fallback;
26 return fallback >= 0;
27 }
28 if (!candidate.isDouble()) {
29 return false;
30 }
31 const qint64 integer = candidate.toInteger(std::numeric_limits<qint64>::min());
32 if (integer < 0 || integer > std::numeric_limits<int>::max()) {
33 return false;
34 }
35 *value = static_cast<int>(integer);
36 return true;
37}

47bool dataUriBytes(const QString &uri, QByteArray *result) {
48 if (result == nullptr || !uri.startsWith(QStringLiteral("data:"), Qt::CaseInsensitive)) {
49 return false;
50 }
51 const int separator = uri.indexOf(QLatin1Char(','));
52 if (separator <= 5) {
53 return false;
54 }
55 const QByteArray metadata = uri.left(separator).toUtf8().toLower();
56 const QByteArray payload = uri.mid(separator + 1).toUtf8();
57 QByteArray decoded;
58 if (metadata.split(';').contains(QByteArrayLiteral("base64"))) {
59 decoded = QByteArray::fromBase64(payload);
60 if (!payload.isEmpty() && decoded.isEmpty()) {
61 return false;
62 }
63 } else {
64 decoded = QByteArray::fromPercentEncoding(payload);
65 }
66 if (decoded.size() > MaximumResourceBytes) {
67 return false;
68 }
69 *result = std::move(decoded);
70 return true;
71}

73bool externalUriBytes(const QString &uri, const QString &sourceDirectory, QByteArray *result) {
74 if (dataUriBytes(uri, result)) {
75 return true;
76 }
77 if (result == nullptr || uri.isEmpty() || sourceDirectory.isEmpty()) {
78 // This branch is intentionally kept below the data-URI case: external resources are
79 // resolved only for a source `.gltf` file, never against the application directory.
80 return false;
81 }
82 const QUrl parsed(uri);
83 if (!parsed.isValid() || !parsed.scheme().isEmpty() || !parsed.host().isEmpty()) {
84 return false;
85 }
86 const QString decodedPath =
87 QString::fromUtf8(QByteArray::fromPercentEncoding(uri.toUtf8()));
88 if (decodedPath.isEmpty() || decodedPath.contains(QChar::Null)) {
89 return false;
90 }
91 QFile file(QDir(sourceDirectory).filePath(decodedPath));
92 if (!file.open(QIODevice::ReadOnly) || file.size() <= 0 ||
93 file.size() > MaximumResourceBytes) {
94 return false;
95 }
96 const QByteArray bytes = file.readAll();
97 if (bytes.size() != file.size()) {
98 return false;
99 }
100 *result = bytes;
101 return true;
102}

104bool bufferViewData(const QJsonArray &bufferViews, const std::vector<QByteArray> &binaries,
105 int viewIndex, QByteArray *result) {
106 if (result == nullptr || viewIndex < 0 || viewIndex >= bufferViews.size()) {
107 return false;
108 }
109 const QJsonObject view = bufferViews[viewIndex].toObject();
110 int bufferIndex = -1;
111 int byteOffset = 0;
112 int byteLength = -1;
113 if (!integerValue(view, QStringLiteral("buffer"), &bufferIndex) || bufferIndex < 0 ||
114 bufferIndex >= static_cast<int>(binaries.size()) ||
115 !integerValue(view, QStringLiteral("byteOffset"), &byteOffset, 0) ||
116 !integerValue(view, QStringLiteral("byteLength"), &byteLength) || byteLength <= 0) {
117 return false;
118 }
119 const QByteArray &binary = binaries[static_cast<std::size_t>(bufferIndex)];
120 const qint64 end = static_cast<qint64>(byteOffset) + byteLength;
121 if (end > binary.size()) {
122 return false;
123 }
124 *result = binary.mid(byteOffset, byteLength);
125 return result->size() == byteLength;
126}

182bool imagePayload(const QJsonObject &imageObject, const QJsonArray &bufferViews,
183 const std::vector<QByteArray> &binaries, const QString &sourceDirectory,
184 QByteArray *encoded) {
185 if (encoded == nullptr) {
186 return false;
187 }
188 const QJsonValue viewValue = imageObject.value(QStringLiteral("bufferView"));
189 const QJsonValue uriValue = imageObject.value(QStringLiteral("uri"));
190 if (!viewValue.isUndefined()) {
191 int viewIndex = -1;
192 if (!uriValue.isUndefined() || !integerValue(imageObject, QStringLiteral("bufferView"),
193 &viewIndex)) {
194 return false;
195 }
196 return bufferViewData(bufferViews, binaries, viewIndex, encoded);
197 }
198 return uriValue.isString() &&
199 externalUriBytes(uriValue.toString(), sourceDirectory, encoded);
200}

216bool decodeImages(const QJsonArray &bufferViews, const QJsonArray &imageTable,
217 const std::vector<QByteArray> &binaries, const QString &sourceDirectory,
218 std::vector<QImage> *images) {
219 images->reserve(static_cast<std::size_t>(imageTable.size()));
220 qint64 totalPixels = 0;
221 for (const QJsonValue &candidate : imageTable) {
222 const QJsonObject imageObject = candidate.toObject();
223 if (!candidate.isObject()) {
224 return false;
225 }
226 QByteArray encoded;
227 if (!imagePayload(imageObject, bufferViews, binaries, sourceDirectory, &encoded) ||
228 encoded.isEmpty() || encoded.size() > MaximumResourceBytes) {
229 return false;
230 }
231 QBuffer buffer(&encoded);
232 if (!buffer.open(QIODevice::ReadOnly)) {
233 return false;
234 }
235 QImageReader reader(&buffer, imageFormat(imageObject));
236 const QSize size = reader.size();
237 const qint64 pixels = static_cast<qint64>(size.width()) * size.height();
238 if (!size.isValid() || size.width() > MaximumTextureDimension ||
239 size.height() > MaximumTextureDimension || pixels <= 0 ||
240 totalPixels + pixels > MaximumTexturePixels) {
241 return false;
242 }
243 QImage decoded = reader.read();
244 if (decoded.isNull() || decoded.size() != size) {
245 return false;
246 }
247 decoded = decoded.convertToFormat(QImage::Format_RGBA8888);
248 if (decoded.isNull()) {
249 return false;
250 }
251 totalPixels += pixels;
252 images->push_back(std::move(decoded));
253 }
254 return true;
255}

266bool GlbTextureReader::load(const QJsonArray &bufferViews, const QJsonArray &imageTable,
267 const QJsonArray &textureTable, const QJsonArray &samplers,
268 const std::vector<QByteArray> &binaries,
269 const QString &sourceDirectory, PlaneModelMesh *mesh) {
270 if (mesh == nullptr || imageTable.size() > MaximumTextureCount ||
271 textureTable.size() > MaximumTextureCount || samplers.size() > MaximumTextureCount) {
272 return false;
273 }
274 std::vector<QImage> images;
275 if (!decodeImages(bufferViews, imageTable, binaries, sourceDirectory, &images)) {
276 return false;
277 }
278
279 mesh->textures.reserve(static_cast<std::size_t>(textureTable.size()));
280 for (const QJsonValue &candidate : textureTable) {
281 const QJsonObject textureObject = candidate.toObject();
282 int sourceIndex = -1;
283 int samplerIndex = -1;
284 if (!candidate.isObject() ||
285 !integerValue(textureObject, QStringLiteral("source"), &sourceIndex) ||
286 sourceIndex >= static_cast<int>(images.size()) ||
287 !optionalIndex(textureObject, QStringLiteral("sampler"), &samplerIndex) ||
288 samplerIndex >= samplers.size()) {
289 return false;
290 }
291
292 PlaneModelTexture texture;
293 texture.image = images[static_cast<std::size_t>(sourceIndex)];
294 if (samplerIndex >= 0) {
295 const QJsonObject sampler = samplers[samplerIndex].toObject();
296 int wrapS = 10497;
297 int wrapT = 10497;
298 int minFilter = 9987;
299 int magFilter = 9729;
300 if (!integerValue(sampler, QStringLiteral("wrapS"), &wrapS, 10497) ||
301 !integerValue(sampler, QStringLiteral("wrapT"), &wrapT, 10497) ||
302 !integerValue(sampler, QStringLiteral("minFilter"), &minFilter, 9987) ||
303 !integerValue(sampler, QStringLiteral("magFilter"), &magFilter, 9729) ||
304 !textureWrap(wrapS, &texture.wrapS) || !textureWrap(wrapT, &texture.wrapT) ||
305 !textureMinFilter(minFilter, &texture.minFilter) ||
306 !textureMagFilter(magFilter, &texture.magFilter)) {
307 return false;
308 }
309 }
310 mesh->textures.push_back(std::move(texture));
311 }
312 return true;
