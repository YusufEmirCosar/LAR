136void RecordingPipelineCoordinator::recordingInputDrained(quint64 token,
137 qint64 boundaryNanoseconds) {
138 if (token == 0 || token != m_activeDrainToken)
139 return;
140 m_activeDrainToken = 0;
141
142 const PendingOperation operation = m_pendingOperation;
143 const RuntimeRequestId request = m_pendingRequest;
144 const QString targetPath = m_pendingTargetPath;
145 m_pendingOperation = PendingOperation::None;
146 m_pendingRequest = {};
147 m_pendingTargetPath.clear();
148
149 switch (operation) {
150 case PendingOperation::Pause:
151 m_recording.pauseRecordingAt(boundaryNanoseconds);
152 completeOperation(operation, request,
153 m_recording.state() == RecordingService::State::Paused,
154 m_recording.state() == RecordingService::State::Paused
155 ? QString{}
156 : QStringLiteral("The recording could not be paused"));
157 break;
158 case PendingOperation::Reset: {
159 QString error;
160 const bool reset = m_recording.resetRecordingAt(boundaryNanoseconds, &error);
161 completeOperation(operation, request, reset, error);
162 if (m_resumeAfterOperation && m_recording.state() == RecordingService::State::Recording)
163 emit recordingInputEnabled(true);
164 break;
165 }
166 case PendingOperation::Snapshot:
167 preparePersistence(false, request, targetPath);
168 if (m_resumeAfterOperation)
169 emit recordingInputEnabled(true);
170 break;
171 case PendingOperation::FinalSave:
172 m_recording.pauseRecordingAt(boundaryNanoseconds);
173 preparePersistence(true, request, targetPath);
174 break;
175 case PendingOperation::Discard:
176 m_recording.cancelRecording();
177 completeOperation(operation, request, true);
178 break;
179 case PendingOperation::None:
180 break;
181 }
182 m_resumeAfterOperation = false;
183}

185void RecordingPipelineCoordinator::recordingInputFailed(const QString &error) {
186 const PendingOperation operation = m_pendingOperation;
187 const RuntimeRequestId request = m_pendingRequest;
188 const QString targetPath = m_pendingTargetPath;
189 m_pendingOperation = PendingOperation::None;
190 m_pendingRequest = {};
191 m_pendingTargetPath.clear();
192 m_activeDrainToken = 0;
193 m_resumeAfterOperation = false;
194 if (m_recording.state() == RecordingService::State::Recording)
195 m_recording.pauseRecording();
196 emit recordingInputEnabled(false);
197 if (operation == PendingOperation::None)
198 emit runtimeError(error);
199 else
200 completeOperation(operation, request, false, error, targetPath);
201}

203void RecordingPipelineCoordinator::persistenceFinished(quint64 requestId, bool finalSave,
204 bool saved, const QString &targetPath,
205 const QString &error) {

216 m_recording.completeRecording();

231bool RecordingPipelineCoordinator::beginDrain(PendingOperation operation, RuntimeRequestId request,
232 const QString &targetPath) {
233 if (m_pendingOperation != PendingOperation::None || m_activeDrainToken != 0) {
234 completeOperation(operation, request, false,
235 QStringLiteral("Another recording operation is already pending"),
236 targetPath);
237 return false;
238 }
239 m_resumeAfterOperation = (m_recording.state() == RecordingService::State::Recording);
240 m_pendingOperation = operation;
241 m_pendingRequest = request;
242 m_pendingTargetPath = targetPath;
243 m_activeDrainToken = ++m_nextDrainToken;
244 const bool preserveIncoming =
245 m_resumeAfterOperation &&
246 (operation == PendingOperation::Snapshot || operation == PendingOperation::Reset);
247 emit recordingDrainRequested(m_activeDrainToken, preserveIncoming);
248 return true;
249}

251void RecordingPipelineCoordinator::completeOperation(PendingOperation operation,
252 RuntimeRequestId request, bool succeeded,
253 const QString &error,
254 const QString &targetPath) {
255 switch (operation) {
256 case PendingOperation::Pause:
257 emit commandFinished({request, RuntimeCommandKind::RecordingPause, succeeded, error});
258 break;
259 case PendingOperation::Reset:
260 emit recordingResetFinished({request, succeeded, error});
261 break;
262 case PendingOperation::Snapshot:
263 emit recordingSaveFinished({request, false, succeeded, targetPath, error});
264 break;
265 case PendingOperation::FinalSave:
266 emit recordingSaveFinished({request, true, succeeded, targetPath, error});
267 break;
268 case PendingOperation::Discard:
269 emit commandFinished({request, RuntimeCommandKind::RecordingDiscard, succeeded, error});
270 break;
271 case PendingOperation::None:
272 break;
273 }
274}

276void RecordingPipelineCoordinator::publishState() {
277 const bool hasSession = m_recording.isRecording();
278 const bool paused = m_recording.isPaused();
279 const uint32_t recordCount = m_recording.recordCount();
280 const SessionTimestamp duration = m_recording.recordingDuration();
281 if (m_hasPublishedState && hasSession == m_lastHasSession && paused == m_lastPaused &&
282 recordCount == m_lastRecordCount && duration == m_lastRecordingDuration) {
283 return;
284 }
285 m_hasPublishedState = true;
286 m_lastHasSession = hasSession;
287 m_lastPaused = paused;
288 m_lastRecordCount = recordCount;
289 m_lastRecordingDuration = duration;
290 emit recordingStateChanged(hasSession, paused, recordCount);
291}

293bool RecordingPipelineCoordinator::preparePersistence(bool finalSave, RuntimeRequestId request,
294 const QString &targetPath) {
295 SessionSnapshot snapshot;
296 QString error;

298 emit recordingSaveFinished({request, finalSave, false, targetPath, error});
299 return false;
300 }
301 m_persistenceInFlight = true;
302 m_finalSaveInFlight = finalSave;

304 m_activePersistenceRequest = ++m_nextPersistenceRequest;

306 return true;
