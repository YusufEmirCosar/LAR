196double flatFloatCoordinateError(const lar::map::MapCamera &camera,
197 const GeoCoordinateRadians &point, int width, int height) {
198 const auto viewport = camera.mercatorViewport(width, height);
199 const QPointF center = camera.mercatorCenter();
200 const double longitudeDegrees = qRadiansToDegrees(point.longitude);
201 const double unwrapped = lar::map::MapProjection::unwrapLongitude(longitudeDegrees, center.x());
202 const double worldOffset =
203 std::round((unwrapped - longitudeDegrees) / lar::map::MapProjection::WorldWidthDegrees) *
204 lar::map::MapProjection::WorldWidthDegrees;
205 const double mercatorY =
206 lar::map::MapProjection::projectLatitude(qRadiansToDegrees(point.latitude));
207 const double exactX = unwrapped - center.x();
208 const double exactY = mercatorY - center.y();
209 const double floatX =
210 static_cast<double>(static_cast<float>(longitudeDegrees) + static_cast<float>(worldOffset) -
211 static_cast<float>(center.x()));
212 const double floatY =
213 static_cast<double>(static_cast<float>(mercatorY) - static_cast<float>(center.y()));
214 const double bearing = qDegreesToRadians(static_cast<double>(camera.bearingDegrees()));
215 const double cosine = std::cos(bearing);
216 const double sine = std::sin(bearing);
217 const double errorX = floatX - exactX;
218 const double errorY = floatY - exactY;
219 const double viewErrorX = errorX * cosine - errorY * sine;
220 const double viewErrorY = errorX * sine + errorY * cosine;
221 return std::hypot(viewErrorX * static_cast<double>(width) / (2.0 * viewport.halfWidth),
222 viewErrorY * static_cast<double>(height) / (2.0 * viewport.halfHeight));
223}

225bool fixedTopologyMeetsAccuracy(const LarZoneDefinition &zone, const lar::map::MapCamera &camera,
226 int width, int height) {
227 if (width <= 0 || height <= 0)
228 return false;
229 const GeodesicZoneSampleGrid adaptive =
230 GeodesicZoneSampler().sample(zone, camera, width, height);
231 if (adaptive.rowCount() < 2U || adaptive.columnCount() < 2U ||
232 adaptive.rowCount() - 1U > static_cast<std::size_t>(RadialSegments / 2) ||
233 adaptive.columnCount() - 1U > static_cast<std::size_t>(AngularSegments / 2)) {
234 return false;
235 }
236
237 constexpr double RoundingBudgetPixels = LarZoneMeshLimits::CurveErrorPixels * 0.35;
238 LarZoneDefinition rounded = zone;
239 rounded.center.latitude = static_cast<double>(static_cast<float>(zone.center.latitude));
240 rounded.center.longitude = static_cast<double>(static_cast<float>(zone.center.longitude));
241 rounded.innerRadiusMeters = static_cast<double>(static_cast<float>(zone.innerRadiusMeters));
242 rounded.outerRadiusMeters = static_cast<double>(static_cast<float>(zone.outerRadiusMeters));
243 rounded.startBearingRadians = static_cast<double>(static_cast<float>(zone.startBearingRadians));
244 rounded.spanRadians = static_cast<double>(static_cast<float>(zone.spanRadians));
245
246 for (const double radius : {zone.innerRadiusMeters, zone.outerRadiusMeters}) {
247 const double roundedRadius = radius == zone.innerRadiusMeters ? rounded.innerRadiusMeters
248 : rounded.outerRadiusMeters;
249 for (int sample = 0; sample <= 16; ++sample) {
250 const double fraction = static_cast<double>(sample) / 16.0;
251 const GeoCoordinateRadians precisePoint = LarGeodesicGeometry::destination(
252 zone.center, radius, zone.startBearingRadians + zone.spanRadians * fraction);
253 const GeoCoordinateRadians roundedPoint = LarGeodesicGeometry::destination(
254 rounded.center, roundedRadius,
255 rounded.startBearingRadians + rounded.spanRadians * fraction);
256 QPointF preciseScreen;
257 QPointF roundedScreen;
258 if (!projectPoint(camera, precisePoint, width, height, &preciseScreen) ||
259 !projectPoint(camera, roundedPoint, width, height, &roundedScreen) ||
260 std::hypot(preciseScreen.x() - roundedScreen.x(),
261 preciseScreen.y() - roundedScreen.y()) > RoundingBudgetPixels) {
262 return false;
263 }
264 if (camera.presentation() == lar::map::MapPresentation::Mercator &&
265 flatFloatCoordinateError(camera, precisePoint, width, height) >
266 RoundingBudgetPixels) {
267 return false;
268 }
269 }
270 }
271 return true;
272}

279void appendGrid(std::vector<float> &vertices, std::vector<std::uint32_t> &indices,
280 LarZoneDrawRange &fill, LarZoneDrawRange &lines, bool annular, bool fullCircle,
281 LarZoneDrawRange *fullCircleLines = nullptr) {
282 const std::uint32_t firstVertex = static_cast<std::uint32_t>(vertices.size() / 2U);
283 for (int row = 0; row <= RadialSegments; ++row) {
284 for (int column = 0; column <= AngularSegments; ++column) {
285 vertices.push_back(float(row) / float(RadialSegments));
286 vertices.push_back(float(column) / float(AngularSegments));
287 }
288 }
289 fill.firstIndex = indices.size();
290 for (int row = 0; row < RadialSegments; ++row) {
291 for (int column = 0; column < AngularSegments; ++column) {
292 const std::uint32_t a =
293 firstVertex + std::uint32_t(row * (AngularSegments + 1) + column);
294 const std::uint32_t b = a + 1U;
295 const std::uint32_t c = a + std::uint32_t(AngularSegments + 1);
296 const std::uint32_t d = c + 1U;
297 indices.insert(indices.end(), {a, c, b, c, d, b});
298 }
299 }
300 fill.indexCount = indices.size() - fill.firstIndex;
301
302 lines.firstIndex = indices.size();
303 const std::uint32_t outer = firstVertex + std::uint32_t(RadialSegments * (AngularSegments + 1));
304 for (int column = 0; column < AngularSegments; ++column)
305 indices.insert(indices.end(),
306 {outer + std::uint32_t(column), outer + std::uint32_t(column + 1)});
307 if (annular) {
308 const std::uint32_t inner = firstVertex;
309 for (int column = 0; column < AngularSegments; ++column) {
310 indices.insert(indices.end(),
311 {inner + std::uint32_t(column), inner + std::uint32_t(column + 1)});
312 }
313 }
314 if (fullCircleLines != nullptr) {
315 fullCircleLines->firstIndex = lines.firstIndex;
316 fullCircleLines->indexCount = indices.size() - fullCircleLines->firstIndex;
317 }
318 if (!fullCircle) {
319 for (int column : {0, AngularSegments}) {
320 for (int row = 0; row < RadialSegments; ++row) {
321 const std::uint32_t a =
322 firstVertex + std::uint32_t(row * (AngularSegments + 1) + column);
323 const std::uint32_t b = a + std::uint32_t(AngularSegments + 1);
324 indices.insert(indices.end(), {a, b});
325 }
326 }
327 }
328 lines.indexCount = indices.size() - lines.firstIndex;
329}

338bool LarParametricZoneGpuLayer::setZones(const Target &target, const QBitArray &availableFields,
339 const lar::map::MapCamera &camera, int viewportWidth,
340 int viewportHeight) {
341 m_inRange = {};
342 m_inZone = {};
343 m_hasZones = false;
344 const LarZoneValidationResult validated =
345 LarZoneInputValidator().validate(target, availableFields);
346 m_eligible = !validated.inputRejected;
347 m_minimumLongitude = 0.0;
348 m_maximumLongitude = 0.0;
349
350 const auto regular = [&camera, viewportWidth, viewportHeight](
351 const LarZoneDefinition &zone, LarParametricZoneState *destination) {
352 const double radiusAngle = zone.outerRadiusMeters / EarthRadius;
353 if (std::abs(zone.center.latitude) + radiusAngle >= Pi * 0.5 - qDegreesToRadians(0.05) ||
354 (camera.presentation() == lar::map::MapPresentation::Mercator &&
355 std::abs(zone.center.latitude) + radiusAngle >=
356 MercatorLimitRadians - qDegreesToRadians(0.05)) ||
357 !fixedTopologyMeetsAccuracy(zone, camera, viewportWidth, viewportHeight)) {
358 return false;
359 }
360 destination->valid = true;
361 destination->centerRadians = {float(zone.center.latitude), float(zone.center.longitude)};
362 destination->radiiMeters = {float(zone.innerRadiusMeters), float(zone.outerRadiusMeters)};
363 destination->startAndSpanRadians = {float(zone.startBearingRadians),
364 float(zone.spanRadians)};
365 destination->fullCircle = zone.spanRadians >= TwoPi - 1.0e-9;
366 return true;
367 };
368
369 if (validated.inRange) {
370 m_hasZones = true;
371 m_eligible = regular(*validated.inRange, &m_inRange) && m_eligible;
372 }
373 if (validated.inZone) {
374 m_hasZones = true;
375 m_eligible = regular(*validated.inZone, &m_inZone) && m_eligible;
376 }
377 if (m_hasZones) {
378 const auto include = [this](const LarParametricZoneState &zone) {
379 if (!zone.valid)
380 return;
381 const double center = qRadiansToDegrees(double(zone.centerRadians.y()));
382 const double radiusAngle = double(zone.radiiMeters.y()) / EarthRadius;
383 // A longitude extent based only on radiusAngle is too narrow near
384 // the poles. Use the exact small-circle extremum when the circle
385 // does not reach a pole, and conservatively cover the full world
386 // once it does. This keeps periodic copies gap-free for the
387 // parametric path just as they are for the CPU mesh path.
388 const double centerLatitude = std::abs(double(zone.centerRadians.x()));
389 const double longitudeDelta =
390 centerLatitude + radiusAngle >= Pi * 0.5
391 ? Pi
392 : std::asin(std::clamp(std::sin(radiusAngle) /
393 std::max(1.0e-9, std::cos(centerLatitude)),
394 -1.0, 1.0));
395 const double radius = qRadiansToDegrees(longitudeDelta);
396 if (m_minimumLongitude == 0.0 && m_maximumLongitude == 0.0) {
397 m_minimumLongitude = center - radius;
398 m_maximumLongitude = center + radius;
399 } else {
400 m_minimumLongitude = std::min(m_minimumLongitude, center - radius);
401 m_maximumLongitude = std::max(m_maximumLongitude, center + radius);
402 }
403 };
404 include(m_inRange);
405 include(m_inZone);
406 }
407 return m_eligible;
408}

437bool LarParametricZoneGpuLayer::compileProgram() {
438 m_program = new QOpenGLShaderProgram(this);
439 bool vertex = m_program->addShaderFromSourceCode(QOpenGLShader::Vertex, vertexShaderLegacy);
440 bool fragment =
441 m_program->addShaderFromSourceCode(QOpenGLShader::Fragment, fragmentShaderLegacy);
442 m_program->bindAttributeLocation("aParam", 0);
443 bool linked = m_program->link();
444 if (!vertex || !fragment || !linked) {
445 delete m_program;
446 m_program = new QOpenGLShaderProgram(this);
447 vertex = m_program->addShaderFromSourceCode(QOpenGLShader::Vertex, vertexShaderCore);
448 fragment = m_program->addShaderFromSourceCode(QOpenGLShader::Fragment, fragmentShaderCore);
449 m_program->bindAttributeLocation("aParam", 0);
450 linked = m_program->link();
451 }
452 if (!vertex || !fragment || !linked) {
453 emit diagnosticRaised(QStringLiteral("Parametric LAR shader initialization failed: %1")
454 .arg(m_program->log()));
455 return false;
456 }
457 m_positionAttribute = m_program->attributeLocation("aParam");
458 m_sphereModeUniform = m_program->uniformLocation("uSphereMode");
459 m_worldOffsetUniform = m_program->uniformLocation("uWorldOffsetX");
460 m_sphereCenterHighUniform = m_program->uniformLocation("uSphereCenterHigh");
461 m_sphereCenterLowUniform = m_program->uniformLocation("uSphereCenterLow");
462 m_sphereLatitudeSinCosUniform = m_program->uniformLocation("uSphereLatitudeSinCos");
463 m_cameraBearingUniform = m_program->uniformLocation("uCameraBearing");
464 m_flatCenterUniform = m_program->uniformLocation("uFlatCenter");
465 m_projectionUniform = m_program->uniformLocation("uProjection");
466 m_zoneCenterUniform = m_program->uniformLocation("uZoneCenter");
467 m_zoneRadiiUniform = m_program->uniformLocation("uZoneRadii");
468 m_startSpanUniform = m_program->uniformLocation("uStartSpan");
469 m_colorUniform = m_program->uniformLocation("uColor");
470 return m_positionAttribute >= 0 && m_sphereModeUniform >= 0 && m_worldOffsetUniform >= 0 &&
471 m_sphereCenterHighUniform >= 0 && m_sphereCenterLowUniform >= 0 &&
472 m_sphereLatitudeSinCosUniform >= 0 && m_cameraBearingUniform >= 0 &&
473 m_flatCenterUniform >= 0 && m_projectionUniform >= 0 && m_zoneCenterUniform >= 0 &&
474 m_zoneRadiiUniform >= 0 && m_startSpanUniform >= 0 && m_colorUniform >= 0;
475}

477bool LarParametricZoneGpuLayer::buildTopology() {
478 std::vector<float> vertices;
479 std::vector<std::uint32_t> indices;
480 vertices.reserve(2U * (RadialSegments + 1) * (AngularSegments + 1) * 2U);
481 indices.reserve(2U * RadialSegments * AngularSegments * 6U);
482 appendGrid(vertices, indices, m_inRangeFill, m_inRangeLines, false, true);
483 appendGrid(vertices, indices, m_inZoneFill, m_inZoneLines, true, false, &m_inZoneFullLines);
484 if (!m_vertexBuffer.create() || !m_indexBuffer.create() || !m_vertexArray.create()) {
485 emit diagnosticRaised(QStringLiteral("Could not create parametric LAR buffers."));
486 return false;
487 }
488 QOpenGLVertexArrayObject::Binder binder(&m_vertexArray);
489 m_vertexBuffer.bind();
490 m_vertexBuffer.setUsagePattern(QOpenGLBuffer::StaticDraw);
491 m_vertexBuffer.allocate(vertices.data(), int(vertices.size() * sizeof(float)));
492 m_indexBuffer.bind();
493 m_indexBuffer.setUsagePattern(QOpenGLBuffer::StaticDraw);
494 m_indexBuffer.allocate(indices.data(), int(indices.size() * sizeof(std::uint32_t)));
495 m_program->enableAttributeArray(m_positionAttribute);
496 m_program->setAttributeBuffer(m_positionAttribute, GL_FLOAT, 0, 2, 2 * int(sizeof(float)));
497 m_indexCount = indices.size();
498 return true;
499}

518void LarParametricZoneGpuLayer::draw(const LarZoneRenderState &state) {
519 if (!m_initialized || !m_eligible || !m_hasZones || m_program == nullptr)
520 return;
521 m_program->bind();
522 m_program->setUniformValue(m_sphereModeUniform, state.sphere);
523 m_program->setUniformValue(m_sphereCenterHighUniform, state.sphereProjection.centerHighDegrees);
524 m_program->setUniformValue(m_sphereCenterLowUniform, state.sphereProjection.centerLowDegrees);
525 m_program->setUniformValue(m_sphereLatitudeSinCosUniform,
526 state.sphereProjection.latitudeSinCos);
527 m_program->setUniformValue(m_cameraBearingUniform, state.bearingRadians);
528 m_program->setUniformValue(m_flatCenterUniform,
529 QVector2D(float(state.flatCenter.x()), float(state.flatCenter.y())));
530 m_program->setUniformValue(m_projectionUniform, state.projection);
531 QOpenGLVertexArrayObject::Binder binder(&m_vertexArray);
532 m_indexBuffer.bind();
533 glEnable(GL_BLEND);
534 glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
535 drawRange(m_inRange, m_inRangeFill, GL_TRIANGLES, QColor(52, 109, 145, 40), state.worldCopies);
