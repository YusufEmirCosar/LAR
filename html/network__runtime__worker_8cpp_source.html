63void NetworkRuntimeWorker::initialize() {
64 if (m_capture)
65 return;
66
67 m_source = m_sourceFactory(this);
68 if (!m_source) {
69 m_source = new UnavailableDatagramSource(this);
70 emit runtimeError({{},
71 std::nullopt,
72 RuntimeFailureCode::Construction,
73 QStringLiteral("Datagram source factory returned null")});
74 }
75 m_metrics = new MetricsService(this);
76 m_capture = new OnlineCaptureService(*m_source, *m_decoder, *m_metrics, this);
77 m_batchTimer = new QTimer(this);
78 m_batchTimer->setInterval(RecordingBatchIntervalMs);
79 m_batchTimer->setTimerType(Qt::PreciseTimer);
80
81 connect(m_capture, &OnlineCaptureService::stateReceived, this,
82 [this](const DecodedState &state) {
83 emit stateReady({{RuntimeStateSource::Online, m_stateGeneration}, state});
84 });
85 connect(m_capture, &OnlineCaptureService::rawPacketReceived, this,
86 &NetworkRuntimeWorker::queueRecordingPacket);
87 connect(m_capture, &OnlineCaptureService::captureError, this, [this](const QString &message) {
88 emit runtimeError({{},
89 RuntimeSourceEpoch{RuntimeStateSource::Online, m_stateGeneration},
90 RuntimeFailureCode::Operational,
91 message});
92 });
93 connect(m_metrics, &MetricsService::rateChanged, this, [this](quint64 rate) {
94 if (!m_capture->isListening())
95 return;
96 emit metricsChanged({{RuntimeStateSource::Online, m_stateGeneration},
97 m_metrics->totalProcessedPackets(),
98 rate});
99 });
100 connect(m_batchTimer, &QTimer::timeout, this, &NetworkRuntimeWorker::flushRecordingBatch);
101 m_batchTimer->start();
102}

172void NetworkRuntimeWorker::setRecordingInputEnabled(bool enabled) {
173 initialize();
174 m_recordingInputEnabled = enabled && m_pendingRecordingFailure.isEmpty();
175 if (enabled) {
176 m_bufferPostDrainPackets = false;
177 if (!m_postDrainBatch.isEmpty()) {
178 m_recordingBatch.reserve(m_recordingBatch.size() + m_postDrainBatch.size());
179 for (CapturedPacket &packet : m_postDrainBatch)
180 m_recordingBatch.append(std::move(packet));
181 m_recordingBatchBytes += m_postDrainBatchBytes;
182 m_postDrainBatch.clear();
183 m_postDrainBatchBytes = 0;
184 flushRecordingBatch();
185 }
186 } else {
187 m_bufferPostDrainPackets = false;
188 m_postDrainBatch.clear();
189 m_postDrainBatchBytes = 0;
190 flushRecordingBatch();
191 }
192}

230void NetworkRuntimeWorker::queueRecordingPacket(const CapturedPacket &packet) {
231 if (!m_recordingInputEnabled) {
232 if (!m_bufferPostDrainPackets)
233 return;
234 m_postDrainBatch.append(packet);
235 m_postDrainBatchBytes += packet.data.size();
236 if (m_postDrainBatchBytes > MaximumBufferedRecordingBytes) {
237 m_bufferPostDrainPackets = false;
238 m_pendingRecordingFailure =
239 QStringLiteral("Recording snapshot barrier exceeded its bounded post-snapshot "
240 "buffer; recording was paused");
241 }
242 return;
243 }
244 m_recordingBatch.append(packet);
245 m_recordingBatchBytes += packet.data.size();
246
247 if (m_recordingBatch.size() >= MaximumPacketsPerBatch ||
248 m_recordingBatchBytes >= MaximumBytesPerBatch) {
249 flushRecordingBatch();
250 }
251
252 if (m_recordingBatchBytes > MaximumBufferedRecordingBytes) {
253 m_recordingInputEnabled = false;
254 m_pendingRecordingFailure =
255 QStringLiteral("Recording pipeline cannot keep up; capture recording was paused before "
256 "unbounded memory growth");
257 flushRecordingBatch();
258 }

261void NetworkRuntimeWorker::flushRecordingBatch() {
262 if (m_recordingBatch.isEmpty() || m_outstandingBatchIds.size() >= MaximumOutstandingBatches) {
263 reportDrainIfComplete();
264 return;
265 }
266
267 QVector<CapturedPacket> packets;
268 packets.reserve(qMin(qsizetype(MaximumPacketsPerBatch), m_recordingBatch.size()));
269 qsizetype batchBytes = 0;
270 qsizetype count = 0;
271 while (count < m_recordingBatch.size() && count < MaximumPacketsPerBatch) {
272 const qsizetype nextSize = m_recordingBatch.at(count).data.size();
273 if (count > 0 && batchBytes + nextSize > MaximumBytesPerBatch) {
274 break;
275 }
276 batchBytes += nextSize;
277 ++count;
278 }
279 for (qsizetype index = 0; index < count; ++index)
280 packets.append(std::move(m_recordingBatch[index]));
281 m_recordingBatch.erase(m_recordingBatch.begin(), m_recordingBatch.begin() + count);
282 m_recordingBatchBytes -= batchBytes;
283 const quint64 batchId = ++m_nextBatchId;
284 m_outstandingBatchIds.insert(batchId);
285 emit recordingBatchReady(batchId, packets);
286}

288void NetworkRuntimeWorker::reportDrainIfComplete() {
289 if (!m_recordingBatch.isEmpty() || !m_outstandingBatchIds.isEmpty()) {
290 return;
291 }
292 if (!m_pendingRecordingFailure.isEmpty()) {
293 const QString error = m_pendingRecordingFailure;
294 m_pendingRecordingFailure.clear();
295 emit recordingInputFailed(error);
296 }
297
298 if (m_pendingDrainToken != 0) {
299 const quint64 token = m_pendingDrainToken;
300 const qint64 boundaryNanoseconds = m_pendingDrainBoundaryNanoseconds;
301 m_pendingDrainToken = 0;
302 m_pendingDrainBoundaryNanoseconds = 0;
303 emit recordingInputDrained(token, boundaryNanoseconds);
304 }
