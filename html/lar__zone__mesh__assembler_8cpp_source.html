71std::vector<double> unwrappedLongitudes(const GeodesicZoneSampleGrid &samples, std::size_t row,
72 double reference) {
73 std::vector<double> result;
74 result.reserve(samples.columnCount());
75 double previous = reference;
76 for (std::size_t column = 0; column < samples.columnCount(); ++column) {
77 const double longitude = lar::map::MapProjection::unwrapLongitude(
78 samples.point(row, column).longitude * ToDegrees, previous);
79 result.push_back(longitude);
80 previous = longitude;
81 }
82 if (!result.empty()) {
83 const auto bounds = std::minmax_element(result.begin(), result.end());
84 const double midpoint = (*bounds.first + *bounds.second) * 0.5;
85 const double shift =
86 std::round((reference - midpoint) / lar::map::MapProjection::WorldWidthDegrees) *
87 lar::map::MapProjection::WorldWidthDegrees;
88 for (double &longitude : result)
89 longitude += shift;
90 }
91 return result;
92}

104bool LarZoneMeshAssembler::append(const LarZoneDefinition &zone,
105 const GeodesicZoneSampleGrid &samples,
106 lar::map::MapPresentation presentation, LarZoneMesh &mesh,
107 LarZoneDrawRange &fillRange, LarZoneDrawRange &lineRange) const {
108 const std::size_t rows = samples.rowCount();
109 const std::size_t columns = samples.columnCount();
110 if (rows < 2U || columns < 2U || samples.points.size() != rows * columns)
111 return false;
112 MeshAppender appender(mesh);
113 const auto firstVertex = static_cast<std::uint32_t>(mesh.vertices.size() / 3U);
114 const double reference = zone.center.longitude * ToDegrees;
115 for (std::size_t row = 0; row < rows; ++row) {
116 const auto longitudes = unwrappedLongitudes(samples, row, reference);
117 for (std::size_t column = 0; column < columns; ++column) {
118 const auto &point = samples.point(row, column);
119 if (presentation == lar::map::MapPresentation::Mercator &&
120 std::abs(point.latitude * ToDegrees) >
121 lar::map::MapProjection::MaximumMercatorLatitudeDegrees) {
122 mesh.mercatorGeometryClipped = true;
123 }
124 appender.vertex(point.latitude, longitudes[column]);
125 }
126 }
127
128 fillRange.firstIndex = mesh.indices.size();
129 for (std::size_t row = 0; row + 1U < rows; ++row) {
130 for (std::size_t column = 0; column + 1U < columns; ++column) {
131 const std::uint32_t inner0 =
132 firstVertex + static_cast<std::uint32_t>(row * columns + column);
133 const std::uint32_t inner1 = inner0 + 1U;
134 const std::uint32_t outer0 = inner0 + static_cast<std::uint32_t>(columns);
135 const std::uint32_t outer1 = outer0 + 1U;
136 if (samples.radii[row] <= LarZoneMeshLimits::SampleTolerance) {
137 appender.triangle(inner0, outer0, outer1);
138 } else {
139 appender.triangle(inner0, outer0, inner1);
140 appender.triangle(outer0, outer1, inner1);
141 }
142 }
143 }
144 fillRange.indexCount = mesh.indices.size() - fillRange.firstIndex;
145
146 lineRange.firstIndex = mesh.indices.size();
147 const auto outerFirst = firstVertex + static_cast<std::uint32_t>((rows - 1U) * columns);
148 appender.lineStrip(outerFirst, static_cast<int>(columns));
149 if (zone.innerRadiusMeters > LarZoneMeshLimits::SampleTolerance) {
150 appender.lineStrip(firstVertex, static_cast<int>(columns));
151 }
152 if (!samples.fullCircle) {
153 appendRadialLine(appender, firstVertex, rows, columns, 0U, 0U);
154 appendRadialLine(appender, firstVertex, rows, columns, columns - 1U, 0U);
155 } else if (presentation == lar::map::MapPresentation::Mercator &&
156 zone.outerRadiusMeters >
157 samples.seamPoleDistanceMeters + LarZoneMeshLimits::SampleTolerance) {
158 const auto seamStart =
159 std::lower_bound(samples.radii.begin(), samples.radii.end(),
160 std::max(zone.innerRadiusMeters, samples.seamPoleDistanceMeters) -
161 LarZoneMeshLimits::SampleTolerance);
162 const std::size_t firstSeamRow =
163 static_cast<std::size_t>(std::distance(samples.radii.begin(), seamStart));
164 appendRadialLine(appender, firstVertex, rows, columns, 0U, firstSeamRow);
165 appendRadialLine(appender, firstVertex, rows, columns, columns - 1U, firstSeamRow);
166 }
167 lineRange.indexCount = mesh.indices.size() - lineRange.firstIndex;
168 return !appender.failed();
169}
