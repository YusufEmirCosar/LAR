97 m_program = std::make_unique<QOpenGLShaderProgram>();
98 bool vertexCompiled =
99 m_program->addShaderFromSourceCode(QOpenGLShader::Vertex, shaders::VertexLegacy);
100 bool fragmentCompiled =
101 m_program->addShaderFromSourceCode(QOpenGLShader::Fragment, shaders::FragmentLegacy);
102 m_program->bindAttributeLocation("aPosition", 0);
103 bool linked = m_program->link();
104
105 if (!vertexCompiled || !fragmentCompiled || !linked) {
106 m_program = std::make_unique<QOpenGLShaderProgram>();
107 vertexCompiled =
108 m_program->addShaderFromSourceCode(QOpenGLShader::Vertex, shaders::VertexCore);
109 fragmentCompiled =

111 m_program->bindAttributeLocation("aPosition", 0);
112 linked = m_program->link();
113 }
114
115 if (!vertexCompiled || !fragmentCompiled || !linked) {
116 setError(errorMessage,

179bool EarthMapGpuRenderer::createOceanDisc(QString *errorMessage) {
180 std::vector<float> circle;
181 constexpr int SegmentCount = 96;
182 circle.reserve(static_cast<std::size_t>(SegmentCount + 2) * 3U);
183 circle.insert(circle.end(), {0.0F, 0.0F, 0.0F});
184 for (int index = 0; index <= SegmentCount; ++index) {
185 const float angle = static_cast<float>(index) * 2.0F * static_cast<float>(M_PI) /
186 static_cast<float>(SegmentCount);
187 circle.push_back(std::cos(angle));
188 circle.push_back(std::sin(angle));
189 circle.push_back(0.0F);
190 }
191 m_oceanVertexCount = circle.size() / 3U;
192
193 if ((!m_oceanVertexBuffer.isCreated() && !m_oceanVertexBuffer.create()) ||
194 (!m_oceanVertexArray.isCreated() && !m_oceanVertexArray.create())) {
195
196 setError(errorMessage, QStringLiteral("Failed to create the ocean GPU geometry."));
197 return false;
198 }
199
200 QOpenGLVertexArrayObject::Binder binder(&m_oceanVertexArray);
201 m_oceanVertexBuffer.bind();
202 m_oceanVertexBuffer.allocate(circle.data(), static_cast<int>(circle.size() * sizeof(float)));
203 m_program->enableAttributeArray(m_positionAttribute);
204
205 m_program->setAttributeBuffer(m_positionAttribute, GL_FLOAT, 0, VertexStrideFloats,
206 VertexStrideBytes);
207
208 return true;
209}

211bool EarthMapGpuRenderer::uploadPending(QString *errorMessage) {
212 if (!m_pendingUpload) {
213 return true;
214 }
215 if (!m_shaderValid || m_program == nullptr || m_mesh == nullptr) {
216 setError(errorMessage, QStringLiteral("Map GPU resources are not ready for upload."));
217 return false;
218 }
219 if ((!m_vertexBuffer.isCreated() && !m_vertexBuffer.create()) ||
220 (!m_mapVertexArray.isCreated() && !m_mapVertexArray.create())) {
221
222 setError(errorMessage, QStringLiteral("Failed to create map GPU geometry."));
223 return false;
224 }
225
226 m_program->bind();
227 {
228 QOpenGLVertexArrayObject::Binder binder(&m_mapVertexArray);
229 m_vertexBuffer.bind();
230 m_vertexBuffer.allocate(m_mesh->vertices.data(),
231 static_cast<int>(m_mesh->vertices.size() * sizeof(float)));
232 m_program->enableAttributeArray(m_positionAttribute);
233
234 m_program->setAttributeBuffer(m_positionAttribute, GL_FLOAT, 0, VertexStrideFloats,
235 VertexStrideBytes);
236 }
237
238 if (!uploadIndexBuffer(m_mercatorIndexBuffer, m_mesh->mercatorFillIndices,
239 QStringLiteral("Mercator fill"), errorMessage) ||
240 !uploadIndexBuffer(m_sphereIndexBuffer, m_mesh->sphereFillIndices,
241 QStringLiteral("sphere fill"), errorMessage) ||
242 !uploadIndexBuffer(m_borderIndexBuffer, m_mesh->borderIndices, QStringLiteral("border"),
243 errorMessage) ||
244 !createOceanDisc(errorMessage)) {
245
246 m_program->release();
247 return false;
248 }
249
250 m_mercatorIndexCount = m_mesh->mercatorFillIndices.size();
251 m_sphereIndexCount = m_mesh->sphereFillIndices.size();
252 m_borderIndexCount = m_mesh->borderIndices.size();
253 m_pendingUpload = false;
254 m_program->release();
255 return true;

266void EarthMapGpuRenderer::draw(const MapCamera &camera, int width, int height) {
267 const bool sphere = camera.presentation() == MapPresentation::Sphere;
268 const QVector4D &clearColor = sphere ? palette::Space : palette::Ocean;
269
270 glClearColor(clearColor.x(), clearColor.y(), clearColor.z(), clearColor.w());
271
272 glClear(GL_COLOR_BUFFER_BIT);
273
274 const std::size_t fillCount = fillIndexCount(camera.presentation());
275
276 if (!m_shaderValid || m_program == nullptr || fillCount == 0U || m_pendingUpload) {
277
278 return;
279 }
280
281 m_program->bind();
282 m_program->setUniformValue(m_screenSpaceUniform, false);
283 m_program->setUniformValue(m_sphereUniform, sphere);
284 m_program->setUniformValue(m_worldOffsetUniform, 0.0F);
285 const SphereProjectionParameters sphereProjection = camera.sphereProjectionParameters();
286 m_program->setUniformValue(m_sphereCenterHighUniform, sphereProjection.centerHighDegrees);
287 m_program->setUniformValue(m_sphereCenterLowUniform, sphereProjection.centerLowDegrees);
288 m_program->setUniformValue(m_sphereLatitudeSinCosUniform, sphereProjection.latitudeSinCos);
289 m_program->setUniformValue(m_bearingUniform, qDegreesToRadians(camera.bearingDegrees()));
290 m_program->setUniformValue(m_mercatorCenterUniform,
291 QVector2D(static_cast<float>(camera.mercatorCenter().x()),
292 static_cast<float>(camera.mercatorCenter().y())));
293 m_program->setUniformValue(m_projectionUniform, camera.projectionMatrix(width, height));
294
295 if (sphere && m_oceanVertexBuffer.isCreated() && m_oceanVertexArray.isCreated()) {
296
297 m_program->setUniformValue(m_keepBackUniform, true);
298 m_program->setUniformValue(m_screenSpaceUniform, true);
299 m_program->setUniformValue(m_colorUniform, palette::Ocean);
300 QOpenGLVertexArrayObject::Binder oceanBinder(&m_oceanVertexArray);
301 glDrawArrays(GL_TRIANGLE_FAN, 0, static_cast<GLsizei>(m_oceanVertexCount));
302 m_program->setUniformValue(m_screenSpaceUniform, false);
303 m_program->setUniformValue(m_keepBackUniform, false);
304 }
305
306 QOpenGLVertexArrayObject::Binder mapBinder(&m_mapVertexArray);
307 fillIndexBuffer(camera.presentation()).bind();
308 m_program->setUniformValue(m_colorUniform, palette::Land);
309 m_program->setUniformValue(m_keepBackUniform, false);
310 const WorldCopyRange copies =
311 sphere ? WorldCopyRange{0, 0} : camera.visibleWorldCopies(width, height);
312 for (int copy = copies.first; copy <= copies.last; ++copy) {
313 m_program->setUniformValue(m_worldOffsetUniform,
314 static_cast<float>(copy * MapProjection::WorldWidthDegrees));
315
316 glDrawElements(GL_TRIANGLES, static_cast<GLsizei>(fillCount), GL_UNSIGNED_INT, nullptr);
317 }
318
319 if (m_borderIndexBuffer.isCreated() && m_borderIndexCount > 0U) {
320 m_borderIndexBuffer.bind();
321 m_program->setUniformValue(m_colorUniform, palette::Border);
322 glLineWidth(1.0F);
323 for (int copy = copies.first; copy <= copies.last; ++copy) {
324 m_program->setUniformValue(m_worldOffsetUniform,
325 static_cast<float>(copy * MapProjection::WorldWidthDegrees));
326
327 glDrawElements(GL_LINES, static_cast<GLsizei>(m_borderIndexCount), GL_UNSIGNED_INT,
328 nullptr);
329 }
330 }
331 m_program->setUniformValue(m_worldOffsetUniform, 0.0F);
332 m_program->release();
333}
