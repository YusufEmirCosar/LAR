27bool compileProgram(QOpenGLShaderProgram &program, const char *legacyVertex,
28 const char *legacyFragment, const char *coreVertex, const char *coreFragment,
29 const char *attribute, QString *errorMessage) {
30 bool vertex = program.addShaderFromSourceCode(QOpenGLShader::Vertex, legacyVertex);
31 bool fragment = program.addShaderFromSourceCode(QOpenGLShader::Fragment, legacyFragment);
32 program.bindAttributeLocation(attribute, 0);
33 bool linked = program.link();
34 if (!vertex || !fragment || !linked) {
35 program.removeAllShaders();
36 vertex = program.addShaderFromSourceCode(QOpenGLShader::Vertex, coreVertex);
37 fragment = program.addShaderFromSourceCode(QOpenGLShader::Fragment, coreFragment);
38 program.bindAttributeLocation(attribute, 0);
39 linked = program.link();
40 }
41 if (!vertex || !fragment || !linked) {
42 setError(
43 errorMessage,
44 QStringLiteral("Plane surface shader initialization failed: %1").arg(program.log()));
45 return false;
46 }
47 return true;
48}

61void gridLine(std::vector<float> &vertices, int coordinate, bool alongZ) {
62 const float value = static_cast<float>(coordinate);
63 const float edge = static_cast<float>(GridHalfLineCount);
64 if (alongZ) {
65 vertex(vertices, value, 0.0F, -edge);
66 vertex(vertices, value, 0.0F, edge);
67 } else {
68 vertex(vertices, -edge, 0.0F, value);
69 vertex(vertices, edge, 0.0F, value);

73bool validZone(const PlaneSurfaceZone &zone) noexcept {
74 constexpr float TwoPi = 6.28318530717958647692F;
75 return zone.visible && std::isfinite(zone.centerXZ.x()) && std::isfinite(zone.centerXZ.y()) &&
76 std::isfinite(zone.innerRadius) && std::isfinite(zone.outerRadius) &&
77 std::isfinite(zone.startBearingRadians) && std::isfinite(zone.spanRadians) &&
78 zone.innerRadius >= 0.0F && zone.outerRadius >= zone.innerRadius &&
79 zone.outerRadius <= PlaneSurfaceMaximumCoordinate && zone.spanRadians > 0.0F &&
80 zone.spanRadians <= TwoPi + 0.0001F;
81}

102bool PlaneSurfaceGpuLayer::createShapeGeometry(QString *errorMessage) {
103 std::vector<float> vertices;
104 vertices.reserve(8000U);
105 const auto capture = [&vertices](std::size_t firstVertex) {
106 return Range{firstVertex, vertices.size() / 3U - firstVertex};
107 };
108 std::size_t first = vertices.size() / 3U;
109 triangle(vertices, {-1.0F, 0.0F, -1.0F}, {1.0F, 0.0F, -1.0F}, {1.0F, 0.0F, 1.0F});
110 triangle(vertices, {-1.0F, 0.0F, -1.0F}, {1.0F, 0.0F, 1.0F}, {-1.0F, 0.0F, 1.0F});
111 m_ground = capture(first);
112
113 first = vertices.size() / 3U;
114 for (int coordinate = -GridHalfLineCount; coordinate <= GridHalfLineCount; ++coordinate) {
115 if (coordinate % 5 != 0) {
116 gridLine(vertices, coordinate, true);
117 gridLine(vertices, coordinate, false);
118 }
119 }
120 m_minorGrid = capture(first);
121 first = vertices.size() / 3U;
122 for (int coordinate = -GridHalfLineCount; coordinate <= GridHalfLineCount; ++coordinate) {
123 // Zero is the phase-wrapped major line nearest the aircraft, not the highlighted
124 // Earth origin. The Earth-fixed axes are drawn separately with groundOriginXZ.
125 if (coordinate % 5 == 0) {
126 gridLine(vertices, coordinate, true);
127 gridLine(vertices, coordinate, false);
128 }
129 }
130 m_majorGrid = capture(first);
131 first = vertices.size() / 3U;
132 gridLine(vertices, 0, true);
133 gridLine(vertices, 0, false);
134 m_gridAxes = capture(first);
135
136 first = vertices.size() / 3U;
137 constexpr QVector3D PyramidApex{0.0F, 1.5F, 0.0F};
138 constexpr QVector3D PyramidNorthWest{-1.0F, 0.0F, -1.0F};
139 constexpr QVector3D PyramidNorthEast{1.0F, 0.0F, -1.0F};
140 constexpr QVector3D PyramidSouthEast{1.0F, 0.0F, 1.0F};
141 constexpr QVector3D PyramidSouthWest{-1.0F, 0.0F, 1.0F};
142 triangle(vertices, PyramidApex, PyramidNorthWest, PyramidNorthEast);
143 triangle(vertices, PyramidApex, PyramidNorthEast, PyramidSouthEast);
144 triangle(vertices, PyramidApex, PyramidSouthEast, PyramidSouthWest);
145 triangle(vertices, PyramidApex, PyramidSouthWest, PyramidNorthWest);
146 m_targetPyramid = capture(first);
147
148 if (!m_shapeVertices.create() || !m_shapeArray.create() ||
149 vertices.size() >
150 static_cast<std::size_t>(std::numeric_limits<int>::max()) / sizeof(float)) {
151 setError(errorMessage, QStringLiteral("Plane surface geometry could not be created."));
152 return false;
153 }
154 m_shapeProgram->bind();
155 QOpenGLVertexArrayObject::Binder binder(&m_shapeArray);
156 m_shapeVertices.bind();
157 m_shapeVertices.setUsagePattern(QOpenGLBuffer::StaticDraw);
158 m_shapeVertices.allocate(vertices.data(), static_cast<int>(vertices.size() * sizeof(float)));
159 m_shapeProgram->enableAttributeArray(0);
160 m_shapeProgram->setAttributeBuffer(0, GL_FLOAT, 0, 3, 3 * static_cast<int>(sizeof(float)));
161 m_shapeProgram->release();
162 return true;
163}

165bool PlaneSurfaceGpuLayer::createZoneGeometry(QString *errorMessage) {
166 std::vector<float> vertices;
167 vertices.reserve(static_cast<std::size_t>((AngularSegments + 1) * 4));
168 for (int row = 0; row <= 1; ++row) {
169 for (int column = 0; column <= AngularSegments; ++column) {
170 vertices.push_back(static_cast<float>(row));
171 vertices.push_back(static_cast<float>(column) / static_cast<float>(AngularSegments));
172 }
173 }
174 std::vector<std::uint32_t> indices;
175 indices.reserve(2048U);
176 auto capture = [&indices](std::size_t first) { return Range{first, indices.size() - first}; };
177 std::size_t first = indices.size();
178 const std::uint32_t outerStart = static_cast<std::uint32_t>(AngularSegments + 1);
179 for (int column = 0; column < AngularSegments; ++column) {
180 const std::uint32_t inner = static_cast<std::uint32_t>(column);
181 const std::uint32_t outer = outerStart + inner;
182 indices.insert(indices.end(), {inner, outer, inner + 1U, outer, outer + 1U, inner + 1U});
183 }
184 m_zoneFill = capture(first);
185 const auto appendArc = [&indices](std::uint32_t start) {
186 for (int column = 0; column < AngularSegments; ++column) {
187 const std::uint32_t current = start + static_cast<std::uint32_t>(column);
188 indices.insert(indices.end(), {current, current + 1U});
189 }
190 };
191 first = indices.size();
192 appendArc(outerStart);
193 m_circleLines = capture(first);
194 first = indices.size();
195 appendArc(outerStart);
196 appendArc(0U);
197 m_annulusLines = capture(first);
198 first = indices.size();
199 appendArc(outerStart);
200 appendArc(0U);
201 indices.insert(indices.end(), {0U, outerStart, static_cast<std::uint32_t>(AngularSegments),
202 outerStart + static_cast<std::uint32_t>(AngularSegments)});
203 m_sectorLines = capture(first);
204
205 const bool tooLarge =
206 vertices.size() >
207 static_cast<std::size_t>(std::numeric_limits<int>::max()) / sizeof(float) ||
208 indices.size() >
209 static_cast<std::size_t>(std::numeric_limits<int>::max()) / sizeof(std::uint32_t);
210 if (tooLarge || !m_zoneVertices.create() || !m_zoneIndices.create() || !m_zoneArray.create()) {
211 setError(errorMessage, QStringLiteral("Plane LAR geometry could not be created."));
212 return false;
213 }
214 m_zoneProgram->bind();
215 QOpenGLVertexArrayObject::Binder binder(&m_zoneArray);
216 m_zoneVertices.bind();
217 m_zoneVertices.setUsagePattern(QOpenGLBuffer::StaticDraw);
218 m_zoneVertices.allocate(vertices.data(), static_cast<int>(vertices.size() * sizeof(float)));
219 m_zoneIndices.bind();
220 m_zoneIndices.setUsagePattern(QOpenGLBuffer::StaticDraw);
221 m_zoneIndices.allocate(indices.data(),
222 static_cast<int>(indices.size() * sizeof(std::uint32_t)));
223 m_zoneProgram->enableAttributeArray(0);
224 m_zoneProgram->setAttributeBuffer(0, GL_FLOAT, 0, 2, 2 * static_cast<int>(sizeof(float)));
225 m_zoneProgram->release();
226 return true;
227}

285void PlaneSurfaceGpuLayer::draw(const PlaneSurfaceState &state, const QMatrix4x4 &view,
286 const QMatrix4x4 &projection, bool drawGround, bool drawOverlays) {
287 if (!m_ready || (!drawGround && !drawOverlays)) {
288 return;
289 }
290 const float halfExtent =
291 std::clamp(state.surfaceHalfExtent, 64.0F, PlaneSurfaceMaximumCoordinate);
292 const float maximumGridSpacing =
293 PlaneSurfaceMaximumCoordinate / static_cast<float>(GridHalfLineCount);
294 const float gridSpacing = std::clamp(state.gridSpacingSceneUnits, 0.25F, maximumGridSpacing);
295 if (drawGround) {
296 glEnable(GL_DEPTH_TEST);
297 glDepthMask(GL_TRUE);
298 glDisable(GL_BLEND);
299 glDisable(GL_CULL_FACE);
300 drawShape(m_ground, GL_TRIANGLES, view, projection, halfExtent, {}, state.surfaceHeight,
301 halfExtent, {0.20F, 0.29F, 0.25F, 1.0F}, false);
302 }
303 if (!drawOverlays) {
304 return;
305 }
306
307 glEnable(GL_BLEND);
308 glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
309 glDepthMask(GL_FALSE);
310 glDisable(GL_DEPTH_TEST);
311 drawShape(m_minorGrid, GL_LINES, view, projection, gridSpacing, state.gridPhaseXZ,
312 state.surfaceHeight + 0.015F, halfExtent, {0.64F, 0.72F, 0.68F, 0.34F}, true);
313 drawShape(m_majorGrid, GL_LINES, view, projection, gridSpacing, state.gridPhaseXZ,
314 state.surfaceHeight + 0.020F, halfExtent, {0.75F, 0.82F, 0.79F, 0.58F}, true);
315 drawShape(m_gridAxes, GL_LINES, view, projection, gridSpacing, state.groundOriginXZ,
316 state.surfaceHeight + 0.025F, halfExtent, {0.88F, 0.92F, 0.90F, 0.80F}, true);
317 drawZone(state.inRange, m_zoneFill, GL_TRIANGLES, view, projection,
318 state.surfaceHeight + 0.035F, {0.20F, 0.43F, 0.57F, 0.35F});
319 drawZone(state.inZone, m_zoneFill, GL_TRIANGLES, view, projection, state.surfaceHeight + 0.045F,
320 {0.08F, 0.51F, 0.40F, 0.46F});
321
322 glLineWidth(2.0F);
323 drawZone(state.inRange, m_circleLines, GL_LINES, view, projection, state.surfaceHeight + 0.055F,
324 {0.35F, 0.70F, 0.91F, 1.0F});
325 const Range &inZoneLines = state.inZone.fullCircle ? m_annulusLines : m_sectorLines;
326 drawZone(state.inZone, inZoneLines, GL_LINES, view, projection, state.surfaceHeight + 0.060F,
327 {0.16F, 0.82F, 0.64F, 1.0F});
328 if (state.targetVisible && std::isfinite(state.targetXZ.x()) &&
329 std::isfinite(state.targetXZ.y())) {
330 const float targetScale =
331 std::isfinite(state.targetMarkerScale)
332 ? std::clamp(state.targetMarkerScale, 1.0F, PlaneSurfaceMaximumCoordinate)
333 : 1.0F;
334 const float targetHeight =
335 std::isfinite(state.targetHeight) ? state.targetHeight : state.surfaceHeight;
336 drawShape(m_targetPyramid, GL_TRIANGLES, view, projection, targetScale, state.targetXZ,
337 targetHeight + 0.070F, halfExtent, {0.90F, 0.20F, 0.14F, 1.0F}, false,
338 targetScale);
339 }
340 glLineWidth(1.0F);
341 glDisable(GL_BLEND);
342 glDepthMask(GL_TRUE);
343}
