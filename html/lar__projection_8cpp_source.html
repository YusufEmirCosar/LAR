7QPointF LarProjection::geographicToPlaneWorld(const double position[3],
8 const double planeLocation[3], double yaw,
9 double originLatitude, bool hasOrigin) noexcept {
10 const double latRef = hasOrigin ? originLatitude : planeLocation[0];
11 const double cosineLatitude = qMax(0.01, std::cos(latRef));
12 double longitudeDelta = position[1] - planeLocation[1];
13 while (longitudeDelta > Pi)
14 longitudeDelta -= 2.0 * Pi;
15 while (longitudeDelta < -Pi)
16 longitudeDelta += 2.0 * Pi;
17 const double east = EarthRadiusMeters * longitudeDelta * cosineLatitude;
18 const double north = EarthRadiusMeters * (position[0] - planeLocation[0]);
19 return {east * std::cos(yaw) - north * std::sin(yaw),
20 east * std::sin(yaw) + north * std::cos(yaw)};
21}
