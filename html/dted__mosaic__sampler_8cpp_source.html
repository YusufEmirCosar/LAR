90DtedMosaicSampler::positionFor(double latitudeRadians, double longitudeRadians) {
91 const std::optional<DtedCellKey> key =
92 DtedTileSource::keyForRadians(latitudeRadians, longitudeRadians);
93 if (!key) {
94 m_lastError = QStringLiteral("Terrain coordinate is outside the WGS84 DTED range.");
95 return {};
96 }
97 const std::shared_ptr<const DtedCell> cell = cellFor(*key);
98 if (cell == nullptr || !cell->valid()) {
99 return {};
100 }
101
102 const double latitudeDegrees = latitudeRadians * 180.0 / Pi;
103 const double longitudeDegrees = normalizedLongitudeDegrees(longitudeRadians);
104 const double longitudeCoordinate =
105 std::clamp((longitudeDegrees - static_cast<double>(key->longitudeDegrees)) /
106 cell->longitudeIntervalDegrees,
107 0.0, static_cast<double>(cell->longitudeSampleCount - 1));
108 const double latitudeCoordinate =
109 std::clamp((latitudeDegrees - static_cast<double>(key->latitudeDegrees)) /
110 cell->latitudeIntervalDegrees,
111 0.0, static_cast<double>(cell->latitudeSampleCount - 1));
112 SamplePosition result;
113 result.key = *key;
114 result.cell = cell;
115 result.longitude0 = static_cast<int>(std::floor(longitudeCoordinate));
116 result.latitude0 = static_cast<int>(std::floor(latitudeCoordinate));
117 result.longitude1 = std::min(result.longitude0 + 1, cell->longitudeSampleCount - 1);
118 result.latitude1 = std::min(result.latitude0 + 1, cell->latitudeSampleCount - 1);
119 result.longitudeFraction = longitudeCoordinate - static_cast<double>(result.longitude0);
120 result.latitudeFraction = latitudeCoordinate - static_cast<double>(result.latitude0);
121 return result;
122}

124std::optional<double> DtedMosaicSampler::interpolatedElevation(const SamplePosition &position,
125 const DtedWaterMaskCell *mask,
126 std::optional<bool> water) {
127 const DtedCell &cell = *position.cell;
128 const int longitude0 = position.longitude0;
129 const int longitude1 = position.longitude1;
130 const int latitude0 = position.latitude0;
131 const int latitude1 = position.latitude1;
132 const double longitudeFraction = position.longitudeFraction;
133 const double latitudeFraction = position.latitudeFraction;
134
135 const std::array<std::optional<double>, 4> samples{
136 cell.elevation(longitude0, latitude0), cell.elevation(longitude1, latitude0),
137 cell.elevation(longitude0, latitude1), cell.elevation(longitude1, latitude1)};
138 const std::array<std::optional<bool>, 4> classifications{
139 mask != nullptr ? mask->water(longitude0, latitude0) : std::nullopt,
140 mask != nullptr ? mask->water(longitude1, latitude0) : std::nullopt,
141 mask != nullptr ? mask->water(longitude0, latitude1) : std::nullopt,
142 mask != nullptr ? mask->water(longitude1, latitude1) : std::nullopt};
143 const std::array<double, 4> weights{(1.0 - longitudeFraction) * (1.0 - latitudeFraction),
144 longitudeFraction * (1.0 - latitudeFraction),
145 (1.0 - longitudeFraction) * latitudeFraction,
146 longitudeFraction * latitudeFraction};
147 double weightedElevation = 0.0;
148 double validWeight = 0.0;
149 for (std::size_t index = 0; index < samples.size(); ++index) {
150 const bool classificationMatches =
151 !water || (classifications[index] && *classifications[index] == *water);
152 if (samples[index] && classificationMatches) {
153 weightedElevation += *samples[index] * weights[index];
154 validWeight += weights[index];
155 }
156 }
157 if (validWeight <= std::numeric_limits<double>::epsilon()) {
158 return std::nullopt;
159 }
160 return weightedElevation / validWeight;
161}
