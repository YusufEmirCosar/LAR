27std::shared_ptr<const DtedCell> DtedMosaicSampler::cellFor(const DtedCellKey &key) {
28 ++m_accessCounter;
29 const auto found = m_cache.find(key);
30 if (found != m_cache.end()) {
31 found->second.access = m_accessCounter;
32 return found->second.cell;
33 }

35 if (!result.succeeded()) {
36 m_lastError = result.message;

39 const std::size_t bytes = cell != nullptr ? cell->storageBytes() : 0U;
40 m_cachedTerrainBytes += bytes;
41 m_cache.emplace(key, CacheEntry{cell, m_accessCounter, bytes});

46void DtedMosaicSampler::evictIfNeeded() {
47 while (m_cache.size() > m_cacheCapacity ||
48 (m_cachedTerrainBytes > m_cacheByteCapacity && m_cache.size() > 1U)) {
49 auto oldest = m_cache.begin();
50 for (auto iterator = std::next(m_cache.begin()); iterator != m_cache.end(); ++iterator) {
51 if (iterator->second.access < oldest->second.access) {
52 oldest = iterator;
53 }
54 }

79 const double latitudeCoordinate =

84 result.key = *key;

87 result.latitude0 = static_cast<int>(std::floor(latitudeCoordinate));
88 result.longitude1 = std::min(result.longitude0 + 1, cell->longitudeSampleCount - 1);
89 result.latitude1 = std::min(result.latitude0 + 1, cell->latitudeSampleCount - 1);
90 result.longitudeFraction = longitudeCoordinate - static_cast<double>(result.longitude0);

95std::optional<double> DtedMosaicSampler::interpolatedElevation(const SamplePosition &position) {
96 const DtedCell &cell = *position.cell;
97 const int longitude0 = position.longitude0;
98 const int longitude1 = position.longitude1;
99 const int latitude0 = position.latitude0;
100 const int latitude1 = position.latitude1;
101 const double longitudeFraction = position.longitudeFraction;
102 const double latitudeFraction = position.latitudeFraction;
103
104 const std::array<std::optional<double>, 4> samples{
105 cell.elevation(longitude0, latitude0), cell.elevation(longitude1, latitude0),
106 cell.elevation(longitude0, latitude1), cell.elevation(longitude1, latitude1)};
107 const std::array<double, 4> weights{(1.0 - longitudeFraction) * (1.0 - latitudeFraction),
108 longitudeFraction * (1.0 - latitudeFraction),
109 (1.0 - longitudeFraction) * latitudeFraction,
110 longitudeFraction * latitudeFraction};
111 double weightedElevation = 0.0;
112 double validWeight = 0.0;
113 for (std::size_t index = 0; index < samples.size(); ++index) {
114 if (samples[index]) {
115 weightedElevation += *samples[index] * weights[index];
116 validWeight += weights[index];
117 }
118 }
119 if (validWeight <= std::numeric_limits<double>::epsilon()) {
120 return std::nullopt;
121 }
122 // Void posts contribute neither value nor weight, so divide by the
123 // remaining weight rather than depressing the surface toward zero.
124 return weightedElevation / validWeight;
125}
