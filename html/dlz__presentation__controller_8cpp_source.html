40void PresentationController::update(const dlz::Solution &solution, double rawRangeNm,
41 double rawRangeRateKnots, dlz::HudMode mode,
42 double dtSeconds) noexcept {
43 if (mode != dlz::HudMode::Prelaunch) {
44 clear();
45 m_state.mode = mode;
46 return;
47 }
48 if (!std::isfinite(rawRangeNm) || rawRangeNm < 0.0 || !std::isfinite(rawRangeRateKnots)) {
49 clear();
50 return;
51 }
52 const double dt = std::clamp(std::isfinite(dtSeconds) ? dtSeconds : 0.0, 0.0, 0.25);
53 m_elapsedSeconds += dt;
54 m_lastSolution = solution;
55 m_hasSolution = true;
56
57 const double decisionRange = std::max(rawRangeNm, solution.aerodynamicMaximumRangeNm);
58 if (!m_scale.initialized()) {
59 m_scale.reset(decisionRange, m_forcedScaleMaximumNm);
60 m_forcedScaleMaximumNm = 0.0;
61 } else {
62 m_scale.update(decisionRange);
63 }
64 m_state.scaleMinimumNm = 0.0;
65 m_state.scaleMaximumNm = m_scale.scaleMaximumNm();
66 m_state.currentRangeNm = rawRangeNm;
67 m_state.rangeRateKnots = rawRangeRateKnots;
68 m_state.caretParked = rawRangeNm > m_state.scaleMaximumNm;
69 m_state.mode = mode;
70 m_state.timeToImpactRemainingSeconds = std::max(0.0, solution.timeOfFlightSeconds);
71
72 if (!m_filterInitialized) {
73 m_displayedRangeNm = rawRangeNm;
74 m_filterInitialized = true;
75 } else {
76 const double gain = std::clamp(dt / FilterTimeConstantSeconds, 0.0, 1.0);
77 m_displayedRangeNm += (rawRangeNm - m_displayedRangeNm) * gain;
78 }
79 updateShootPresentation(solution, rawRangeNm, dt);
80}

82void PresentationController::updateShootPresentation(const dlz::Solution &solution,
83 double rawRangeNm, double dtSeconds) noexcept {
84 const bool entry = solution.shootCue &&
85 rawRangeNm > solution.minimumRangeNm + ShootEntryMinimumMarginNm &&
86 rawRangeNm < solution.interceptRangeNm - ShootEntryMaximumMarginNm;
87 const bool exit = rawRangeNm <= solution.minimumRangeNm - ShootExitMinimumMarginNm ||
88 rawRangeNm > solution.interceptRangeNm + ShootExitMaximumMarginNm;
89
90 if (!m_shootActive && entry) {
91 m_shootActive = true;
92 m_shootVisibleUntil = m_elapsedSeconds + ShootMinimumVisibleSeconds;
93 } else if (m_shootActive && exit && m_elapsedSeconds >= m_shootVisibleUntil) {
94 m_shootActive = false;
95 }
96 (void)dtSeconds;
97 const double phase = std::fmod(std::max(0.0, m_elapsedSeconds), ShootFlashPeriodSeconds);
98 m_state.shootFlash = m_shootActive && phase < ShootFlashPeriodSeconds * 0.5;
99}
