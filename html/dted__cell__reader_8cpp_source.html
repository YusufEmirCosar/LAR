22bool parseUnsigned(const QByteArray &bytes, qsizetype offset, qsizetype length, int *value) {
23 if (value == nullptr || offset < 0 || length <= 0 || offset + length > bytes.size()) {
24 return false;
25 }
26 int parsed = 0;
27 for (qsizetype index = 0; index < length; ++index) {
28 const char character = bytes.at(offset + index);
29 if (character < '0' || character > '9') {
30 return false;
31 }
32 if (parsed > (std::numeric_limits<int>::max() - 9) / 10) {
33 return false;
34 }
35 parsed = parsed * 10 + static_cast<int>(character - '0');
36 }
37 *value = parsed;
38 return true;
39}

41bool parseOrigin(const QByteArray &bytes, qsizetype offset, double *degrees) {
42 int wholeDegrees = 0;
43 int minutes = 0;
44 int seconds = 0;
45 if (degrees == nullptr || !parseUnsigned(bytes, offset, 3, &wholeDegrees) ||
46 !parseUnsigned(bytes, offset + 3, 2, &minutes) ||
47 !parseUnsigned(bytes, offset + 5, 2, &seconds) || minutes >= 60 || seconds >= 60) {
48 return false;
49 }
50 const char hemisphere = bytes.at(offset + 7);
51 if (hemisphere != 'E' && hemisphere != 'W' && hemisphere != 'N' && hemisphere != 'S') {
52 return false;
53 }
54 double result = static_cast<double>(wholeDegrees) + static_cast<double>(minutes) / 60.0 +
55 static_cast<double>(seconds) / 3600.0;
56 if (hemisphere == 'W' || hemisphere == 'S') {
57 result = -result;
58 }

71qint16 decodeSignedMagnitude(quint16 encoded) noexcept {
72 if (encoded == 0xFFFFU) {
73 return DtedCell::NoDataElevation;
74 }
75 if ((encoded & 0x8000U) == 0U) {
76 return static_cast<qint16>(encoded);
77 }
78 const int magnitude = static_cast<int>(encoded & 0x7FFFU);
79 const int signedMagnitude = -magnitude;
80 // A small number of producers wrote two's-complement negatives. Values below the physical
81 // range of Earth expose that variant unambiguously while preserving standard DTED data.
82 if (signedMagnitude < -12000) {
83 return static_cast<qint16>(encoded);
84 }
85 return static_cast<qint16>(signedMagnitude);
86}

126 !parseUnsigned(header, 24, 4, &latitudeIntervalTenths) ||

143 longitudeIntervalMultiplier == 6);

149 static_cast<double>(longitudeIntervalTenths) / 36000.0;

151 static_cast<double>(latitudeIntervalTenths) / 36000.0;
152 if (std::abs(longitudeSpanDegrees - 1.0) > 1.0e-9 ||
153 std::abs(latitudeSpanDegrees - 1.0) > 1.0e-9) {
154 return failure(path, expectedLevel,
155 QStringLiteral("sample intervals do not span exactly one degree"));
156 }

158 const double roundedLatitude = std::round(latitudeOrigin);
159 if (std::abs(longitudeOrigin - roundedLongitude) > 1.0e-9 ||
160 std::abs(latitudeOrigin - roundedLatitude) > 1.0e-9 || roundedLongitude < -180.0 ||
161 roundedLongitude > 179.0 || roundedLatitude < -90.0 || roundedLatitude > 89.0) {
162 return failure(path, expectedLevel,
163 QStringLiteral("cell origin is not a valid degree boundary"));
164 }

167 const qint64 expectedLength =
168 static_cast<qint64>(HeaderLength) + static_cast<qint64>(longitudeCount) * recordLength;
169 if (expectedLength != fileSize) {
170 return failure(path, expectedLevel,
171 QStringLiteral("profile records do not match declared dimensions"));
172 }
173

176 cell->level = expectedLevel;
177 cell->longitudeSampleCount = longitudeCount;
178 cell->latitudeSampleCount = latitudeCount;
179 cell->longitudeIntervalDegrees = static_cast<double>(longitudeIntervalTenths) / 36000.0;
180 cell->latitudeIntervalDegrees = static_cast<double>(latitudeIntervalTenths) / 36000.0;
181 cell->elevations.resize(static_cast<std::size_t>(longitudeCount) *
182 static_cast<std::size_t>(latitudeCount));

185 const QByteArray record = file.read(recordLength);
186 if (static_cast<qint64>(record.size()) != recordLength) {
187 return failure(path, expectedLevel, QStringLiteral("profile record is truncated"));
188 }
189 if (static_cast<uchar>(record.at(0)) != 0xAAU) {
190 return failure(path, expectedLevel, QStringLiteral("profile sentinel is missing"));
191 }

194 calculatedChecksum += static_cast<uchar>(record.at(index));

197 if (calculatedChecksum != storedChecksum) {

199 }

207 }
208 }
209 if (!cell->valid()) {
210 return failure(path, expectedLevel, QStringLiteral("decoded cell is inconsistent"));
211 }
212 return {std::move(cell), {}};
