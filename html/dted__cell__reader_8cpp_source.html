23bool parseUnsigned(const QByteArray &bytes, qsizetype offset, qsizetype length, int *value) {
24 if (value == nullptr || offset < 0 || length <= 0 || offset + length > bytes.size()) {
25 return false;
26 }
27 int parsed = 0;
28 for (qsizetype index = 0; index < length; ++index) {
29 const char character = bytes.at(offset + index);
30 if (character < '0' || character > '9') {
31 return false;
32 }
33 if (parsed > (std::numeric_limits<int>::max() - 9) / 10) {
34 return false;
35 }
36 parsed = parsed * 10 + static_cast<int>(character - '0');
37 }
38 *value = parsed;
39 return true;
40}

42bool parseOrigin(const QByteArray &bytes, qsizetype offset, double *degrees) {
43 int wholeDegrees = 0;
44 int minutes = 0;
45 int seconds = 0;
46 if (degrees == nullptr || !parseUnsigned(bytes, offset, 3, &wholeDegrees) ||
47 !parseUnsigned(bytes, offset + 3, 2, &minutes) ||
48 !parseUnsigned(bytes, offset + 5, 2, &seconds) || minutes >= 60 || seconds >= 60) {
49 return false;
50 }
51 const char hemisphere = bytes.at(offset + 7);
52 if (hemisphere != 'E' && hemisphere != 'W' && hemisphere != 'N' && hemisphere != 'S') {
53 return false;
54 }
55 double result = static_cast<double>(wholeDegrees) + static_cast<double>(minutes) / 60.0 +
56 static_cast<double>(seconds) / 3600.0;
57 if (hemisphere == 'W' || hemisphere == 'S') {
58 result = -result;
59 }
60 *degrees = result;
61 return true;
62}

72qint16 decodeSignedMagnitude(quint16 encoded) noexcept {
73 if (encoded == 0xFFFFU) {
74 return DtedCell::NoDataElevation;
75 }
76 if ((encoded & 0x8000U) == 0U) {
77 return static_cast<qint16>(encoded);
78 }
79 const int magnitude = static_cast<int>(encoded & 0x7FFFU);
80 const int signedMagnitude = -magnitude;
81 // A small number of producers wrote two's-complement negatives. Values below the physical
82 // range of Earth expose that variant unambiguously while preserving standard DTED data.
83 if (signedMagnitude < -12000) {
84 return static_cast<qint16>(encoded);
85 }
86 return static_cast<qint16>(signedMagnitude);
87}

92 QFile file(path);
93 if (!file.open(QIODevice::ReadOnly)) {
94 return {nullptr, QStringLiteral("DTED0 tile '%1' could not be opened: %2")
95 .arg(path, file.errorString())};
96 }
97 const qint64 fileSize = file.size();
98 if (fileSize < HeaderLength || fileSize > MaximumLevelZeroFileBytes) {
99 return failure(path, QStringLiteral("file size is outside Level-0 limits"));
100 }
101 const QByteArray bytes = file.readAll();
102 if (bytes.size() != fileSize || bytes.size() < HeaderLength) {
103 return failure(path, QStringLiteral("file could not be read completely"));
104 }
105 if (bytes.first(4) != QByteArrayLiteral("UHL1") ||
106 bytes.mid(UhlLength, 3) != QByteArrayLiteral("DSI") ||
107 bytes.mid(UhlLength + DsiLength, 3) != QByteArrayLiteral("ACC")) {
108 return failure(path, QStringLiteral("required UHL/DSI/ACC headers are missing"));
109 }
110
111 double longitudeOrigin = 0.0;
112 double latitudeOrigin = 0.0;
113 int longitudeIntervalTenths = 0;
114 int latitudeIntervalTenths = 0;
115 int longitudeCount = 0;
116 int latitudeCount = 0;
117 if (!parseOrigin(bytes, 4, &longitudeOrigin) || !parseOrigin(bytes, 12, &latitudeOrigin) ||
118 !parseUnsigned(bytes, 20, 4, &longitudeIntervalTenths) ||
119 !parseUnsigned(bytes, 24, 4, &latitudeIntervalTenths) ||
120 !parseUnsigned(bytes, 47, 4, &longitudeCount) ||
121 !parseUnsigned(bytes, 51, 4, &latitudeCount)) {
122 return failure(path, QStringLiteral("UHL coordinate or dimension fields are malformed"));
123 }
124 if (longitudeCount < 2 || longitudeCount > MaximumLevelZeroSamples || latitudeCount < 2 ||
125 latitudeCount > MaximumLevelZeroSamples || longitudeIntervalTenths <= 0 ||
126 latitudeIntervalTenths <= 0) {
127 return failure(path, QStringLiteral("declared dimensions exceed Level-0 limits"));
128 }
129 const double longitudeSpanDegrees = static_cast<double>(longitudeCount - 1) *
130 static_cast<double>(longitudeIntervalTenths) / 36000.0;
131 const double latitudeSpanDegrees = static_cast<double>(latitudeCount - 1) *
132 static_cast<double>(latitudeIntervalTenths) / 36000.0;
133 if (std::abs(longitudeSpanDegrees - 1.0) > 1.0e-9 ||
134 std::abs(latitudeSpanDegrees - 1.0) > 1.0e-9) {
135 return failure(path, QStringLiteral("sample intervals do not span exactly one degree"));
136 }
137 const double roundedLongitude = std::round(longitudeOrigin);
138 const double roundedLatitude = std::round(latitudeOrigin);
139 if (std::abs(longitudeOrigin - roundedLongitude) > 1.0e-9 ||
140 std::abs(latitudeOrigin - roundedLatitude) > 1.0e-9 || roundedLongitude < -180.0 ||
141 roundedLongitude > 179.0 || roundedLatitude < -90.0 || roundedLatitude > 89.0) {
142 return failure(path, QStringLiteral("cell origin is not a valid degree boundary"));
143 }
144
145 const qsizetype recordLength = 8 + static_cast<qsizetype>(latitudeCount) * 2 + 4;
146 const qsizetype expectedLength =
147 HeaderLength + static_cast<qsizetype>(longitudeCount) * recordLength;
148 if (expectedLength != bytes.size()) {
149 return failure(path, QStringLiteral("profile records do not match declared dimensions"));
150 }
151
152 auto cell = std::make_shared<DtedCell>();
153 cell->key = {static_cast<int>(roundedLongitude), static_cast<int>(roundedLatitude)};
154 cell->longitudeSampleCount = longitudeCount;
155 cell->latitudeSampleCount = latitudeCount;
156 cell->longitudeIntervalDegrees = static_cast<double>(longitudeIntervalTenths) / 36000.0;
157 cell->latitudeIntervalDegrees = static_cast<double>(latitudeIntervalTenths) / 36000.0;
158 cell->elevations.reserve(static_cast<std::size_t>(longitudeCount) *
159 static_cast<std::size_t>(latitudeCount));
160
161 for (int longitudeIndex = 0; longitudeIndex < longitudeCount; ++longitudeIndex) {
162 const qsizetype recordOffset =
163 HeaderLength + static_cast<qsizetype>(longitudeIndex) * recordLength;
164 if (static_cast<uchar>(bytes.at(recordOffset)) != 0xAAU) {
165 return failure(path, QStringLiteral("profile sentinel is missing"));
166 }
167 quint32 calculatedChecksum = 0U;
168 for (qsizetype index = 0; index < recordLength - 4; ++index) {
169 calculatedChecksum += static_cast<uchar>(bytes.at(recordOffset + index));
170 }
171 const quint32 storedChecksum =
172 bigEndianU32(bytes.constData() + recordOffset + recordLength - 4);
173 if (calculatedChecksum != storedChecksum) {
174 return failure(path, QStringLiteral("profile checksum mismatch"));
175 }
176 const qsizetype sampleOffset = recordOffset + 8;
177 for (int latitudeIndex = 0; latitudeIndex < latitudeCount; ++latitudeIndex) {
178 const quint16 encoded =
179 bigEndianU16(bytes.constData() + sampleOffset + latitudeIndex * 2);
180 cell->elevations.push_back(decodeSignedMagnitude(encoded));
181 }
182 }
183 if (!cell->valid()) {
184 return failure(path, QStringLiteral("decoded cell is inconsistent"));
185 }
186 return {std::move(cell), {}};
187}
