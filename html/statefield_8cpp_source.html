13template <int Id>
14double readField(const Plane &plane, const Target &target,
15 const dlz::TelemetryInputs &dlzInputs) noexcept {
16 if constexpr (Id >= StateField::Location0 && Id <= StateField::Location2) {
17 return plane.location[Id - StateField::Location0];
18 } else if constexpr (Id >= StateField::Euler0 && Id <= StateField::Euler2) {
19 return plane.euler[Id - StateField::Euler0];
20 } else if constexpr (Id >= StateField::Velocity0 && Id <= StateField::Velocity2) {
21 return plane.velocity[Id - StateField::Velocity0];
22 } else if constexpr (Id >= StateField::IzPos0 && Id <= StateField::IzPos2) {
23 return target.iz_pos[Id - StateField::IzPos0];
24 } else if constexpr (Id >= StateField::IrPos0 && Id <= StateField::IrPos2) {
25 return target.ir_pos[Id - StateField::IrPos0];
26 } else if constexpr (Id == StateField::IzTheta1) {
27 return target.iz_theta1;
28 } else if constexpr (Id == StateField::IzTheta2) {
29 return target.iz_theta2;
30 } else if constexpr (Id == StateField::IzR1) {
31 return target.iz_r1;
32 } else if constexpr (Id == StateField::IzR2) {
33 return target.iz_r2;
34 } else if constexpr (Id == StateField::IrR) {
35 return target.ir_r;
36 } else if constexpr (Id == StateField::Time) {
37 return target.time;
38 } else if constexpr (Id == StateField::DlzRangeNm) {
39 return dlzInputs.rangeNm;
40 } else if constexpr (Id == StateField::DlzAspectDegrees) {
41 return dlzInputs.aspectDegrees;
42 } else {
43 static_assert(Id == StateField::DlzAltitudeFeet, "Unhandled StateField getter");
44 return dlzInputs.altitudeFeet;

141constexpr bool hasUniqueMappingKeys() {
142 for (std::size_t left = 0; left < Descriptors.size(); ++left) {

165int StateField::resolve(const QString &name, int index) {
166 if (index < 0)
167 return -1;
168 for (std::size_t id = 0; id < Descriptors.size(); ++id) {
169 const Descriptor &candidate = Descriptors[id];
170 if (candidate.mappingIndex == index && name == QLatin1String(candidate.mappingName)) {
171 return int(id);
172 }
173 }
174 return -1;
175}
