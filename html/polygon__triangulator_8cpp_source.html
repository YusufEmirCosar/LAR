22SourceRing prepareRing(const SourceRing &source, double referenceLongitude) {
23 SourceRing ring;
24 ring.reserve(source.size());
25 double previousLongitude = referenceLongitude;
26 // Unwrap relative to the preceding vertex so an antimeridian edge remains
27 // short in the planar domain. The final whole-world shift keeps every ring
28 // near the exterior's reference without changing its shape.
29 for (const SourceCoordinate &coordinate : source) {
30 SourceCoordinate unwrapped{lar::map::MapProjection::unwrapLongitude(
31 coordinate.longitudeDegrees, previousLongitude),
32 coordinate.latitudeDegrees};
33 previousLongitude = unwrapped.longitudeDegrees;
34 if (ring.empty() || !samePoint(ring.back(), unwrapped)) {
35 ring.push_back(unwrapped);
36 }
37 }
38 if (ring.size() > 1U && samePoint(ring.front(), ring.back())) {
39 ring.pop_back();
40 }
41 if (ring.size() < 3U) {
42 return {};
43 }
44
45 const double shift =
46 360.0 * std::round((referenceLongitude - ring.front().longitudeDegrees) / 360.0);
47 for (SourceCoordinate &coordinate : ring) {
48 coordinate.longitudeDegrees += shift;
49 }
50 return ring;
51}

53double signedArea(const SourceRing &ring) {
54 double twiceArea = 0.0;
55 for (std::size_t index = 0U; index < ring.size(); ++index) {
56 const SourceCoordinate &current = ring[index];
57 const SourceCoordinate &next = ring[(index + 1U) % ring.size()];
58 twiceArea += current.longitudeDegrees * next.latitudeDegrees -
59 next.longitudeDegrees * current.latitudeDegrees;
60 }
61 return twiceArea * 0.5;
62}

115RingNode *removeRedundantNodes(RingNode *start) {
116 if (start == nullptr) {
117 return nullptr;
118 }
119 RingNode *node = start;
120 bool changed = false;
121 do {
122 changed = false;
123 RingNode *next = node->next;
124 const bool duplicate = std::abs(node->x - next->x) <= CoordinateTolerance &&
125 std::abs(node->y - next->y) <= CoordinateTolerance;
126 const bool collinear =
127 std::abs(cross(*node->previous, *node, *node->next)) <= AreaTolerance;
128 if (node != node->next && node->previous != node->next && (duplicate || collinear)) {
129 if (node == start) {
130 start = next;
131 }
132 detach(*node);
133 node = next;
134 changed = true;
135 } else {
136 node = next;
137 }
138 } while (changed || node != start);
139 return start;
140}

142bool pointInsideTriangle(const RingNode &a, const RingNode &b, const RingNode &c,
143 const RingNode &point) {
144 const double ab = cross(a, b, point);
145 const double bc = cross(b, c, point);
146 const double ca = cross(c, a, point);
147 return ab >= -AreaTolerance && bc >= -AreaTolerance && ca >= -AreaTolerance;
148}

150bool isEar(const RingNode &candidate) {
151 const RingNode &a = *candidate.previous;
152 const RingNode &b = candidate;
153 const RingNode &c = *candidate.next;
154 if (cross(a, b, c) <= AreaTolerance) {
155 return false;
156 }
157
158 const RingNode *point = candidate.next->next;
159 while (point != candidate.previous) {
160 if (pointInsideTriangle(a, b, c, *point) &&
161 cross(*point->previous, *point, *point->next) <= AreaTolerance) {
162 return false;
163 }
164 point = point->next;
165 }
166 return true;
167}

169int orientation(double ax, double ay, double bx, double by, double cx, double cy) {
170 const double value = (bx - ax) * (cy - ay) - (by - ay) * (cx - ax);
171 if (std::abs(value) <= AreaTolerance) {
172 return 0;
173 }
174 return value > 0.0 ? 1 : -1;

177bool segmentsIntersect(const RingNode &a, const RingNode &b, const RingNode &c, const RingNode &d) {
178 const int first = orientation(a.x, a.y, b.x, b.y, c.x, c.y);
179 const int second = orientation(a.x, a.y, b.x, b.y, d.x, d.y);
180 const int third = orientation(c.x, c.y, d.x, d.y, a.x, a.y);
181 const int fourth = orientation(c.x, c.y, d.x, d.y, b.x, b.y);
182 return first != 0 && second != 0 && third != 0 && fourth != 0 && first != second &&
183 third != fourth;

186bool bridgeCrossesRing(const RingNode &first, const RingNode &second, const RingNode &ring) {
187 const RingNode *edge = &ring;
188 do {
189 const RingNode *next = edge->next;
190 const bool sharesEndpoint =
191 edge == &first || edge == &second || next == &first || next == &second;
192 if (!sharesEndpoint && segmentsIntersect(first, second, *edge, *next)) {
193 return true;
194 }
195 edge = next;
196 } while (edge != &ring);
197 return false;
198}

200bool pointInsideRing(double x, double y, const RingNode &ring) {
201 bool inside = false;
202 const RingNode *first = &ring;
203 do {
204 const RingNode *second = first->next;
205 if ((first->y > y) != (second->y > y)) {
206 const double intersectionX =
207 first->x + (second->x - first->x) * (y - first->y) / (second->y - first->y);
208 if (x < intersectionX) {
209 inside = !inside;
210 }
211 }
212 first = second;
213 } while (first != &ring);
214 return inside;
215}

229RingNode *joinHole(RingNode &outer, RingNode &hole, NodePool &pool) {
230 // The nearest bridge is used only when it crosses neither boundary and its
231 // midpoint lies in the polygon but outside the hole. Duplicating both bridge
232 // endpoints turns the two rings into one walk without adding new vertices.
233 RingNode *holeAnchor = leftmost(hole);
234 RingNode *outerAnchor = nullptr;
235 double bestDistance = std::numeric_limits<double>::max();
236 RingNode *candidate = &outer;
237 do {
238 const double dx = candidate->x - holeAnchor->x;
239 const double dy = candidate->y - holeAnchor->y;
240 const double distance = dx * dx + dy * dy;
241 const double midpointX = (candidate->x + holeAnchor->x) * 0.5;
242 const double midpointY = (candidate->y + holeAnchor->y) * 0.5;
243 if (distance < bestDistance && !bridgeCrossesRing(*candidate, *holeAnchor, outer) &&
244 !bridgeCrossesRing(*candidate, *holeAnchor, hole) &&
245 pointInsideRing(midpointX, midpointY, outer) &&
246 !pointInsideRing(midpointX, midpointY, hole)) {
247 outerAnchor = candidate;
248 bestDistance = distance;
249 }
250 candidate = candidate->next;
251 } while (candidate != &outer);
252 if (outerAnchor == nullptr) {
253 return nullptr;
254 }
255
256 RingNode *outerAfter = outerAnchor->next;
257 RingNode *holeBefore = holeAnchor->previous;
258 RingNode *outerCopy = pool.duplicate(*outerAnchor);
259 RingNode *holeCopy = pool.duplicate(*holeAnchor);
260
261 outerAnchor->next = holeAnchor;
262 holeAnchor->previous = outerAnchor;
263 holeBefore->next = holeCopy;
264 holeCopy->previous = holeBefore;
265 holeCopy->next = outerCopy;
266 outerCopy->previous = holeCopy;
267 outerCopy->next = outerAfter;
268 outerAfter->previous = outerCopy;
269 return outerAnchor;
270}

272RingNode *makeRing(const SourceRing &ring, std::uint32_t firstVertex, bool counterClockwise,
273 NodePool &pool) {
274 const bool reverse = (signedArea(ring) > 0.0) != counterClockwise;
275 RingNode *last = nullptr;
276 RingNode *first = nullptr;
277 for (std::size_t offset = 0U; offset < ring.size(); ++offset) {
278 const std::size_t sourceIndex = reverse ? ring.size() - 1U - offset : offset;
279 last = pool.append(last, firstVertex + static_cast<std::uint32_t>(sourceIndex),
280 ring[sourceIndex]);
281 if (first == nullptr) {
282 first = last;
283 }
284 }
285 return first;

305bool splitAndTriangulate(RingNode &start, std::vector<std::uint32_t> &indices, NodePool &pool,
306 int depth) {
307 if (depth >= 16) {
308 return false;
309 }
310 std::vector<RingNode *> nodes;
311 RingNode *node = &start;
312 do {
313 nodes.push_back(node);
314 node = node->next;
315 } while (node != &start);
316 if (nodes.size() < 4U) {
317 return false;
318 }
319
320 for (std::size_t first = 0U; first + 2U < nodes.size(); ++first) {
321 for (std::size_t second = first + 2U; second < nodes.size(); ++second) {
322 if (first == 0U && second + 1U == nodes.size()) {
323 continue;
324 }
325 RingNode &a = *nodes[first];
326 RingNode &b = *nodes[second];
327 const double midpointX = (a.x + b.x) * 0.5;
328 const double midpointY = (a.y + b.y) * 0.5;
329 if (samePoint({a.x, a.y}, {b.x, b.y}) || bridgeCrossesRing(a, b, start) ||
330 !pointInsideRing(midpointX, midpointY, start)) {
331 continue;
332 }
333
334 RingNode *firstRing = clonePath(nodes, first, second, pool);
335 std::vector<RingNode *> wrapped;
336 wrapped.reserve(nodes.size() - second + first + 1U);
337 wrapped.insert(wrapped.end(), nodes.begin() + static_cast<std::ptrdiff_t>(second),
338 nodes.end());
339 wrapped.insert(wrapped.end(), nodes.begin(),
340 nodes.begin() + static_cast<std::ptrdiff_t>(first + 1U));
341 RingNode *secondRing = clonePath(wrapped, 0U, wrapped.size() - 1U, pool);
342 return triangulateLinkedRing(firstRing, indices, pool, depth + 1) &&
343 triangulateLinkedRing(secondRing, indices, pool, depth + 1);
344 }
345 }
346 return false;
347}

349bool triangulateLinkedRing(RingNode *start, std::vector<std::uint32_t> &indices, NodePool &pool,
350 int depth) {
351 start = removeRedundantNodes(start);
352 if (start == nullptr) {
353 return false;
354 }
355 RingNode *candidate = start;
356 RingNode *stop = start;
357 std::size_t stalledPasses = 0U;
358 while (candidate->previous != candidate->next) {
359 RingNode *next = candidate->next;
360 if (isEar(*candidate)) {
361 indices.push_back(candidate->previous->vertex);
362 indices.push_back(candidate->vertex);
363 indices.push_back(candidate->next->vertex);
364 detach(*candidate);
365 candidate = next->next;
366 stop = candidate;
367 stalledPasses = 0U;
368 continue;
369 }
370 candidate = next;
371 if (candidate == stop) {
372 candidate = removeRedundantNodes(candidate);
373 stop = candidate;
374 if (++stalledPasses > 1U) {
375 // Ear clipping can stall on numerically awkward concavity. A
376 // valid interior diagonal decomposes the ring into smaller
377 // problems; recursion is capped by splitAndTriangulate().
378 return candidate != nullptr &&
379 splitAndTriangulate(*candidate, indices, pool, depth);
380 }
381 }
382 }
383 return true;
384}

388PlanarPolygonMesh PolygonTriangulator::triangulate(const SourcePolygon &polygon) {
389 PlanarPolygonMesh mesh;
390 if (polygon.exterior.empty()) {
391 return mesh;
392 }
393 const double referenceLongitude = polygon.exterior.front().longitudeDegrees;
394 SourceRing exterior = prepareRing(polygon.exterior, referenceLongitude);
395 if (exterior.size() < 3U) {
396 return mesh;
397 }
398
399 std::vector<SourceRing> holes;
400 holes.reserve(polygon.holes.size());
401 std::size_t coordinateCount = exterior.size();
402 for (const SourceRing &sourceHole : polygon.holes) {
403 SourceRing hole = prepareRing(sourceHole, referenceLongitude);
404 if (hole.size() >= 3U) {
405 coordinateCount += hole.size();
406 holes.push_back(std::move(hole));
407 }
408 }
409 if (coordinateCount > static_cast<std::size_t>(std::numeric_limits<std::uint32_t>::max())) {
410 return {};
411 }
412
413 mesh.coordinates.reserve(coordinateCount);
414 mesh.borderIndices.reserve(coordinateCount * 2U);
415 const auto appendCoordinates = [&mesh](const SourceRing &ring) {
416 const std::uint32_t first = static_cast<std::uint32_t>(mesh.coordinates.size());
417 mesh.coordinates.insert(mesh.coordinates.end(), ring.begin(), ring.end());
418 for (std::size_t index = 0U; index < ring.size(); ++index) {
419 mesh.borderIndices.push_back(first + static_cast<std::uint32_t>(index));
420 mesh.borderIndices.push_back(first +
421 static_cast<std::uint32_t>((index + 1U) % ring.size()));
422 }
423 return first;
424 };
425
426 NodePool pool;
427 const std::uint32_t exteriorFirst = appendCoordinates(exterior);
428 RingNode *outer = makeRing(exterior, exteriorFirst, true, pool);
429 struct PendingHole final {
430 RingNode *ring = nullptr;
431 double leftmostX = 0.0;
432 };
433 std::vector<PendingHole> pendingHoles;
434 pendingHoles.reserve(holes.size());
435 for (const SourceRing &hole : holes) {
436 const std::uint32_t holeFirst = appendCoordinates(hole);
437 RingNode *holeRing = makeRing(hole, holeFirst, false, pool);
438 pendingHoles.push_back({holeRing, leftmost(*holeRing)->x});
439 }
440 std::sort(pendingHoles.begin(), pendingHoles.end(),
441 [](const PendingHole &first, const PendingHole &second) {
442 return first.leftmostX < second.leftmostX;
443 });
444 for (const PendingHole &hole : pendingHoles) {
445 outer = joinHole(*outer, *hole.ring, pool);
446 if (outer == nullptr) {
447 return {};
448 }
449 }
450
451 mesh.fillIndices.reserve(mesh.coordinates.size() >= 3U ? (mesh.coordinates.size() - 2U) * 3U
452 : 0U);
453 if (!triangulateLinkedRing(outer, mesh.fillIndices, pool) || mesh.fillIndices.empty()) {
454 return {};
455 }
456 return mesh;
457}
