32bool appendRing(const QJsonArray &coordinates, SourceMap &map, SourceRing &destination,
33 SourceMapError &error) {
34 if (coordinates.size() < 4) {
35 error = SourceMapError::InvalidSchema;
36 return false;
37 }
38 if (static_cast<std::size_t>(coordinates.size()) > source_limits::MaximumCoordinatesPerRing ||
39 map.coordinateCount >
40 source_limits::MaximumCoordinateCount - static_cast<std::size_t>(coordinates.size())) {
41 error = SourceMapError::ResourceLimit;
42 return false;
43 }
44
45 destination.reserve(static_cast<std::size_t>(coordinates.size()));
46 for (const QJsonValue &coordinateValue : coordinates) {
47 if (!coordinateValue.isArray()) {
48 error = SourceMapError::InvalidCoordinate;
49 return false;
50 }
51 const QJsonArray coordinate = coordinateValue.toArray();
52 if (coordinate.size() < 2 || !coordinate.at(0).isDouble() || !coordinate.at(1).isDouble()) {
53 error = SourceMapError::InvalidCoordinate;
54 return false;
55 }
56 const double longitude = coordinate.at(0).toDouble();
57 const double latitude = coordinate.at(1).toDouble();
58 if (!std::isfinite(longitude) || !std::isfinite(latitude) || longitude < -180.0 ||
59 longitude > 180.0 || latitude < -90.0 || latitude > 90.0) {
60 error = SourceMapError::InvalidCoordinate;
61 return false;
62 }
63 destination.push_back({longitude, latitude});
64 }
65 if (destination.front().longitudeDegrees != destination.back().longitudeDegrees ||
66 destination.front().latitudeDegrees != destination.back().latitudeDegrees) {
67 error = SourceMapError::InvalidSchema;
68 return false;
69 }
70 map.coordinateCount += destination.size();
71 return true;
72}

74bool appendPolygon(const QJsonArray &coordinates, SourceMap &map, std::size_t &ringCount,
75 SourceMapError &error) {
76 if (coordinates.isEmpty() || map.polygons.size() >= source_limits::MaximumPolygonCount) {
77 error =
78 coordinates.isEmpty() ? SourceMapError::InvalidSchema : SourceMapError::ResourceLimit;
79 return false;
80 }
81
82 SourcePolygon polygon;
83 polygon.holes.reserve(static_cast<std::size_t>(std::max<qsizetype>(coordinates.size() - 1, 0)));
84 for (qsizetype index = 0; index < coordinates.size(); ++index) {
85 if (!checkedIncrement(ringCount, source_limits::MaximumRingCount)) {
86 error = SourceMapError::ResourceLimit;
87 return false;
88 }
89 if (!coordinates.at(index).isArray()) {
90 error = SourceMapError::InvalidSchema;
91 return false;
92 }
93 SourceRing ring;
94 if (!appendRing(coordinates.at(index).toArray(), map, ring, error)) {
95 return false;
96 }
97 if (index == 0) {
98 polygon.exterior = std::move(ring);
99 } else {
100 polygon.holes.push_back(std::move(ring));
101 }
102 }
103 map.polygons.push_back(std::move(polygon));
104 return true;
105}

124SourceMapReadResult GeoJsonSourceReader::read(const QString &path) {
125 QFile file(path);
126 if (!file.open(QIODevice::ReadOnly)) {
127 return failure(SourceMapError::OpenFailed,
128 QStringLiteral("The map source could not be opened."));
129 }
130 if (file.size() <= 0 ||
131 static_cast<std::uint64_t>(file.size()) > source_limits::MaximumSourceBytes) {
132 return failure(SourceMapError::SizeRejected,
133 QStringLiteral("The map source file size is outside the allowed range."));
134 }
135 const qint64 expectedSize = file.size();
136 const QByteArray bytes = file.readAll();
137 if (bytes.size() != expectedSize) {
138 return failure(SourceMapError::ReadFailed,
139 QStringLiteral("The complete map source could not be read."));
140 }
141 return parse(bytes);
142}

144SourceMapReadResult GeoJsonSourceReader::parse(const QByteArray &bytes) {
145 if (bytes.isEmpty() ||
146 static_cast<std::size_t>(bytes.size()) > source_limits::MaximumSourceBytes) {
147 return failure(SourceMapError::SizeRejected,
148 QStringLiteral("The map source byte count is outside the allowed range."));
149 }
150
151 QJsonParseError parseError;
152 const QJsonDocument document = QJsonDocument::fromJson(bytes, &parseError);
153 if (parseError.error != QJsonParseError::NoError || !document.isObject()) {
154 return failure(SourceMapError::InvalidJson,
155 QStringLiteral("The map source is not valid JSON."));
156 }
157
158 const QJsonObject root = document.object();
159 if (root.value(QStringLiteral("type")).toString() != QStringLiteral("FeatureCollection") ||
160 !root.value(QStringLiteral("features")).isArray()) {
161 return failure(SourceMapError::InvalidSchema, messageFor(SourceMapError::InvalidSchema));
162 }
163 const QJsonArray features = root.value(QStringLiteral("features")).toArray();
164 if (features.isEmpty() ||
165 static_cast<std::size_t>(features.size()) > source_limits::MaximumFeatureCount) {
166 return failure(SourceMapError::ResourceLimit, messageFor(SourceMapError::ResourceLimit));
167 }
168
169 SourceMap map;
170 std::size_t ringCount = 0U;
171 SourceMapError error = SourceMapError::None;
172 for (const QJsonValue &featureValue : features) {
173 if (!featureValue.isObject()) {
174 return failure(SourceMapError::InvalidSchema,
175 messageFor(SourceMapError::InvalidSchema));
176 }
177 const QJsonObject feature = featureValue.toObject();
178 if (feature.value(QStringLiteral("type")).toString() != QStringLiteral("Feature") ||
179 !feature.value(QStringLiteral("geometry")).isObject()) {
180 return failure(SourceMapError::InvalidSchema,
181 messageFor(SourceMapError::InvalidSchema));
182 }
183 const QJsonObject geometry = feature.value(QStringLiteral("geometry")).toObject();
184 const QString type = geometry.value(QStringLiteral("type")).toString();
185 if (!geometry.value(QStringLiteral("coordinates")).isArray()) {
186 return failure(SourceMapError::InvalidSchema,
187 messageFor(SourceMapError::InvalidSchema));
188 }
189 const QJsonArray coordinates = geometry.value(QStringLiteral("coordinates")).toArray();
190 if (type == QStringLiteral("Polygon")) {
191 if (!appendPolygon(coordinates, map, ringCount, error)) {
192 return failure(error, messageFor(error));
193 }
194 } else if (type == QStringLiteral("MultiPolygon")) {
195 if (coordinates.isEmpty()) {
196 return failure(SourceMapError::InvalidSchema,
197 messageFor(SourceMapError::InvalidSchema));
198 }
199 for (const QJsonValue &polygonValue : coordinates) {
200 if (!polygonValue.isArray() ||
201 !appendPolygon(polygonValue.toArray(), map, ringCount, error)) {
202 if (error == SourceMapError::None) {
203 error = SourceMapError::InvalidSchema;
204 }
205 return failure(error, messageFor(error));
206 }
207 }
208 } else {
209 return failure(SourceMapError::UnsupportedGeometry,
210 messageFor(SourceMapError::UnsupportedGeometry));
211 }
212 }
213
214 if (map.empty()) {
215 return failure(SourceMapError::InvalidSchema,
216 QStringLiteral("The map source contains no renderable polygons."));
217 }
218 return {std::move(map), SourceMapError::None, {}};
219}
