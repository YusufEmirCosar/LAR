43template <typename Value>
44void appendArray(QByteArray &destination, const std::vector<Value> &values) {
45#if Q_BYTE_ORDER == Q_LITTLE_ENDIAN
46 if (!values.empty()) {
47 destination.append(reinterpret_cast<const char *>(values.data()),
48 static_cast<qsizetype>(values.size() * sizeof(Value)));
49 }
50#else
51 for (const Value value : values) {
52 if constexpr (std::is_same_v<Value, float>) {
53 std::uint32_t bits = 0U;
54 std::memcpy(&bits, &value, sizeof(bits));
55 appendUint32(destination, bits);
56 } else {
57 appendUint32(destination, value);
58 }
59 }
60#endif
61}

63bool validate(const MapMesh &mesh) {
64 if (mesh.empty() || mesh.vertices.size() % 3U != 0U ||
65 mesh.vertexCount() > limits::MaximumVertexCount || mesh.mercatorFillIndices.empty() ||
66 mesh.mercatorFillIndices.size() > limits::MaximumMercatorIndexCount ||
67 mesh.mercatorFillIndices.size() % 3U != 0U || mesh.sphereFillIndices.empty() ||
68 mesh.sphereFillIndices.size() > limits::MaximumSphereIndexCount ||
69 mesh.sphereFillIndices.size() % 3U != 0U ||
70 mesh.borderIndices.size() > limits::MaximumBorderIndexCount ||
71 mesh.borderIndices.size() % 2U != 0U || mesh.byteSize() > limits::MaximumPayloadBytes ||
72 mesh.byteSize() > static_cast<std::size_t>(std::numeric_limits<qsizetype>::max()) ||
73 mesh.landTriangleReferences.size() > limits::MaximumLandTriangleReferenceCount ||
74 !mapLandIndexIsValid(mesh)) {
75 return false;
76 }
77
78 for (std::size_t offset = 0U; offset + 2U < mesh.vertices.size(); offset += 3U) {
79 if (!std::isfinite(mesh.vertices[offset]) || !std::isfinite(mesh.vertices[offset + 1U]) ||
80 !std::isfinite(mesh.vertices[offset + 2U]) ||
81 std::abs(mesh.vertices[offset]) > limits::MaximumAbsoluteLongitude ||
82 std::abs(mesh.vertices[offset + 1U]) > limits::MaximumAbsoluteLatitude ||
83 std::abs(mesh.vertices[offset + 2U]) > limits::MaximumAbsoluteMercatorY) {
84 return false;
85 }
86 }
87
88 const auto indicesAreValid =
89 [vertexCount = mesh.vertexCount()](const std::vector<std::uint32_t> &indices) {
90 for (const std::uint32_t index : indices) {
91 if (static_cast<std::size_t>(index) >= vertexCount) {
92 return false;
93 }
94 }
95 return true;
96 };
97 return indicesAreValid(mesh.mercatorFillIndices) && indicesAreValid(mesh.sphereFillIndices) &&
98 indicesAreValid(mesh.borderIndices);
99}

115bool capturePreviousFile(const QString &path, PreviousFile *previous, QString *errorMessage) {
116 const QFileInfo info(path);
117 if (!info.exists()) {
118 *previous = {};
119 return true;
120 }
121 if (!info.isFile() || info.isSymLink() || info.size() < 0 ||
122 info.size() > limits::MaximumAssetBytes) {
123 setError(errorMessage,
124 QStringLiteral("The previous map asset cannot be backed up safely."));
125 return false;
126 }
127 QFile file(path);
128 if (!file.open(QIODevice::ReadOnly)) {
129 setError(errorMessage, file.errorString());
130 return false;
131 }
132 const QByteArray bytes = file.read(limits::MaximumAssetBytes + 1);
133 if (bytes.size() != info.size()) {
134 setError(errorMessage,
135 QStringLiteral("The previous map asset changed while it was backed up."));
136 return false;
137 }
138 *previous = {true, bytes};
139 return true;
140}

142bool restorePreviousFile(const QString &path, const PreviousFile &previous, QString *errorMessage) {
143 if (previous.existed) {
144 return writeSingleFile(path, previous.bytes, errorMessage);
145 }
146 if (!QFileInfo::exists(path) || QFile::remove(path)) {
147 return true;
148 }
149 setError(errorMessage, QStringLiteral("The newly committed map asset could not be removed."));
150 return false;

153bool writeFilePair(const QString &assetPath, const QByteArray &assetBytes,
154 const QString &manifestPath, const QByteArray &manifestBytes,
155 QString *errorMessage) {
156 if (QFileInfo(assetPath).absoluteFilePath() == QFileInfo(manifestPath).absoluteFilePath()) {
157 setError(errorMessage, QStringLiteral("Map asset and manifest paths must be different."));
158 return false;
159 }
160 PreviousFile previousAsset;
161 if (!capturePreviousFile(assetPath, &previousAsset, errorMessage)) {
162 return false;
163 }
164
165 QSaveFile assetFile(assetPath);
166 QSaveFile manifestFile(manifestPath);
167 if (!assetFile.open(QIODevice::WriteOnly) || assetFile.write(assetBytes) != assetBytes.size()) {
168 setError(errorMessage, assetFile.errorString());
169 return false;
170 }
171 if (!manifestFile.open(QIODevice::WriteOnly) ||
172 manifestFile.write(manifestBytes) != manifestBytes.size()) {
173 setError(errorMessage, manifestFile.errorString());
174 return false;
175 }
176 if (!assetFile.commit()) {
177 setError(errorMessage, assetFile.errorString());
178 return false;
179 }
180 if (manifestFile.commit()) {
181 return true;
182 }
183
184 const QString commitError = manifestFile.errorString();
185 QString rollbackError;
186 if (!restorePreviousFile(assetPath, previousAsset, &rollbackError)) {
187 setError(errorMessage,
188 QStringLiteral("Manifest commit failed (%1) and asset rollback failed (%2).")
189 .arg(commitError, rollbackError));
190 } else {
191 setError(errorMessage,
192 QStringLiteral("Manifest commit failed; the previous asset was restored: %1")
193 .arg(commitError));
194 }
195 return false;
196}

200bool MapAssetWriter::write(const QString &assetPath, const QString &manifestPath,
201 const MapMesh &mesh, QString *errorMessage) {
202 if (!validate(mesh)) {
203 setError(errorMessage, QStringLiteral("The map mesh is invalid or exceeds asset limits."));
204 return false;
205 }
206
207 QByteArray payload;
208 payload.reserve(static_cast<qsizetype>(mesh.byteSize()));
209 appendArray(payload, mesh.vertices);
210 appendArray(payload, mesh.mercatorFillIndices);
211 appendArray(payload, mesh.sphereFillIndices);
212 appendArray(payload, mesh.borderIndices);
213 for (const MapLandCellRange &range : mesh.landCellRanges) {
214 appendUint32(payload, range.firstReference);
215 appendUint32(payload, range.referenceCount);
216 }
217 appendArray(payload, mesh.landTriangleReferences);
218
219 QByteArray asset;
220 asset.reserve(static_cast<qsizetype>(format::HeaderSize) + payload.size());
221 asset.append(format::Magic.data(), format::Magic.size());
222 appendUint32(asset, format::Version);
223 appendUint32(asset, format::HeaderSize);
224 appendUint32(asset, format::SupportedFlags);
225 appendUint64(asset, mesh.vertexCount());
226 appendUint64(asset, mesh.mercatorFillIndices.size());
227 appendUint64(asset, mesh.sphereFillIndices.size());
228 appendUint64(asset, mesh.borderIndices.size());
229 appendUint64(asset, mesh.landCellRanges.size());
230 appendUint64(asset, mesh.landTriangleReferences.size());
231 appendUint64(asset, mesh.byteSize());
232 appendUint32(asset,
233 MapChecksum::crc32(reinterpret_cast<const unsigned char *>(payload.constData()),
234 static_cast<std::size_t>(payload.size())));
235 appendUint32(asset, 0U);
236 if (asset.size() != static_cast<qsizetype>(format::HeaderSize)) {
237 setError(errorMessage, QStringLiteral("The map asset header could not be generated."));
238 return false;
239 }
240 asset.append(payload);
241
242 const QJsonObject manifest{
243 {QStringLiteral("formatVersion"), static_cast<int>(format::Version)},
244 {QStringLiteral("bytes"), static_cast<qint64>(asset.size())},
245 {QStringLiteral("sha256"),
246 QString::fromLatin1(QCryptographicHash::hash(asset, QCryptographicHash::Sha256).toHex())}};
247 const QByteArray manifestBytes = QJsonDocument(manifest).toJson(QJsonDocument::Indented);
248 return writeFilePair(assetPath, asset, manifestPath, manifestBytes, errorMessage);
249}
