42template <typename Value>
43void appendArray(QByteArray &destination, const std::vector<Value> &values) {
44#if Q_BYTE_ORDER == Q_LITTLE_ENDIAN
45 if (!values.empty()) {
46 destination.append(reinterpret_cast<const char *>(values.data()),
47 static_cast<qsizetype>(values.size() * sizeof(Value)));
48 }
49#else
50 for (const Value value : values) {
51 if constexpr (std::is_same_v<Value, float>) {
52 std::uint32_t bits = 0U;
53 std::memcpy(&bits, &value, sizeof(bits));
54 appendUint32(destination, bits);
55 } else {
56 appendUint32(destination, value);
57 }
58 }
59#endif
60}

62bool validate(const MapMesh &mesh) {
63 if (mesh.empty() || mesh.vertices.size() % 3U != 0U ||
64 mesh.vertexCount() > limits::MaximumVertexCount || mesh.mercatorFillIndices.empty() ||
65 mesh.mercatorFillIndices.size() > limits::MaximumMercatorIndexCount ||
66 mesh.mercatorFillIndices.size() % 3U != 0U || mesh.sphereFillIndices.empty() ||
67 mesh.sphereFillIndices.size() > limits::MaximumSphereIndexCount ||
68 mesh.sphereFillIndices.size() % 3U != 0U ||
69 mesh.borderIndices.size() > limits::MaximumBorderIndexCount ||
70 mesh.borderIndices.size() % 2U != 0U || mesh.byteSize() > limits::MaximumPayloadBytes ||
71 mesh.byteSize() > static_cast<std::size_t>(std::numeric_limits<qsizetype>::max())) {
72 return false;
73 }
74
75 for (std::size_t offset = 0U; offset + 2U < mesh.vertices.size(); offset += 3U) {
76 if (!std::isfinite(mesh.vertices[offset]) || !std::isfinite(mesh.vertices[offset + 1U]) ||
77 !std::isfinite(mesh.vertices[offset + 2U]) ||
78 std::abs(mesh.vertices[offset]) > limits::MaximumAbsoluteLongitude ||
79 std::abs(mesh.vertices[offset + 1U]) > limits::MaximumAbsoluteLatitude ||
80 std::abs(mesh.vertices[offset + 2U]) > limits::MaximumAbsoluteMercatorY) {
81 return false;
82 }
83 }
84
85 const auto indicesAreValid =
86 [vertexCount = mesh.vertexCount()](const std::vector<std::uint32_t> &indices) {
87 for (const std::uint32_t index : indices) {
88 if (static_cast<std::size_t>(index) >= vertexCount) {
89 return false;
90 }
91 }
92 return true;
93 };
94 return indicesAreValid(mesh.mercatorFillIndices) && indicesAreValid(mesh.sphereFillIndices) &&
95 indicesAreValid(mesh.borderIndices);
96}

112bool capturePreviousFile(const QString &path, PreviousFile *previous, QString *errorMessage) {
113 const QFileInfo info(path);
114 if (!info.exists()) {
115 *previous = {};
116 return true;
117 }
118 if (!info.isFile() || info.isSymLink() || info.size() < 0 ||
119 info.size() > limits::MaximumAssetBytes) {
120 setError(errorMessage,
121 QStringLiteral("The previous map asset cannot be backed up safely."));
122 return false;
123 }
124 QFile file(path);
125 if (!file.open(QIODevice::ReadOnly)) {
126 setError(errorMessage, file.errorString());
127 return false;
128 }
129 const QByteArray bytes = file.read(limits::MaximumAssetBytes + 1);
130 if (bytes.size() != info.size()) {
131 setError(errorMessage,
132 QStringLiteral("The previous map asset changed while it was backed up."));
133 return false;
134 }
135 *previous = {true, bytes};
136 return true;
137}

139bool restorePreviousFile(const QString &path, const PreviousFile &previous, QString *errorMessage) {
140 if (previous.existed) {
141 return writeSingleFile(path, previous.bytes, errorMessage);
142 }
143 if (!QFileInfo::exists(path) || QFile::remove(path)) {
144 return true;
145 }
146 setError(errorMessage, QStringLiteral("The newly committed map asset could not be removed."));
147 return false;

150bool writeFilePair(const QString &assetPath, const QByteArray &assetBytes,
151 const QString &manifestPath, const QByteArray &manifestBytes,
152 QString *errorMessage) {
153 if (QFileInfo(assetPath).absoluteFilePath() == QFileInfo(manifestPath).absoluteFilePath()) {
154 setError(errorMessage, QStringLiteral("Map asset and manifest paths must be different."));
155 return false;
156 }
157 PreviousFile previousAsset;
158 if (!capturePreviousFile(assetPath, &previousAsset, errorMessage)) {
159 return false;
160 }
161
162 QSaveFile assetFile(assetPath);
163 QSaveFile manifestFile(manifestPath);
164 if (!assetFile.open(QIODevice::WriteOnly) || assetFile.write(assetBytes) != assetBytes.size()) {
165 setError(errorMessage, assetFile.errorString());
166 return false;
167 }
168 if (!manifestFile.open(QIODevice::WriteOnly) ||
169 manifestFile.write(manifestBytes) != manifestBytes.size()) {
170 setError(errorMessage, manifestFile.errorString());
171 return false;
172 }
173 if (!assetFile.commit()) {
174 setError(errorMessage, assetFile.errorString());
175 return false;
176 }
177 if (manifestFile.commit()) {
178 return true;
179 }
180
181 const QString commitError = manifestFile.errorString();
182 QString rollbackError;
183 if (!restorePreviousFile(assetPath, previousAsset, &rollbackError)) {
184 setError(errorMessage,
185 QStringLiteral("Manifest commit failed (%1) and asset rollback failed (%2).")
186 .arg(commitError, rollbackError));
187 } else {
188 setError(errorMessage,
189 QStringLiteral("Manifest commit failed; the previous asset was restored: %1")
190 .arg(commitError));
191 }
192 return false;
193}

197bool MapAssetWriter::write(const QString &assetPath, const QString &manifestPath,
198 const MapMesh &mesh, QString *errorMessage) {
199 if (!validate(mesh)) {
200 setError(errorMessage, QStringLiteral("The map mesh is invalid or exceeds asset limits."));
201 return false;
202 }
203
204 QByteArray payload;
205 payload.reserve(static_cast<qsizetype>(mesh.byteSize()));
206 appendArray(payload, mesh.vertices);
207 appendArray(payload, mesh.mercatorFillIndices);
208 appendArray(payload, mesh.sphereFillIndices);
209 appendArray(payload, mesh.borderIndices);
210
211 QByteArray asset;
212 asset.reserve(static_cast<qsizetype>(format::HeaderSize) + payload.size());
213 asset.append(format::Magic.data(), format::Magic.size());
214 appendUint32(asset, format::Version);
215 appendUint32(asset, format::HeaderSize);
216 appendUint32(asset, format::SupportedFlags);
217 appendUint64(asset, mesh.vertexCount());
218 appendUint64(asset, mesh.mercatorFillIndices.size());
219 appendUint64(asset, mesh.sphereFillIndices.size());
220 appendUint64(asset, mesh.borderIndices.size());
221 appendUint64(asset, mesh.byteSize());
222 appendUint32(asset,
223 MapChecksum::crc32(reinterpret_cast<const unsigned char *>(payload.constData()),
224 static_cast<std::size_t>(payload.size())));
225 appendUint32(asset, 0U);
226 if (asset.size() != static_cast<qsizetype>(format::HeaderSize)) {
227 setError(errorMessage, QStringLiteral("The map asset header could not be generated."));
228 return false;
229 }
230 asset.append(payload);
231
232 const QJsonObject manifest{
233 {QStringLiteral("formatVersion"), static_cast<int>(format::Version)},
234 {QStringLiteral("bytes"), static_cast<qint64>(asset.size())},
235 {QStringLiteral("sha256"),
236 QString::fromLatin1(QCryptographicHash::hash(asset, QCryptographicHash::Sha256).toHex())}};
237 const QByteArray manifestBytes = QJsonDocument(manifest).toJson(QJsonDocument::Indented);
238 return writeFilePair(assetPath, asset, manifestPath, manifestBytes, errorMessage);
239}
