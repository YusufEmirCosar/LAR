21bool EarthMapWidget::setMapMesh(std::shared_ptr<const MapMesh> mesh, QString *errorMessage) {
22 if (mesh == nullptr) {
23 if (errorMessage != nullptr) {
24 *errorMessage = QStringLiteral("The map mesh is unavailable.");
25 }
26 return false;
27 }
28
29 double minimumLongitude = 180.0;
30 double maximumLongitude = -180.0;
31 for (std::size_t offset = 0U; offset + 2U < mesh->vertices.size(); offset += 3U) {
32 minimumLongitude = std::min(minimumLongitude, static_cast<double>(mesh->vertices[offset]));
33 maximumLongitude = std::max(maximumLongitude, static_cast<double>(mesh->vertices[offset]));
34 }
35 m_camera.setGeometryLongitudeBounds(minimumLongitude, maximumLongitude);
36 if (!m_renderer.setMesh(std::move(mesh), errorMessage)) {
37 return false;
38 }
39
40 if (isValid()) {
41 makeCurrent();
42 const bool uploaded = m_renderer.uploadPending(errorMessage);
43 doneCurrent();
44 if (!uploaded) {
45 return false;
46 }
47 emit rendererReady();
48 }
49 update();
50 return true;
51}

201void EarthMapWidget::mouseMoveEvent(QMouseEvent *event) {
202 if (event->buttons() & Qt::LeftButton) {
203 const QPoint delta = event->pos() - m_lastMousePosition;
204 m_lastMousePosition = event->pos();
205 if (mapPresentation() == MapPresentation::Mercator) {
206 panFlatMap(delta);
207 } else {
208 emit mapDragged(delta);
209 }
210 event->accept();
211 return;
212 }
213
214 if (mapPresentation() == MapPresentation::Mercator) {
215 const QPointF projected = screenToFlatProjected(event->position(), width(), height());
216 const double longitude = MapProjection::wrapLongitude(projected.x());
217 const double latitude = MapProjection::unprojectLatitude(projected.y());
218 const bool insideMap = (projected.y() >= -MapProjection::HalfWorldDegrees) &&
219 (projected.y() <= MapProjection::HalfWorldDegrees);
220 emit geoCoordinateChanged(longitude, latitude, activeZoom(), insideMap);
221 } else {
222 double longitude = 0.0;
223 double latitude = 0.0;
224 const bool insideMap = sphereCoordinateAt(event->position(), longitude, latitude);
225 emit geoCoordinateChanged(longitude, latitude, activeZoom(), insideMap);
226 }
227 QOpenGLWidget::mouseMoveEvent(event);
228}
