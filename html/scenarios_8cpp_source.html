44double wrapLongitude(double longitude) {
45 double wrapped = std::fmod(longitude + Pi, TwoPi);
46 if (wrapped < 0.0)
47 wrapped += TwoPi;
48 wrapped -= Pi;
49 constexpr double BoundaryMargin = 0.000001;
50 return std::clamp(wrapped, -Pi + BoundaryMargin, Pi - BoundaryMargin);
51}

53double wrapPositiveAngle(double angle) {
54 double wrapped = std::fmod(angle, TwoPi);
55 return wrapped < 0.0 ? wrapped + TwoPi : wrapped;
56}

58double triangleWave(double phase) {
59 return (2.0 / Pi) * std::asin(std::sin(phase));

62void setPosition(double baseLatitude, double baseLongitude, double northMeters, double eastMeters,
63 double altitude, double position[3]) {
64 position[0] = baseLatitude + northMeters / EarthRadiusMeters;
65 position[1] =
66 wrapLongitude(baseLongitude + eastMeters / (EarthRadiusMeters * std::cos(baseLatitude)));
67 position[2] = altitude;

152void updateCoordinatedTurn(int, double elapsedSeconds, Plane *plane, Target *) {
153 constexpr double Radius = 6000.0;
154 constexpr double AngularRate = 0.035;
155 const double phase = AngularRate * elapsedSeconds;
156 setPosition(BaseLatitude, BaseLongitude, Radius * std::cos(phase), Radius * std::sin(phase),
157 3300.0, plane->location);
158 plane->euler[0] = wrapPositiveAngle(phase + Pi * 0.5);
159 plane->euler[1] = 0.01 * std::sin(phase * 2.0);
160 plane->euler[2] = std::atan((Radius * AngularRate * AngularRate) / 9.80665);
161 plane->velocity[0] = Radius * AngularRate * 3.6;
162}

164void updateFigureEight(int, double elapsedSeconds, Plane *plane, Target *) {
165 constexpr double AngularRate = 0.045;
166 const double phase = AngularRate * elapsedSeconds;
167 const double north = 7000.0 * std::sin(phase * 2.0);
168 const double east = 11000.0 * std::sin(phase);
169 const double northRate = 14000.0 * AngularRate * std::cos(phase * 2.0);
170 const double eastRate = 11000.0 * AngularRate * std::cos(phase);
171 setPosition(BaseLatitude, BaseLongitude, north, east, 3200.0 + 450.0 * std::sin(phase * 3.0),
172 plane->location);
173 plane->euler[0] = wrapPositiveAngle(std::atan2(eastRate, northRate));
174 plane->euler[1] = 0.10 * std::cos(phase * 3.0);
175 plane->euler[2] = 0.42 * std::sin(phase * 2.0);
176 plane->velocity[0] = std::hypot(northRate, eastRate) * 3.6;
177 plane->velocity[2] = 450.0 * 3.0 * AngularRate * std::cos(phase * 3.0) * 3.6;

229void updatePoleCrossingRoute(int packetIndex, double, Plane *plane, Target *) {
230 const double phase = std::remainder(
231 70.0 * DegreesToRadians + static_cast<double>(packetIndex) * 5.0 * DegreesToRadians, TwoPi);
232 const double direction = std::cos(phase);
233 plane->location[0] = std::asin(std::sin(phase));
234 plane->location[1] = wrapLongitude(BaseLongitude + (direction < 0.0 ? Pi : 0.0));
235 plane->location[2] = 5000.0 + 250.0 * std::sin(phase * 2.0);
236 plane->euler[0] = direction < 0.0 ? Pi : 0.0;
237 plane->velocity[0] = 900.0;
238}

240void updateMixedHighDynamics(int, double elapsedSeconds, Plane *plane, Target *target) {
241 const double phase = 0.085 * elapsedSeconds;
242 const double north = 12500.0 * std::sin(phase * 1.7);
243 const double east = 18000.0 * std::sin(phase);
244 const double northRate = 12500.0 * 1.7 * 0.085 * std::cos(phase * 1.7);
245 const double eastRate = 18000.0 * 0.085 * std::cos(phase);
246
247 setPosition(BaseLatitude, BaseLongitude, north, east, 4200.0 + 1300.0 * std::sin(phase * 2.3),
248 plane->location);
249
250 plane->euler[0] = wrapPositiveAngle(std::atan2(eastRate, northRate));
251 plane->euler[1] = 0.24 * std::sin(phase * 2.3);
252 plane->euler[2] = 0.68 * std::sin(phase * 1.7 + 0.5);
253 plane->velocity[0] = std::hypot(northRate, eastRate) * 3.6;
254 plane->velocity[1] = 180.0 * std::sin(phase * 1.3);
255 plane->velocity[2] = 1300.0 * 2.3 * 0.085 * std::cos(phase * 2.3) * 3.6;
256 setTargetPositions(10000.0 * std::sin(phase * 0.73 + 1.0), 15000.0 * std::cos(phase * 0.61),
257 target);
258 setLarPulse(elapsedSeconds, target);
259}

421void initialize(const QString &name, Plane *plane, Target *target,
422 dlz::TelemetryInputs *dlzInputs) {
423 if (!plane || !target)
424 return;
425 const ScenarioDefinition *definition = findDefinition(name);
426 if (definition && definition->initialize) {
427 definition->initialize(plane, target);
428 if (dlzInputs)
429 *dlzInputs = {20.0, 0.0, 30000.0};
430 return;
431 }
432 const DlzScenarioDefinition *dlzDefinition = findDlzDefinition(name);
433 if (dlzDefinition) {
434 // DLZ-only mappings do not require LAR values, but a valid common
435 // baseline keeps the sender safe when a combined mapping is selected.
436 initializeCommon(plane, target);
437 if (dlzInputs && dlzDefinition->initialize)
438 dlzDefinition->initialize(dlzInputs);
439 }
440}
441
442void update(const QString &name, int packetIndex, double elapsedSeconds, Plane *plane,
443 Target *target, dlz::TelemetryInputs *dlzInputs) {
444 if (!plane || !target)
445 return;
446 const ScenarioDefinition *definition = findDefinition(name);
447 if (definition && definition->update) {
448 target->time = elapsedSeconds;
449 definition->update(packetIndex, elapsedSeconds, plane, target);
450 return;
451 }
452 const DlzScenarioDefinition *dlzDefinition = findDlzDefinition(name);
453 if (dlzDefinition) {
454 target->time = elapsedSeconds;
455 if (dlzInputs && dlzDefinition->update)
456 dlzDefinition->update(packetIndex, elapsedSeconds, dlzInputs);
457 }
458}
