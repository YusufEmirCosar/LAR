141void PlaneViewWorkspace::resizeEvent(QResizeEvent *event) {

145 m_sceneWidget->setGeometry(rect());

150 constexpr int ViewportMargin = 18;
151 constexpr int ControlGap = 8;
152 m_uploadModelButton->adjustSize();
153 m_terrainButton->adjustSize();
154 m_surfaceButton->adjustSize();
155 m_changeSkyboxButton->adjustSize();
156 const int controlHeight =
157 std::max({m_uploadModelButton->height(), m_terrainButton->height(),
158 m_surfaceButton->height(), m_changeSkyboxButton->height()});
159 const int controlsY = std::max(0, height() - controlHeight - ViewportMargin);
160 const int skyboxX = std::max(0, width() - m_changeSkyboxButton->width() - ViewportMargin);
161 const int surfaceX = std::max(0, skyboxX - ControlGap - m_surfaceButton->width());
162 const int terrainX = std::max(0, surfaceX - ControlGap - m_terrainButton->width());
163 const int unwrappedUploadX = terrainX - ControlGap - m_uploadModelButton->width();
164 const bool wrapUpload = unwrappedUploadX < ViewportMargin;
165 const int uploadX = wrapUpload
166 ? std::max(0, width() - m_uploadModelButton->width() - ViewportMargin)
167 : unwrappedUploadX;
168 const int uploadY = wrapUpload ? std::max(0, controlsY - controlHeight - ControlGap) : controlsY;
169 m_uploadModelButton->setGeometry(uploadX, uploadY, m_uploadModelButton->width(),
170 m_uploadModelButton->height());
171 m_terrainButton->move(terrainX, controlsY);
172 m_surfaceButton->move(surfaceX, controlsY);
173 m_changeSkyboxButton->move(skyboxX, controlsY);
174 if (m_openGlFallback != nullptr) {
175 m_openGlFallback->raise();
176 } else {
177 m_sceneWidget->raise();
178 }
179 m_surfaceButton->raise();
180 m_terrainButton->raise();
181 m_changeSkyboxButton->raise();
182 m_uploadModelButton->raise();
183}
