34std::vector<double> bearingSamples(double start, double span, int segments) {
35 const int safeSegments = std::max(1, segments);
36 std::vector<double> samples;
37 samples.reserve(static_cast<std::size_t>(safeSegments + 3));
38 for (int index = 0; index <= safeSegments; ++index) {
39 samples.push_back(start +
40 span * static_cast<double>(index) / static_cast<double>(safeSegments));
41 }
42 const double end = start + span;
43 for (const double poleBearing : {0.0, Pi}) {
44 const double turns = std::ceil((start - poleBearing) / TwoPi);
45 const double candidate = poleBearing + turns * TwoPi;
46 if (candidate > start + LarZoneMeshLimits::SampleTolerance &&
47 candidate < end - LarZoneMeshLimits::SampleTolerance) {
48 appendUnique(samples, candidate);
49 }
50 }
51 std::sort(samples.begin(), samples.end());
52 return samples;
53}

55std::vector<double> radialSamples(const LarZoneDefinition &zone, int segments) {
56 const int safeSegments = std::max(1, segments);
57 std::vector<double> samples;
58 samples.reserve(static_cast<std::size_t>(safeSegments + 3));
59 for (int index = 0; index <= safeSegments; ++index) {
60 samples.push_back(zone.innerRadiusMeters +
61 (zone.outerRadiusMeters - zone.innerRadiusMeters) *
62 static_cast<double>(index) / static_cast<double>(safeSegments));
63 }
64 for (const double distance : {poleDistance(zone.center.latitude, Pi * 0.5),
65 poleDistance(zone.center.latitude, -Pi * 0.5)}) {
66 if (distance > zone.innerRadiusMeters + LarZoneMeshLimits::SampleTolerance &&
67 distance < zone.outerRadiusMeters - LarZoneMeshLimits::SampleTolerance) {
68 appendUnique(samples, distance);
69 }
70 }
71 std::sort(samples.begin(), samples.end());
72 return samples;
73}

75int angularSegmentCount(const LarZoneDefinition &zone, const lar::map::MapCamera &camera, int width,
76 int height) {
77 const int minimum = std::max(8, static_cast<int>(std::ceil(64.0 * zone.spanRadians / TwoPi)));
78 if (zone.outerRadiusMeters <= 0.0 || width <= 0 || height <= 0) {
79 return minimum;
80 }
81 QPointF centerScreen;
82 bool centerVisible = false;
83 if (!camera.projectGeoToScreen(zone.center.longitude * ToDegrees,
84 zone.center.latitude * ToDegrees, width, height, centerScreen,
85 centerVisible)) {
86 return minimum;
87 }
88 double radiusPixels = 0.0;
89 for (const double bearing : {0.0, Pi * 0.5, Pi, Pi * 1.5}) {
90 const auto point =
91 LarGeodesicGeometry::destination(zone.center, zone.outerRadiusMeters, bearing);
92 QPointF pointScreen;
93 bool pointVisible = false;
94 if (camera.projectGeoToScreen(point.longitude * ToDegrees, point.latitude * ToDegrees,
95 width, height, pointScreen, pointVisible)) {
96 radiusPixels = std::max(radiusPixels, std::hypot(pointScreen.x() - centerScreen.x(),
97 pointScreen.y() - centerScreen.y()));
98 }
99 }
100 if (!std::isfinite(radiusPixels) || radiusPixels <= LarZoneMeshLimits::CurveErrorPixels) {
101 return minimum;
102 }
103 const double cosineArgument =
104 std::clamp(1.0 - LarZoneMeshLimits::CurveErrorPixels / radiusPixels, -1.0, 1.0);
105 const double step = 2.0 * std::acos(cosineArgument);
106 if (!std::isfinite(step) || step <= 1.0e-6) {
107 return LarZoneMeshLimits::MaximumAngularSegmentCount;
108 }
109 return std::clamp(static_cast<int>(std::ceil(zone.spanRadians / step)), minimum,
110 LarZoneMeshLimits::MaximumAngularSegmentCount);
111}

113int radialSegmentCount(const LarZoneDefinition &zone) noexcept {
114 const double thickness =
115 (zone.outerRadiusMeters - zone.innerRadiusMeters) / LarGeodesicGeometry::EarthRadiusMeters;
116 return std::clamp(
117 static_cast<int>(std::ceil(thickness / LarZoneMeshLimits::MaximumRadialStepRadians)), 1,
118 LarZoneMeshLimits::MaximumRadialSegmentCount);

121double stableStartBearing(const LarZoneDefinition &zone) noexcept {
122 if (!isFullCircle(zone.spanRadians))
123 return zone.startBearingRadians;
124 const double north = poleDistance(zone.center.latitude, Pi * 0.5);
125 const double south = poleDistance(zone.center.latitude, -Pi * 0.5);
126 const bool crossesNorth = zone.outerRadiusMeters > north + LarZoneMeshLimits::SampleTolerance;
127 const bool crossesSouth = zone.outerRadiusMeters > south + LarZoneMeshLimits::SampleTolerance;
128 if (crossesNorth && (!crossesSouth || north <= south))
129 return 0.0;
130 if (crossesSouth)
131 return Pi;
132 return zone.startBearingRadians;
133}

137GeodesicZoneSampleGrid GeodesicZoneSampler::sample(const LarZoneDefinition &zone,
138 const lar::map::MapCamera &camera,
139 int viewportWidth, int viewportHeight) const {
140 int radialSegments = radialSegmentCount(zone);
141 int angularSegments = angularSegmentCount(zone, camera, viewportWidth, viewportHeight);
142 const int minimumAngular =
143 std::max(8, static_cast<int>(std::ceil(64.0 * zone.spanRadians / TwoPi)));
144 if (angularSegments * radialSegments > LarZoneMeshLimits::MaximumFillCellsPerZone) {
145 angularSegments =
146 std::max(minimumAngular, LarZoneMeshLimits::MaximumFillCellsPerZone / radialSegments);
147 }
148 if (angularSegments * radialSegments > LarZoneMeshLimits::MaximumFillCellsPerZone) {
149 radialSegments = std::max(1, LarZoneMeshLimits::MaximumFillCellsPerZone / angularSegments);
150 }
151
152 GeodesicZoneSampleGrid result;
153 result.fullCircle = isFullCircle(zone.spanRadians);
154 result.startBearingRadians = stableStartBearing(zone);
155 result.bearings = bearingSamples(result.startBearingRadians, zone.spanRadians, angularSegments);
156 result.radii = radialSamples(zone, radialSegments);
157 const double seamPole = std::abs(std::remainder(result.startBearingRadians, TwoPi)) <
158 LarZoneMeshLimits::FullCircleTolerance
159 ? Pi * 0.5
160 : -Pi * 0.5;
161 result.seamPoleDistanceMeters = poleDistance(zone.center.latitude, seamPole);
162 result.points.reserve(result.rowCount() * result.columnCount());
163 for (double radius : result.radii) {
164 for (double bearing : result.bearings) {
165 result.points.push_back(LarGeodesicGeometry::destination(zone.center, radius, bearing));
166 }
167 }
168 return result;
169}
