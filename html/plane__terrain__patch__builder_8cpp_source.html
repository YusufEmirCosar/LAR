161 static_cast<float>(-eastSlope / normalLength),

174 const std::size_t southEast = vertexIndex(row, column + 1, resolution);
175 const std::size_t northWest = vertexIndex(row + 1, column, resolution);
176 const std::size_t northEast = vertexIndex(row + 1, column + 1, resolution);

178 valid[northEast] == 0U) {
179 continue;
180 }
181 patch->indices.insert(
182 patch->indices.end(),
183 {static_cast<std::uint32_t>(southWest), static_cast<std::uint32_t>(southEast),

186 }
187 }

190 QStringLiteral("Available DTED0 samples do not form terrain triangles."));
191 return nullptr;
192 }
193 setError(errorMessage, {});
