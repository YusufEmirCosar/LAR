24bool validRequest(const PlaneTerrainBuildRequest &request) noexcept {
25 constexpr double HalfPi = LarProjection::Pi * 0.5;
26 const bool validProjectionOrigin = std::isnan(request.projectionOriginLatitudeRadians) ||
27 (std::isfinite(request.projectionOriginLatitudeRadians) &&
28 request.projectionOriginLatitudeRadians >= -HalfPi &&
29 request.projectionOriginLatitudeRadians <= HalfPi);
30 return std::isfinite(request.latitudeRadians) && request.latitudeRadians >= -HalfPi &&
31 request.latitudeRadians <= HalfPi && std::isfinite(request.longitudeRadians) &&
32 request.longitudeRadians >= -LarProjection::Pi &&
33 request.longitudeRadians <= LarProjection::Pi &&
34 std::isfinite(request.halfExtentMeters) &&
35 request.halfExtentMeters >= MinimumHalfExtentMeters &&
36 request.halfExtentMeters <= MaximumHalfExtentMeters && validProjectionOrigin &&
37 std::isfinite(request.metersPerSceneUnit) && request.metersPerSceneUnit > 0.0 &&
38 request.resolution >= MinimumResolution && request.resolution <= MaximumResolution;
39}

59 QString *errorMessage,

62 setError(errorMessage, QStringLiteral("Terrain patch request is outside bounded limits."));
63 return nullptr;

66 const std::size_t sampleCount =
67 static_cast<std::size_t>(resolution) * static_cast<std::size_t>(resolution);

69 std::vector<double> waterDepths(sampleCount, 0.0);
70 std::vector<unsigned char> valid(sampleCount, 0U);

72 const double sampleSpacing =

74 const double anchorPosition[3]{request.latitudeRadians, request.longitudeRadians, 0.0};
75 const double projectionOriginLatitude = std::isfinite(request.projectionOriginLatitudeRadians)
76 ? request.projectionOriginLatitudeRadians
77 : request.latitudeRadians;

79 m_landMaskBuilder.build(request.latitudeRadians, request.longitudeRadians,
80 projectionOriginLatitude, request.halfExtentMeters, cancelled);
81 if (!landMask.valid()) {
82 setError(errorMessage,
83 cancelled && cancelled()
84 ? QStringLiteral("Terrain patch request was superseded.")
85 : QStringLiteral("The packaged vector land map is unavailable."));
86 return nullptr;

89 double minimumElevation = std::numeric_limits<double>::infinity();
90 double maximumElevation = -std::numeric_limits<double>::infinity();
91 double maximumWaterDepth = 0.0;
92 std::size_t validSamples = 0U;
93 std::size_t waterSamples = 0U;
94 for (int row = 0; row < resolution; ++row) {
95 if (cancelled && cancelled()) {
96 setError(errorMessage, QStringLiteral("Terrain patch request was superseded."));
97 return nullptr;
98 }
99 const double northMeters = -request.halfExtentMeters + sampleSpacing * row;
100 for (int column = 0; column < resolution; ++column) {
101 const double eastMeters = -request.halfExtentMeters + sampleSpacing * column;
102 // The Plane surface, target marker, and LAR rings all use this local flat metric.
103 // Sampling with a spherical destination while retaining east/north vertex coordinates
104 // shifts terrain against those overlays (up to hundreds of metres at patch corners).
105 // Invert the exact renderer projection so the DTED elevation belongs to the
106 // geographic point represented by this vertex.
107 const std::optional<GeoCoordinateRadians> coordinate =
108 LarProjection::planeWorldToGeographic({eastMeters, northMeters}, anchorPosition,
109 0.0, projectionOriginLatitude, true);
110 if (!coordinate) {
111 continue;
112 }
113 const std::optional<double> sampledElevation =
114 m_sampler.sampleRadians(coordinate->latitude, coordinate->longitude);
115 const std::size_t index = vertexIndex(row, column, resolution);
116 if (!sampledElevation || !std::isfinite(*sampledElevation)) {
117 continue;
118 }
119 const bool isLand = landMask.landAtLocal(eastMeters, northMeters);
120 const double surfaceElevation = isLand ? *sampledElevation : 0.0;
121 const double waterDepth = isLand ? 0.0 : std::max(0.0, -*sampledElevation);
122 elevations[index] = *sampledElevation;
123 waterDepths[index] = waterDepth;
124 valid[index] = 1U;
125 water[index] = isLand ? 0U : 1U;
126 minimumElevation = std::min(minimumElevation, surfaceElevation);
127 maximumElevation = std::max(maximumElevation, surfaceElevation);
128 maximumWaterDepth = std::max(maximumWaterDepth, waterDepth);
129 waterSamples += isLand ? 0U : 1U;
130 ++validSamples;
131 }
132 }
133
134 const int centerIndex = resolution / 2;
135 const std::size_t centerOffset = vertexIndex(centerIndex, centerIndex, resolution);
136 if (validSamples == 0U || valid[centerOffset] == 0U) {
137 const QString detail = m_sampler.lastError();
138 setError(errorMessage,
139 detail.isEmpty()
140 ? QStringLiteral("No %1 elevation is available near the aircraft.")
141 .arg(dtedLevelDisplayName(m_level))
142 : detail);
143 return nullptr;
144 }

146 auto patch = std::make_shared<PlaneTerrainPatch>();
147 patch->anchorLatitudeRadians = request.latitudeRadians;
148 patch->anchorLongitudeRadians = request.longitudeRadians;
149 patch->projectionOriginLatitudeRadians = projectionOriginLatitude;
150 patch->halfExtentMeters = request.halfExtentMeters;
151 patch->metersPerSceneUnit = request.metersPerSceneUnit;
152 patch->minimumElevationMeters = minimumElevation;
153 patch->maximumElevationMeters = maximumElevation;
154 patch->centerElevationMeters = water[centerOffset] != 0U ? 0.0 : elevations[centerOffset];
155 patch->maximumWaterDepthMeters = maximumWaterDepth;
156 patch->validSampleCount = validSamples;
157 patch->waterSampleCount = waterSamples;
158 patch->resolution = resolution;
159 patch->sourceLevel = m_level;
160 patch->landMask = std::move(landMask);
161 patch->vertices.reserve(sampleCount * PlaneTerrainVertexStrideFloats);
162
163 for (int row = 0; row < resolution; ++row) {
164 for (int column = 0; column < resolution; ++column) {
165 const std::size_t index = vertexIndex(row, column, resolution);
166 const double center = elevations[index];
167 const auto usableNeighbor = [&valid, &water, index, resolution](int neighborRow,
168 int neighborColumn) {
169 const std::size_t neighbor = vertexIndex(neighborRow, neighborColumn, resolution);
170 return valid[neighbor] != 0U && water[neighbor] == water[index];
171 };
172 const int leftColumn =
173 column > 0 && usableNeighbor(row, column - 1) ? column - 1 : column;
174 const int rightColumn =
175 column + 1 < resolution && usableNeighbor(row, column + 1) ? column + 1 : column;
176 const int southRow = row > 0 && usableNeighbor(row - 1, column) ? row - 1 : row;
177 const int northRow =
178 row + 1 < resolution && usableNeighbor(row + 1, column) ? row + 1 : row;
179 const double eastSpan =
180 std::max(sampleSpacing, (rightColumn - leftColumn) * sampleSpacing);
181 const double northSpan = std::max(sampleSpacing, (northRow - southRow) * sampleSpacing);
182 const double eastSlope = valid[index] != 0U && water[index] == 0U
183 ? (elevations[vertexIndex(row, rightColumn, resolution)] -
184 elevations[vertexIndex(row, leftColumn, resolution)]) /
185 eastSpan
186 : 0.0;
187 const double northSlope =
188 valid[index] != 0U && water[index] == 0U
189 ? (elevations[vertexIndex(northRow, column, resolution)] -
190 elevations[vertexIndex(southRow, column, resolution)]) /
191 northSpan
192 : 0.0;
193 const double normalLength =
194 std::sqrt(eastSlope * eastSlope + 1.0 + northSlope * northSlope);
195 const double eastMeters = -request.halfExtentMeters + sampleSpacing * column;
196 const double northMeters = -request.halfExtentMeters + sampleSpacing * row;
197 patch->vertices.insert(patch->vertices.end(),
198 {static_cast<float>(eastMeters / request.metersPerSceneUnit),
199 static_cast<float>(center / request.metersPerSceneUnit),
200 static_cast<float>(-northMeters / request.metersPerSceneUnit),
201 static_cast<float>(-eastSlope / normalLength),
202 static_cast<float>(1.0 / normalLength),
203 static_cast<float>(northSlope / normalLength),
204 static_cast<float>(waterDepths[index])});
205 }
206 }
207
208 patch->indices.reserve(static_cast<std::size_t>(resolution - 1) *
209 static_cast<std::size_t>(resolution - 1) * 6U);
210 for (int row = 0; row + 1 < resolution; ++row) {
211 for (int column = 0; column + 1 < resolution; ++column) {
212 const std::size_t southWest = vertexIndex(row, column, resolution);
213 const std::size_t southEast = vertexIndex(row, column + 1, resolution);
214 const std::size_t northWest = vertexIndex(row + 1, column, resolution);
215 const std::size_t northEast = vertexIndex(row + 1, column + 1, resolution);
216 if (valid[southWest] == 0U || valid[southEast] == 0U || valid[northWest] == 0U ||
217 valid[northEast] == 0U) {
218 continue;
219 }
220 patch->indices.insert(
221 patch->indices.end(),
222 {static_cast<std::uint32_t>(southWest), static_cast<std::uint32_t>(southEast),
223 static_cast<std::uint32_t>(northWest), static_cast<std::uint32_t>(southEast),
224 static_cast<std::uint32_t>(northEast), static_cast<std::uint32_t>(northWest)});
225 }
226 }
227 if (patch->empty()) {
228 setError(errorMessage, QStringLiteral("Available %1 samples do not form terrain triangles.")
229 .arg(dtedLevelDisplayName(m_level)));
230 return nullptr;
231 }
232 patch->sampleValidity = std::move(valid);
233 setError(errorMessage, {});
234 return patch;
