23bool finitePatch(const PlaneTerrainPatch &patch) {
24 const bool finiteVertices = std::all_of(patch.vertices.cbegin(), patch.vertices.cend(),
25 [](float value) { return std::isfinite(value); });
26 bool validWaterDepth = patch.vertices.size() % PlaneTerrainVertexStrideFloats == 0U;
27 for (std::size_t offset = 0; validWaterDepth && offset < patch.vertices.size();
28 offset += PlaneTerrainVertexStrideFloats) {
29 validWaterDepth = patch.vertices[offset + 6U] >= 0.0F;
30 }
31 return finiteVertices && validWaterDepth && patch.landMask.valid() &&
32 patch.sampleValidity.size() == patch.vertexCount() &&
33 std::isfinite(patch.minimumElevationMeters) &&
34 std::isfinite(patch.maximumElevationMeters) &&
35 std::isfinite(patch.maximumWaterDepthMeters) && patch.maximumWaterDepthMeters >= 0.0 &&
36 patch.waterSampleCount <= patch.validSampleCount &&
37 std::isfinite(patch.metersPerSceneUnit) && patch.metersPerSceneUnit > 0.0;
38}

246PlaneTerrainGpuLayer::groundHeightAt(const QVector2D &currentXZ, const QVector2D &offsetXZ,
247 const QVector2D &scaleXZ,
248 float aircraftAltitudeScene) const noexcept {
249 if (!hasRenderableGeometry() || !std::isfinite(currentXZ.x()) ||
250 !std::isfinite(currentXZ.y()) || !std::isfinite(offsetXZ.x()) ||
251 !std::isfinite(offsetXZ.y()) || !std::isfinite(scaleXZ.x()) ||
252 !std::isfinite(scaleXZ.y()) || std::abs(scaleXZ.x()) <= 1.0e-6F ||
253 std::abs(scaleXZ.y()) <= 1.0e-6F || !std::isfinite(aircraftAltitudeScene) ||
254 m_patch->resolution < 2 || m_patch->sampleValidity.size() != m_patch->vertexCount()) {
255 return std::nullopt;
256 }
257
258 const double patchX = (static_cast<double>(currentXZ.x()) - offsetXZ.x()) / scaleXZ.x();
259 const double patchZ = (static_cast<double>(currentXZ.y()) - offsetXZ.y()) / scaleXZ.y();
260 const double eastMeters = patchX * m_patch->metersPerSceneUnit;
261 const double northMeters = -patchZ * m_patch->metersPerSceneUnit;
262 if (eastMeters < -m_patch->halfExtentMeters || eastMeters > m_patch->halfExtentMeters ||
263 northMeters < -m_patch->halfExtentMeters || northMeters > m_patch->halfExtentMeters) {
264 return std::nullopt;
265 }
266 if (!m_patch->landMask.landAtLocal(eastMeters, northMeters)) {
267 return -aircraftAltitudeScene;
268 }
269
270 const double scale =
271 static_cast<double>(m_patch->resolution - 1) / (2.0 * m_patch->halfExtentMeters);
272 const double columnPosition = (eastMeters + m_patch->halfExtentMeters) * scale;
273 const double rowPosition = (northMeters + m_patch->halfExtentMeters) * scale;
274 const int column0 =
275 std::clamp(static_cast<int>(std::floor(columnPosition)), 0, m_patch->resolution - 1);
276 const int row0 =
277 std::clamp(static_cast<int>(std::floor(rowPosition)), 0, m_patch->resolution - 1);
278 const int column1 = std::min(column0 + 1, m_patch->resolution - 1);
279 const int row1 = std::min(row0 + 1, m_patch->resolution - 1);
280 const auto vertex = [this](int row, int column) {
281 return static_cast<std::size_t>(row) * static_cast<std::size_t>(m_patch->resolution) +
282 static_cast<std::size_t>(column);
283 };
284 const std::size_t southWest = vertex(row0, column0);
285 const std::size_t southEast = vertex(row0, column1);
286 const std::size_t northWest = vertex(row1, column0);
287 const std::size_t northEast = vertex(row1, column1);
288 for (const std::size_t sample : {southWest, southEast, northWest, northEast}) {
289 if (m_patch->sampleValidity[sample] == 0U) {
290 return std::nullopt;
291 }
292 }
293 const auto height = [this](std::size_t sample) {
294 return static_cast<double>(m_patch->vertices[sample * PlaneTerrainVertexStrideFloats + 1U]);
295 };
296 const double columnFraction = columnPosition - static_cast<double>(column0);
297 const double rowFraction = rowPosition - static_cast<double>(row0);
298 const double south =
299 height(southWest) * (1.0 - columnFraction) + height(southEast) * columnFraction;
300 const double north =
301 height(northWest) * (1.0 - columnFraction) + height(northEast) * columnFraction;
302 return static_cast<float>(south * (1.0 - rowFraction) + north * rowFraction) -
303 aircraftAltitudeScene;
304}
