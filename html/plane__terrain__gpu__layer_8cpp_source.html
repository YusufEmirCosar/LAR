22bool finitePatch(const PlaneTerrainPatch &patch) {
23 const bool finiteVertices = std::all_of(patch.vertices.cbegin(), patch.vertices.cend(),
24 [](float value) { return std::isfinite(value); });
25 bool validWater = patch.vertices.size() % PlaneTerrainVertexStrideFloats == 0U;
26 for (std::size_t offset = 0; validWater && offset < patch.vertices.size();
27 offset += PlaneTerrainVertexStrideFloats) {
28 const float water = patch.vertices[offset + 6U];
29 const float depth = patch.vertices[offset + 7U];
30 validWater = water >= 0.0F && water <= 1.0F && depth >= 0.0F;
31 }
32 return finiteVertices && validWater && std::isfinite(patch.minimumElevationMeters) &&
33 std::isfinite(patch.maximumElevationMeters) &&
34 std::isfinite(patch.maximumWaterDepthMeters) && patch.maximumWaterDepthMeters >= 0.0 &&
35 patch.waterSampleCount <= patch.validSampleCount &&
36 std::isfinite(patch.metersPerSceneUnit) && patch.metersPerSceneUnit > 0.0;
37}

103bool PlaneTerrainGpuLayer::uploadPending(QString *errorMessage) {
104 if (!m_uploadPending) {
105 return true;
106 }
107 m_indexCount = 0;
108 if (m_patch == nullptr || m_patch->empty()) {
109 m_uploadPending = false;
110 return true;
111 }
112 const std::size_t vertexCount = m_patch->vertexCount();
113 const bool invalidSizes =
114 m_patch->vertices.size() % PlaneTerrainVertexStrideFloats != 0U || vertexCount == 0U ||
115 m_patch->vertices.size() >
116 static_cast<std::size_t>(std::numeric_limits<int>::max()) / sizeof(float) ||
117 m_patch->indices.size() >
118 static_cast<std::size_t>(std::numeric_limits<int>::max()) / sizeof(std::uint32_t) ||
119 m_patch->indices.size() > static_cast<std::size_t>(std::numeric_limits<int>::max());
120 const bool invalidIndex =
121 std::any_of(m_patch->indices.cbegin(), m_patch->indices.cend(),
122 [vertexCount](std::uint32_t index) { return index >= vertexCount; });
123 if (invalidSizes || invalidIndex || !finitePatch(*m_patch)) {
124 setError(errorMessage, QStringLiteral("Plane terrain mesh exceeds GPU safety limits."));
125 return false;
126 }
127
128 m_program->bind();
129 QOpenGLVertexArrayObject::Binder binder(&m_array);
130 m_vertices.bind();
131 m_vertices.setUsagePattern(QOpenGLBuffer::DynamicDraw);
132 m_vertices.allocate(m_patch->vertices.data(),
133 static_cast<int>(m_patch->vertices.size() * sizeof(float)));
134 m_indices.bind();
135 m_indices.setUsagePattern(QOpenGLBuffer::DynamicDraw);
136 m_indices.allocate(m_patch->indices.data(),
137 static_cast<int>(m_patch->indices.size() * sizeof(std::uint32_t)));
138 m_program->enableAttributeArray(0);
139 m_program->enableAttributeArray(1);
140 m_program->enableAttributeArray(2);
141 m_program->setAttributeBuffer(0, GL_FLOAT, 0, 3, VertexStrideBytes);
142 m_program->setAttributeBuffer(1, GL_FLOAT, 3 * static_cast<int>(sizeof(float)), 3,
143 VertexStrideBytes);
144 m_program->setAttributeBuffer(2, GL_FLOAT, 6 * static_cast<int>(sizeof(float)), 2,
145 VertexStrideBytes);
146 m_program->release();
147 m_indexCount = static_cast<int>(m_patch->indices.size());
148 m_uploadPending = false;
149 return true;
