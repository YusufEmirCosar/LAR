89CommandDispatch DirectApplicationRuntime::loadMapping(const QString &path) {
90 if (m_shuttingDown)
91 return rejectCommand(shutdownRejection());
92 if (path.isEmpty())
93 return rejectCommand(QStringLiteral("Mapping path is empty"));
94 const CommandDispatch dispatch = acceptCommand();
95 if (m_capture.isListening()) {
96 emit mappingLoadFinished(
97 {dispatch.request,
98 false,
99 path,
100 0,
101 0,
102 {},
103 QStringLiteral("Stop online capture before replacing the packet mapping")});
104 return dispatch;
105 }
106 PacketMapping mapping;
107 QString error;
108 const bool loaded = m_mappingRepository.loadFile(path, &mapping, &error);
109 if (loaded) {
110 m_mapping = std::move(mapping);
111 m_capture.setMapping(m_mapping);
112 }
113 emit mappingLoadFinished({dispatch.request, loaded, path,
114 loaded ? static_cast<int>(m_mapping.bindings().size()) : 0,
115 loaded ? m_mapping.minimumPacketSize() : 0,
116 loaded ? m_mapping.json() : QByteArray{}, error});
117 return dispatch;
118}

120CommandDispatch DirectApplicationRuntime::startOnline(quint16 port) {
121 if (m_shuttingDown)
122 return rejectCommand(shutdownRejection());
123 if (port == 0)
124 return rejectCommand(QStringLiteral("UDP port must be non-zero"));
125 const CommandDispatch dispatch = acceptCommand();
126 const RuntimeSourceEpoch epoch{RuntimeStateSource::Online, ++m_nextStateGeneration};
127 m_activeStateSource = epoch.source;
128 m_activeStateGeneration = epoch.generation;
129 QString error;
130 const bool started = m_capture.start(port, &error);
131 emit onlineStartFinished({dispatch.request, epoch, started, error});
132 emit onlineStateChanged({epoch, m_capture.isListening(), m_capture.port()});
133 if (!started)
134 m_activeStateSource = RuntimeStateSource::None;
135 return dispatch;
136}

138CommandDispatch DirectApplicationRuntime::setIpAccessPolicy(const IpAccessPolicy &policy) {
139 if (m_shuttingDown)
140 return rejectCommand(shutdownRejection());
141 const CommandDispatch dispatch = acceptCommand();
142 QString error;
143 bool applied = false;
144 if (m_capture.isListening()) {
145 error = QStringLiteral("Stop online capture before changing the IP access policy");
146 } else if (!m_capture.setIpAccessPolicy(policy)) {
147 error = QStringLiteral("Datagram source rejected the IP access policy");
148 } else {
149 applied = true;
150 m_ipPolicy = policy;
151 }
152 emit ipAccessPolicyChangeFinished({dispatch.request, applied, m_ipPolicy, error});
153 return dispatch;

406void DirectApplicationRuntime::publishRecordingState() {
407 const bool hasSession = m_recording.isRecording();
408 const bool paused = m_recording.isPaused();
409 const quint32 recordCount = m_recording.recordCount();
410 const SessionTimestamp duration = m_recording.recordingDuration();
411 if (m_hasPublishedRecordingState && hasSession == m_lastHasRecordingSession &&

416 m_lastHasRecordingSession = hasSession;
417 m_lastRecordingPaused = paused;
