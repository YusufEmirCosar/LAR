13QString solveErrorMessage(DlzSolveError error) {
14 switch (error) {
15 case DlzSolveError::InvalidInput:
16 return QStringLiteral("DLZ solver input is invalid");
17 case DlzSolveError::UnsupportedOrderedDomain:
18 return QStringLiteral("DLZ toy model input is outside its ordered domain");
19 case DlzSolveError::NumericOverflow:
20 return QStringLiteral("DLZ toy model calculation exceeded its finite numeric domain");
21 case DlzSolveError::InvalidDerivedOrdering:
22 return QStringLiteral("DLZ toy model result is not strictly ordered");
23 case DlzSolveError::None:
24 break;
25 }
26 return {};
27}

29bool validInputs(const ScenarioInputs &inputs, QString *error) {
30 const auto fail = [error](const QString &message) {
31 if (error != nullptr) {
32 *error = message;
33 }
34 return false;
35 };
36 if (!std::isfinite(inputs.rangeNm) || inputs.rangeNm <= 0.0 ||
37 !std::isfinite(inputs.aspectDegrees) || inputs.aspectDegrees < 0.0 ||
38 inputs.aspectDegrees > 180.0 || !std::isfinite(inputs.altitudeFeet) ||
39 inputs.altitudeFeet < 0.0 || !std::isfinite(inputs.shooterMach) ||
40 inputs.shooterMach < 0.0 || !std::isfinite(inputs.targetMach) || inputs.targetMach < 0.0) {
41 return fail(QStringLiteral("Range, aspect, altitude, and Mach controls are invalid"));
42 }
43 return true;
44}

48bool ScenarioAdapter::build(const ScenarioInputs &inputs, ScenarioFrame *frame, QString *error) {
49 if (frame == nullptr) {
50 if (error != nullptr) {
51 *error = QStringLiteral("DLZ scenario output is null");
52 }
53 return false;
54 }
55 if (!validInputs(inputs, error)) {
56 return false;
57 }
58
59 ScenarioFrame result;
60 result.weapon = WeaponModel{};
61 result.atmosphere = Atmosphere{};
62 result.shooter.mach = inputs.shooterMach;
63 result.shooter.psi = 0.0;
64 result.shooter.theta = 0.0;
65 result.shooter.phi = 0.0;
66 result.shooter.pos = {0.0, 0.0, -inputs.altitudeFeet / units::FeetPerMeter};
67
68 const double aspectRadians = inputs.aspectDegrees * units::RadiansPerDegree;
69 const double targetSpeed = inputs.targetMach * result.atmosphere.speedOfSound;
70 result.shooter.vel = {inputs.shooterMach * result.atmosphere.speedOfSound, 0.0, 0.0};
71 result.target.pos = {inputs.rangeNm * units::MetersPerNauticalMile, 0.0, result.shooter.pos.z};
72 // Target velocity is selected so target-velocity versus target-to-shooter
73 // LOS produces the requested aspect exactly.
74 result.target.vel = {targetSpeed * std::cos(Pi - aspectRadians),
75 targetSpeed * std::sin(Pi - aspectRadians), 0.0};
76 result.target.mach = inputs.targetMach;
77 result.target.psi = Pi - aspectRadians;
78 result.target.nzDefensive = 9.0;
79
80 QString geometryError;
81 const auto geometry = calculateGeometry(result.shooter, result.target, result.atmosphere,
82 result.weapon, &geometryError);
83 if (!geometry.has_value()) {
84 if (error != nullptr) {
85 *error = geometryError;
86 }
87 return false;
88 }
89 if (std::abs(geometry->aspectRadians - aspectRadians) > 1e-9) {
90 if (error != nullptr) {
91 *error = QStringLiteral("Scenario adapter could not realize requested aspect");
92 }
93 return false;
94 }
95
96 result.geometry = *geometry;
97 // Keep the exact formula values available to the UI even when the strict
98 // solver-domain guard rejects this input. The workspace uses these only
99 // as readouts while the invalid frame is withheld from the renderer.
100 result.solution = evaluateToyModel(result.shooter, result.geometry, result.weapon);
101
102 const DlzSolveResult solution =
103 solve(result.shooter, result.target, result.atmosphere, *geometry, result.weapon);
104 if (!solution) {
105 *frame = result;
106 if (error != nullptr) {
107 *error = solveErrorMessage(solution.error);
108 }
109 return false;
110 }
111 result.solution = *solution.solution;
112 *frame = result;
113 return true;
114}
