192void OnlinePanel::renderOnlineState(bool listening) {

194 m_listener->setChecked(listening);
195 m_port->setEnabled(!listening);
196 m_allowAll->setEnabled(!listening && !m_policyPending);
197 m_whitelist->setEnabled(!listening && !m_policyPending);
198 m_listener->setText(listening ? QStringLiteral("Stop Listener")
199 : QStringLiteral("Start Listener"));
200 m_listener->setIcon(
201 style()->standardIcon(listening ? QStyle::SP_MediaStop : QStyle::SP_MediaPlay));

205void OnlinePanel::renderPolicy(const IpAccessPolicy &activePolicy, const QString &label,
206 bool pending) {
207 Q_UNUSED(label);
208 m_policyPending = pending;
209 {
210 const QSignalBlocker allowAllBlocker(m_allowAll);
211 const QSignalBlocker whitelistBlocker(m_whitelist);
212 const bool allowAll = activePolicy.mode() == IpAccessPolicy::Mode::AllowAll;
213 m_allowAll->setChecked(allowAll);
214 m_whitelist->setChecked(!allowAll);
215 }
216 const bool listening = m_application.isListening();

218 m_whitelist->setEnabled(!listening && !pending);

231void OnlinePanel::renderRecordingMetrics() {
232 const SessionTimestamp duration = m_application.viewModel().recordingDuration();
233 const qint64 milliseconds = duration.milliseconds();
234 const qint64 minutes = milliseconds / 60'000;
235 const qint64 seconds = (milliseconds / 1'000) % 60;
236 const qint64 remainder = milliseconds % 1'000;
237 const QString formattedDuration =
238 QStringLiteral("%1:%2:%3")
239 .arg(minutes, 2, 10, QLatin1Char('0'))
240 .arg(seconds, 2, 10, QLatin1Char('0'))
241 .arg(remainder, 3, 10, QLatin1Char('0'));
242 m_recordDuration->setText(QStringLiteral("Duration: %1").arg(formattedDuration));
243 m_recordCount->setText(
244 QStringLiteral("Saved packages: %1")
245 .arg(QString::number(m_application.viewModel().recordedPacketCount())));
246}
