32quint32 crc32(const QByteArray &bytes) noexcept {
33 quint32 result = std::numeric_limits<quint32>::max();
34 for (const char character : bytes) {
35 result ^= static_cast<quint8>(character);
36 for (int bit = 0; bit < 8; ++bit) {
37 const quint32 mask = 0U - (result & 1U);
38 result = (result >> 1U) ^ (CrcPolynomial & mask);
39 }
40 }
41 return result ^ std::numeric_limits<quint32>::max();
42}

133 const bool payloadValid =

147 }
148 if (nextPayloadOffset != declaredFileSize) {

151 return;
152 }

156DtedWaterMaskReadResult DtedWaterMaskSource::load(const DtedCellKey &key, int longitudeSampleCount,
157 int latitudeSampleCount) const {
158 const std::size_t index = indexFor(key);
159 if (!isAvailable() || index >= m_entries.size()) {
160 return {nullptr, m_initializationError.isEmpty()
161 ? QStringLiteral("DTED water-mask source is unavailable.")
162 : m_initializationError};
163 }
164 const IndexEntry &entry = m_entries[index];
165 if (entry.coverage == DtedWaterCoverage::Missing) {
166 return {nullptr, QStringLiteral("DTED water-mask cell is unavailable at %1, %2.")
167 .arg(key.latitudeDegrees)
168 .arg(key.longitudeDegrees)};
169 }
170 if (longitudeSampleCount != entry.longitudeSampleCount ||
171 latitudeSampleCount != entry.latitudeSampleCount) {
172 return {nullptr,
173 QStringLiteral("DTED water-mask dimensions disagree with the terrain cell.")};
174 }
175
176 auto cell = std::make_shared<DtedWaterMaskCell>();
177 cell->key = key;
178 cell->longitudeSampleCount = longitudeSampleCount;
179 cell->latitudeSampleCount = latitudeSampleCount;
180 cell->coverage = entry.coverage;
181 if (entry.coverage == DtedWaterCoverage::Mixed) {
182 QFile file(m_packPath);
183 if (!file.open(QIODevice::ReadOnly) || !file.seek(entry.payloadOffset)) {
184 return {nullptr, QStringLiteral("DTED water-mask payload could not be opened.")};
185 }
186 const QByteArray payload = file.read(entry.payloadBytes);
187 if (payload.size() != entry.payloadBytes || crc32(payload) != entry.checksum) {
188 return {nullptr, QStringLiteral("DTED water-mask payload is truncated or corrupt.")};
189 }
190 cell->packedWaterBits.assign(
191 reinterpret_cast<const quint8 *>(payload.constData()),
192 reinterpret_cast<const quint8 *>(payload.constData() + payload.size()));
193 const std::size_t sampleCount = static_cast<std::size_t>(longitudeSampleCount) *
194 static_cast<std::size_t>(latitudeSampleCount);
195 const std::size_t unusedBits = cell->packedWaterBits.size() * 8U - sampleCount;
196 if (unusedBits > 0U) {
197 const quint8 usedMask = static_cast<quint8>(0xFFU >> unusedBits);
198 if ((cell->packedWaterBits.back() & static_cast<quint8>(~usedMask)) != 0U) {
199 return {nullptr,
200 QStringLiteral("DTED water-mask payload has non-zero padding bits.")};
201 }
202 }
203 }
204 if (!cell->valid()) {
205 return {nullptr, QStringLiteral("DTED water-mask cell is internally inconsistent.")};
206 }
207 return {std::move(cell), {}};
