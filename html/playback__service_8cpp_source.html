128void PlaybackService::onClockTick() {
129 if (!m_isPlaying || !m_reader.isValid() || m_reader.recordCount() == 0)
130 return;
131
132 const long double frameStepMilliseconds = 1000.0L * static_cast<long double>(m_rate) /
133 static_cast<long double>(ReplayFramesPerSecond);
134 const long double durationMilliseconds =
135 static_cast<long double>(m_reader.duration().milliseconds());
136 long double nextPosition = m_exactPositionMilliseconds + frameStepMilliseconds;
137
138 if (m_repeat && durationMilliseconds > 0.0L && std::isfinite(nextPosition)) {
139 nextPosition = std::fmod(nextPosition, durationMilliseconds);
140 int bestIndex = 0;
141 if (!findFrameBefore(nextPosition, &bestIndex) || !publishReplayFrame(bestIndex))
142 return;
143 m_exactPositionMilliseconds = nextPosition;
144 m_position = SessionTimestamp::clampedMilliseconds(
145 static_cast<qint64>(std::round(m_exactPositionMilliseconds)));
146 emit positionChanged(m_position);
147 return;
148 }
149
150 if (durationMilliseconds <= 0.0L || !std::isfinite(nextPosition) ||
151 nextPosition > durationMilliseconds) {
152 int bestIndex = 0;
153 const long double searchPosition =
154 std::isfinite(nextPosition) ? nextPosition : durationMilliseconds + 1.0L;
155 if (!findFrameBefore(searchPosition, &bestIndex) || !publishReplayFrame(bestIndex))
156 return;
157 finishPlayback();
158 return;
159 }
160
161 int bestIndex = 0;
162 if (!findFrameBefore(nextPosition, &bestIndex) || !publishReplayFrame(bestIndex))
163 return;
164 m_exactPositionMilliseconds = nextPosition;
165 m_position = SessionTimestamp::clampedMilliseconds(
166 static_cast<qint64>(std::round(m_exactPositionMilliseconds)));
167 emit positionChanged(m_position);
168}

206bool PlaybackService::findFrameBefore(long double targetMilliseconds, int *index) {
207 if (index == nullptr || !m_reader.isValid() || m_reader.recordCount() <= 0 ||
208 !std::isfinite(targetMilliseconds)) {
209 return false;
210 }
211
212 int low = 0;
213 int high = m_reader.recordCount();
214 while (low < high) {
215 const int middle = low + (high - low) / 2;
216 SessionTimestamp timestamp;
217 if (!readTimestamp(middle, &timestamp))
218 return false;
219 if (static_cast<long double>(timestamp.milliseconds()) < targetMilliseconds)
220 low = middle + 1;
221 else
222 high = middle;
223 }
224 *index = std::max(0, low - 1);
225 return true;
226}
