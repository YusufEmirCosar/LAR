98void PlaybackService::seek(SessionTimestamp position) {
99 if (!m_reader.isValid() || m_reader.recordCount() == 0)
100 return;
101 const SessionTimestamp targetTime = std::min(position, m_reader.duration());
102
103 qint64 bestIndex = 0;
104 // findFrameBefore() uses a strict comparison. Advancing the integer target
105 // by one millisecond makes seeking inclusive without changing replay's
106 // between-tick sampling rule.
107 if (!findFrameBefore(static_cast<long double>(targetTime.milliseconds()) + 1.0L, &bestIndex))
108 return;
109
110 m_position = targetTime;
111 m_exactPositionMilliseconds = static_cast<long double>(targetTime.milliseconds());
112 QString error;
113 if (!publishFrame(bestIndex, &error)) {
114 failPlayback(error);
115 return;
116 }
117 emit positionChanged(m_position);
118}

131void PlaybackService::onClockTick() {
132 if (!m_isPlaying || !m_reader.isValid() || m_reader.recordCount() == 0)
133 return;
134
135 // The clock advances independently of record density. Selecting only the
136 // newest eligible record prevents a dense capture or high rate from
137 // flooding the UI with a catch-up burst.
138 const long double frameStepMilliseconds = 1000.0L * static_cast<long double>(m_rate) /
139 static_cast<long double>(ReplayFramesPerSecond);
140 const long double durationMilliseconds =
141 static_cast<long double>(m_reader.duration().milliseconds());
142 long double nextPosition = m_exactPositionMilliseconds + frameStepMilliseconds;
143
144 if (m_repeat && durationMilliseconds > 0.0L && std::isfinite(nextPosition)) {
145 nextPosition = std::fmod(nextPosition, durationMilliseconds);
146 qint64 bestIndex = 0;
147 if (!findFrameBefore(nextPosition, &bestIndex) || !publishReplayFrame(bestIndex))
148 return;
149 m_exactPositionMilliseconds = nextPosition;
150 m_position = SessionTimestamp::clampedMilliseconds(
151 static_cast<qint64>(std::round(m_exactPositionMilliseconds)));
152 emit positionChanged(m_position);
153 return;
154 }
155
156 if (durationMilliseconds <= 0.0L || !std::isfinite(nextPosition) ||
157 nextPosition > durationMilliseconds) {
158 qint64 bestIndex = 0;
159 const long double searchPosition =
160 std::isfinite(nextPosition) ? nextPosition : durationMilliseconds + 1.0L;
161 if (!findFrameBefore(searchPosition, &bestIndex) || !publishReplayFrame(bestIndex))
162 return;
163 finishPlayback();
164 return;
165 }
166
167 qint64 bestIndex = 0;
168 if (!findFrameBefore(nextPosition, &bestIndex) || !publishReplayFrame(bestIndex))
169 return;
170 m_exactPositionMilliseconds = nextPosition;
171 m_position = SessionTimestamp::clampedMilliseconds(
172 static_cast<qint64>(std::round(m_exactPositionMilliseconds)));
173 emit positionChanged(m_position);
174}

212bool PlaybackService::findFrameBefore(long double targetMilliseconds, qint64 *index) {
213 if (index == nullptr || !m_reader.isValid() || m_reader.recordCount() <= 0 ||
214 !std::isfinite(targetMilliseconds)) {
215 return false;
216 }
217
218 // No record can precede zero. The historical behavior clamps this case to
219 // the first record rather than selecting later records with timestamp zero.
220 if (targetMilliseconds <= 0.0L) {
221 *index = 0;
222 return true;
223 }
224
225 // Stored timestamps are integral milliseconds. Therefore the greatest
226 // timestamp strictly below x is ceil(x) - 1. Passing that inclusive bound
227 // lets the reader use its native resident-or-sparse lookup and materialize
228 // at most one selected record per lookup.
229 const long double inclusiveMilliseconds =
230 std::min(std::ceil(targetMilliseconds) - 1.0L,
231 static_cast<long double>(SessionTimestamp::maximum().milliseconds()));
232 const SessionTimestamp position =
233 SessionTimestamp::clampedMilliseconds(static_cast<qint64>(inclusiveMilliseconds));
234 QString error;
235 if (m_reader.findRecordAtOrBefore(position, index, &error))
236 return true;
237 failPlayback(error);
238 return false;
239}
