33int faceIndex(const QString &suffix) {
34 for (std::size_t index = 0; index < FaceSuffixes.size(); ++index) {
35 if (suffix.compare(QLatin1String(FaceSuffixes[index]), Qt::CaseInsensitive) == 0) {
36 return static_cast<int>(index);

42bool validFaceFile(const QString &path, QSize *expectedSize) {
43 const QFileInfo file(path);
44 if (!file.isFile() || !file.isReadable() || file.isSymLink() || file.size() <= 0 ||
45 file.size() > MaximumFaceBytes) {
46 return false;
47 }
48 QImageReader reader(path);
49 reader.setAutoTransform(false);
50 const QSize size = reader.size();
51 if (!size.isValid() || size.width() != size.height() || size.width() > MaximumFaceSize) {
52 return false;
53 }
54 if (!expectedSize->isValid()) {
55 *expectedSize = size;
56 return true;
57 }
58 return size == *expectedSize;
59}

63CubemapCatalog::CubemapCatalog(QString directory) {
64 const QDir cubemapDirectory(std::move(directory));
65 const QRegularExpression pattern(QStringLiteral(R"(^(.+)_(rt|lf|up|dn|ft|bk)\.png$)"),
66 QRegularExpression::CaseInsensitiveOption);
67 const QFileInfoList files = cubemapDirectory.entryInfoList(
68 {QStringLiteral("*.png")}, QDir::Files | QDir::Readable | QDir::NoSymLinks, QDir::Name);
69 QMap<QString, std::array<QString, 6>> groupedPaths;
70 for (const QFileInfo &file : files) {
71 const auto match = pattern.match(file.fileName());
72 if (!match.hasMatch()) {
73 continue;
74 }
75 const QString key = match.captured(1);
76 if (!groupedPaths.contains(key) && groupedPaths.size() >= MaximumSkyboxCount) {
77 break;
78 }
79 const int index = faceIndex(match.captured(2));
80 if (index < 0) {
81 continue;
82 }
83 groupedPaths[key][static_cast<std::size_t>(index)] = file.absoluteFilePath();
84 }
85
86 for (auto group = groupedPaths.cbegin(); group != groupedPaths.cend(); ++group) {
87 QSize faceSize;
88 bool valid = true;
89 for (const QString &path : group.value()) {
90 if (path.isEmpty() || !validFaceFile(path, &faceSize)) {
91 valid = false;
92 break;
93 }
94 }
95 if (!valid) {
96 ++m_rejectedSetCount;
97 continue;
98 }
99 const int displayNumber = static_cast<int>(m_entries.size()) + 1;
100 m_entries.append(
101 {group.value(), QStringLiteral("Sky %1").arg(displayNumber, 2, 10, QLatin1Char('0'))});
102 }
103}

109bool CubemapCatalog::load(int index, CubemapFaces *faces, QString *errorMessage) const {
110 if (faces == nullptr || index < 0 || index >= m_entries.size()) {
111 setError(errorMessage, QStringLiteral("The requested skybox is unavailable."));
112 return false;
113 }
114 CubemapFaces result;
115 QSize expectedSize;
116 for (std::size_t face = 0; face < result.images.size(); ++face) {
117 const QString &path = m_entries[index].facePaths[face];
118 const QFileInfo fileInfo(path);
119 if (!fileInfo.isFile() || !fileInfo.isReadable() || fileInfo.isSymLink() ||
120 fileInfo.size() <= 0 || fileInfo.size() > MaximumFaceBytes) {
121 setError(errorMessage,
122 QStringLiteral("A cubemap face no longer passes file safety limits."));
123 return false;
124 }
125 QFile file(path);
126 if (!file.open(QIODevice::ReadOnly) || file.size() <= 0 || file.size() > MaximumFaceBytes) {
127 setError(errorMessage, QStringLiteral("A cubemap face could not be opened safely."));
128 return false;
129 }
130 const QFileInfo openedPath(path);
131 if (!openedPath.isFile() || !openedPath.isReadable() || openedPath.isSymLink() ||
132 openedPath.size() != file.size()) {
133 setError(errorMessage,
134 QStringLiteral("A cubemap face changed while it was being opened."));
135 return false;
136 }
137#ifdef Q_OS_UNIX
138 struct stat pathStatus{};
139 struct stat openedStatus{};
140 const QByteArray nativePath = QFile::encodeName(openedPath.absoluteFilePath());
141 if (::lstat(nativePath.constData(), &pathStatus) != 0 ||
142 ::fstat(file.handle(), &openedStatus) != 0 || !S_ISREG(pathStatus.st_mode) ||
143 !S_ISREG(openedStatus.st_mode) || pathStatus.st_dev != openedStatus.st_dev ||
144 pathStatus.st_ino != openedStatus.st_ino) {
145 setError(errorMessage,
146 QStringLiteral("A cubemap face changed while it was being opened."));
147 return false;
148 }
149#endif
150 QImageReader reader(&file);
151 reader.setAutoTransform(false);
152 const QSize declaredSize = reader.size();
153 if (!declaredSize.isValid() || declaredSize.width() != declaredSize.height() ||
154 declaredSize.width() > MaximumFaceSize ||
155 (expectedSize.isValid() && declaredSize != expectedSize)) {
156 setError(errorMessage,
157 QStringLiteral("A cubemap face exceeds the bounded square dimensions."));
158 return false;
159 }
160 const QImage decoded = reader.read();
161 if (decoded.isNull() || decoded.width() != decoded.height() ||
162 decoded.width() > MaximumFaceSize || decoded.size() != declaredSize ||
163 (expectedSize.isValid() && decoded.size() != expectedSize)) {
164 setError(
165 errorMessage,
166 QStringLiteral("A cubemap face is missing, invalid, or has a mismatched size."));
167 return false;
168 }
169 expectedSize = decoded.size();
170 QTransform orientation;
171 if (face == 2U) {
172 orientation.rotate(90.0);
173 } else if (face == 3U) {
174 orientation.rotate(-90.0);
175 }
176 result.images[face] =
177 decoded.convertToFormat(QImage::Format_RGBA8888).transformed(orientation);
178 }
179 result.displayName = m_entries[index].name;
180 *faces = std::move(result);
181 return true;
