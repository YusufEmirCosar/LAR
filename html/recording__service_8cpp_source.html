9qint64 boundedAccumulation(qint64 accumulated, qint64 delta) noexcept {
10 if (accumulated > MaximumSessionNanoseconds ||
11 delta > MaximumSessionNanoseconds - accumulated) {
12 return MaximumSessionNanoseconds + 1;

58void RecordingService::pauseRecordingAt(qint64 boundaryNanoseconds) {
59 if (m_state != State::Recording)

64 m_accumulatedNanoseconds = boundedAccumulation(
65 m_accumulatedNanoseconds, boundaryNanoseconds - m_segmentStartNanoseconds);

137void RecordingService::recordPackets(const QVector<CapturedPacket> &packets) {
138 if (m_state != State::Recording || packets.isEmpty())
139 return;
140 QString error;
141 for (const CapturedPacket &packet : packets) {
142 const auto timestamp = timestampFor(packet);
143 if (!timestamp) {
144 error = QStringLiteral("Session duration exceeds the supported 365-day limit");
145 }
146 if (!timestamp || !m_transaction.append(*timestamp, packet.data, &error)) {
147 if (m_transaction.isActive()) {
148 pauseRecording();
149 } else {
150 m_accumulatedNanoseconds = 0;
151 m_firstRecordedTimestamp.reset();
152 m_lastRecordedTimestamp.reset();
153 setState(State::Idle);
154 emit recordCountChanged(0);
155 }
156 emit recordingError(error);
157 return;
158 }
159 if (!m_firstRecordedTimestamp)
160 m_firstRecordedTimestamp = *timestamp;
161 m_lastRecordedTimestamp = *timestamp;
162 }
163 emit recordCountChanged(m_transaction.recordCount());
164}

173std::optional<SessionTimestamp>
174RecordingService::timestampFor(const CapturedPacket &packet) const noexcept {
175 qint64 activeNanoseconds = 0;
176 if (m_segmentStartNanoseconds >= 0 &&
177 packet.receivedAtNanoseconds > m_segmentStartNanoseconds) {
178 activeNanoseconds = packet.receivedAtNanoseconds - m_segmentStartNanoseconds;
179 }
180 if (m_accumulatedNanoseconds > MaximumSessionNanoseconds ||
181 activeNanoseconds > MaximumSessionNanoseconds - m_accumulatedNanoseconds) {
182 return std::nullopt;
183 }
184 return SessionTimestamp::fromNanoseconds(m_accumulatedNanoseconds + activeNanoseconds);
185}
