98bool accessorSpan(const ReaderContext &context, int accessorIndex, int elementBytes,
99 AccessorSpan *result) {
100 if (result == nullptr || accessorIndex < 0 || accessorIndex >= context.accessors.size()) {
101 return false;
102 }

104 if (!accessor.value(QStringLiteral("sparse")).isUndefined()) {
105 return false;
106 }
107 int viewIndex = -1;
108 int count = -1;
109 int componentType = -1;
110 if (!integerValue(accessor, QStringLiteral("bufferView"), &viewIndex) ||
111 !integerValue(accessor, QStringLiteral("count"), &count) ||
112 !integerValue(accessor, QStringLiteral("componentType"), &componentType) || count <= 0 ||
113 viewIndex >= context.bufferViews.size() ||
114 !accessor.value(QStringLiteral("type")).isString()) {
115 return false;
116 }
117 const QJsonObject view = context.bufferViews[viewIndex].toObject();
118 int bufferIndex = -1;

134 const qint64 viewEnd = static_cast<qint64>(viewOffset) + viewLength;
135 if (first < viewOffset || required <= 0 || first + required > viewEnd ||
136 first + required > binary.size()) {
137 return false;
138 }
139 *result = {binary.constData() + first, count, stride, componentType,
140 accessor.value(QStringLiteral("type")).toString()};
141 return true;

288 !integerValue(attributes, QStringLiteral("NORMAL"), &normalIndex) ||

305 if (indicesIndex >= context.accessors.size() ||

351 normalTransform(0, 2) * normal.z(),

353 normalTransform(1, 2) * normal.z(),
354 normalTransform(2, 0) * normal.x() + normalTransform(2, 1) * normal.y() +
355 normalTransform(2, 2) * normal.z());

357 float textureV = 0.0F;

359 const char *textureData =
360 textureCoordinates.data + static_cast<qint64>(index) * textureCoordinates.stride;
361 textureU = littleFloat(textureData);

363 }
364 const float normalLengthSquared = normal.lengthSquared();
365 if (!std::isfinite(transformedPosition.x()) || !std::isfinite(transformedPosition.y()) ||
366 !std::isfinite(transformedPosition.z()) || !std::isfinite(normal.x()) ||
367 !std::isfinite(normal.y()) || !std::isfinite(normal.z()) ||

374 !std::isfinite(normal.z())) {
375 return false;
376 }
377 mesh->vertices.insert(mesh->vertices.end(),
378 {transformedPosition.x(), transformedPosition.y(),

381 }

385 quint32 local = static_cast<quint32>(index);
386 if (indicesIndex >= 0) {

388 indices.data + static_cast<qint64>(index) * indices.stride;
389 switch (indices.componentType) {

391 local = static_cast<quint32>(*reinterpret_cast<const unsigned char *>(indexData));
392 break;
393 case 5123:

395 break;
396 case 5125:

398 break;

401 }
402 }

405 return false;
406 }
407 mesh->indices.push_back(static_cast<std::uint32_t>(baseVertex + local));
408 }
409 mesh->draws.push_back({firstIndex, static_cast<std::size_t>(indexCount), material.baseColor,
410 material.baseColorTexture, material.doubleSided});
411 return true;
412}

433bool flattenScene(const ReaderContext &context, const QJsonArray &roots, PlaneModelMesh *mesh) {
434 std::vector<bool> visiting(static_cast<std::size_t>(context.nodes.size()), false);
435 std::vector<bool> visited(static_cast<std::size_t>(context.nodes.size()), false);
436 std::function<bool(int, const QMatrix4x4 &, int)> visit;
437 visit = [&](int index, const QMatrix4x4 &parent, int depth) {
438 if (index < 0 || index >= context.nodes.size() || depth > 128 ||
439 visiting[static_cast<std::size_t>(index)] || visited[static_cast<std::size_t>(index)]) {
440 return false;
441 }
442 visiting[static_cast<std::size_t>(index)] = true;
443 const QJsonObject node = context.nodes[index].toObject();
444 bool transformValid = false;
445 const QMatrix4x4 world = parent * nodeTransform(node, &transformValid);
446 if (!transformValid) {
447 return false;
448 }
449 const int meshIndex = node.value(QStringLiteral("mesh")).toInt(-1);
450 if (meshIndex >= 0 && !appendMesh(context, meshIndex, world, mesh)) {
451 return false;
452 }
453 for (const QJsonValue &child : node.value(QStringLiteral("children")).toArray()) {
454 if (!child.isDouble() || !visit(child.toInt(-1), world, depth + 1)) {
455 return false;
456 }
457 }
458 visiting[static_cast<std::size_t>(index)] = false;
459 visited[static_cast<std::size_t>(index)] = true;
460 return true;
461 };
462 for (const QJsonValue &root : roots) {
463 QMatrix4x4 identity;
464 if (!root.isDouble() || !visit(root.toInt(-1), identity, 0)) {
465 return false;
466 }
467 }
468 return !mesh->empty();

471bool normalizeMesh(PlaneModelMesh *mesh) {
472 QVector3D minimum(std::numeric_limits<float>::max(), std::numeric_limits<float>::max(),
473 std::numeric_limits<float>::max());
474 QVector3D maximum(-std::numeric_limits<float>::max(), -std::numeric_limits<float>::max(),
475 -std::numeric_limits<float>::max());
476 for (std::size_t offset = 0; offset + PlaneModelVertexStrideFloats - 1U < mesh->vertices.size();
477 offset += PlaneModelVertexStrideFloats) {
478 const QVector3D value(mesh->vertices[offset], mesh->vertices[offset + 1U],
479 mesh->vertices[offset + 2U]);
480 minimum.setX(std::min(minimum.x(), value.x()));
481 minimum.setY(std::min(minimum.y(), value.y()));
482 minimum.setZ(std::min(minimum.z(), value.z()));
483 maximum.setX(std::max(maximum.x(), value.x()));
484 maximum.setY(std::max(maximum.y(), value.y()));
485 maximum.setZ(std::max(maximum.z(), value.z()));
486 }
487 const QVector3D extent = maximum - minimum;
488 const float largest = std::max({extent.x(), extent.y(), extent.z()});
489 if (!std::isfinite(largest) || largest <= 1.0e-6F) {
490 return false;
491 }
492 const QVector3D center = minimum + extent * 0.5F;
493 const float scale = 2.0F / largest;
494 mesh->forwardExtentSceneUnits = extent.z() * scale;
495 if (!std::isfinite(mesh->forwardExtentSceneUnits) || mesh->forwardExtentSceneUnits <= 1.0e-6F) {
496 return false;
497 }
498 for (std::size_t offset = 0; offset + PlaneModelVertexStrideFloats - 1U < mesh->vertices.size();
499 offset += PlaneModelVertexStrideFloats) {
500 mesh->vertices[offset] = (mesh->vertices[offset] - center.x()) * scale;
501 mesh->vertices[offset + 1U] = (mesh->vertices[offset + 1U] - center.y()) * scale;
502 mesh->vertices[offset + 2U] = (mesh->vertices[offset + 2U] - center.z()) * scale;
503 if (!std::isfinite(mesh->vertices[offset]) || !std::isfinite(mesh->vertices[offset + 1U]) ||
504 !std::isfinite(mesh->vertices[offset + 2U])) {
505 return false;
506 }
507 }
508 return true;
509}

511bool dataUriBytes(const QString &uri, QByteArray *result) {
512 if (result == nullptr || !uri.startsWith(QStringLiteral("data:"), Qt::CaseInsensitive)) {
513 return false;
514 }
515 const int separator = uri.indexOf(QLatin1Char(','));
516 if (separator <= 5) {
517 return false;
518 }
519 const QByteArray metadata = uri.left(separator).toUtf8().toLower();
520 const QByteArray payload = uri.mid(separator + 1).toUtf8();
521 QByteArray decoded;
522 if (metadata.split(';').contains(QByteArrayLiteral("base64"))) {
523 decoded = QByteArray::fromBase64(payload);
524 if (!payload.isEmpty() && decoded.isEmpty()) {
525 return false;
526 }
527 } else {
528 decoded = QByteArray::fromPercentEncoding(payload);
529 }
530 if (decoded.size() > MaximumBufferBytes) {
531 return false;
532 }
533 *result = std::move(decoded);
534 return true;
535}

605bool splitGlb(const QByteArray &bytes, QByteArray *json, QByteArray *binary,
606 bool *hasBinary) {
607 if (json == nullptr || binary == nullptr || hasBinary == nullptr || bytes.size() < 20 ||
608 littleU32(bytes.constData()) != GlbMagic || littleU32(bytes.constData() + 4) != 2U ||
609 littleU32(bytes.constData() + 8) != static_cast<quint32>(bytes.size())) {
610 return false;
611 }
612 bool foundJson = false;
613 *hasBinary = false;
614 qint64 offset = 12;
615 while (offset + 8 <= bytes.size()) {
616 const quint32 chunkLength =
617 littleU32(bytes.constData() + static_cast<qsizetype>(offset));
618 const quint32 chunkType =
619 littleU32(bytes.constData() + static_cast<qsizetype>(offset + 4));
620 offset += 8;
621 if (chunkLength == 0U || offset + static_cast<qint64>(chunkLength) > bytes.size()) {
622 return false;
623 }
624 const QByteArray chunk = bytes.mid(static_cast<qsizetype>(offset),
625 static_cast<qsizetype>(chunkLength));
626 if (chunkType == JsonChunk && !foundJson) {
627 *json = chunk;
628 foundJson = true;
629 } else if (chunkType == BinaryChunk && !*hasBinary) {
630 *binary = chunk;
631 *hasBinary = true;
632 }
633 offset += static_cast<qint64>(chunkLength);
634 }
635 if (offset != bytes.size() || !foundJson) {

644GlbModelReadResult parseModel(const QJsonObject &root, const QByteArray &glbBinary,
645 bool hasGlbBinary, const QString &sourceDirectory) {

647 .toObject()
648 .value(QStringLiteral("version"))
649 .toString() != QStringLiteral("2.0") ||
650 !root.value(QStringLiteral("extensionsRequired")).toArray().isEmpty()) {
651 return failure(QStringLiteral("The selected glTF model requires unsupported extensions."));
652 }
653 const QJsonArray scenes = root.value(QStringLiteral("scenes")).toArray();
654 const int sceneIndex = root.value(QStringLiteral("scene")).toInt(0);
655 if (sceneIndex < 0 || sceneIndex >= scenes.size() || !scenes[sceneIndex].isObject()) {
656 return failure(QStringLiteral("The selected glTF model has no valid default scene."));
657 }
658

660 root.value(QStringLiteral("bufferViews")).toArray(),
661 root.value(QStringLiteral("materials")).toArray(),

664 root.value(QStringLiteral("images")).toArray(),
665 root.value(QStringLiteral("textures")).toArray(),
666 root.value(QStringLiteral("samplers")).toArray(),
667 {},
668 sourceDirectory};
669 if (!resolveBuffers(root.value(QStringLiteral("buffers")).toArray(), sourceDirectory,
670 glbBinary, hasGlbBinary, &context.binaries)) {
671 return failure(QStringLiteral("The selected glTF model has invalid buffer resources."));
672 }
673
674 auto mesh = std::make_shared<PlaneModelMesh>();
675 if (context.accessors.isEmpty() || context.bufferViews.isEmpty() || context.meshes.isEmpty() ||
676 context.nodes.isEmpty() ||
677 !GlbTextureReader::load(context.bufferViews, context.images, context.textures,
678 context.samplers, context.binaries, context.sourceDirectory,
679 mesh.get()) ||
680 !flattenScene(context,
681 scenes[sceneIndex].toObject().value(QStringLiteral("nodes")).toArray(),
682 mesh.get()) ||
683 !normalizeMesh(mesh.get())) {
684 return failure(QStringLiteral(
685 "The selected glTF model exceeds the supported static mesh subset."));
686 }
687 return {std::move(mesh), {}};

692GlbModelReadResult GlbModelReader::readFile(const QString &path) {
693 const QString suffix = QFileInfo(path).suffix().toLower();
694 QFile file(path);
695 if (!file.open(QIODevice::ReadOnly) || file.size() <= 0 || file.size() > MaximumModelBytes) {
696 return failure(QStringLiteral("The selected glTF model is unavailable or too large."));
697 }
698 const QByteArray bytes = file.readAll();
699 if (bytes.size() != file.size()) {
700 return failure(QStringLiteral("The selected glTF model could not be read completely."));
701 }
702
703 QByteArray jsonBytes;
704 QByteArray binaryBytes;
705 bool hasBinary = false;
706 const bool glbFile =
707 suffix == QStringLiteral("glb") ||
708 (bytes.size() >= 4 && littleU32(bytes.constData()) == GlbMagic);
709 if (glbFile) {
710 if (!splitGlb(bytes, &jsonBytes, &binaryBytes, &hasBinary)) {
711 return failure(QStringLiteral("The selected file is not a valid GLB 2.0 model."));
712 }
713 } else {
714 jsonBytes = bytes;
715 }
716 QJsonParseError parseError;
717 const QJsonDocument document = QJsonDocument::fromJson(jsonBytes, &parseError);
718 if (parseError.error != QJsonParseError::NoError || !document.isObject()) {
719 return failure(QStringLiteral("The selected glTF JSON is invalid."));
720 }
721 return parseModel(document.object(), binaryBytes, hasBinary,
722 QFileInfo(path).absolutePath());
723}
