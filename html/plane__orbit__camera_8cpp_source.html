19void PlaneOrbitCamera::orbit(const QPointF &pixelDelta, const QSize &viewportSize) noexcept {
20 const double width = static_cast<double>(std::max(1, viewportSize.width()));
21 const double height = static_cast<double>(std::max(1, viewportSize.height()));
22 const double horizontalDelta =
23 std::clamp(pixelDelta.x(), -width * 0.25, width * 0.25) * HorizontalOrbitDegrees / width;
24 const double verticalDelta =
25 std::clamp(pixelDelta.y(), -height * 0.25, height * 0.25) * VerticalOrbitDegrees / height;
26 m_azimuthOffsetDegrees = std::remainder(m_azimuthOffsetDegrees + horizontalDelta, 360.0);
27 const double minimumElevation =
28 m_groundConstrained ? MinimumSurfaceElevationDegrees : MinimumElevationDegrees;
29 m_elevationDegrees =
30 std::clamp(m_elevationDegrees + verticalDelta, minimumElevation, MaximumElevationDegrees);
31}

51QVector3D PlaneOrbitCamera::position(double headingRadians) const noexcept {
52 const double azimuth = headingRadians + qDegreesToRadians(m_azimuthOffsetDegrees);
53 const double elevation = qDegreesToRadians(m_elevationDegrees);
