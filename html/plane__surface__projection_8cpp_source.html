27double saturatedProduct(double value, double factor) noexcept {
28 const double magnitude = std::abs(value);
29 if (magnitude > std::numeric_limits<double>::max() / factor) {
30 return std::numeric_limits<double>::max();
31 }
32 return magnitude * factor;
33}

41bool validPosition(double latitude, double longitude) noexcept {
42 return std::isfinite(latitude) && std::isfinite(longitude) && latitude >= -Pi * 0.5 &&

46std::optional<QPointF> localMeters(const GeoCoordinateRadians &coordinate,
47 const Plane &plane) noexcept {
48 if (!validPosition(coordinate.latitude, coordinate.longitude)) {
49 return std::nullopt;
50 }
51 const double position[3]{coordinate.latitude, coordinate.longitude, 0.0};
52 const QPointF eastNorth = LarProjection::geographicToPlaneWorld(position, plane.location, 0.0,
53 plane.location[0], true);
54 if (!std::isfinite(eastNorth.x()) || !std::isfinite(eastNorth.y())) {
55 return std::nullopt;
56 }
57 return eastNorth;
58}

80void applySurfaceLayout(double contentExtentMeters, const std::optional<double> &altitudeMeters,
81 PlaneSurfaceState *result) noexcept {
82 if (altitudeMeters) {
83 const double surfaceHeight = -*altitudeMeters / result->metersPerSceneUnit;
84 result->surfaceHeight = static_cast<float>(
85 std::clamp(surfaceHeight, -static_cast<double>(PlaneSurfaceMaximumCoordinate),
86 static_cast<double>(PlaneSurfaceMaximumCoordinate)));
87 }
88 const double altitudeExtentMeters =
89 altitudeMeters ? saturatedProduct(*altitudeMeters, AltitudeExtentFactor) : 0.0;
90 const double requiredHalfExtentMeters =
91 std::max({MinimumExtentMeters, contentExtentMeters * SurfaceMargin, altitudeExtentMeters});
92 const double requiredHalfExtent = requiredHalfExtentMeters / result->metersPerSceneUnit;
93 result->surfaceHalfExtent =
94 std::clamp(static_cast<float>(requiredHalfExtent), MinimumSurfaceHalfExtentSceneUnits,
95 PlaneSurfaceMaximumCoordinate);
96 const double targetGridSpacing =
97 std::max(result->metersPerSceneUnit * 0.5,
98 requiredHalfExtentMeters /
99 (static_cast<double>(PlaneSurfaceGridHalfLineCount) * GridCoverageFraction));
100 const double maximumGridSpacingMeters = static_cast<double>(PlaneSurfaceMaximumCoordinate) *
101 result->metersPerSceneUnit /
102 static_cast<double>(PlaneSurfaceGridHalfLineCount);
103 const double maximumQuantizedSpacing =
104 std::exp2(std::floor(std::log2(maximumGridSpacingMeters)));
105 result->gridSpacingMeters =
106 GridGeometryBuilder::gridStep(std::min(targetGridSpacing, maximumQuantizedSpacing));
107 result->gridSpacingSceneUnits =
108 static_cast<float>(result->gridSpacingMeters / result->metersPerSceneUnit);
109}

111PlaneSurfaceZone projectedZone(const LarZoneDefinition &definition, const QPointF &centerMeters,
112 double metersPerSceneUnit) noexcept {
113 PlaneSurfaceZone zone;
114 zone.centerXZ = {static_cast<float>(centerMeters.x() / metersPerSceneUnit),
115 static_cast<float>(-centerMeters.y() / metersPerSceneUnit)};
116 zone.innerRadius = static_cast<float>(definition.innerRadiusMeters / metersPerSceneUnit);
117 zone.outerRadius = static_cast<float>(definition.outerRadiusMeters / metersPerSceneUnit);
118 zone.startBearingRadians = static_cast<float>(definition.startBearingRadians);
119 zone.spanRadians = static_cast<float>(definition.spanRadians);
120 zone.visible = true;
121 zone.fullCircle = definition.spanRadians >= LarGeodesicGeometry::TwoPi - 1.0e-9;
122 return zone;
123}

125float targetMarkerScale(const PlaneSurfaceState &state) noexcept {
126 const float referenceRadius = state.inZone.visible ? state.inZone.outerRadius
127 : state.inRange.visible ? state.inRange.outerRadius
128 : 0.0F;
129 const float minimumScale =
130 static_cast<float>(MinimumTargetHalfWidthMeters / state.metersPerSceneUnit);
131 const float maximumScale =
132 static_cast<float>(MaximumTargetHalfWidthMeters / state.metersPerSceneUnit);
133 if (!std::isfinite(referenceRadius) || referenceRadius <= 0.0F) {
134 return minimumScale;
135 }
136 return std::clamp(referenceRadius * TargetRadiusFraction, minimumScale, maximumScale);
137}

141PlaneSurfaceState
142PlaneSurfaceProjection::project(const LarSceneState &scene, double metersPerSceneUnit,
143 const std::optional<GeoCoordinateRadians> &groundOrigin) noexcept {
144 PlaneSurfaceState result;
145 result.metersPerSceneUnit = std::isfinite(metersPerSceneUnit) && metersPerSceneUnit > 0.0
146 ? metersPerSceneUnit
147 : PlaneAircraftScale::DefaultMetersPerSceneUnit;
148 std::optional<double> altitudeMeters;
149 if (scene.hasScene && hasFields(scene.availableFields, {StateField::Location2}) &&
150 std::isfinite(scene.plane.location[2])) {
151 altitudeMeters = scene.plane.location[2];
152 }
153 applySurfaceLayout(MinimumExtentMeters, altitudeMeters, &result);
154 if (!scene.hasScene ||
155 !hasFields(scene.availableFields, {StateField::Location0, StateField::Location1}) ||
156 !validPosition(scene.plane.location[0], scene.plane.location[1])) {
157 return result;
158 }
159 result.geographicAnchorValid = true;
160
161 const LarZoneValidationResult validation =
162 LarZoneInputValidator().validate(scene.target, scene.availableFields);
163 std::optional<QPointF> inRangeCenter;
164 std::optional<QPointF> inZoneCenter;
165 std::optional<QPointF> targetCenter;
166 double extentMeters = MinimumExtentMeters;
167 if (hasFields(scene.availableFields, {StateField::IzPos0, StateField::IzPos1})) {
168 targetCenter = localMeters({scene.target.iz_pos[0], scene.target.iz_pos[1]}, scene.plane);
169 }
170 if (!targetCenter &&
171 hasFields(scene.availableFields, {StateField::IrPos0, StateField::IrPos1})) {
172 targetCenter = localMeters({scene.target.ir_pos[0], scene.target.ir_pos[1]}, scene.plane);
173 }
174 if (targetCenter) {
175 includeExtent(*targetCenter, 0.0, &extentMeters);
176 }
177 if (validation.inRange) {
178 inRangeCenter = localMeters(validation.inRange->center, scene.plane);
179 if (inRangeCenter) {
180 includeExtent(*inRangeCenter, validation.inRange->outerRadiusMeters, &extentMeters);
181 }
182 }
183 if (validation.inZone) {
184 inZoneCenter = localMeters(validation.inZone->center, scene.plane);
185 if (inZoneCenter) {
186 includeExtent(*inZoneCenter, validation.inZone->outerRadiusMeters, &extentMeters);
187 }
188 }
189
190 applySurfaceLayout(extentMeters, altitudeMeters, &result);
191
192 if (groundOrigin) {
193 const std::optional<QPointF> aircraftGround =
194 aircraftGroundMeters(scene.plane, *groundOrigin);
195 if (aircraftGround) {
196 result.groundOriginXZ = {
197 static_cast<float>(-aircraftGround->x() / result.metersPerSceneUnit),
198 static_cast<float>(aircraftGround->y() / result.metersPerSceneUnit)};
199 const float majorPeriod = result.gridSpacingSceneUnits * 5.0F;
200 if (std::isfinite(majorPeriod) && majorPeriod > 0.0F) {
201 result.gridPhaseXZ = {std::remainder(result.groundOriginXZ.x(), majorPeriod),
202 std::remainder(result.groundOriginXZ.y(), majorPeriod)};
203 }
204 }
205 }
206
207 if (validation.inRange && inRangeCenter) {
208 result.inRange =
209 projectedZone(*validation.inRange, *inRangeCenter, result.metersPerSceneUnit);
210 }
211 if (validation.inZone && inZoneCenter) {
212 result.inZone = projectedZone(*validation.inZone, *inZoneCenter, result.metersPerSceneUnit);
213 }
214 result.targetMarkerScale = targetMarkerScale(result);
215
216 if (targetCenter) {
217 result.targetXZ = {static_cast<float>(targetCenter->x() / result.metersPerSceneUnit),
218 static_cast<float>(-targetCenter->y() / result.metersPerSceneUnit)};
219 result.targetVisible = true;
220 }
221 return result;
222}
