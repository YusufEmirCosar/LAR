47PacketMapping::PacketMapping(QVector<FieldBinding> bindings, QByteArray json)
48 : m_availableFields(StateField::Count) {
49 QBitArray seenFields(StateField::Count);
50 for (qsizetype i = 0; i < bindings.size(); ++i) {
51 const auto &binding = bindings.at(i);
52 if (binding.fieldId < 0 || binding.fieldId >= StateField::Count || binding.offset < 0 ||
53 (binding.size != 4 && binding.size != 8) ||
54 binding.offset > std::numeric_limits<int>::max() - binding.size ||
55 binding.offset > MaximumPacketSize - binding.size ||
56 seenFields.testBit(binding.fieldId)) {
57 return;
58 }
59 for (qsizetype j = 0; j < i; ++j) {
60 const auto &other = bindings.at(j);
61 if (binding.offset < other.offset + other.size &&
62 binding.offset + binding.size > other.offset) {
63 return;
64 }
65 }
66 seenFields.setBit(binding.fieldId);
67 m_minimumPacketSize = qMax(m_minimumPacketSize, binding.offset + binding.size);
68 }
69 if (bindings.isEmpty() || json.isEmpty())
70 return;
71 const int dlzFieldCount = int(seenFields.testBit(StateField::DlzRangeNm)) +
72 int(seenFields.testBit(StateField::DlzAspectDegrees)) +
73 int(seenFields.testBit(StateField::DlzAltitudeFeet));
74 if (dlzFieldCount != 0 && dlzFieldCount != 3)
75 return;
76 m_bindings = std::move(bindings);
77 m_availableFields = std::move(seenFields);
78 m_json = std::move(json);
79 m_valid = true;
80}

98bool PacketMapping::decode(const QByteArray &packet, DecodedState *state, QString *error) const {
99 if (!isValid() || state == nullptr) {
100 if (error)
101 *error = QStringLiteral("A valid mapping and output state are required");
102 return false;
103 }
104 if (packet.size() < m_minimumPacketSize) {
105 if (error) {
106 *error = QStringLiteral("Mapped packet requires at least %1 bytes; received %2")
107 .arg(m_minimumPacketSize)
108 .arg(packet.size());
109 }
110 return false;
111 }
112 DecodedState decoded;
113 decoded.availableFields = m_availableFields;
114 for (const auto &binding : m_bindings) {
115 StateField::setValue(&decoded, binding.fieldId,
116 readNumber(packet.constData() + binding.offset, binding.size));
117 }
118 QString validationError;
119 if (!StateValidator::validate(decoded, &validationError)) {
120 if (error)
121 *error = validationError;
122 return false;
123 }
124 *state = std::move(decoded);
125 return true;
126}
