23template <typename Worker> bool shutdownAndRehome(Worker *worker, QThread *destination) {
24 if (!worker)
25 return true;
26 std::atomic_bool moved = false;
27 const bool invoked = QMetaObject::invokeMethod(
28 worker,
29 [worker, destination, &moved] {
30 worker->shutdown();
31 const bool rehomed =
32 worker->moveToThread(destination) && worker->thread() == destination;
33 moved.store(rehomed, std::memory_order_release);
34 },
35 Qt::BlockingQueuedConnection);
36 return invoked && moved.load(std::memory_order_acquire);
37}

470void ThreadedApplicationRuntime::shutdown() {
471 Q_ASSERT(QThread::currentThread() == thread());
472 if (m_shuttingDown)
473 return;
474 m_shuttingDown = true;
475 m_activeStateSource = RuntimeStateSource::None;
476 ++m_nextStateGeneration;
477 QThread *const ownerThread = thread();
478 bool workersRehomed = true;
479 if (m_networkThread.isRunning())
480 workersRehomed &= shutdownAndRehome(m_networkWorker, ownerThread);
481 if (m_sessionThread.isRunning()) {
482 workersRehomed &= shutdownAndRehome(m_recordingWorker, ownerThread);
483 workersRehomed &= shutdownAndRehome(m_playbackWorker, ownerThread);
484 }
485 if (m_persistenceThread.isRunning())
486 workersRehomed &= shutdownAndRehome(m_persistenceWorker, ownerThread);
487
488 if (!workersRehomed) {
489 qFatal("A runtime worker could not be rehomed before thread shutdown");
490 }
491 m_networkThread.quit();
492 m_sessionThread.quit();
493 m_persistenceThread.quit();
494 m_networkThread.wait();
495 m_sessionThread.wait();
496 m_persistenceThread.wait();
497 delete std::exchange(m_networkWorker, nullptr);
498 delete std::exchange(m_recordingWorker, nullptr);
499 delete std::exchange(m_playbackWorker, nullptr);
500 delete std::exchange(m_persistenceWorker, nullptr);
501}
