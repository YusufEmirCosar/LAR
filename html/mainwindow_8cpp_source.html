213void MainWindow::updateStateFromViewModel() {
214 const auto &vm = m_app.viewModel();
215 m_valuesPanel->render(vm);
216 if (!vm.hasState()) {
217 m_viewport->clearState();
218 m_viewport->setDlzInputs({}, false);
219 m_viewportControls->setNavigationReadout({}, {}, {});
220 updateCameraControls();
221 return;
222 }
223
224 m_viewport->setState(vm.plane(), vm.target(), vm.availableFields());
225 const auto &fields = vm.availableFields();
226 const auto isAvailable = [&fields](int id) {
227 return id >= 0 && id < fields.size() && fields.testBit(id);
228 };
229 const bool hasDlzTelemetry = isAvailable(StateField::DlzRangeNm) &&
230 isAvailable(StateField::DlzAspectDegrees) &&
231 isAvailable(StateField::DlzAltitudeFeet);
232 m_viewport->setDlzInputs(vm.dlzInputs(), hasDlzTelemetry);
233 m_viewportControls->setNavigationReadout(vm.plane(), vm.target(), vm.availableFields());
234 updateCameraControls();
235}

237void MainWindow::switchMode(int modeIndex) {
238 if (!m_modeStack)
239 return;
240 if (modeIndex == OfflineMode && !m_app.stopOnline()) {
241 m_onlineButton->setChecked(true);
242 return;
243 }
244 m_modeStack->setCurrentIndex(modeIndex);
245 if (modeIndex == OnlineMode) {
246 m_onlineButton->setChecked(true);
247 m_app.closeSession();
248 statusBar()->showMessage(QStringLiteral("Online mode"));
249 } else {
250 m_offlineButton->setChecked(true);
251 m_app.closeSession();
252 statusBar()->showMessage(QStringLiteral("Offline mode: select a .lar file"));
253 }

267void MainWindow::updateCameraControls() {
268 if (m_viewportControls == nullptr) {
269 return;
270 }
271 const auto &viewModel = m_app.viewModel();
272 const bool yawAvailable =
273 !viewModel.hasState() || (StateField::Euler0 < viewModel.availableFields().size() &&
274 viewModel.availableFields().testBit(StateField::Euler0));
275 if (!yawAvailable && m_viewport->turnWithPlane()) {
276 m_viewport->setTurnWithPlane(false);
277 }
278 m_viewportControls->setTurnWithPlaneAvailability(yawAvailable);
279}
