20bool EarthCameraPolicy::fieldsChanged(const Plane &previousPlane, const Target &previousTarget,
21 const QBitArray &previousFields, bool hadPreviousState,
22 const Plane &nextPlane, const Target &nextTarget,
23 const QBitArray &nextFields,
24 std::initializer_list<int> fields) noexcept {
25 for (int field : fields) {
26 const bool wasAvailable = hadPreviousState && field >= 0 && field < previousFields.size() &&
27 previousFields.testBit(field);
28 const bool isAvailable =
29 field >= 0 && field < nextFields.size() && nextFields.testBit(field);
30 if (wasAvailable != isAvailable)
31 return true;
32 if (isAvailable && StateField::value(previousPlane, previousTarget, field) !=
33 StateField::value(nextPlane, nextTarget, field)) {
34 return true;
35 }
36 }
37 return false;
38}

41EarthCameraPolicy::trackedCoordinate(const ViewportCameraState &cameraState,
42 bool hasScene) noexcept {
43 constexpr double Pi = LarGeodesicGeometry::Pi;
44 const double latitude = cameraState.anchorRadians[0];
45 const double longitude = cameraState.anchorRadians[1];
46 if (!hasScene || !cameraState.trackingActive || !cameraState.hasAnchor ||
47 !std::isfinite(latitude) || !std::isfinite(longitude) || latitude < -Pi * 0.5 ||
48 latitude > Pi * 0.5 || longitude < -Pi || longitude > Pi) {
49 return std::nullopt;
50 }

58std::vector<EarthCameraPolicy::Coordinate>
59EarthCameraPolicy::fitCoordinates(const Plane &plane, const Target &target,
60 const QBitArray &available, bool hasScene) {
61 std::vector<Coordinate> coordinates;
62 if (!hasScene)
63 return coordinates;
64 const auto append = [&coordinates](double latitude, double longitude) {
65 coordinates.emplace_back(latitude, longitude);
66 };
67 if (hasFields(available, {StateField::Location0, StateField::Location1})) {
68 append(plane.location[0], plane.location[1]);
69 }
70 if (hasFields(available, {StateField::IzPos0, StateField::IzPos1})) {
71 append(target.iz_pos[0], target.iz_pos[1]);
72 }
73 if (hasFields(available, {StateField::IrPos0, StateField::IrPos1})) {
74 append(target.ir_pos[0], target.ir_pos[1]);
75 }
76 if (hasFields(available, {StateField::IrPos0, StateField::IrPos1, StateField::IrR})) {
77 for (const auto &point :
78 LarGeodesicGeometry::arc({target.ir_pos[0], target.ir_pos[1]}, target.ir_r, 0.0,
79 LarGeodesicGeometry::TwoPi, 64)) {
80 append(point.latitude, point.longitude);
81 }
82 }
83 if (hasFields(available, {StateField::IzPos0, StateField::IzPos1, StateField::IzTheta1,
84 StateField::IzTheta2, StateField::IzR2})) {
85 const double span =
86 LarGeodesicGeometry::positiveAngularSpan(target.iz_theta1, target.iz_theta2);
87 for (const auto &point : LarGeodesicGeometry::arc(
88 {target.iz_pos[0], target.iz_pos[1]}, target.iz_r2, target.iz_theta1, span, 64)) {
89 append(point.latitude, point.longitude);
90 }
91 }
92 return coordinates;
93}

102double EarthCameraPolicy::navigationMetersPerPixel(const lar::map::MapCamera &camera, int width,
103 int height) noexcept {
104 if (width <= 1 || height <= 1)
105 return 0.0;
106 constexpr double HalfPixel = 0.5;
107 const QPointF center(width * 0.5, height * 0.5);
108 double longitudeA = 0.0;
109 double latitudeA = 0.0;
110 double longitudeB = 0.0;
111 double latitudeB = 0.0;
112 if (camera.presentation() == lar::map::MapPresentation::Sphere) {
113 if (!camera.sphereCoordinateAt({center.x() - HalfPixel, center.y()}, width, height,
114 longitudeA, latitudeA) ||
115 !camera.sphereCoordinateAt({center.x() + HalfPixel, center.y()}, width, height,
116 longitudeB, latitudeB)) {
117 return 0.0;
118 }
119 } else {
120 const QPointF geographicA = lar::map::MapProjection::unproject(
121 camera.screenToMercator({center.x() - HalfPixel, center.y()}, width, height));
122 const QPointF geographicB = lar::map::MapProjection::unproject(
123 camera.screenToMercator({center.x() + HalfPixel, center.y()}, width, height));
124 longitudeA = geographicA.x();
125 latitudeA = geographicA.y();
126 longitudeB = geographicB.x();
127 latitudeB = geographicB.y();
128 }
129 if (!std::isfinite(longitudeA) || !std::isfinite(latitudeA) || !std::isfinite(longitudeB) ||
130 !std::isfinite(latitudeB)) {
131 return 0.0;
132 }
133
134 const double latitudeARadians = qDegreesToRadians(latitudeA);
135 const double latitudeBRadians = qDegreesToRadians(latitudeB);
136 const double deltaLatitude = latitudeBRadians - latitudeARadians;
137 const double deltaLongitude =
138 qDegreesToRadians(lar::map::MapProjection::wrapLongitude(longitudeB - longitudeA));
139 const double sineLatitude = std::sin(deltaLatitude * 0.5);
140 const double sineLongitude = std::sin(deltaLongitude * 0.5);
141 const double haversine = std::clamp(
142 sineLatitude * sineLatitude +
143 std::cos(latitudeARadians) * std::cos(latitudeBRadians) * sineLongitude * sineLongitude,
144 0.0, 1.0);
145 const double distance =
146 LarGeodesicGeometry::EarthRadiusMeters * 2.0 *
147 std::atan2(std::sqrt(haversine), std::sqrt(std::max(0.0, 1.0 - haversine)));
148 return std::isfinite(distance) && distance > 0.0 ? distance : 0.0;
149}
