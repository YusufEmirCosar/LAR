32bool validCommonInputs(const ShooterState &shooter, const TargetState &target,

34 const auto fail = [error](const QString &message) {

36 *error = message;
37 }
38 return false;

40 if (!finite(shooter.pos) || !finite(shooter.vel) || !finite(target.pos) ||
41 !finite(target.vel)) {
42 return fail(QStringLiteral("DLZ state vectors must be finite"));

45 !std::isfinite(shooter.phi) || !std::isfinite(target.psi)) {
46 return fail(QStringLiteral("DLZ attitudes must be finite"));

66 return fail(QStringLiteral("Weapon values must be finite and positive"));

73std::optional<Geometry> calculateGeometry(const ShooterState &shooter, const TargetState &target,
74 const Atmosphere &atmosphere, const WeaponModel &weapon,
75 QString *error) {
76 if (!validCommonInputs(shooter, target, atmosphere, weapon, error)) {
77 return std::nullopt;
78 }
79
80 const Vec3 shooterToTarget = subtract(target.pos, shooter.pos);
81 const Vec3 targetToShooter = subtract(shooter.pos, target.pos);
82 const double rangeMeters = norm(shooterToTarget);
83 if (!std::isfinite(rangeMeters) || rangeMeters <= 1e-9) {
84 if (error != nullptr) {
85 *error = QStringLiteral("Shooter and target must have non-zero separation");
86 }
87 return std::nullopt;
88 }
89
90 const Vec3 relativeVelocity = subtract(target.vel, shooter.vel);
91 const double rangeRateMetersPerSecond = dot(shooterToTarget, relativeVelocity) / rangeMeters;
92 const double targetSpeed = norm(target.vel);
93 if (!std::isfinite(targetSpeed) || targetSpeed <= 1e-9) {
94 if (error != nullptr) {
95 *error = QStringLiteral("Target velocity is required for aspect");
96 }
97 return std::nullopt;
98 }
99
100 const double targetToShooterNorm = norm(targetToShooter);
101 const double aspectCosine = std::clamp(
102 dot(target.vel, targetToShooter) / (targetSpeed * targetToShooterNorm), -1.0, 1.0);
103 const double aspect = std::acos(aspectCosine);
104
105 const double horizontalRange = std::hypot(shooterToTarget.x, shooterToTarget.y);
106 const double losAzimuth = std::atan2(shooterToTarget.y, shooterToTarget.x);
107 const double losElevation = std::atan2(-shooterToTarget.z, horizontalRange);
108 const Vec3 forward = forwardVector(shooter);
109 const double boresightCosine =
110 std::clamp(dot(forward, shooterToTarget) / rangeMeters, -1.0, 1.0);
111 const double offBoresight = std::acos(boresightCosine);
112
113 Geometry geometry;
114 geometry.rangeNm = rangeMeters / units::MetersPerNauticalMile;
115 geometry.rangeRateKnots = rangeRateMetersPerSecond / units::MetersPerSecondPerKnot;
116 geometry.aspectRadians = aspect;
117 geometry.antennaTrainAngleRadians = offBoresight;
118 geometry.offBoresightRadians = offBoresight;
119 geometry.losAzimuthRadians = losAzimuth;
120 geometry.losElevationRadians = losElevation;
121 geometry.altitudeDifferenceFeet = -(target.pos.z - shooter.pos.z) * units::FeetPerMeter;
122
123 if (!std::isfinite(geometry.rangeNm) || geometry.rangeNm <= 0.0 ||
124 !std::isfinite(geometry.rangeRateKnots) || !std::isfinite(geometry.aspectRadians) ||
125 !std::isfinite(geometry.antennaTrainAngleRadians) ||
126 !std::isfinite(geometry.offBoresightRadians) ||
127 !std::isfinite(geometry.losAzimuthRadians) ||
128 !std::isfinite(geometry.losElevationRadians) ||
129 !std::isfinite(geometry.altitudeDifferenceFeet) || aspect < 0.0 || aspect > Pi) {
130 if (error != nullptr) {
131 *error = QStringLiteral("Derived DLZ geometry is invalid");
132 }
133 return std::nullopt;
134 }
135 return geometry;
136}
