95bool ApplicationFacade::loadMapping(const QString &path) {
96 if (m_state.listening || m_state.hasRecordingSession) {
97 return reportError(
98 QStringLiteral("Stop online capture before replacing the packet mapping"));

106 if (synchronous)
107 applyResult(*synchronous);
108 if (!dispatch.accepted && !dispatch.rejectionReason.isEmpty()) {
109 reportError(dispatch.rejectionReason);

396void ApplicationFacade::applyResult(const RecordingSaveResult &result) {
397 if (!result.saved) {
398 setRecordingOperationState(RecordingOperationState::FailedRetained);
399 reportError(result.error);
400 return;
401 }
402
403 setRecordingOperationState(RecordingOperationState::Idle);
404 if (!result.finalSave)
405 return;
406
407 m_stopOnlineAfterFinalSave = true;
408 if (!m_state.hasRecordingSession) {
409 m_stopOnlineAfterFinalSave = false;
410 if (m_state.listening)
411 m_sources->stopOnline();
412 }
413 m_viewModel.setStatusText(QStringLiteral("Recording saved successfully"));
414}
