28std::array<double, 3> unitSphere(const SourceCoordinate &coordinate) {
29 const double longitude = coordinate.longitudeDegrees * Pi / 180.0;
30 const double latitude = coordinate.latitudeDegrees * Pi / 180.0;
31 const double cosineLatitude = std::cos(latitude);
32 return {cosineLatitude * std::sin(longitude), std::sin(latitude),
33 cosineLatitude * std::cos(longitude)};
34}

36bool edgeNeedsRefinement(const SourceCoordinate &first, const SourceCoordinate &second) {
37 const std::array<double, 3> firstUnit = unitSphere(first);
38 const std::array<double, 3> secondUnit = unitSphere(second);
39 const double dot = std::clamp(firstUnit[0] * secondUnit[0] + firstUnit[1] * secondUnit[1] +
40 firstUnit[2] * secondUnit[2],
41 -1.0, 1.0);
42 if (std::acos(dot) * 180.0 / Pi > MaximumSphereEdgeDegrees) {
43 return true;
44 }
45
46 const SourceCoordinate midpoint{(first.longitudeDegrees + second.longitudeDegrees) * 0.5,
47 (first.latitudeDegrees + second.latitudeDegrees) * 0.5};
48 const std::array<double, 3> midpointUnit = unitSphere(midpoint);
49 const double deltaX = midpointUnit[0] - (firstUnit[0] + secondUnit[0]) * 0.5;
50 const double deltaY = midpointUnit[1] - (firstUnit[1] + secondUnit[1]) * 0.5;
51 const double deltaZ = midpointUnit[2] - (firstUnit[2] + secondUnit[2]) * 0.5;
52 return std::sqrt(deltaX * deltaX + deltaY * deltaY + deltaZ * deltaZ) > MaximumSphereChordError;
53}

55std::uint64_t edgeKey(std::uint32_t first, std::uint32_t second) {
56 const std::uint32_t low = std::min(first, second);
57 const std::uint32_t high = std::max(first, second);
58 return (static_cast<std::uint64_t>(low) << 32U) | static_cast<std::uint64_t>(high);
59}

70bool refineForSphere(PlanarPolygonMesh &mesh, std::size_t remainingVertexCount,
71 std::size_t remainingIndexCount) {
72 if (mesh.empty() || mesh.fillIndices.size() % 3U != 0U ||
73 mesh.coordinates.size() > remainingVertexCount ||
74 mesh.fillIndices.size() > remainingIndexCount) {
75 return false;
76 }
77
78 std::vector<std::uint32_t> current = mesh.fillIndices;
79 for (std::size_t pass = 0U; pass < MaximumRefinementPasses; ++pass) {
80 bool subdivided = false;
81 std::vector<std::uint32_t> next;
82 next.reserve(std::min(remainingIndexCount, current.size() <= remainingIndexCount / 2U
83 ? current.size() * 2U
84 : remainingIndexCount));
85 std::unordered_map<std::uint64_t, std::uint32_t> midpoints;
86 midpoints.reserve(current.size() / 2U);
87 bool limitExceeded = false;
88
89 const auto midpointIndex = [&](std::uint32_t first, std::uint32_t second) {
90 const std::uint64_t key = edgeKey(first, second);
91 const auto existing = midpoints.find(key);
92 if (existing != midpoints.end()) {
93 return existing->second;
94 }
95 if (mesh.coordinates.size() >= remainingVertexCount ||
96 mesh.coordinates.size() >=
97 static_cast<std::size_t>(std::numeric_limits<std::uint32_t>::max())) {
98 limitExceeded = true;
99 return first;
100 }
101 const SourceCoordinate &a = mesh.coordinates[first];
102 const SourceCoordinate &b = mesh.coordinates[second];
103 const std::uint32_t result = static_cast<std::uint32_t>(mesh.coordinates.size());
104 mesh.coordinates.push_back({(a.longitudeDegrees + b.longitudeDegrees) * 0.5,
105 (a.latitudeDegrees + b.latitudeDegrees) * 0.5});
106 midpoints.emplace(key, result);
107 return result;
108 };
109
110 for (std::size_t offset = 0U; offset < current.size(); offset += 3U) {
111 const std::uint32_t a = current[offset];
112 const std::uint32_t b = current[offset + 1U];
113 const std::uint32_t c = current[offset + 2U];
114 if (a >= mesh.coordinates.size() || b >= mesh.coordinates.size() ||
115 c >= mesh.coordinates.size()) {
116 return false;
117 }
118 const bool splitAb = edgeNeedsRefinement(mesh.coordinates[a], mesh.coordinates[b]);
119 const bool splitBc = edgeNeedsRefinement(mesh.coordinates[b], mesh.coordinates[c]);
120 const bool splitCa = edgeNeedsRefinement(mesh.coordinates[c], mesh.coordinates[a]);
121 const unsigned mask = (splitAb ? 1U : 0U) | (splitBc ? 2U : 0U) | (splitCa ? 4U : 0U);
122 if (mask == 0U) {
123 if (!appendIndices(next, {a, b, c}, remainingIndexCount)) {
124 return false;
125 }
126 continue;
127 }
128
129 subdivided = true;
130 const std::uint32_t ab = splitAb ? midpointIndex(a, b) : 0U;
131 const std::uint32_t bc = splitBc ? midpointIndex(b, c) : 0U;
132 const std::uint32_t ca = splitCa ? midpointIndex(c, a) : 0U;
133 if (limitExceeded) {
134 return false;
135 }
136 bool appended = false;
137 switch (mask) {
138 case 1U:
139 appended = appendIndices(next, {a, ab, c, ab, b, c}, remainingIndexCount);
140 break;
141 case 2U:
142 appended = appendIndices(next, {a, b, bc, a, bc, c}, remainingIndexCount);
143 break;
144 case 3U:
145 appended =
146 appendIndices(next, {b, bc, ab, a, ab, c, ab, bc, c}, remainingIndexCount);
147 break;
148 case 4U:
149 appended = appendIndices(next, {a, b, ca, ca, b, c}, remainingIndexCount);
150 break;
151 case 5U:
152 appended =
153 appendIndices(next, {a, ab, ca, ab, b, ca, b, c, ca}, remainingIndexCount);
154 break;
155 case 6U:
156 appended =
157 appendIndices(next, {c, ca, bc, a, b, ca, b, bc, ca}, remainingIndexCount);
158 break;
159 case 7U:
160 appended = appendIndices(next, {a, ab, ca, ab, b, bc, ca, bc, c, ab, bc, ca},
161 remainingIndexCount);
162 break;
163 default:
164 break;
165 }
166 if (!appended) {
167 return false;
168 }
169 }
170
171 current = std::move(next);
172 if (!subdivided) {
173 break;
174 }
175 }
176 mesh.fillIndices = std::move(current);
177 return true;
178}

201bool forEachTriangleCell(const MapMesh &mesh, std::size_t triangle, Callback callback) {
202 const std::size_t indexOffset = triangle * 3U;
203 if (indexOffset + 2U >= mesh.mercatorFillIndices.size()) {
204 return false;
205 }
206 std::array<double, 3> longitude{};
207 std::array<double, 3> latitude{};
208 for (std::size_t corner = 0U; corner < 3U; ++corner) {
209 const std::size_t vertex = mesh.mercatorFillIndices[indexOffset + corner];
210 const std::size_t vertexOffset = vertex * 3U;
211 if (vertexOffset + 1U >= mesh.vertices.size()) {
212 return false;
213 }
214 longitude[corner] = mesh.vertices[vertexOffset];
215 latitude[corner] = mesh.vertices[vertexOffset + 1U];
216 }
217 longitude[1] = MapProjection::unwrapLongitude(longitude[1], longitude[0]);
218 longitude[2] = MapProjection::unwrapLongitude(longitude[2], longitude[0]);
219 const auto [minimumLongitude, maximumLongitude] =
220 std::minmax_element(longitude.cbegin(), longitude.cend());
221 const auto [minimumLatitude, maximumLatitude] =
222 std::minmax_element(latitude.cbegin(), latitude.cend());
223 if (*maximumLongitude - *minimumLongitude > 360.0 || *minimumLatitude < -90.0 ||
224 *maximumLatitude > 90.0) {
225 return false;
226 }
227
228 const int west = static_cast<int>(std::floor(*minimumLongitude));
229 const int east = static_cast<int>(std::floor(*maximumLongitude));
230 const int south = boundedLatitudeCell(*minimumLatitude);
231 const int north = boundedLatitudeCell(*maximumLatitude);
232 for (int latitudeCell = south; latitudeCell <= north; ++latitudeCell) {
233 for (int longitudeCell = west; longitudeCell <= east; ++longitudeCell) {
234 const std::size_t index =
235 static_cast<std::size_t>(latitudeCell) * MapLandLongitudeCellCount +
236 static_cast<std::size_t>(normalizedLongitudeCell(longitudeCell));
237 if (!callback(index)) {
238 return false;
239 }
240 }
241 }
242 return true;
243}

245bool buildLandIndex(MapMesh &mesh) {
246 if (mesh.mercatorFillIndices.empty() || mesh.mercatorFillIndices.size() % 3U != 0U) {
247 return false;
248 }
249 std::vector<std::uint32_t> counts(MapLandCellCount, 0U);
250 std::size_t referenceCount = 0U;
251 const std::size_t triangleCount = mesh.mercatorFillIndices.size() / 3U;
252 for (std::size_t triangle = 0U; triangle < triangleCount; ++triangle) {
253 const bool counted = forEachTriangleCell(mesh, triangle, [&](std::size_t cell) {
254 if (counts[cell] == std::numeric_limits<std::uint32_t>::max() ||
255 referenceCount >= limits::MaximumLandTriangleReferenceCount) {
256 return false;
257 }
258 ++counts[cell];
259 ++referenceCount;
260 return true;
261 });
262 if (!counted) {
263 return false;
264 }
265 }
266
267 mesh.landCellRanges.resize(MapLandCellCount);
268 std::size_t firstReference = 0U;
269 for (std::size_t cell = 0U; cell < MapLandCellCount; ++cell) {
270 if (firstReference > std::numeric_limits<std::uint32_t>::max()) {
271 return false;
272 }
273 mesh.landCellRanges[cell] = {static_cast<std::uint32_t>(firstReference), counts[cell]};
274 firstReference += counts[cell];
275 }
276 if (firstReference != referenceCount) {
277 return false;
278 }
279
280 mesh.landTriangleReferences.resize(referenceCount);
281 std::vector<std::uint32_t> cursors(MapLandCellCount, 0U);
282 for (std::size_t cell = 0U; cell < MapLandCellCount; ++cell) {
283 cursors[cell] = mesh.landCellRanges[cell].firstReference;
284 }
285 for (std::size_t triangle = 0U; triangle < triangleCount; ++triangle) {
286 const bool stored = forEachTriangleCell(mesh, triangle, [&](std::size_t cell) {
287 const std::size_t destination = cursors[cell]++;
288 if (destination >= mesh.landTriangleReferences.size() ||
289 triangle > std::numeric_limits<std::uint32_t>::max()) {
290 return false;
291 }
292 mesh.landTriangleReferences[destination] = static_cast<std::uint32_t>(triangle);
293 return true;
294 });
295 if (!stored) {
296 return false;
297 }
298 }
299 return mapLandIndexIsValid(mesh);
300}

304MapMeshCompileResult MapMeshCompiler::compile(const SourceMap &source) {
305 if (source.empty()) {
306 return {{}, QStringLiteral("The map source contains no geometry.")};
307 }
308
309 MapMesh output;
310 try {
311 output.vertices.reserve(std::min(source.coordinateCount, limits::MaximumVertexCount) * 3U);
312 output.mercatorFillIndices.reserve(
313 std::min(source.coordinateCount * 3U, limits::MaximumMercatorIndexCount));
314 output.sphereFillIndices.reserve(
315 std::min(source.coordinateCount * 4U, limits::MaximumSphereIndexCount));
316 output.borderIndices.reserve(
317 std::min(source.coordinateCount * 2U, limits::MaximumBorderIndexCount));
318
319 for (std::size_t polygonIndex = 0; polygonIndex < source.polygons.size(); ++polygonIndex) {
320 const SourcePolygon &polygon = source.polygons[polygonIndex];
321 PlanarPolygonMesh planar = PolygonTriangulator::triangulate(polygon);
322 if (planar.empty()) {
323 return {{},
324 QStringLiteral("Source polygon %1 could not be triangulated completely.")
325 .arg(polygonIndex)};
326 }
327 const std::vector<std::uint32_t> mercatorIndices = planar.fillIndices;
328 const std::size_t usedVertices = output.vertexCount();
329 if (usedVertices > limits::MaximumVertexCount ||
330 !refineForSphere(planar, limits::MaximumVertexCount - usedVertices,
331 limits::MaximumSphereIndexCount -
332 output.sphereFillIndices.size()) ||
333 !checkedAppendCount(output.mercatorFillIndices.size(), mercatorIndices.size(),
334 limits::MaximumMercatorIndexCount) ||
335 !checkedAppendCount(output.sphereFillIndices.size(), planar.fillIndices.size(),
336 limits::MaximumSphereIndexCount) ||
337 !checkedAppendCount(output.borderIndices.size(), planar.borderIndices.size(),
338 limits::MaximumBorderIndexCount)) {
339 return {{}, QStringLiteral("Map triangulation exceeded its resource budget.")};
340 }
341
342 const std::uint32_t baseVertex = static_cast<std::uint32_t>(usedVertices);
343 for (const SourceCoordinate &coordinate : planar.coordinates) {
344 const double mercatorY =
345 lar::map::MapProjection::projectLatitude(coordinate.latitudeDegrees);
346 if (!std::isfinite(coordinate.longitudeDegrees) ||
347 !std::isfinite(coordinate.latitudeDegrees) || !std::isfinite(mercatorY) ||
348 std::abs(coordinate.longitudeDegrees) > limits::MaximumAbsoluteLongitude ||
349 std::abs(coordinate.latitudeDegrees) > limits::MaximumAbsoluteLatitude ||
350 std::abs(mercatorY) > limits::MaximumAbsoluteMercatorY) {
351 return {{},
352 QStringLiteral("Map triangulation produced "
353 "an invalid coordinate.")};
354 }
355 output.vertices.push_back(static_cast<float>(coordinate.longitudeDegrees));
356 output.vertices.push_back(static_cast<float>(coordinate.latitudeDegrees));
357 output.vertices.push_back(static_cast<float>(mercatorY));
358 }
359 for (const std::uint32_t index : mercatorIndices) {
360 output.mercatorFillIndices.push_back(baseVertex + index);
361 }
362 for (const std::uint32_t index : planar.fillIndices) {
363 output.sphereFillIndices.push_back(baseVertex + index);
364 }
365 for (const std::uint32_t index : planar.borderIndices) {
366 output.borderIndices.push_back(baseVertex + index);
367 }
368 }
369 } catch (const std::bad_alloc &) {
370 return {{}, QStringLiteral("There is not enough memory to compile the map.")};
371 } catch (const std::length_error &) {
372 return {{}, QStringLiteral("The compiled map dimensions are invalid.")};
373 }
374
375 if (output.empty() || !buildLandIndex(output) ||
376 output.byteSize() > limits::MaximumPayloadBytes) {
377 return {{}, QStringLiteral("Map compilation produced no valid bounded mesh.")};
378 }
379 return {std::move(output), {}};
380}
