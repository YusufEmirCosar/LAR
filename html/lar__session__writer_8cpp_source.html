32bool LarSessionWriter::append(SessionTimestamp timestamp, const QByteArray &packet,
33 QString *error) {
34 if (!m_file) {
35 if (error)
36 *error = QStringLiteral("Session writer is not open");
37 return false;
38 }
39 if (packet.isEmpty() || packet.size() > int(lar::session::MaximumPacketSize)) {
40 if (error)
41 *error = QStringLiteral("Validated UDP packet has an unsupported size");
42 return false;
43 }
44 if (m_recordCount >= lar::session::MaximumRecordCount) {
45 if (error)
46 *error = QStringLiteral("Session contains too many UDP packages");
47 return false;
48 }
49
50 DecodedState state;
51 QString packetError;
52 if (!m_mapping.decode(packet, &state, &packetError)) {
53 if (error)
54 *error = QStringLiteral("Cannot record invalid UDP package: %1").arg(packetError);
55 return false;
56 }
57
58 const quint64 timestampMs = static_cast<quint64>(timestamp.milliseconds());
59 if (m_recordCount > 0 && timestampMs < m_lastTimestampMs) {
60 if (error)
61 *error = QStringLiteral("Session package time is not monotonic");
62 return false;
63 }
64
65 QByteArray encoded;
66 encoded.reserve(int(sizeof(quint64) + sizeof(quint32)) + packet.size());
67 QDataStream stream(&encoded, QIODevice::WriteOnly);
68 configureStream(stream);
69 stream << timestampMs << quint32(packet.size());
70 if (stream.writeRawData(packet.constData(), packet.size()) != packet.size() ||
71 stream.status() != QDataStream::Ok) {
72
73 if (error)
74 *error = QStringLiteral("Cannot encode UDP package record");
75 return false;
76 }
77
78 const qint64 previousSize = m_file->size();
79 if (!m_file->seek(previousSize) || m_file->write(encoded) != encoded.size()) {
80 const QString writeError = m_file->errorString();
81 const bool rolledBack = m_file->resize(previousSize) && m_file->seek(previousSize);
82 if (!rolledBack) {
83 const QString rollbackError = m_file->errorString();
84 cancel();
85 if (error) {
86 *error = QStringLiteral("Session append failed (%1) and rollback failed (%2); "
87 "the recording transaction was discarded")
88 .arg(writeError, rollbackError);
89 }
90 return false;
91 }
92 if (error)
93 *error = writeError;
94 return false;
95 }
96 m_lastTimestampMs = timestampMs;
97 ++m_recordCount;
98 return true;
99}

149bool LarSessionWriter::openSession(const QByteArray &mappingJson, QString *error) {
150 PacketMapping mapping;
151 QString mappingError;
152 if (!JsonMappingRepository().loadJson(mappingJson, &mapping, &mappingError)) {
153 if (error)
154 *error = QStringLiteral("Invalid session mapping: %1").arg(mappingError);
155 return false;
156 }
157
158 auto candidate = std::make_shared<QTemporaryFile>();
159 candidate->setAutoRemove(true);
160 if (!candidate->open()) {
161 if (error)
162 *error = candidate->errorString();
163 return false;
164 }
165
166 const QByteArray canonicalMapping = mapping.json();
167 const quint64 packetOffset =
168 quint64(lar::session::HeaderSize) + quint64(canonicalMapping.size());
169 if (packetOffset > std::numeric_limits<quint32>::max()) {
170 if (error)
171 *error = QStringLiteral("Session mapping exceeds the uint32 package offset");
172 return false;
173 }
174
175 QByteArray header;
176 header.reserve(qsizetype(packetOffset));
177 header.append("LAR1", 4);
178 QDataStream stream(&header, QIODevice::Append);
179 configureStream(stream);
180 stream << quint32(packetOffset);
181 header.append(canonicalMapping);
182 if (stream.status() != QDataStream::Ok || candidate->write(header) != header.size() ||
183 !candidate->flush()) {
184
185 if (error)
186 *error = candidate->errorString();
187 return false;
188 }
189
190 m_file = std::move(candidate);
191 m_mapping = std::move(mapping);
192 m_mappingJson = canonicalMapping;
193 m_lastTimestampMs = 0;
194 m_recordCount = 0;
195 return true;
196}
