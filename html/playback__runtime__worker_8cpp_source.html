42void PlaybackRuntimeWorker::initialize() {
43 if (m_playback)
44 return;
45 m_clock = m_clockFactory(this);
46 if (!m_clock) {
47 m_clock = new InertPlaybackClock(this);
48 emit runtimeError({{},
49 std::nullopt,
50 RuntimeFailureCode::Construction,
51 QStringLiteral("Playback clock factory returned null")});
52 }
53 m_playback = new PlaybackService(*m_reader, *m_clock, this);
54 m_metrics = new PlaybackMetrics(this);
55
56 // PlaybackService already samples at the presentation rate. Forwarding the
57 // sampled frame on that same tick avoids beating a separate 16 ms timer
58 // against the 60 Hz clock and silently coalescing otherwise valid frames.
59 connect(m_playback, &PlaybackService::frameReady, this, [this](const DecodedState &state) {
60 if (m_publicationDeferred) {
61 m_deferredState = state;
62 return;
63 }
64 emit stateReady({{RuntimeStateSource::Playback, m_stateGeneration}, state});
65 });
66 connect(m_playback, &PlaybackService::positionChanged, this, [this](SessionTimestamp position) {
67 if (m_publicationDeferred) {
68 m_deferredPosition = position;
69 return;
70 }
71 emit playbackPositionChanged({{RuntimeStateSource::Playback, m_stateGeneration}, position});
72 });
73 connect(m_playback, &PlaybackService::recordsProcessed, m_metrics, &PlaybackMetrics::record);
74 connect(m_playback, &PlaybackService::playingChanged, this, [this](bool playing) {
75 m_metrics->setActive(playing);
76 emit playbackPlayingChanged({{RuntimeStateSource::Playback, m_stateGeneration}, playing});
77 });
78 connect(m_playback, &PlaybackService::playbackFinished, this, [this] {
79 m_metrics->finish();
80 emit playbackFinished({{RuntimeStateSource::Playback, m_stateGeneration}});
81 });
82 connect(m_playback, &PlaybackService::playbackError, this, [this](const QString &message) {
83 emit runtimeError({{},
84 RuntimeSourceEpoch{RuntimeStateSource::Playback, m_stateGeneration},
85 RuntimeFailureCode::Operational,
86 message});
87 });
88 connect(m_metrics, &PlaybackMetrics::metricsChanged, this, [this](quint64 count, quint64 rate) {
89 emit metricsChanged({{RuntimeStateSource::Playback, m_stateGeneration}, count, rate});
90 });
91}

93void PlaybackRuntimeWorker::loadSession(const QString &path, quint64 generation,
94 RuntimeRequestId request) {
95 initialize();
96 m_stateGeneration = generation != 0 ? generation : m_stateGeneration + 1;
97 m_publicationDeferred = true;
98 clearDeferredPublication();
99 m_playback->closeSession();
100 clearDeferredPublication();
101 m_metrics->reset();
102
103 QString error;
104 const bool loaded = m_playback->loadSession(path, &error);
105 emit sessionLoadFinished({request,
106 {RuntimeStateSource::Playback, m_stateGeneration},
107 loaded,
108 path,
109 loaded ? m_playback->recordCount() : 0,
110 loaded ? m_playback->duration() : SessionTimestamp{},
111 error});
112
113 m_publicationDeferred = false;
114 if (loaded)
115 flushDeferredPublication();
116 else
117 clearDeferredPublication();
118}

120void PlaybackRuntimeWorker::closeSession(quint64 generation, RuntimeRequestId request) {
121 initialize();
122 const RuntimeSourceEpoch epoch{RuntimeStateSource::Playback,
123 generation != 0 ? generation : m_stateGeneration};
124 m_publicationDeferred = true;
125 m_playback->closeSession();
126 m_metrics->setActive(false);
127 clearDeferredPublication();
128 m_publicationDeferred = false;
129 emit sessionClosed({request, epoch, true, {}});
130}

207void PlaybackRuntimeWorker::flushDeferredPublication() {
208 const auto state = std::exchange(m_deferredState, std::nullopt);
209 const auto position = std::exchange(m_deferredPosition, std::nullopt);
210 if (state)
211 emit stateReady({{RuntimeStateSource::Playback, m_stateGeneration}, *state});
212 if (position) {
213 emit playbackPositionChanged(
214 {{RuntimeStateSource::Playback, m_stateGeneration}, *position});
