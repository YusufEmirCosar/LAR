17double angularDistance(double latitudeA, double longitudeA, double latitudeB, double longitudeB) {
18 const double deltaLongitude = longitudeB - longitudeA;
19 const double cosine =
20 std::clamp(std::sin(latitudeA) * std::sin(latitudeB) +
21 std::cos(latitudeA) * std::cos(latitudeB) * std::cos(deltaLongitude),
22 -1.0, 1.0);
23 return std::acos(cosine);
24}

28std::optional<LarMercatorFit>
29LarViewFit::mercator(const std::vector<Coordinate> &coordinatesRadians,
30 const std::optional<Coordinate> &trackedCenterRadians, double bearingDegrees,
31 const QSize &viewportSize) {
32 if (coordinatesRadians.empty() || viewportSize.width() <= 0 || viewportSize.height() <= 0) {
33
34 return std::nullopt;
35 }
36
37 const double referenceLongitude = (trackedCenterRadians)
38 ? (trackedCenterRadians->second * RadiansToDegrees)
39 : (coordinatesRadians.front().second * RadiansToDegrees);
40 std::vector<QPointF> projected;
41 projected.reserve(coordinatesRadians.size());
42 double previousLongitude = referenceLongitude;
43 for (const Coordinate &coordinate : coordinatesRadians) {
44 const double longitude = lar::map::MapProjection::unwrapLongitude(
45 coordinate.second * RadiansToDegrees, previousLongitude);
46 previousLongitude = longitude;
47 projected.emplace_back(longitude, lar::map::MapProjection::projectLatitude(
48 coordinate.first * RadiansToDegrees));
49 }
50
51 QPointF center;
52 if (trackedCenterRadians) {
53 center = {referenceLongitude, lar::map::MapProjection::projectLatitude(
54 trackedCenterRadians->first * RadiansToDegrees)};
55 } else {
56 double minimumX = projected.front().x();
57 double maximumX = minimumX;
58 double minimumY = projected.front().y();
59 double maximumY = minimumY;
60 for (const QPointF &point : projected) {
61 minimumX = std::min(minimumX, point.x());
62 maximumX = std::max(maximumX, point.x());
63 minimumY = std::min(minimumY, point.y());
64 maximumY = std::max(maximumY, point.y());
65 }
66 center = {(minimumX + maximumX) * 0.5, (minimumY + maximumY) * 0.5};
67 }
68
69 const double bearing = qDegreesToRadians(bearingDegrees);
70 const double cosine = std::cos(bearing);
71 const double sine = std::sin(bearing);
72 double requiredX = 0.001;
73 double requiredY = 0.001;
74 for (const QPointF &point : projected) {
75 const double deltaX = point.x() - center.x();
76 const double deltaY = point.y() - center.y();
77 requiredX = std::max(requiredX, std::abs(deltaX * cosine - deltaY * sine));
78 requiredY = std::max(requiredY, std::abs(deltaX * sine + deltaY * cosine));
79 }
80
81 const double aspect =
82 static_cast<double>(viewportSize.width()) / static_cast<double>(viewportSize.height());
83 double baseHalfWidth = 180.0;
84 double baseHalfHeight = 180.0;
85 if (aspect >= 1.0) {
86 baseHalfWidth = baseHalfHeight * aspect;
87 } else if (aspect > 0.0) {
88 baseHalfHeight = baseHalfWidth / aspect;
89 }
90
91 return LarMercatorFit{center, static_cast<float>(0.82 * std::min(baseHalfWidth / requiredX,
92 baseHalfHeight / requiredY))};
93}

95std::optional<LarSphereFit>
96LarViewFit::sphere(const std::vector<Coordinate> &coordinatesRadians,
97 const std::optional<Coordinate> &trackedCenterRadians) {
98 if (coordinatesRadians.empty()) {
99 return std::nullopt;
100 }
101
102 double centerLatitude = 0.0;
103 double centerLongitude = 0.0;
104 if (trackedCenterRadians) {
105 centerLatitude = trackedCenterRadians->first;
106 centerLongitude = trackedCenterRadians->second;
107 } else {
108 double x = 0.0;
109 double y = 0.0;
110 double z = 0.0;
111 for (const Coordinate &coordinate : coordinatesRadians) {
112 const double cosine = std::cos(coordinate.first);
113 x += cosine * std::cos(coordinate.second);
114 y += cosine * std::sin(coordinate.second);
115 z += std::sin(coordinate.first);
116 }
117 const double horizontal = std::hypot(x, y);
118 if (horizontal > 1.0e-12 || std::abs(z) > 1.0e-12) {
119 centerLatitude = std::atan2(z, horizontal);
120 centerLongitude = std::atan2(y, x);
121 } else {
122 centerLatitude = coordinatesRadians.front().first;
123 centerLongitude = coordinatesRadians.front().second;
124 }
125 }
126
127 double maximumSine = 1.0e-5;
128 bool crossesHorizon = false;
129 for (const Coordinate &coordinate : coordinatesRadians) {
130 const double angle =
131 angularDistance(centerLatitude, centerLongitude, coordinate.first, coordinate.second);
132 maximumSine = std::max(maximumSine, std::sin(std::min(angle, Pi * 0.5)));
133 crossesHorizon = crossesHorizon || angle >= Pi * 0.5;
134 }
135
136 return LarSphereFit{centerLongitude * RadiansToDegrees, centerLatitude * RadiansToDegrees,
137 static_cast<float>(0.82 / maximumSine), crossesHorizon};
138}
