13double aspectFactor(double aspectRadians) {

47DlzSolveError orderingError(double aspectRadians, double altitudeFeet, double shooterMach,
48 const WeaponModel &weapon) noexcept {
49 const double aero = weapon.referenceRangeNm * aspectFactor(aspectRadians) *
50 altitudeFactor(altitudeFeet) * machFactor(shooterMach);
51 const double noEscape = aero * (0.44 * aspectFactor(aspectRadians) + 0.08);
52 const double turnAndRun = 0.60 * noEscape;
53 if (!std::isfinite(aero) || !std::isfinite(noEscape) || !std::isfinite(turnAndRun)) {
54 return DlzSolveError::NumericOverflow;
55 }
56 if (turnAndRun <= weapon.baseMinimumRangeNm) {
57 return DlzSolveError::UnsupportedOrderedDomain;
58 }
59 return DlzSolveError::None;
60}

64bool supportsOrdering(double aspectRadians, double altitudeFeet, double shooterMach,
65 const WeaponModel &weapon) {
66 if (!std::isfinite(aspectRadians) || aspectRadians < 0.0 ||
67 aspectRadians > 3.14159265358979323846 || !std::isfinite(altitudeFeet) ||
68 altitudeFeet < 0.0 || !std::isfinite(shooterMach) || shooterMach < 0.0 ||
69 !std::isfinite(weapon.referenceRangeNm) || !std::isfinite(weapon.referenceAltitudeFeet) ||
70 !std::isfinite(weapon.referenceShooterMach) || !std::isfinite(weapon.baseMinimumRangeNm) ||
71 !std::isfinite(weapon.averageMissileSpeedKnots) || weapon.referenceRangeNm <= 0.0 ||
72 weapon.referenceAltitudeFeet <= 0.0 || weapon.referenceShooterMach < 0.0 ||
73 weapon.baseMinimumRangeNm <= 0.0 || weapon.averageMissileSpeedKnots <= 0.0) {
74 return false;
75 }
76 return orderingError(aspectRadians, altitudeFeet, shooterMach, weapon) == DlzSolveError::None;
77}

79Solution evaluateToyModel(const ShooterState &shooter, const Geometry &geometry,
80 const WeaponModel &weapon) noexcept {
81 const double shooterAltitudeFeet = -shooter.pos.z * units::FeetPerMeter;
82 const double fAspect = aspectFactor(geometry.aspectRadians);
83 const double fAlt = altitudeFactor(shooterAltitudeFeet);
84 const double fMach = machFactor(shooter.mach);
85
86 Solution solution;
87 solution.aerodynamicMaximumRangeNm = weapon.referenceRangeNm * fAspect * fAlt * fMach;
88 solution.interceptRangeNm = 0.83 * solution.aerodynamicMaximumRangeNm;
89 solution.noEscapeRangeNm = solution.aerodynamicMaximumRangeNm * (0.44 * fAspect + 0.08);
90 solution.turnAndRunRangeNm = 0.60 * solution.noEscapeRangeNm;
91 solution.minimumRangeNm = weapon.baseMinimumRangeNm;
92 solution.timeOfFlightSeconds = (geometry.rangeNm / weapon.averageMissileSpeedKnots) * 3600.0;
93 solution.inEnvelope = geometry.rangeNm > solution.minimumRangeNm && geometry.rangeNm <= solution.interceptRangeNm;
94 solution.shootCue = solution.inEnvelope;
95 return solution;
96}

98DlzSolveResult solve(const ShooterState &shooter, const TargetState &target,
99 const Atmosphere &atmosphere, const Geometry &geometry,
100 const WeaponModel &weapon) noexcept {
101 if (!finiteVector(shooter.pos) || !finiteVector(shooter.vel) || !finiteVector(target.pos) ||
102 !finiteVector(target.vel) || !std::isfinite(shooter.mach) || shooter.mach < 0.0 ||
103 !std::isfinite(target.mach) || target.mach < 0.0 || !std::isfinite(target.nzDefensive) ||
104 target.nzDefensive < 0.0 || !std::isfinite(shooter.psi) || !std::isfinite(shooter.theta) ||
105 !std::isfinite(shooter.phi) || !std::isfinite(target.psi) ||
106 !std::isfinite(atmosphere.rho) || atmosphere.rho <= 0.0 ||
107 !std::isfinite(atmosphere.temperature) || atmosphere.temperature <= 0.0 ||
108 !std::isfinite(atmosphere.pressure) || atmosphere.pressure <= 0.0 ||
109 !std::isfinite(atmosphere.speedOfSound) || atmosphere.speedOfSound <= 0.0 ||
110 !finiteGeometry(geometry) || geometry.rangeNm <= 0.0 ||
111 !std::isfinite(weapon.referenceRangeNm) || weapon.referenceRangeNm <= 0.0 ||
112 !std::isfinite(weapon.referenceAltitudeFeet) || weapon.referenceAltitudeFeet <= 0.0 ||
113 !std::isfinite(weapon.referenceShooterMach) || weapon.referenceShooterMach < 0.0 ||
114 !std::isfinite(weapon.baseMinimumRangeNm) || weapon.baseMinimumRangeNm <= 0.0 ||
115 !std::isfinite(weapon.averageMissileSpeedKnots) || weapon.averageMissileSpeedKnots <= 0.0) {
116 return {{}, DlzSolveError::InvalidInput};
117 }
118 // The altitude input is carried by the scenario adapter in the shooter's
119 // NED z coordinate. The solver uses the declared mean-sea-level datum.
120 const double shooterAltitudeFeet = -shooter.pos.z * units::FeetPerMeter;
121 const DlzSolveError orderError =
122 orderingError(geometry.aspectRadians, shooterAltitudeFeet, shooter.mach, weapon);
123 if (orderError != DlzSolveError::None) {
124 return {{}, orderError};
125 }
126
127 const Solution solution = evaluateToyModel(shooter, geometry, weapon);
128
129 if (!std::isfinite(solution.aerodynamicMaximumRangeNm) ||
130 !std::isfinite(solution.interceptRangeNm) || !std::isfinite(solution.noEscapeRangeNm) ||
131 !std::isfinite(solution.turnAndRunRangeNm) || !std::isfinite(solution.minimumRangeNm) ||
132 !std::isfinite(solution.timeOfFlightSeconds)) {
133 return {{}, DlzSolveError::NumericOverflow};
134 }
135 if (!validSolutionOrder(solution)) {
136 return {{}, DlzSolveError::InvalidDerivedOrdering};
137 }
138 return {solution, DlzSolveError::None};
139}
