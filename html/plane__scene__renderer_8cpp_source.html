78bool compileProgram(QOpenGLShaderProgram &program, const char *legacyVertex,
79 const char *legacyFragment, const char *coreVertex, const char *coreFragment,
80 bool normalAttribute, QString *errorMessage) {
81 bool vertexCompiled = program.addShaderFromSourceCode(QOpenGLShader::Vertex, legacyVertex);
82 bool fragmentCompiled =
83 program.addShaderFromSourceCode(QOpenGLShader::Fragment, legacyFragment);
84 program.bindAttributeLocation("aPosition", 0);
85 if (normalAttribute) {
86 program.bindAttributeLocation("aNormal", 1);
87 program.bindAttributeLocation("aTexCoord", 2);
88 }
89 bool linked = program.link();
90 if (!vertexCompiled || !fragmentCompiled || !linked) {
91 program.removeAllShaders();
92 vertexCompiled = program.addShaderFromSourceCode(QOpenGLShader::Vertex, coreVertex);
93 fragmentCompiled = program.addShaderFromSourceCode(QOpenGLShader::Fragment, coreFragment);
94 program.bindAttributeLocation("aPosition", 0);
95 if (normalAttribute) {
96 program.bindAttributeLocation("aNormal", 1);
97 program.bindAttributeLocation("aTexCoord", 2);
98 }
99 linked = program.link();
100 }
101 if (!vertexCompiled || !fragmentCompiled || !linked) {
102 setError(errorMessage,
103 QStringLiteral("Plane shader initialization failed: %1").arg(program.log()));
104 return false;
105 }
106 return true;
107}

182bool PlaneSceneRenderer::uploadInterleaved(const std::vector<float> &vertices,
183 const std::vector<std::uint32_t> &indices,
184 QOpenGLBuffer &vertexBuffer, QOpenGLBuffer &indexBuffer,
185 QOpenGLVertexArrayObject &vertexArray,
186 QString *errorMessage) {
187 if (vertices.empty() || vertices.size() % PlaneModelVertexStrideFloats != 0U ||
188 vertices.size() >
189 static_cast<std::size_t>(std::numeric_limits<int>::max()) / sizeof(float) ||
190 indices.size() >
191 static_cast<std::size_t>(std::numeric_limits<int>::max()) / sizeof(std::uint32_t)) {
192 setError(errorMessage, QStringLiteral("Plane geometry exceeds GPU buffer limits."));
193 return false;
194 }
195 if ((!vertexBuffer.isCreated() && !vertexBuffer.create()) ||
196 (!vertexArray.isCreated() && !vertexArray.create()) ||
197 (!indices.empty() && !indexBuffer.isCreated() && !indexBuffer.create())) {
198 setError(errorMessage, QStringLiteral("Plane GPU geometry could not be created."));
199 return false;
200 }
201 m_meshProgram->bind();
202 QOpenGLVertexArrayObject::Binder binder(&vertexArray);
203 vertexBuffer.bind();
204 vertexBuffer.allocate(vertices.data(), static_cast<int>(vertices.size() * sizeof(float)));
205 m_meshProgram->enableAttributeArray(0);
206 m_meshProgram->enableAttributeArray(1);
207 m_meshProgram->enableAttributeArray(2);
208 m_meshProgram->setAttributeBuffer(0, GL_FLOAT, 0, 3, VertexStrideBytes);
209 m_meshProgram->setAttributeBuffer(1, GL_FLOAT, 3 * static_cast<int>(sizeof(float)), 3,
210 VertexStrideBytes);
211 m_meshProgram->setAttributeBuffer(2, GL_FLOAT, 6 * static_cast<int>(sizeof(float)), 2,
212 VertexStrideBytes);
213 if (!indices.empty()) {
214 indexBuffer.bind();
215 indexBuffer.allocate(indices.data(),
216 static_cast<int>(indices.size() * sizeof(std::uint32_t)));
217 }
218 m_meshProgram->release();
219 return true;
220}

244bool PlaneSceneRenderer::uploadModel(QString *errorMessage) {
245 if (m_model == nullptr || m_model->empty()) {
246 setError(errorMessage,
247 QStringLiteral("The selected plane model has no renderable geometry."));
248 return false;
249 }
250 if (!uploadInterleaved(m_model->vertices, m_model->indices, m_modelVertices, m_modelIndices,
251 m_modelArray, errorMessage)) {
252 return false;
253 }
254 m_modelTextures.clear();
255 m_whiteModelTexture = std::make_unique<QOpenGLTexture>(QOpenGLTexture::Target2D);
256 m_whiteModelTexture->setSize(1, 1);
257 m_whiteModelTexture->setFormat(QOpenGLTexture::RGBA8_UNorm);
258 m_whiteModelTexture->setMipLevels(1);
259 m_whiteModelTexture->allocateStorage(QOpenGLTexture::RGBA, QOpenGLTexture::UInt8);
260 constexpr std::uint32_t WhitePixel = 0xFFFFFFFFU;
261 m_whiteModelTexture->setData(0, QOpenGLTexture::RGBA, QOpenGLTexture::UInt8, &WhitePixel);
262 m_whiteModelTexture->setMipLevelRange(0, 0);
263 m_whiteModelTexture->setMinMagFilters(QOpenGLTexture::Linear, QOpenGLTexture::Linear);
264 if (!m_whiteModelTexture->isCreated() || !m_whiteModelTexture->isStorageAllocated()) {
265 setError(errorMessage, QStringLiteral("The fallback plane texture could not be uploaded."));
266 return false;
267 }
268 m_modelTextures.reserve(m_model->textures.size());
269 for (const PlaneModelTexture &source : m_model->textures) {
270 if (source.image.isNull() || source.image.format() != QImage::Format_RGBA8888) {
271 setError(
272 errorMessage,
273 QStringLiteral("The selected plane model contains an invalid decoded texture."));
274 return false;
275 }
276 auto texture = std::make_unique<QOpenGLTexture>(QOpenGLTexture::Target2D);
277 texture->setSize(source.image.width(), source.image.height());
278 texture->setFormat(QOpenGLTexture::RGBA8_UNorm);
279 const bool mipmapped = usesMipmaps(source.minFilter);
280 const int mipLevels = mipmapped ? mipLevelCount(source.image.size()) : 1;
281 texture->setMipLevels(mipLevels);
282 texture->allocateStorage(QOpenGLTexture::RGBA, QOpenGLTexture::UInt8);
283 texture->setData(0, QOpenGLTexture::RGBA, QOpenGLTexture::UInt8, source.image.constBits());
284 texture->setMipLevelRange(0, mipLevels - 1);
285 texture->setWrapMode(QOpenGLTexture::DirectionS, textureWrapMode(source.wrapS));
286 texture->setWrapMode(QOpenGLTexture::DirectionT, textureWrapMode(source.wrapT));
287 texture->setMinificationFilter(textureMinificationFilter(source.minFilter));
288 texture->setMagnificationFilter(textureMagnificationFilter(source.magFilter));
289 if (mipmapped) {
290 texture->generateMipMaps(0, false);
291 }
292 if (!texture->isCreated() || !texture->isStorageAllocated()) {
293 setError(errorMessage,
294 QStringLiteral("A plane-model base-color texture could not be uploaded."));
295 return false;
296 }
297 m_modelTextures.push_back(std::move(texture));
298 }
299 m_modelPending = false;
300 return true;
301}

346 const QMatrix4x4 &model, const QMatrix4x4 &view,
347 const QMatrix4x4 &projection, const DrawRange &range, bool unlit,
348 float pointSize) {

351 return;
352 }
353 m_meshProgram->bind();
354 m_meshProgram->setUniformValue("uModel", model);
355 m_meshProgram->setUniformValue("uView", view);
356 m_meshProgram->setUniformValue("uProjection", projection);
357 m_meshProgram->setUniformValue("uNormalMatrix", model.normalMatrix());
358 m_meshProgram->setUniformValue("uColor", range.color);
359 m_meshProgram->setUniformValue("uLightDirection", QVector3D(0.35F, 0.85F, 0.4F));

361 m_meshProgram->setUniformValue("uPointSize", pointSize);
362 QOpenGLTexture *texture = nullptr;
363 if (range.textureIndex >= 0 &&
364 static_cast<std::size_t>(range.textureIndex) < m_modelTextures.size()) {
365 texture = m_modelTextures[static_cast<std::size_t>(range.textureIndex)].get();
366 }
367 m_meshProgram->setUniformValue("uHasBaseColorTexture", texture != nullptr);
368 constexpr int ModelTextureUnit = 1;
369 m_meshProgram->setUniformValue("uBaseColorTexture", ModelTextureUnit);
370 QOpenGLTexture *boundTexture = texture != nullptr ? texture : m_whiteModelTexture.get();
371 boundTexture->bind(ModelTextureUnit);
372 QOpenGLVertexArrayObject::Binder binder(&vertexArray);
373 indexBuffer.bind();
374 glDrawElements(range.primitive, static_cast<GLsizei>(range.indexCount), GL_UNSIGNED_INT,
375 reinterpret_cast<const void *>(range.firstIndex * sizeof(std::uint32_t)));
376 boundTexture->release(ModelTextureUnit);
377 m_meshProgram->release();
378}

428bool PlaneSceneRenderer::draw(const QMatrix4x4 &view, const QMatrix4x4 &projection,
429 const QQuaternion &aircraftAttitude, QString *errorMessage) {
430 if (!m_ready) {
431 setError(errorMessage, QStringLiteral("Plane GPU resources are not ready."));
432 return false;
433 }
434 if ((m_modelPending && !uploadModel(errorMessage)) || !uploadSkybox(errorMessage)) {
435 return false;
436 }
437 const QVector4D clearColor(0.08F, 0.1F, 0.14F, 1.0F);
438 glClearColor(clearColor.x(), clearColor.y(), clearColor.z(), clearColor.w());
439 glEnable(GL_DEPTH_TEST);
440 glDepthMask(GL_TRUE);
441 glDepthFunc(GL_LESS);
442 glDisable(GL_BLEND);
443 glDisable(GL_CULL_FACE);
444 glFrontFace(GL_CCW);
445 glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
446 drawSkybox(view, projection);
447 bool terrainDrawn = false;
448 if (m_terrainVisible) {
449 if (!m_terrainLayer.draw(view, projection, m_terrainOffsetXZ, m_terrainScaleXZ,
450 m_aircraftAltitudeScene, errorMessage)) {
451 return false;
452 }
453 terrainDrawn = m_terrainLayer.hasRenderableGeometry();
454 }
455 // A terrain-enabled scene must not silently fall back to land-colored flat ground while the
456 // asynchronous patch is loading. That transient surface made a near-shore target appear on
457 // land in one frame and at sea after the patch arrived. Keep the tactical overlays available,
458 // but only draw the flat ground when terrain is not the selected source.
459 const bool drawFlatGround = !m_terrainVisible && !terrainDrawn;
460 if (m_surfaceVisible || drawFlatGround) {
461 PlaneSurfaceState surfaceState = m_surfaceState;
462 if (terrainDrawn) {
463 const std::optional<float> groundHeight =
464 m_terrainLayer.centerGroundHeight(m_aircraftAltitudeScene);
465 if (groundHeight) {
466 surfaceState.surfaceHeight = *groundHeight;
467 }
468 if (surfaceState.targetVisible) {
469 const std::optional<float> targetHeight =
470 m_terrainLayer.groundHeightAt(surfaceState.targetXZ, m_terrainOffsetXZ,
471 m_terrainScaleXZ, m_aircraftAltitudeScene);
472 if (targetHeight) {
473 surfaceState.targetHeight = *targetHeight;
474 }
475 }
476 }
477 m_surfaceLayer.draw(surfaceState, view, projection, drawFlatGround, m_surfaceVisible);
478 }
479 drawAircraft(view, projection, aircraftAttitude);
480 return true;
481}
