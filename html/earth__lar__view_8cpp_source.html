122void EarthLarView::installMapAsset(quint64 revision, const lar::map::MapAssetReadResult &result) {
123 if (!result.succeeded()) {
124 setRendererDiagnostic(result.message.isEmpty()
125 ? QStringLiteral("The packaged world map could not be loaded.")
126 : result.message);
127 emit mapUnavailable();
128 return;
129 }
130
131 QString error;
132 if (!setMapMesh(result.mesh, &error)) {
133 const QString message = QStringLiteral("The GPU rejected the world map: %1").arg(error);
134 m_mapLoader->completeInstallation(revision, false, message);
135 setRendererDiagnostic(message);
136 emit mapUnavailable();
137 return;
138 }
139
140 m_mapLoader->completeInstallation(revision, true);
141 m_rendererMessage.clear();
142 applyTrackedCamera();
143 markFallbackGeometryDirty();
144 refreshOverlays();
145 update();
146}

188void EarthLarView::setState(const Plane &plane, const Target &target,
189 const QBitArray &availableFields) {
190 const bool geometryChanged =

192 EarthCameraPolicy::fieldsChanged(
193 m_plane, m_target, m_availableFields, m_hasState, plane, target, availableFields,
194 {StateField::IrPos0, StateField::IrPos1, StateField::IrR, StateField::IzPos0,
195 StateField::IzPos1, StateField::IzTheta1, StateField::IzTheta2, StateField::IzR1,
196 StateField::IzR2});
197 const bool markerChanged =
198 !m_hasState || EarthCameraPolicy::fieldsChanged(m_plane, m_target, m_availableFields,
199 m_hasState, plane, target, availableFields,
200 {StateField::Location0,
201 StateField::Location1, StateField::Euler0,
202 StateField::IzPos0, StateField::IzPos1});
203 m_plane = plane;
204 m_target = target;
205 m_availableFields = availableFields;
206 m_hasState = true;
207 updateZoneBackend();
208 if (geometryChanged)
209 markFallbackGeometryDirty();
210 if (geometryChanged || markerChanged) {
211 update();

213 }
