84void GridCameraTransform::pan(const QPoint &pixelDelta) noexcept {
85 if (!m_hasLocation || m_scale <= 0.0) {
86 return;
87 }
88 const double eastMeters = -static_cast<double>(pixelDelta.x()) / m_scale;
89 const double northMeters = static_cast<double>(pixelDelta.y()) / m_scale;
90 m_location[0] =
91 std::clamp(m_location[0] + northMeters / EarthRadiusMeters, -Pi * 0.5, Pi * 0.5);
92 const double cosineLatitude = qMax(0.01, std::cos(m_location[0]));
93 m_location[1] += eastMeters / (EarthRadiusMeters * cosineLatitude);
94 while (m_location[1] > Pi) {
95 m_location[1] -= 2.0 * Pi;
96 }
97 while (m_location[1] < -Pi) {
98 m_location[1] += 2.0 * Pi;
99 }
100}
