80bool SourceLifecycleCoordinator::startOnline(quint16 port) {
81 if (m_state == State::ShuttingDown || port == 0)
82 return false;
83 if (m_state == State::Online && m_applicationState.listening &&
84 m_applicationState.listeningPort == port) {
85 return true;
86 }
87 m_desiredSource = DesiredSource::Online;
88 m_desiredPort = port;
89 transitionBoundary();
90 if (m_state == State::StartingOnline || m_state == State::StoppingOnlineForPlayback ||
91 m_state == State::LoadingPlayback || m_state == State::ClosingPlaybackForOnline) {
92 return true;
93 }
94 if (m_state == State::Online)
95 return beginStopOnline();
96 if (m_state == State::SessionLoaded || m_state == State::Playing ||
97 m_state == State::PlaybackPaused) {
98 return beginClosePlayback();
99 }
100 return beginStartOnline();
101}

297void SourceLifecycleCoordinator::handle(const OnlineStartResult &result) {
298 if (result.request != m_pendingStart)
299 return;
300 m_pendingStart = {};
301 const quint16 completedPort = m_pendingStartPort;
302 m_pendingStartPort = 0;
303 if (!result.started) {
304 setState(State::Idle);
305 if (!result.error.isEmpty())
306 emit errorRaised(result.error);
307 if (m_desiredSource == DesiredSource::Online && m_desiredPort == completedPort) {
308 m_desiredSource = DesiredSource::None;
309 }
310 m_transitionBoundaryActive = false;
311 advanceIntent();
312 return;
313 }
314 if (m_desiredSource != DesiredSource::Online || m_desiredPort != completedPort) {
315 clearDeferredPublications();
316 beginStopOnline();
317 return;
318 }
319 m_activeEpoch = result.epoch;
320 m_transitionBoundaryActive = false;
321 replayDeferredPublications();
322}

342void SourceLifecycleCoordinator::handle(const SessionLoadResult &result) {
343 if (result.request != m_pendingLoad)
344 return;
345 m_pendingLoad = {};
346 const QString completedPath = m_pendingLoadPath;
347 m_pendingLoadPath.clear();
348 if (!result.loaded) {
349 setState(State::Idle);
350 if (!result.error.isEmpty())
351 emit errorRaised(result.error);
352 if (m_desiredSource == DesiredSource::Playback && m_desiredSessionPath == completedPath) {
353 m_desiredSource = DesiredSource::None;
354 }
355 m_transitionBoundaryActive = false;
356 advanceIntent();
357 return;
358 }
359 if (m_desiredSource != DesiredSource::Playback || result.path != m_desiredSessionPath) {
360 clearDeferredPublications();
361 beginClosePlayback();
362 return;
363 }
364 m_activeEpoch = result.epoch;
365 m_applicationState.sessionLoaded = true;
366 m_applicationState.loadedRecordCount = result.recordCount;
367 m_viewModel.setPlaybackDuration(result.duration);
368 setState(State::SessionLoaded);
369 transitionTo(ApplicationMode::SessionLoaded);
370 emit sessionLoaded(result.path, result.recordCount);
371 m_transitionBoundaryActive = false;
372 replayDeferredPublications();
373}

410void SourceLifecycleCoordinator::accept(const OnlineStateEvent &event) {
411 if (m_dispatching != DispatchKind::None) {
412 m_deferredOnlineState = event;
413 return;
414 }
415 if (!accepts(event.epoch) || m_desiredSource != DesiredSource::Online)
416 return;
417 m_applicationState.listening = event.listening;
418 m_applicationState.listeningPort = event.listening ? event.port : 0;
419 setState(event.listening ? State::Online : State::Idle);
420 if (event.listening) {
421 transitionTo(ApplicationMode::Online);
422 m_viewModel.setStatusText(QStringLiteral("Listening on UDP port %1").arg(event.port));
423 } else {
424 m_activeEpoch = {};
425 m_desiredSource = DesiredSource::None;
426 }
427 emit onlineStateChanged(event.listening);
428}
