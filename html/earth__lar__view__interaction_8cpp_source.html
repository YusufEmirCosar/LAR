13void EarthLarView::applyTrackedCamera() {
14 const auto tracked = EarthCameraPolicy::trackedCoordinate(m_cameraState, m_hasState);
15 if (tracked) {
16 constexpr double ToDegrees = 180.0 / LarGeodesicGeometry::Pi;
17 const double latitude = tracked->first * ToDegrees;
18 const double longitude = tracked->second * ToDegrees;
19 if (m_viewMode == LarViewMode::Mercator) {
20 setFlatCenter({longitude, lar::map::MapProjection::projectLatitude(latitude)});
21 } else {
22 setRotation(longitude, latitude);
23 }
24 }
25 setCameraBearing(static_cast<float>(EarthCameraPolicy::bearingDegrees(m_cameraState)));
26}

28void EarthLarView::rotateSphere(const QPoint &delta) {
29 if (m_viewMode != LarViewMode::Sphere || m_cameraState.mode != CameraTrackingMode::Free)
30 return;
31 const QPointF center = rotation();
32 const double sensitivity = 0.25 / std::max(0.2, static_cast<double>(activeZoom()));
33 const double longitude =
34 lar::map::MapProjection::wrapLongitude(center.x() - delta.x() * sensitivity);
35 const double latitude = std::clamp(center.y() + delta.y() * sensitivity, -90.0, 90.0);
36 setRotation(longitude, latitude);
37 markFallbackGeometryDirty();
38 refreshOverlays();
39}

45void EarthLarView::fitToData() {
46 if (!m_hasState || width() <= 0 || height() <= 0)
47 return;
48 applyTrackedCamera();
49 const auto tracked = EarthCameraPolicy::trackedCoordinate(m_cameraState, m_hasState);
50 const auto coordinates =
51 EarthCameraPolicy::fitCoordinates(m_plane, m_target, m_availableFields, m_hasState);
52 if (m_viewMode == LarViewMode::Sphere) {
53 const auto fit = LarViewFit::sphere(coordinates, tracked);
54 if (!fit)
55 return;
56 setRotation(fit->longitudeDegrees, fit->latitudeDegrees);
57 setSphereZoom(fit->zoom);
58 if (fit->crossesHorizon) {
59 emit diagnosticRaised(
60 QStringLiteral("The selected tracking anchor cannot show all LAR geometry "
61 "on one globe hemisphere."));
62 }
63 } else {
64 const auto fit = LarViewFit::mercator(coordinates, tracked,
65 static_cast<double>(cameraBearing()), size());
66 if (!fit)
67 return;
68 setFlatZoom(fit->zoom);
69 setFlatCenter(fit->center);
70 }
71 updateZoneBackend();
72 markFallbackGeometryDirty();
73 update();
74 refreshOverlays();
75}
